    static readonly ConditionalWeakTable<HeroFullSceneViewBase,State> states=new ConditionalWeakTable<HeroFullSceneViewBase,State>();
    sealed class LayoutState { public int Width,Height,RefreshThrough; public Transform Root,Parent; public Camera Camera; }
    static readonly ConditionalWeakTable<BaseWindow,LayoutState> layouts=new ConditionalWeakTable<BaseWindow,LayoutState>();
    public static bool IsScreenLayout(Transform root) {
        if(root==null)return false;
        var rect=root.GetComponent<UIRect>();
        if(rect==null)return false;
        var left=rect.leftAnchor;var right=rect.rightAnchor;
        if(left.target==null || left.target!=right.target || left.relative!=0 || right.relative!=1 || left.absolute!=0 || right.absolute!=0)return false;
        var panel=left.target.GetComponent<UIPanel>();
        return left.target.GetComponent<Camera>()!=null || panel!=null && panel.clipping==UIDrawCall.Clipping.None;
    }
    public static void InvalidateLayout(BaseWindow window) {
        if(window==null)return;
        layouts.Remove(window);
        State state;
        var preview=window as HeroFullSceneViewBase;
        if(preview!=null && states.TryGetValue(preview,out state))RestoreShadows(state);
    }
    public static Transform LayoutRoot(BaseWindow window) {
        // Lobby menus anchor their children directly to panels/cameras and do
        // not all have a stretchable Panel/Root widget.
        if(window is LobbyTopmostMenu || window is LobbyRightMenu || window is LobbyRightGridWindow)return window.transform;
        var root=window.transform.Find("Panel/Root");
        return window is HeroDeck2 || SupportsWindow(window.GetType()) || IsScreenLayout(root)?root:null;
    }
    public static void RefreshLayout(BaseWindow window) {
        if(window==null || !window.isActiveAndEnabled || !window.IsOpen())return;
        var root=LayoutRoot(window);
        if(root==null || !root.gameObject.activeInHierarchy)return;
        var state=layouts.GetOrCreateValue(window);
        var camera=NGUITools.FindCameraForLayer(root.gameObject.layer);
        if(state.Root!=root || state.Parent!=root.parent || state.Camera!=camera || state.Width!=Screen.width || state.Height!=Screen.height)
            state.RefreshThrough=Time.frameCount+2;
        if(Time.frameCount>state.RefreshThrough)return;
        RefreshLayoutRoot(root);
        state.Root=root;state.Parent=root.parent;state.Camera=camera;state.Width=Screen.width;state.Height=Screen.height;
    }
    public static void RefreshLayoutRoot(Transform root) {
        // These prefabs size their root only at Start. Recycled windows and
        // fullscreen changes otherwise retain a centered 1920-wide canvas.
        // Re-evaluate native anchors, including the frame, tiled background,
        // scroll viewport and edge-aligned controls, without scaling content.
        foreach(var rect in root.GetComponentsInChildren<UIRect>(true))rect.ResetAnchors();
        foreach(var rect in root.GetComponentsInChildren<UIRect>(true))rect.ForceUpdateAnchors(false);
    }

    public static bool SupportsWindow(Type type) { return typeof(HeroModelView).IsAssignableFrom(type)||typeof(HeroCostumeView).IsAssignableFrom(type)||typeof(HeroDetailView).IsAssignableFrom(type); }
    static bool Eligible(HeroFullSceneViewBase window) {
        if(window==null || !SupportsWindow(window.GetType()) || !window.isActiveAndEnabled || !window.IsOpen())return false;
        return true;
    }
    static State Get(HeroFullSceneViewBase window) {
        var state=states.GetOrCreateValue(window);
        if(!ReferenceEquals(state.Controller,window.CameraController)){RestoreShadows(state);state.Reset();state.Controller=window.CameraController;state.AnchorModel=null;state.Head=null;}
        return state;
    }
    static float ReadSensitivity(string key) { return Mathf.Clamp(SprkTestPrefs.GetInt("Sprk.Graphics."+key,SprkTestPrefs.GetInt("Sprk.Graphics.CameraSensitivity",100)),50,150)/100f; }
    static float Sensitivity { get { return ReadSensitivity("ModelCameraSensitivity"); } }
    static float LobbySensitivity { get { return ReadSensitivity("LobbyCameraSensitivity"); } }
    static Vector2 ScaleDrag(Vector2 delta) { return delta*Sensitivity; }
    static void ZoomStep(State state,float wheel) { state.Zoom=Mathf.Clamp(state.Zoom*Mathf.Exp(-wheel*.15f*Sensitivity),.02f,8); }
    public static Vector2 ScaleLobbyDrag(Vector2 delta) {
        return Application.isMobilePlatform || Input.touchCount>0?delta:delta*LobbySensitivity;
    }
    public static float ScaleLobbyZoom(float delta) {
        return Application.isMobilePlatform || Input.touchCount>0?delta:delta*LobbySensitivity;
    }
    public static void Reset(HeroFullSceneViewBase window) {
        if(window==null)return;
        State state;if(states.TryGetValue(window,out state))RestoreShadows(state);
        states.Remove(window);layouts.Remove(window);
    }
    public static bool InputCamera(HeroFullSceneViewBase window) {
        if(!Eligible(window)||Application.isMobilePlatform||Input.touchCount>0)return false;
        var state=Get(window);
        if(window.CameraController==null || !Application.isFocused || UICamera.inputHasFocus){state.Button=-1;return true;}
        var texture=window.Hero3DTexture;
        var hovered=UICamera.hoveredObject;
        bool over=OwnsPointer(texture,hovered);
        Vector2 mouse=Input.mousePosition;
        if(state.Button>=0 && !Input.GetMouseButton(state.Button))state.Button=-1;
        if(state.Button<0 && over) {
            for(int button=0;button<3;button++)if(Input.GetMouseButtonDown(button)){state.Button=button;state.Last=mouse;break;}
        }
        if(state.Button>=0) {
            Vector2 delta=ScaleDrag(mouse-state.Last);state.Last=mouse;
            if(state.Button==0 && !Input.GetKey(KeyCode.LeftShift) && !Input.GetKey(KeyCode.RightShift)) {
                Orbit(state,delta);
            } else {
                var cinema=cinemaField.GetValue(window) as RenderTargetCinema;
                if(cinema!=null && cinema.CinemaCamera!=null) {
                    var info=cinema.CinemaCamera.CameraInfo;
                    var rotation=Quaternion.LookRotation(info.LookDir);
                    float units=2*Mathf.Max(.15f,info.TargetDistance)*Mathf.Tan(info.FieldOfView*Mathf.Deg2Rad*.5f)/Mathf.Max(1,Screen.height);
                    state.Pan=Vector3.ClampMagnitude(state.Pan-rotation*new Vector3(delta.x,delta.y,0)*units,40);
                }
            }
        }
        if(over) {
            ZoomStep(state,Input.mouseScrollDelta.y);
            if(Input.GetKeyDown(KeyCode.R))state.Reset();
        }
        return true;
    }
    public static bool OwnsPointer(UITexture texture,GameObject hovered) {
        return texture!=null && hovered!=null && (hovered==texture.gameObject || (hovered.transform.IsChildOf(texture.transform) && hovered.GetComponentInParent<UIButton>()==null && hovered.GetComponentInParent<UIInput>()==null));
    }
    static void Orbit(State state,Vector2 delta) {
        state.Yaw=Mathf.Repeat(state.Yaw+delta.x*.3f+180,360)-180;
        state.Pitch=Mathf.Clamp(state.Pitch-delta.y*.3f,-85,85);
    }

        bool[] hadKeys=Array.ConvertAll(sensitivityKeys,k=>SprkTestPrefs.HasKey(k));

        check(Sensitivity==1 && LobbySensitivity==1,"Default camera sensitivities");
        SprkTestPrefs.SetInt("Sprk.Graphics.CameraSensitivity",75);
        check(Sensitivity==.75f && LobbySensitivity==.75f,"Legacy sensitivity fallback for both cameras");
        SprkTestPrefs.SetInt("Sprk.Graphics.LobbyCameraSensitivity",50);
        foreach(int percent in new[]{50,75,100,125,150}) {
            SprkTestPrefs.SetInt("Sprk.Graphics.ModelCameraSensitivity",percent);
            var sample=new State();var drag=ScaleDrag(new Vector2(100,40));Orbit(sample,drag);
            check(Mathf.Abs(sample.Yaw-30*percent/100f)<.001f && Mathf.Abs(sample.Pitch+12*percent/100f)<.001f,"Scaled orbit sensitivity");
            check(Vector2.Distance(drag,new Vector2(percent,percent*.4f))<.001f,"Scaled pan input");
            check(LobbySensitivity==.5f,"Model setting does not change lobby");
            ZoomStep(sample,1);check(Mathf.Abs(sample.Zoom-Mathf.Exp(-.15f*percent/100f))<.001f,"Scaled wheel sensitivity");
            sample.Reset();check(Sensitivity==percent/100f,"Reset preserves selected sensitivity");

        // Invoke the native lobby camera methods so scaling is verified at

    public static Vector3 Direction(Vector3 original,float yaw,float pitch) {
        original=original.sqrMagnitude<.0001f?Vector3.forward:original.normalized;
        float baseYaw=Mathf.Atan2(original.x,original.z)*Mathf.Rad2Deg;
        float basePitch=-Mathf.Asin(Mathf.Clamp(original.y,-1,1))*Mathf.Rad2Deg;
        return Quaternion.Euler(Mathf.Clamp(basePitch+pitch,-85,85),baseYaw+yaw,0)*Vector3.forward;
    }

    public static void FitPreviewAspect(Camera camera,UITexture texture) {
        if(camera==null || texture==null || texture.mainTexture==null)return;
        var uiCamera=texture.anchorCamera;
        if(uiCamera==null)return;
        // The render target can retain the startup resolution after switching
        // fullscreen. Its pixel ratio is not the ratio of the displayed quad.
        var rect=texture.drawingDimensions;
        var tm=texture.transform;
        var bottomLeft=uiCamera.WorldToScreenPoint(tm.TransformPoint(rect.x,rect.y,0));
        var bottomRight=uiCamera.WorldToScreenPoint(tm.TransformPoint(rect.z,rect.y,0));
        var topLeft=uiCamera.WorldToScreenPoint(tm.TransformPoint(rect.x,rect.w,0));
        float width=Vector3.Distance(bottomLeft,bottomRight),height=Vector3.Distance(bottomLeft,topLeft);
        var uv=texture.uvRect;
        if(width<.001f || height<.001f || Mathf.Abs(uv.width)<.0001f || Mathf.Abs(uv.height)<.0001f)return;
        camera.aspect=width/height*Mathf.Abs(uv.height/uv.width);
    }
    public static void Apply(HeroFullSceneViewBase window) {
        if(!Eligible(window))return;
        var state=Get(window);
        var cinema=cinemaField.GetValue(window) as RenderTargetCinema;
        if(cinema==null)return;
        FitPreviewAspect(cinema.MainCamera,window.Hero3DTexture);
        if(cinema.CinemaCamera==null || window.CameraController==null)return;
        var info=cinema.CinemaCamera.CameraInfo;
        var facade=((MotionSequence.IDataProvider)window).GetRenderFacade();
        if(facade!=null && facade.AssetCreatureAsync.ISDONE() && facade.Pivot!=null) {
            if(!ReferenceEquals(state.ShadowModel,facade.Pivot)) {
                RestoreShadows(state);
                state.ShadowModel=facade.Pivot;
                var entries=new System.Collections.Generic.List<ShadowBounds>();
                foreach(var renderer in facade.GetComponentsInChildren<Renderer>(true)) {
                    var entry=ShadowBounds.Create(renderer);
                    if(entry!=null)entries.Add(entry);
                }
                state.Shadows=entries.ToArray();
            }
            var shadow=Shader.GetGlobalVector("_ShadowDirection");
            if(state.Shadows!=null && Mathf.Abs(shadow.y)>.001f)
                foreach(var entry in state.Shadows)entry.Apply(facade.transform.position.y,new Vector3(shadow.x,shadow.y,shadow.z));
        }
        // ShapeBounds is the stable body volume, not animated mesh/weapon bounds.
        // Do not use the UI-shifted native target as the orbit center.
        var pivot=facade!=null && facade.ShapeBounds.size.y>.01f
            ? facade.transform.TransformPoint(facade.ShapeBounds.center)
            : info.TargetPos-window.CameraController.ExtraOffsetByDistance*info.TargetDistance;
        if(facade!=null && facade.AssetCreatureAsync.ISDONE() && facade.Pivot!=null && facade.ShapeBounds.size.y>.01f) {
            var head=ReferenceEquals(state.AnchorModel,facade) && state.Head!=null?state.Head:facade.Pivot.FindTransform("Bip001 Head");
            pivot=StabilizeOrbit(state,facade,pivot,head,info,window.CameraController.ZoomOutRate);
        }
        ApplyCamera(cinema.CinemaCamera,state.Pan,state.Yaw,state.Pitch,state.Zoom,pivot);
    }
    static Vector3 StabilizeOrbit(State state,object model,Vector3 pivot,Transform head,NGame2.NUI.NRender3D.NRenderTargetCinema.CinemaCamera.CameraInfoData info,float zoomOutRate) {
        if(!ReferenceEquals(state.AnchorModel,model) || state.Head!=head) {
            state.AnchorModel=model;state.AnchorPivot=pivot;state.Head=head;
            if(head!=null)state.AnchorHead=head.position;
        }
        // Native GetZoomInInfo follows the animated head, blended with the
        // fixed zoom-out target. Remove only that animation displacement;
        // retain native zoom, framing offsets and user camera controls.
        if(head!=null)info.TargetPos-=(head.position-state.AnchorHead)*(1-Mathf.Clamp01(zoomOutRate));
        return state.AnchorPivot;
    }
    static void ApplyCamera(NGame2.NUI.NRender3D.NRenderTargetCinema.CinemaCamera camera,Vector3 pan,float yaw,float pitch,float zoom,Vector3 pivot) {
        var info=camera.CameraInfo;
        var originalRotation=Quaternion.LookRotation(info.LookDir);
        var direction=Direction(info.LookDir,yaw,pitch);
        var orbit=Quaternion.LookRotation(direction)*Quaternion.Inverse(originalRotation);
        float distance=Distance(info.TargetDistance,zoom);
        float ratio=distance/Mathf.Max(.0001f,info.TargetDistance);
        // Rotate/scale the framing offset with the camera. This preserves the
        // hero's right-side screen position, but the physical orbit is around
        // the hero rather than the center of the window.
        info.TargetPos=pivot+orbit*(info.TargetPos-pivot)*ratio+pan;
        info.LookDir=direction;
        info.TargetDistance=distance;
        camera.ChainUpdate();
    }

    // Planar shadows are drawn by the character's own shader pass. Unity culls
    // that entire renderer using its unprojected bounds, including its shadow.
    // Preview-only bounds must contain both the model and its floor projection.
    sealed class ShadowBounds {
        public SkinnedMeshRenderer Skin;
        public MeshFilter Filter;
        public Mesh Original,Copy;
        public Bounds Source,Applied;
        public bool HasApplied;
        public bool UpdateWhenOffscreen;
        public Transform[] Bones;
        public Matrix4x4[] Bindposes;
        public Bounds MeshBounds;
        public static ShadowBounds Create(Renderer renderer) {
            var skin=renderer as SkinnedMeshRenderer;
            if(skin!=null)return new ShadowBounds{Skin=skin,Source=skin.localBounds,UpdateWhenOffscreen=skin.updateWhenOffscreen,
                Bones=skin.bones,Bindposes=skin.sharedMesh!=null?skin.sharedMesh.bindposes:null,MeshBounds=skin.sharedMesh!=null?skin.sharedMesh.bounds:skin.localBounds};
            var mesh=renderer as MeshRenderer;
            var filter=mesh!=null?mesh.GetComponent<MeshFilter>():null;
            if(filter==null || filter.sharedMesh==null)return null;
            return new ShadowBounds{Filter=filter,Original=filter.sharedMesh,Source=filter.sharedMesh.bounds};
        }
        public void Apply(float floor,Vector3 direction) {
            Matrix4x4 matrix;
            if(Skin!=null) {
                Skin.updateWhenOffscreen=false;
                var current=Skin.localBounds;
                if(!HasApplied || !Same(current,Applied))Source=current;
                matrix=Skin.rootBone!=null?Skin.rootBone.localToWorldMatrix:Skin.localToWorldMatrix;
            } else {
                if(Filter==null || Original==null || (Filter.sharedMesh!=Original && Filter.sharedMesh!=Copy))return;
                matrix=Filter.transform.localToWorldMatrix;
            }
            var posed=Source;
            // Unity otherwise replaces localBounds during skinning/culling.
            // Bound the current bone transforms conservatively without baking
            // vertices on the CPU. Each weighted vertex lies in the union of
            // its bones' transformed bind-pose mesh bounds.
            if(Skin!=null && Bones!=null && Bindposes!=null && Bones.Length==Bindposes.Length) {
                var inverse=matrix.inverse;
                for(int i=0;i<Bones.Length;i++)if(Bones[i]!=null) {
                    var pose=inverse*Bones[i].localToWorldMatrix*Bindposes[i];
                    for(int j=0;j<8;j++)posed.Encapsulate(pose.MultiplyPoint3x4(Corner(MeshBounds,j)));
                }
            }
            var expanded=ProjectedBounds(posed,matrix,floor,direction);
            if(Skin!=null)Skin.localBounds=expanded;
            else {
                if(Copy==null){Copy=UnityEngine.Object.Instantiate(Original);Copy.name=Original.name+" (preview shadow bounds)";Filter.sharedMesh=Copy;}
                Copy.bounds=expanded;
            }
            Applied=expanded;HasApplied=true;
        }
        public void Restore() {
            if(Skin!=null && HasApplied && Same(Skin.localBounds,Applied))Skin.localBounds=Source;
            if(Skin!=null)Skin.updateWhenOffscreen=UpdateWhenOffscreen;
            if(Filter!=null && Filter.sharedMesh==Copy)Filter.sharedMesh=Original;
            if(Copy!=null)UnityEngine.Object.Destroy(Copy);
            Copy=null;HasApplied=false;
        }
        static bool Same(Bounds a,Bounds b) {return (a.center-b.center).sqrMagnitude<1e-10f && (a.size-b.size).sqrMagnitude<1e-10f;}
    }
    static Bounds ProjectedBounds(Bounds source,Matrix4x4 matrix,float floor,Vector3 direction) {
        var inverse=matrix.inverse;
        var expanded=source;
        for(int i=0;i<8;i++) {
            var corner=Corner(source,i);
            var world=matrix.MultiplyPoint3x4(corner);
            world-=direction*((world.y-floor)/direction.y);
            expanded.Encapsulate(inverse.MultiplyPoint3x4(world));
        }
        expanded.Expand(.05f);
        return expanded;
    }
    static Vector3 Corner(Bounds bounds,int i) {
        return bounds.center+Vector3.Scale(bounds.extents,new Vector3((i&1)==0?-1:1,(i&2)==0?-1:1,(i&4)==0?-1:1));
    }
    static void RestoreShadows(State state) {
        if(state.Shadows!=null)foreach(var entry in state.Shadows)entry.Restore();
        state.Shadows=null;state.ShadowModel=null;
    }
