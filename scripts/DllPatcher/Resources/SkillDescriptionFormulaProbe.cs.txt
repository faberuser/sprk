    static object Member(object owner,string name) {
        for(var type=owner.GetType();type!=null;type=type.BaseType) {
            var p=type.GetProperty(name,Flags|BindingFlags.DeclaredOnly);if(p!=null)return p.GetValue(owner,null);
            var f=type.GetField(name,Flags|BindingFlags.DeclaredOnly);if(f!=null)return f.GetValue(owner);
        }
        throw new Exception("Missing member "+name);
    }

    static void CheckDescriptionLayout(MonoBehaviour owner,UILabel label,bool shortText) {
        var scroll=label.GetComponentInParent<UIScrollView>();var panel=scroll.GetComponent<UIPanel>();
        var backdrop=owner.GetComponentsInChildren<UISprite>(true).Single(s=>s.name=="Sprite_DescBack");
        var frame=NGUIMath.CalculateRelativeWidgetBounds(scroll.transform,backdrop.transform,true);
        var start=SprkSkillDescriptionFormula.TextStart(label,scroll.transform);
        Check(Mathf.Abs((start.x-frame.min.x)-(frame.max.y-start.y))<.6f,"equal visible top/left padding: "+owner.GetType().Name);
        Check((bool)Member(scroll,"shouldMove"),"drag enabled for fitting descriptions");
        Check(scroll.dragEffect==UIScrollView.DragEffect.MomentumAndSpring && scroll.restrictWithinPanel,"native rubber-band configuration");
        if(shortText) {
            Check(scroll.bounds.size.y<=panel.finalClipRegion.w+1,"short text fits viewport");
            var rest=scroll.transform.localPosition;
            scroll.MoveRelative(new Vector3(0,45,0));
            Check(scroll.RestrictWithinBounds(false,false,true),"fitting text springs back after drag");
            var spring=scroll.GetComponent<SpringPanel>();
            Check(spring!=null && spring.enabled && Vector3.Distance(spring.target,rest)<1,"spring target preserves padded resting position");
            scroll.DisableSpring();scroll.RestrictWithinBounds(true,false,true);
            Check(Vector3.Distance(scroll.transform.localPosition,rest)<1,"fitting text returns to rest");
        } else {
            float overflow=scroll.bounds.size.y-panel.finalClipRegion.w;
            if(overflow>2) {
                var rest=scroll.transform.localPosition;float amount=Mathf.Min(40,overflow*.5f);
                scroll.MoveRelative(new Vector3(0,amount,0));scroll.RestrictWithinBounds(true,false,true);
                Check(Mathf.Abs(scroll.transform.localPosition.y-rest.y-amount)<1,"overflowing text remains scrollable");
                scroll.ResetPosition();
            }
        }
    }
