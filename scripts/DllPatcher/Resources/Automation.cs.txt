// Separate account preferences for simultaneous test clients. Without both
// environment variables these wrappers retain the original PlayerPrefs calls.
public static class SprkTestPrefs {
    public static string DeviceId() {
        string prefix=Prefix();
        return prefix=="" ? SystemInfo.deviceUniqueIdentifier : "sprk-test-"+Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE");
    }
    static string Prefix() {
        if(String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR"))) return "";
        string profile=Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE");
        return String.IsNullOrEmpty(profile) ? "" : "SprkAutomation/"+profile+"/";
    }
    static string Key(string key) { return Prefix()+key; }
    public static string GetString(string key) { return PlayerPrefs.GetString(Key(key)); }
    public static string GetString(string key,string value) { return PlayerPrefs.GetString(Key(key),value); }
    public static int GetInt(string key) { return PlayerPrefs.GetInt(Key(key)); }
    public static int GetInt(string key,int value) { return PlayerPrefs.GetInt(Key(key),value); }
    public static float GetFloat(string key) { return PlayerPrefs.GetFloat(Key(key)); }
    public static float GetFloat(string key,float value) { return PlayerPrefs.GetFloat(Key(key),value); }
    public static void SetString(string key,string value) { PlayerPrefs.SetString(Key(key),value); }
    public static void SetInt(string key,int value) { PlayerPrefs.SetInt(Key(key),value); }
    public static void SetFloat(string key,float value) { PlayerPrefs.SetFloat(Key(key),value); }
    public static bool HasKey(string key) { return PlayerPrefs.HasKey(Key(key)); }
    public static void DeleteKey(string key) { PlayerPrefs.DeleteKey(Key(key)); }
    public static void Save() { PlayerPrefs.Save(); }
    public static void DeleteAll() {
        if(Prefix()!="") throw new InvalidOperationException("DeleteAll is disabled for test profiles; choose a new profile name");
        PlayerPrefs.DeleteAll();
    }
}

// Local, opt-in test instrumentation. No listener, arbitrary eval, or gameplay overrides.
public sealed class SprkAutomation : MonoBehaviour
{
    static SprkAutomation instance;
    string root;
    bool busy;
    float nextPoll;
    readonly List<Failure> errors = new List<Failure>();
    [Serializable] public class Command { public string id; public string action; public int target; public string text; }
    [Serializable] public class Failure { public string type; public string message; public string stack; }
    [Serializable] public class Element { public int id; public string path; public string kind; public string text; public bool enabled; public float x; public float y; }
    [Serializable] public class Reply {
        public string id; public bool ok; public string error; public string scene; public int pid;
        public int width; public int height; public int frame; public string screenshot;
        public List<Element> elements = new List<Element>(); public List<string> behaviours = new List<string>();
        public List<Failure> errors = new List<Failure>();
        public List<object> materials = new List<object>();
    }
    public static void Ensure()
    {
        string dir = Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR");
        if (instance != null || String.IsNullOrEmpty(dir)) return;
        try {
            dir = Path.GetFullPath(dir);
            Directory.CreateDirectory(dir);
            var go = new GameObject("SprkAutomation");
            DontDestroyOnLoad(go);
            instance = go.AddComponent<SprkAutomation>();
            instance.root = dir;
            Application.runInBackground = true;
            Application.logMessageReceived += instance.Log;
            File.WriteAllText(Path.Combine(dir, "ready.json"), "{\"pid\":" + System.Diagnostics.Process.GetCurrentProcess().Id + "}");
        } catch (Exception e) { Debug.LogError("SPRK automation init: " + e); }
    }
    void Log(string message, string stack, LogType type) {
        if (type != LogType.Error && type != LogType.Exception && type != LogType.Assert) return;
        // Some original combat diagnostics omit Unity's stack trace. Retain
        // the synchronous caller chain in this opt-in test client only.
        if (String.IsNullOrEmpty(stack) && message != null && message.IndexOf("Duplicate passive",StringComparison.OrdinalIgnoreCase)>=0)
            stack=Environment.StackTrace;
        if (errors.Count == 100) errors.RemoveAt(0);
        errors.Add(new Failure { type=type.ToString(), message=message, stack=stack });
    }
    static string Hierarchy(Transform t) {
        string path=t.name; while(t.parent != null) { t=t.parent; path=t.name + "/" + path; } return path;
    }
    static Element Describe(Component c, string kind, string text, bool enabled) {
        Camera camera=NGUITools.FindCameraForLayer(c.gameObject.layer);
        Vector3 p=camera == null ? Vector3.zero : camera.WorldToScreenPoint(c.transform.position);
        return new Element { id=c.gameObject.GetInstanceID(), path=Hierarchy(c.transform), kind=kind, text=text,
            enabled=enabled, x=p.x, y=Screen.height-p.y };
    }
    Reply Snapshot(string id) {
        var r=new Reply {id=id,ok=true,scene=SceneManager.GetActiveScene().name,
            pid=System.Diagnostics.Process.GetCurrentProcess().Id,width=Screen.width,height=Screen.height,frame=Time.frameCount};
        foreach(UIButton b in UnityEngine.Object.FindObjectsOfType<UIButton>()) {
            var labels=b.GetComponentsInChildren<UILabel>(); string text="";
            foreach(var label in labels) text += label.text + " ";
            r.elements.Add(Describe(b,"button",text.Trim(),b.isEnabled && b.enabled));
        }
        foreach(Collider c in UnityEngine.Object.FindObjectsOfType<Collider>()) {
            if(c.GetComponent<UIButton>() != null || NGUITools.FindCameraForLayer(c.gameObject.layer) == null) continue;
            var widget=c.GetComponent<UIWidget>();
            if(widget == null && c.GetComponent<UIEventListener>() == null) continue;
            r.elements.Add(Describe(c,"button","",c.enabled && (widget == null || widget.alpha > 0)));
        }
        foreach(UILabel l in UnityEngine.Object.FindObjectsOfType<UILabel>())
            r.elements.Add(Describe(l,"label",l.text,l.enabled && l.alpha > 0));
        foreach(UIInput i in UnityEngine.Object.FindObjectsOfType<UIInput>())
            r.elements.Add(Describe(i,"input","",i.enabled)); // Never export input values.
        foreach(var node in UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NWorldMap.NMapLayout.MapNode>()) {
            var collider=node.GetComponent<Collider>();
            if(node.NodeData==null || node.NodeData.Hidden) continue;
            r.elements.Add(Describe(node,"map-node",node.ChapterIndex+":"+node.DungeonIndex+":"+node.NodeData.EntryNPCCode,
                node.isActiveAndEnabled && collider!=null && collider.enabled));
        }
        var seen=new HashSet<string>();
        foreach(MonoBehaviour b in UnityEngine.Object.FindObjectsOfType<MonoBehaviour>()) {
            if(b == null || !b.enabled) continue;
            string name=b.GetType().FullName;
            if((name.Contains("Sprk") || name.Contains("Window") || name.Contains("Battle") || name.Contains("Lobby") || name.Contains("State")) && seen.Add(name)) r.behaviours.Add(name);
        }
        r.errors.AddRange(errors); return r;
    }
    GameObject Target(int id) {
        foreach(Transform t in UnityEngine.Object.FindObjectsOfType<Transform>()) if(t.gameObject.GetInstanceID()==id) return t.gameObject;
        throw new InvalidOperationException("Target is no longer active: " + id);
    }
    void Update() {
        if(root == null || busy || Time.realtimeSinceStartup < nextPoll) return;
        nextPoll=Time.realtimeSinceStartup+0.1f;
        string inbox=Path.Combine(root,"command.json"); if(!File.Exists(inbox)) return;
        Command cmd=null; Reply reply=null;
        try {
            cmd=JsonUtility.FromJson<Command>(File.ReadAllText(inbox)); File.Delete(inbox);
            Guid parsed;
            if(cmd == null || !Guid.TryParseExact(cmd.id,"N",out parsed)) throw new InvalidOperationException("Invalid command ID");
        if(cmd.action=="click" || cmd.action=="press" || cmd.action=="drag") {
                var go=Target(cmd.target); var button=go.GetComponent<UIButton>();
                var collider=go.GetComponent<Collider>();
                if(button != null ? (!button.enabled || !button.isEnabled) : (collider == null || !collider.enabled)) throw new InvalidOperationException("Button is disabled");
                // Some native handlers inspect currentTouch.current to identify
                // the button. Preserve the real event context outside this call.
                var oldTouch=UICamera.currentTouch; var oldCamera=UICamera.currentCamera;
                var oldCurrent=UICamera.current; int oldId=UICamera.currentTouchID;
                var oldWorld=UICamera.lastWorldPosition;
                try {
                    var camera=NGUITools.FindCameraForLayer(go.layer);
                    UICamera.currentCamera=camera;
                    UICamera.current=camera == null ? null : camera.GetComponent<UICamera>();
                    UICamera.currentTouchID=-1;
                    UICamera.currentTouch=new UICamera.MouseOrTouch {current=go,pressed=go,last=go};
                    if(cmd.action=="drag") {
                        var parts=(cmd.text ?? "").Split(',');
                        float dx,dy;
                        if(parts.Length!=2 || !float.TryParse(parts[0],System.Globalization.NumberStyles.Float,System.Globalization.CultureInfo.InvariantCulture,out dx)
                            || !float.TryParse(parts[1],System.Globalization.NumberStyles.Float,System.Globalization.CultureInfo.InvariantCulture,out dy)
                            || float.IsNaN(dx) || float.IsNaN(dy) || float.IsInfinity(dx) || float.IsInfinity(dy)
                            || Mathf.Abs(dx)>1000 || Mathf.Abs(dy)>1000 || camera==null)
                            throw new InvalidOperationException("Expected a drag delta dx,dy within 1000 pixels");
                        var pos=(Vector2)camera.WorldToScreenPoint(go.transform.position);
                        UICamera.currentTouch.pos=pos;
                        UICamera.lastWorldPosition=go.transform.position;
                        UICamera.Notify(go,UICamera.ENotifyType.OnPress,true);
                        try {
                            for(int step=1;step<=12;step++) {
                                var delta=new Vector2(dx,dy)/12f;
                                UICamera.currentTouch.pos=pos+delta*step;
                                UICamera.currentTouch.delta=delta;
                                UICamera.currentTouch.totalDelta=delta*step;
                                UICamera.Notify(go,UICamera.ENotifyType.OnDrag,delta);
                            }
                        } finally { UICamera.Notify(go,UICamera.ENotifyType.OnPress,false); }
                    } else if(cmd.action=="press") {
                        UICamera.Notify(go,UICamera.ENotifyType.OnPress,true);
                        UICamera.Notify(go,UICamera.ENotifyType.OnPress,false);
                    } else UICamera.Notify(go,UICamera.ENotifyType.OnClick,null);
                } finally { UICamera.currentTouch=oldTouch; UICamera.currentCamera=oldCamera; UICamera.current=oldCurrent; UICamera.currentTouchID=oldId; UICamera.lastWorldPosition=oldWorld; }
            } else if(cmd.action=="enter-raid-node") {
                var nodes=UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NComponent2.PunishmentRaidComponent>();
                NGame2.NUI.NComponent2.PunishmentRaidComponent selected=null;
                foreach(var node in nodes) if(node.gameObject.activeInHierarchy && node.PunishmentRaidData!=null && node.PunishmentRaidData.RaidIndex==cmd.target) {
                    if(selected!=null)throw new InvalidOperationException("Ambiguous raid node");
                    selected=node;
                }
                if(selected==null || !selected.OpenRoot.gameObject.activeInHierarchy)
                    throw new InvalidOperationException("Raid node is not open");
                selected.OnClickEnter();
            } else if(cmd.action=="text") {
                var input=Target(cmd.target).GetComponent<UIInput>();

                input.value=cmd.text ?? "";
                input.Submit();
            } else if(cmd.action=="clear-errors") errors.Clear();

                if(cmd.target==0) UICamera.selectedObject=null;
                else {
                    var input=Target(cmd.target).GetComponent<UIInput>();
                    if(input==null || !input.isActiveAndEnabled) throw new InvalidOperationException("Input is disabled");
                    input.isSelected=true;
                }
            }
            else if(cmd.action=="scroll") {
                var go=Target(cmd.target);
                var scroll=NGUITools.FindInParents<UIScrollView>(go);
                float amount;
                if(scroll==null || !float.TryParse(cmd.text,System.Globalization.NumberStyles.Float,
                    System.Globalization.CultureInfo.InvariantCulture,out amount) || float.IsNaN(amount) || float.IsInfinity(amount) || Mathf.Abs(amount)>10f)
                    throw new InvalidOperationException("Expected a scroll view and an amount between -10 and 10");
                scroll.Scroll(amount);
            }
            else if(cmd.action=="preview-perk") {
                // Read-only UI preview, including already learned perks. The

                NGame2.NUI.NWindow.LearnTranscendSkillMessagePopup.Open(hero,icon.TranscendSkill.TranscendSkillIndex,delegate {});

                    popup.Label_SkillDesc.text=text+"\n\n"+text+"\n\n"+text;

            else if(cmd.action!="snapshot" && cmd.action!="screenshot" && cmd.action!="quit" && cmd.action!="renderers" && cmd.action!="probe-renderer" && cmd.action!="soul-scene" && cmd.action!="dispatch-probe" && cmd.action!="party-ui" && cmd.action!="inn-scene" && cmd.action!="input-focus" && cmd.action!="perk-popup" && cmd.action!="skill-probe" && cmd.action!="combat-probe" && cmd.action!="combat-simulation" && cmd.action!="skill-table-probe" && cmd.action!="combat-table-probe") throw new InvalidOperationException("Unknown action");
            reply=Snapshot(cmd.id);
            if(cmd.action=="combat-table-probe") {
                var requests=(System.Collections.IList)MiniJSON.Json.Deserialize(cmd.text);
                if(requests.Count>1000)throw new InvalidOperationException("Too many diagnostic rows");
                foreach(Dictionary<string,object> request in requests) {
                    string table=(string)request["table"],key=(string)request["key"];
                    object container=typeof(NShared.GameRoot).GetProperty(table+"DataContainer").GetValue(NShared.GameRoot.Instance,null);
                    var lookup=container.GetType().GetMethods().Single(m=>m.Name=="GetData" && m.GetParameters().Length==1);
                    var type=lookup.GetParameters()[0].ParameterType;object index;
                    if(table=="CreatureStarGradeStat") {
                        var parts=key.Split(':');index=new NShared.CreatureStarGradeStatKey(int.Parse(parts[0]),int.Parse(parts[1]),int.Parse(parts[2]));
                    } else index=type==typeof(string)?(object)key:type.IsEnum?Enum.ToObject(type,int.Parse(key)):(object)int.Parse(key);
                    object row;
                    if(request.ContainsKey("enumeration") && (bool)request["enumeration"]) {
                        var rows=(System.Collections.IEnumerable)container.GetType().GetMethod("GetAllData").Invoke(container,null);
                        row=null;
                        foreach(var candidate in rows) {
                            var property=candidate.GetType().GetProperty(table=="State"?"StateType":"Index");
                            if(property==null)throw new InvalidOperationException("Unsupported enumeration key: "+table);
                            if(Convert.ToString(property.GetValue(candidate,null))==key) {row=candidate;break;}
                        }
                    } else row=table=="Creature"?(object)RestoredPortal.CombatCreature(NShared.GameRoot.Instance.CreatureDataContainer,(int)index):
                        table=="State"?(object)RestoredPortal.CombatState(NShared.GameRoot.Instance.StateDataContainer,(string)index):
                        table=="SkillProjectile"?(object)RestoredPortal.CombatProjectile(NShared.GameRoot.Instance.SkillProjectileDataContainer,(int)index):lookup.Invoke(container,new[]{index});
                    if(row==null)throw new InvalidOperationException("Missing diagnostic row: "+table+"/"+key);
                    var values=new Dictionary<string,object>();
                    foreach(string field in (System.Collections.IList)request["fields"])
                        values[field]=table=="CreatureStarGradeStat" && Enum.IsDefined(typeof(NShared.StatType),field)?((NShared.CreatureStarGradeStatData)row).GetStat((NShared.StatType)Enum.Parse(typeof(NShared.StatType),field)):
                            row.GetType().GetProperty(field).GetValue(row,null);
                    reply.materials.Add(new Dictionary<string,object>{{"table",table},{"key",key},{"values",values}});
                }
            }
            if(cmd.action=="skill-table-probe") {
                var ids=(cmd.text??"").Split(',').Select(s=>Int32.Parse(s)).ToArray();
                if(ids.Length>1000)throw new InvalidOperationException("Too many diagnostic rows");
                foreach(int id in ids) {
                    var skill=RestoredPortal.PunishmentSkill(NShared.GameRoot.Instance.SkillDataContainer,id);
                    if(skill==null)continue;
                    var operations=new List<object>();
                    for(int op=1;op<=8;op++) {
                        var data=skill.GetOperationData(op);if(data!=null)operations.Add(TableProperties(data));
                    }
                    reply.materials.Add(new Dictionary<string,object>{{"index",id},{"skill",TableProperties(skill)},{"operations",operations}});
                }
            }
            if(cmd.action=="combat-simulation") {
                int chapter=70000,dungeon=cmd.target,difficulty=0;
                string heroList=cmd.text??"";
                if(heroList.Contains(";")) {
                    var parts=heroList.Split(';');if(parts.Length!=4)throw new InvalidOperationException("Expected chapter;dungeon;difficulty;heroes");
                    chapter=int.Parse(parts[0]);dungeon=int.Parse(parts[1]);difficulty=int.Parse(parts[2]);heroList=parts[3];
                } else if(cmd.target<1 || cmd.target>5)throw new InvalidOperationException("Expected Apocalypsion dragon dungeon 1 to 5");
                var account=NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.instance;
                var ids=heroList.Split(',').Select(s=>Int32.Parse(s)).ToArray();
                if(ids.Length<1 || ids.Length>8 || ids.Distinct().Count()!=ids.Length) throw new InvalidOperationException("Expected 1 to 8 owned heroes");
                var heroes=new Dictionary<int,NShared.CachedHeroInfo>();
                foreach(int id in ids) {

                var simulation=new NShared.BattleSimulationInstance();

                    ExtraData=new NShared.BattleInstanceExtraData{InfinityLevel=1},
                    AccountInfos=new NShared.BattleSimulationAccountInfo[]{new NShared.BattleSimulationAccountInfo{

                        AccountBuffs=account.GetActiveAccountBuffDataBaseList(false).ToArray()}}});
                // Detached local engine only: no entry, dispatch, reward or result API.
                try {
                    simulation.BattleStart(false);long nextSample=0;bool karmaInitialized=false;
                    var samples=new List<object>();
                    for(int step=0;step<30000 && !simulation.IsBattleEnd;step++) {
                        simulation.Update();var battle=simulation.BattleInstance;long time=battle.GlobalTimer.GetCurrentTimeMs();
                        // Rendered KarmaUpdater normally sends KarmaInitBattleLog
                        // after player creation. The detached simulator needs the
                        // same initialization before it can finish a Karma wave.

                        }
                        if(time>=nextSample) {
                            nextSample=time+1000;var creatures=new List<object>();
                            foreach(var c in battle.CreatureContainer.GetAllCreatures()) {
                                var stats=new Dictionary<string,object>();

                            }
                            samples.Add(new Dictionary<string,object>{{"timeMs",time},{"creatures",creatures}});
                        }
                        if(time>=60000)break;
                    }
                    reply.materials.Add(new Dictionary<string,object>{{"kind","combatSimulation"},{"chapter",chapter},{"dungeon",dungeon},{"difficulty",difficulty},{"ended",simulation.IsBattleEnd},

            }
            if(cmd.action=="combat-probe") {
                var account=NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.instance;
                foreach(var hero in account.GetHeroInfoList(true)) {
                    var equipped=NShared.StatCalculator.GetEquippedHeroStat(hero);
                    var equipment=new List<object>();
                    foreach(var pair in hero.GetSlotIndexEquipItemPair())
                        equipment.Add(new Dictionary<string,object>{{"slot",pair.Key},{"item",pair.Value.ItemIndex},{"star",pair.Value.Star}});
                    reply.materials.Add(new Dictionary<string,object>{{"kind","heroStats"},{"hero",hero.HeroIndex},
                        {"level",hero.Level},{"star",hero.Star},{"transcended",hero.Transcended},
                        {"stats",CombatStats(equipped)},{"equipment",equipment}});
                }
                if(NVespa.SingletonAuto<NGame2.NBattleField.BattleFacade>.isValidInstance) {
                    string diagnosticFile=Path.Combine(root,"combat-thread.json");
                    NVespa.SingletonAuto<NGame2.NBattleField.BattleFacade>.instance.AddThreadCommonAction(delegate(NShared.BattleInstance running) {
                        try {
                            var creatures=new List<object>();
                            foreach(var creature in running.CreatureContainer.GetAllCreatures()) {
                                var stats=new Dictionary<string,object>();
                                foreach(var stat in CombatStatTypes()) stats[stat.ToString()]=creature.StatController.GetFinalStat(stat);
                                creatures.Add(new Dictionary<string,object>{{"hero",creature.CreatureIndex},{"tier",creature.CreateInfo.Tier},
                                    {"level",creature.CreateInfo.HeroInfo.Level},{"hp",creature.StatController.Hp},
                                    {"resting",creature.CreateInfo.IsInitialResting},{"isAi",creature.CreateInfo.IsAi},
                                    {"initialStats",CombatStats(creature.CreateInfo.StatData)},{"stats",stats}});

                            File.WriteAllText(diagnosticFile,MiniJSON.Json.Serialize(new Dictionary<string,object>{
                                {"chapter",running.ChapterIndex},{"dungeon",running.DungeonIndex},
                                {"timeMs",running.GlobalTimer.GetCurrentTimeMs()},{"creatures",creatures}}));
                        } catch(Exception e) { File.WriteAllText(diagnosticFile,MiniJSON.Json.Serialize(new Dictionary<string,object>{{"error",e.ToString()}})); }
                    });
                    var battle=NVespa.SingletonAuto<NGame2.NBattleField.BattleFacade>.instance.GetStoppedBattleInstance();
                    if(battle!=null) {
                        foreach(var creature in battle.CreatureContainer.GetAllCreatures()) {

                                {"level",creature.CreateInfo.HeroInfo.Level},{"hp",creature.StatController.Hp},
                                {"initialStats",CombatStats(creature.CreateInfo.StatData)},{"stats",stats}});

                    probe.SkillExtend1=0;

                        {"canEnhance",NGame2.NAccount.Helper.CanAnySkillExtend(hero)},
                        {"unmaxedReminder",NGame2.NAccount.Helper.CanAnySkillExtend(probe)},{"slots",slots}});
                }
            }
            if(cmd.action=="input-focus" || cmd.action=="focus-input") {
                var input=UIInput.selection;
                var helper=typeof(NGame2.NScene.LobbyScene2).Assembly.GetType("RestoredPortal");
                var shortcut=helper==null?null:helper.GetMethod("MenuShortcutKeyUp");
                reply.materials.Add(new Dictionary<string,object>{{"kind","inputFocus"},
                    {"selected",input==null?null:Hierarchy(input.transform)},
                    {"active",input!=null && input.isActiveAndEnabled},{"cameraFocus",UICamera.inputHasFocus},
                    {"heroesKey",shortcut==null?null:shortcut.Invoke(null,new object[]{KeyCode.H})},
                    {"inventoryKey",shortcut==null?null:shortcut.Invoke(null,new object[]{KeyCode.B})},
                    {"shopKey",shortcut==null?null:shortcut.Invoke(null,new object[]{KeyCode.P})}});
            }
            if(cmd.action=="perk-popup") {
                foreach(var popup in UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NWindow.LearnTranscendSkillMessagePopup>()) {
                    var label=popup.Label_SkillDesc;
                    var scroll=label.GetComponentInParent<UIScrollView>();
                    var printed=typeof(UILabel).GetField("mPrintedSize",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic);
                    reply.materials.Add(new Dictionary<string,object>{{"title",popup.Label_SkillName.text},{"text",label.text},
                        {"fontSize",label.fontSize},{"printedFontSize",printed.GetValue(label)},{"width",label.width},{"height",label.height},
                        {"overflow",label.overflowMethod.ToString()},{"scroll",scroll!=null},{"canScroll",scroll!=null && scroll.shouldMoveVertically},
                        {"labelY",label.transform.localPosition.y},{"viewportY",scroll==null?0:scroll.transform.localPosition.y},
                        {"drag",label.GetComponent<UIDragScrollView>()!=null},{"warning",popup.Label_Warning.gameObject.activeInHierarchy}});
                }
            }
            if(cmd.action=="inn-scene") {
                var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
                foreach(var window in UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NWindow.HeroInnView>()) {
                    var facade=typeof(NGame2.NUI.NWindow.HeroRecruitView).GetField("_renderFacade",flags).GetValue(window);

                    reply.materials.Add(new Dictionary<string,object>{{"kind","manager"},{"refresh",manager.GetType().GetField("_requestRefresh",flags).GetValue(manager)}});
            }
            if(cmd.action=="dispatch-probe") {
                var helper=AppDomain.CurrentDomain.GetAssemblies().First(a=>a.GetName().Name=="SprkDispatch").GetType("SprkDispatch");
                var flags=System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.NonPublic;
                reply.materials.Add(new Dictionary<string,object>{{"location",helper.Assembly.Location},{"ready",helper.GetMethod("Ready",flags).Invoke(null,null)},{"time",Time.realtimeSinceStartup},{"frame",Time.frameCount},{"lastFrame",helper.GetField("lastFrame",flags).GetValue(null)},{"lastStatus",helper.GetField("LastStatus").GetValue(helper.GetField("instance",flags).GetValue(null))},{"lastError",helper.GetField("LastError").GetValue(helper.GetField("instance",flags).GetValue(null))},{"host",NVespa.SingletonUpdater<NVespa.NGlobal.WebServiceManager>.instance.Requester.Host}});
            }
            if(cmd.action=="soul-scene") {
                foreach(var window in UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NWindow.NSoulWeapon.SoulStoneRestoreWindow>()) {
                    reply.materials.Add(new Dictionary<string,object> {
                        {"kind","SoulJudgmentScene"},
                        {"state",window.GetType().GetField("_nodeTreeMainState",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(window).ToString()},
                        {"pack",window.BgNodeTreeUIPlayer.NodeTreePackCode},
                        {"tree",window.BgNodeTreeUIPlayer.NodeTreeName}
                    });

            }
            if(cmd.action=="renderers") {
                foreach(Renderer renderer in UnityEngine.Object.FindObjectsOfType<Renderer>()) {
                    if(!renderer.enabled || !renderer.gameObject.activeInHierarchy) continue;
                    foreach(Material material in renderer.sharedMaterials) {
                        Shader shader=material == null ? null : material.shader;
                        var properties=new Dictionary<string,object>();
                        if(material != null) {
                            foreach(string key in new string[] {"_MainTex","_MainTex_Color","_NoiseTex_Nor"}) {
                                if(!material.HasProperty(key)) continue;
                                Texture tex=material.GetTexture(key);
                                properties[key]=tex == null ? null : tex.name + " " + tex.width + "x" + tex.height;
                            }
                            foreach(string key in new string[] {"_Color","_TintColor"})
                                if(material.HasProperty(key)) properties[key]=material.GetColor(key).ToString();

                            {"renderer",Hierarchy(renderer.transform)},

                            {"properties",properties}
                        });
                    }
                }

                foreach(Renderer renderer in UnityEngine.Object.FindObjectsOfType<Renderer>())

                        if(selected != null) throw new InvalidOperationException("Ambiguous renderer path");
                        selected=renderer;
                    }
                if(selected==null) throw new InvalidOperationException("Active renderer not found");
                busy=true; StartCoroutine(Capture(reply,selected)); return;
            }
            if(cmd.action=="party-ui") {
                foreach(var party in UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NWindow.BattlePartySetting>()) {
                    if(!party.gameObject.activeInHierarchy) continue;
                    var raid=party as NGame2.NUI.NWindow.RaidPartySetting;
                    reply.materials.Add(new Dictionary<string,object>{{"kind","raid"},{"path",Hierarchy(party.transform)},
                        {"index",raid==null || raid.RaidData==null ? 0 : raid.RaidData.Index},
                        {"minimumLevel",raid==null || raid.RaidData==null ? 0 : raid.RaidData.ReqHeroLevel}});
                    foreach(var portrait in party.GetComponentsInChildren<NGame2.NUI.NComponent2.HeroPortraitComponent>(true))
                        if(portrait.HeroInfo!=null) reply.materials.Add(new Dictionary<string,object>{{"kind","hero"},
                            {"id",portrait.gameObject.GetInstanceID()},{"path",Hierarchy(portrait.transform)},
                            {"hero",portrait.HeroInfo.HeroIndex},{"level",portrait.HeroInfo.Level},
                            {"active",portrait.gameObject.activeInHierarchy}});
                    foreach(var sprite in party.GetComponentsInChildren<UISprite>(true))
                        if(sprite.name.Contains("Frame") || sprite.name=="Sprite_Top") reply.materials.Add(new Dictionary<string,object>{
                            {"kind","frame"},{"path",Hierarchy(sprite.transform)},{"sprite",sprite.spriteName},
                            {"active",sprite.gameObject.activeInHierarchy},{"alpha",sprite.alpha}});
                }
            }
            if(cmd.action=="screenshot") { busy=true; StartCoroutine(Capture(reply)); return; }
        } catch(Exception e) { reply=new Reply {id=cmd==null ? "invalid" : cmd.id,ok=false,error=e.ToString()}; }
        Write(reply);
        if(cmd != null && cmd.action=="quit" && reply.ok) Application.Quit();
    }
    IEnumerator Capture(Reply reply,Renderer hidden=null) {
        if(hidden != null) hidden.enabled=false;
        yield return new WaitForEndOfFrame();
        Texture2D image=null;
        try {
            image=ScreenCapture.CaptureScreenshotAsTexture();
            reply.screenshot=reply.id+".png";
            File.WriteAllBytes(Path.Combine(root,reply.screenshot),image.EncodeToPNG());
        } catch(Exception e) { reply.ok=false; reply.error=e.ToString(); }
        finally { if(hidden!=null) hidden.enabled=true; if(image!=null) Destroy(image); busy=false; }
        Write(reply);
    }
    static NShared.StatType[] CombatStatTypes() {
        return new NShared.StatType[]{NShared.StatType.MaxHp,NShared.StatType.PhysicalAttack,NShared.StatType.MagicalAttack,
            NShared.StatType.PhysicalDefense,NShared.StatType.MagicalDefense,NShared.StatType.PhysicalDodgeChance,
            NShared.StatType.MagicalDodgeChance,NShared.StatType.PhysicalToughness,NShared.StatType.MagicalToughness};
    }
    static Dictionary<string,object> TableProperties(object row) {
        var result=new Dictionary<string,object>();
        foreach(var property in row.GetType().GetProperties(System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public)) {
            if(!property.CanRead || property.GetIndexParameters().Length!=0)continue;
            var type=property.PropertyType;
            if(!(type.IsPrimitive || type.IsEnum || type==typeof(string) || type.IsArray))continue;
            var value=property.GetValue(row,null);
            result[property.Name]=type.IsEnum ? Convert.ToInt64(value) : value;
        }
        return result;
    }
    static Dictionary<string,object> CombatStats(NShared.StatData data) {
        var result=new Dictionary<string,object>();
        if(data!=null) foreach(var stat in CombatStatTypes()) result[stat.ToString()]=data.GetStat(stat);
        return result;
    }
    void Write(Reply reply) {
        // Use the client's serializer: Unity's type registry does not include
        // nested serializable types from this separately loaded helper assembly.
        var elements=new List<object>();
        foreach(var e in reply.elements) elements.Add(new Dictionary<string,object> {
            {"id",e.id},{"path",e.path},{"kind",e.kind},{"text",e.text},{"enabled",e.enabled},{"x",e.x},{"y",e.y} });
        var failures=new List<object>();
        foreach(var e in reply.errors) failures.Add(new Dictionary<string,object> { {"type",e.type},{"message",e.message},{"stack",e.stack} });
        var data=new Dictionary<string,object> { {"id",reply.id},{"ok",reply.ok},{"error",reply.error},
            {"scene",reply.scene},{"pid",reply.pid},{"width",reply.width},{"height",reply.height},{"frame",reply.frame},
            {"screenshot",reply.screenshot},{"elements",elements},{"behaviours",reply.behaviours},{"errors",failures},{"materials",reply.materials} };
        string path=Path.Combine(root,reply.id+".json");
        File.WriteAllText(path+".tmp",MiniJSON.Json.Serialize(data)); File.Move(path+".tmp",path);
    }
