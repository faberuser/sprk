// Separate account preferences for simultaneous test clients. Without both
// environment variables these wrappers retain the original PlayerPrefs calls.
public static class SprkTestPrefs {
    struct IntPreference { public int Default,Value; }
    struct StringPreference { public string Default,Value; }
    static readonly Dictionary<string,IntPreference> graphicsInts=new Dictionary<string,IntPreference>();
    static readonly Dictionary<string,StringPreference> graphicsStrings=new Dictionary<string,StringPreference>();
    static readonly Dictionary<string,bool> graphicsKeys=new Dictionary<string,bool>();
    public static int GraphicsStorageReads { get; private set; }
    static bool CacheGraphics(string key) { return key.StartsWith("Sprk.Graphics.",StringComparison.Ordinal) || key.StartsWith("KingsRaid_SystemOption_graphic",StringComparison.Ordinal) || key.StartsWith("KingsRaid_GameOption_render_",StringComparison.Ordinal) || key.StartsWith("KingsRaid_SystemOption_render",StringComparison.Ordinal); }
    static void Invalidate(string key) { graphicsInts.Remove(key);graphicsStrings.Remove(key);graphicsKeys.Remove(key); }
    public static void ClearGraphicsCache() { graphicsInts.Clear();graphicsStrings.Clear();graphicsKeys.Clear(); }
    public static string DeviceId() {
        string prefix=Prefix();
        return prefix=="" ? SystemInfo.deviceUniqueIdentifier : "sprk-test-"+Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE");
    }
    static string Prefix() {
        SprkStorageMigration.Ensure();
        if(String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR"))) return "";
        string profile=Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE");
        return String.IsNullOrEmpty(profile) ? "" : "SprkAutomation/"+profile+"/";
    }
    static string Key(string key) { return Prefix()+key; }
    public static string GetString(string key) { return GetString(key,""); }
    public static string GetString(string key,string value) {
        string full=Key(key);if(!CacheGraphics(key))return PlayerPrefs.GetString(full,value);
        StringPreference saved;
        if(graphicsStrings.TryGetValue(full,out saved) && saved.Default==value)return saved.Value;
        GraphicsStorageReads++;string result=PlayerPrefs.GetString(full,value);
        graphicsStrings[full]=new StringPreference{Default=value,Value=result};return result;
    }
    public static int GetInt(string key) { return GetInt(key,0); }
    public static int GetInt(string key,int value) {
        string full=Key(key);if(!CacheGraphics(key))return PlayerPrefs.GetInt(full,value);
        IntPreference saved;
        if(graphicsInts.TryGetValue(full,out saved) && saved.Default==value)return saved.Value;
        GraphicsStorageReads++;int result=PlayerPrefs.GetInt(full,value);
        graphicsInts[full]=new IntPreference{Default=value,Value=result};return result;
    }
    public static float GetFloat(string key) { return PlayerPrefs.GetFloat(Key(key)); }
    public static float GetFloat(string key,float value) { return PlayerPrefs.GetFloat(Key(key),value); }
    public static void SetString(string key,string value) { string full=Key(key);PlayerPrefs.SetString(full,value);Invalidate(full); }
    public static void SetInt(string key,int value) { string full=Key(key);PlayerPrefs.SetInt(full,value);Invalidate(full); }
    public static void SetFloat(string key,float value) { string full=Key(key);PlayerPrefs.SetFloat(full,value);Invalidate(full); }
    public static bool HasKey(string key) {
        string full=Key(key);if(!CacheGraphics(key))return PlayerPrefs.HasKey(full);
        bool result;if(graphicsKeys.TryGetValue(full,out result))return result;
        GraphicsStorageReads++;result=PlayerPrefs.HasKey(full);graphicsKeys[full]=result;return result;
    }
    public static void DeleteKey(string key) { string full=Key(key);PlayerPrefs.DeleteKey(full);Invalidate(full); }
    public static void Save() { PlayerPrefs.Save(); }
    public static void DeleteAll() {
        if(Prefix()!="") throw new InvalidOperationException("DeleteAll is disabled for test profiles; choose a new profile name");
        PlayerPrefs.DeleteAll();ClearGraphicsCache();
    }
}

// Local, opt-in test instrumentation. No listener, arbitrary eval, or gameplay overrides.
public sealed class SprkAutomation : MonoBehaviour
{
    static SprkAutomation instance;
    string root;
    bool busy;
    float nextPoll;
    readonly List<Failure> errors = new List<Failure>();
    [Serializable] public class Command { public string id; public string action; public int target; public string text; }
    [Serializable] public class Failure { public string type; public string message; public string stack; }
    [Serializable] public class Element { public int id; public string path; public string kind; public string text; public bool enabled; public float x; public float y; }
    [Serializable] public class Reply {
        public string id; public bool ok; public string error; public string scene; public int pid;
        public int width; public int height; public int frame; public string screenshot;
        public List<Element> elements = new List<Element>(); public List<string> behaviours = new List<string>();
        public List<Failure> errors = new List<Failure>();
        public List<object> materials = new List<object>();
    }
    public static void Ensure()
    {
        SprkStorageMigration.StartReporter();
        string dir = Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR");
        if (instance != null || String.IsNullOrEmpty(dir)) return;
        try {
            dir = Path.GetFullPath(dir);
            Directory.CreateDirectory(dir);
            var go = new GameObject("SprkAutomation");
            DontDestroyOnLoad(go);
            instance = go.AddComponent<SprkAutomation>();
            instance.root = dir;
            Application.runInBackground = true;
            Application.logMessageReceived += instance.Log;
            File.WriteAllText(Path.Combine(dir, "ready.json"), "{\"pid\":" + System.Diagnostics.Process.GetCurrentProcess().Id + "}");
        } catch (Exception e) { Debug.LogError("SPRK automation init: " + e); }
    }
    void Log(string message, string stack, LogType type) {
        if (type != LogType.Error && type != LogType.Exception && type != LogType.Assert) return;
        // Some original combat diagnostics omit Unity's stack trace. Retain
        // the synchronous caller chain in this opt-in test client only.
        if (String.IsNullOrEmpty(stack) && message != null && message.IndexOf("Duplicate passive",StringComparison.OrdinalIgnoreCase)>=0)
            stack=Environment.StackTrace;
        if (errors.Count == 100) errors.RemoveAt(0);
        errors.Add(new Failure { type=type.ToString(), message=message, stack=stack });
    }
    static string Hierarchy(Transform t) {
        string path=t.name; while(t.parent != null) { t=t.parent; path=t.name + "/" + path; } return path;
    }
    static Element Describe(Component c, string kind, string text, bool enabled) {
        Camera camera=NGUITools.FindCameraForLayer(c.gameObject.layer);
        Vector3 p=camera == null ? Vector3.zero : camera.WorldToScreenPoint(c.transform.position);
        return new Element { id=c.gameObject.GetInstanceID(), path=Hierarchy(c.transform), kind=kind, text=text,
            enabled=enabled, x=p.x, y=Screen.height-p.y };
    }
    Reply Snapshot(string id) {
        var r=new Reply {id=id,ok=true,scene=SceneManager.GetActiveScene().name,
            pid=System.Diagnostics.Process.GetCurrentProcess().Id,width=Screen.width,height=Screen.height,frame=Time.frameCount};
        foreach(UIButton b in UnityEngine.Object.FindObjectsOfType<UIButton>()) {
            var labels=b.GetComponentsInChildren<UILabel>(); string text="";
            foreach(var label in labels) text += label.text + " ";
            r.elements.Add(Describe(b,"button",text.Trim(),b.isEnabled && b.enabled));
        }
        foreach(Collider c in UnityEngine.Object.FindObjectsOfType<Collider>()) {
            if(c.GetComponent<UIButton>() != null || NGUITools.FindCameraForLayer(c.gameObject.layer) == null) continue;
            var widget=c.GetComponent<UIWidget>();
            if(widget == null && c.GetComponent<UIEventListener>() == null) continue;
            r.elements.Add(Describe(c,"button","",c.enabled && (widget == null || widget.alpha > 0)));
        }
        foreach(UILabel l in UnityEngine.Object.FindObjectsOfType<UILabel>())
            r.elements.Add(Describe(l,"label",l.text,l.enabled && l.alpha > 0));
        foreach(UIInput i in UnityEngine.Object.FindObjectsOfType<UIInput>())
            r.elements.Add(Describe(i,"input","",i.enabled)); // Never export input values.
        foreach(var node in UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NWorldMap.NMapLayout.MapNode>()) {
            var collider=node.GetComponent<Collider>();
            if(node.NodeData==null || node.NodeData.Hidden) continue;
            r.elements.Add(Describe(node,"map-node",node.ChapterIndex+":"+node.DungeonIndex+":"+node.NodeData.EntryNPCCode,
                node.isActiveAndEnabled && collider!=null && collider.enabled));
        }
        var seen=new HashSet<string>();
        foreach(MonoBehaviour b in UnityEngine.Object.FindObjectsOfType<MonoBehaviour>()) {
            if(b == null || !b.enabled) continue;
            string name=b.GetType().FullName;
            if((name.Contains("Sprk") || name.Contains("Window") || name.Contains("Battle") || name.Contains("Lobby") || name.Contains("State")) && seen.Add(name)) r.behaviours.Add(name);
        }
        r.errors.AddRange(errors); return r;
    }
    GameObject Target(int id) {
        foreach(Transform t in UnityEngine.Object.FindObjectsOfType<Transform>()) if(t.gameObject.GetInstanceID()==id) return t.gameObject;
        throw new InvalidOperationException("Target is no longer active: " + id);
    }
    void Update() {
        if(root == null || busy || Time.realtimeSinceStartup < nextPoll) return;
        nextPoll=Time.realtimeSinceStartup+0.1f;
        string inbox=Path.Combine(root,"command.json"); if(!File.Exists(inbox)) return;
        Command cmd=null; Reply reply=null;
        try {
            cmd=JsonUtility.FromJson<Command>(File.ReadAllText(inbox)); File.Delete(inbox);
            Guid parsed;
            if(cmd == null || !Guid.TryParseExact(cmd.id,"N",out parsed)) throw new InvalidOperationException("Invalid command ID");
            if(cmd.action=="essence-craft-probe") {
                string profile=Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE");
                if(SceneManager.GetActiveScene().name!="Login" || String.IsNullOrEmpty(profile) || !profile.StartsWith("essence-craft-"))
                    throw new InvalidOperationException("Essence crafting fixture requires isolated Login and essence-craft profile");
                var results=ProbeEssenceCraft();reply=Snapshot(cmd.id);reply.materials.AddRange(results);Write(reply);return;
            }
            if(cmd.action=="inventory-rune-probe") {
                if(SceneManager.GetActiveScene().name!="Login")throw new InvalidOperationException("Inventory rune probe requires isolated Login");
                ProbeInventoryRunes();reply=Snapshot(cmd.id);reply.materials.Add("PASS: native inventory prefab exposes Normal Rune and Rune of Blasphemy subcategories, and both select correctly.");Write(reply);return;
            }
            if(cmd.action=="awakening-material-probe") {
                if(SceneManager.GetActiveScene().name!="Login")throw new InvalidOperationException("Awakening material fixture requires isolated Login");
                ProbeAwakeningMaterials();reply=Snapshot(cmd.id);reply.materials.Add("PASS: native awakening icon visibility, compact grid positions, hide-all, restore-all and eligibility refresh round trips.");Write(reply);return;
            }
            if(cmd.action=="tower-continuous-probe") {
                if(SceneManager.GetActiveScene().name!="Login")throw new InvalidOperationException("Continuous Battle prefab probe requires isolated Login");
                busy=true;StartCoroutine(ProbeTowerContinuous(cmd));return;
            }
            if(cmd.action=="native-stop-cycle") {
                if(SceneManager.GetActiveScene().name!="Login" || cmd.target<0 || cmd.target>10)
                    throw new InvalidOperationException("Native overlay lifecycle probe requires Login and a 0..10 frame delay");
                busy=true; StartCoroutine(ProbeNativeStop(cmd)); return;
            }
        if(cmd.action=="settings-probe") {
            if(!String.IsNullOrEmpty(cmd.text)) {
                string profile=Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE");
                if(String.IsNullOrEmpty(profile) || !profile.StartsWith("settings-persistence-"))
                    throw new InvalidOperationException("Settings writes require an isolated settings-persistence profile");
                var values=cmd.text.Split(',');
                NVespa.SystemOptionData.Graphic_FrameControl=(NShared.EFrameControl)Int32.Parse(values[0]);
                NVespa.SystemOptionData.Spec=(NGameAsset.ESpecType)Int32.Parse(values[1]);
                NVespa.SystemOptionData.Sound_BgmVolume=Single.Parse(values[2],System.Globalization.CultureInfo.InvariantCulture);
                NVespa.SystemOptionData.Render_EnableOutline=Boolean.Parse(values[3]);
                // Seed a legacy preference without changing the active FPS, to test migration.
                if(values.Length>4)SprkTestPrefs.SetString("KingsRaid_GameOption_device_ConsumeBattery",values[4]);
            }
        }
        else if(cmd.action=="click" || cmd.action=="click-and-stop-story" || cmd.action=="press" || cmd.action=="drag") {
                var go=Target(cmd.target); var button=go.GetComponent<UIButton>();
                var collider=go.GetComponent<Collider>();
                if(button != null ? (!button.enabled || !button.isEnabled) : (collider == null || !collider.enabled)) throw new InvalidOperationException("Button is disabled");
                // Some native handlers inspect currentTouch.current to identify
                // the button. Preserve the real event context outside this call.
                var oldTouch=UICamera.currentTouch; var oldCamera=UICamera.currentCamera;
                var oldCurrent=UICamera.current; int oldId=UICamera.currentTouchID;
                var oldWorld=UICamera.lastWorldPosition;
                try {
                    var camera=NGUITools.FindCameraForLayer(go.layer);
                    UICamera.currentCamera=camera;
                    UICamera.current=camera == null ? null : camera.GetComponent<UICamera>();
                    UICamera.currentTouchID=-1;
                    UICamera.currentTouch=new UICamera.MouseOrTouch {current=go,pressed=go,last=go};
                    if(cmd.action=="drag") {
                        var parts=(cmd.text ?? "").Split(',');
                        float dx,dy;
                        if(parts.Length!=2 || !float.TryParse(parts[0],System.Globalization.NumberStyles.Float,System.Globalization.CultureInfo.InvariantCulture,out dx)
                            || !float.TryParse(parts[1],System.Globalization.NumberStyles.Float,System.Globalization.CultureInfo.InvariantCulture,out dy)
                            || float.IsNaN(dx) || float.IsNaN(dy) || float.IsInfinity(dx) || float.IsInfinity(dy)
                            || Mathf.Abs(dx)>1000 || Mathf.Abs(dy)>1000 || camera==null)
                            throw new InvalidOperationException("Expected a drag delta dx,dy within 1000 pixels");
                        var pos=(Vector2)camera.WorldToScreenPoint(go.transform.position);
                        UICamera.currentTouch.pos=pos;
                        UICamera.lastWorldPosition=go.transform.position;
                        UICamera.Notify(go,UICamera.ENotifyType.OnPress,true);
                        try {
                            for(int step=1;step<=12;step++) {
                                var delta=new Vector2(dx,dy)/12f;
                                UICamera.currentTouch.pos=pos+delta*step;
                                UICamera.currentTouch.delta=delta;
                                UICamera.currentTouch.totalDelta=delta*step;
                                UICamera.Notify(go,UICamera.ENotifyType.OnDrag,delta);
                            }
                        } finally { UICamera.Notify(go,UICamera.ENotifyType.OnPress,false); }
                    } else if(cmd.action=="press") {
                        UICamera.Notify(go,UICamera.ENotifyType.OnPress,true);
                        UICamera.Notify(go,UICamera.ENotifyType.OnPress,false);
                    } else UICamera.Notify(go,UICamera.ENotifyType.OnClick,null);
                    if(cmd.action=="click-and-stop-story" && NGame2.NUI.NWindow.NBattle.AutoNextBattle.IsValidInstance)
                        NGame2.NUI.NWindow.NBattle.AutoNextBattle.instance.OnClickStop();
                } finally { UICamera.currentTouch=oldTouch; UICamera.currentCamera=oldCamera; UICamera.current=oldCurrent; UICamera.currentTouchID=oldId; UICamera.lastWorldPosition=oldWorld; }
            } else if(cmd.action=="enter-raid-node") {
                var nodes=UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NComponent2.PunishmentRaidComponent>();
                NGame2.NUI.NComponent2.PunishmentRaidComponent selected=null;
                foreach(var node in nodes) if(node.gameObject.activeInHierarchy && node.PunishmentRaidData!=null && node.PunishmentRaidData.RaidIndex==cmd.target) {
                    if(selected!=null)throw new InvalidOperationException("Ambiguous raid node");
                    selected=node;
                }
                if(selected==null || !selected.OpenRoot.gameObject.activeInHierarchy)
                    throw new InvalidOperationException("Raid node is not open");
                selected.OnClickEnter();
            } else if(cmd.action=="video-probe") {
                if(!String.IsNullOrEmpty(cmd.text)) {
                    if(String.IsNullOrEmpty(NShared.GameRoot.Instance.VideoDataContainer.GetVideoPath(cmd.text)))
                        throw new InvalidOperationException("Unknown cinematic video code");
                    if(NGame2.NUI.NWindow.VideoManager.IsPlaying())
                        throw new InvalidOperationException("A cinematic is already running");
                    NGame2.NUI.NWindow.VideoManager.Open_VideoCode(cmd.text);
                } else if(cmd.target==1 && NGame2.NUI.NWindow.VideoManager.IsPlaying()) {
                    NGame2.NUI.NWindow.VideoManager.SkipToClose(null);
                }
            } else if(cmd.action=="story-auto-probe") {
                if(!String.IsNullOrEmpty(cmd.text)) {
                    var parts=cmd.text.Split(':');
                    int chapter=int.Parse(parts[0]), dungeon=int.Parse(parts[1]);
                    var campaign=NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.instance.CampaignManager;
                    if(!campaign.CanGoToChapter(chapter) || !campaign.CanVisit(chapter,dungeon))
                        throw new InvalidOperationException("Requested story test node is locked");
                    int linked=campaign.GetLinkedDungeonIndex(chapter,dungeon);
                    NGame2.NAccount.UserCampaignManager.SetChapterDungeonIndex(chapter,dungeon,true);
                    NVespa.NSingleton.SceneAutoBehaviour<NGame2.NUI.NManager.NLobby.LobbyCampaignDungeon>.instance.OpenPartySetting(
                        new NGame2.NBattleContext.EnterDungeonData(chapter,dungeon,linked,NShared.DungeonDifficultyType.Normal,
                            campaign.CanPlayScenario(chapter,dungeon) && !campaign.IsCompleted(chapter,dungeon)));
                }
            } else if(cmd.action=="text" || cmd.action=="fill-input") {
                var input=Target(cmd.target).GetComponent<UIInput>();
                if(input == null || !input.enabled) throw new InvalidOperationException("Input is disabled");
                input.value=cmd.text ?? "";
                if(cmd.action=="text")input.Submit();
            } else if(cmd.action=="clear-errors") errors.Clear();
            else if(cmd.action=="focus-input") {
                if(cmd.target==0) UICamera.selectedObject=null;
                else {
                    var input=Target(cmd.target).GetComponent<UIInput>();
                    if(input==null || !input.isActiveAndEnabled) throw new InvalidOperationException("Input is disabled");
                    input.isSelected=true;
                }
            }
            else if(cmd.action=="scroll") {
                var go=Target(cmd.target);
                var scroll=NGUITools.FindInParents<UIScrollView>(go);
                float amount;
                if(scroll==null || !float.TryParse(cmd.text,System.Globalization.NumberStyles.Float,
                    System.Globalization.CultureInfo.InvariantCulture,out amount) || float.IsNaN(amount) || float.IsInfinity(amount) || Mathf.Abs(amount)>10f)
                    throw new InvalidOperationException("Expected a scroll view and an amount between -10 and 10");
                scroll.Scroll(amount);
            }
            else if(cmd.action=="jane-legend-probe" || cmd.action=="maria-legend-probe" || cmd.action=="ui-navigation-probe") { }
            else if(cmd.action=="preview-perk") {
                // Read-only UI preview, including already learned perks. The
                // confirmation callback intentionally cannot change the save.
                var icon=Target(cmd.target).GetComponent<NGame2.NUI.NComponent2.HeroTranscendSkillIconComponent>();
                if(icon==null || icon.TranscendSkill==null) throw new InvalidOperationException("Expected a text perk icon");
                var hero=(NShared.HeroInfo)icon.GetType().GetField("_heroInfo",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(icon);
                NGame2.NUI.NWindow.LearnTranscendSkillMessagePopup.Open(hero,icon.TranscendSkill.TranscendSkillIndex,delegate {});
                if(cmd.text=="repeat-description") {
                    var popup=UnityEngine.Object.FindObjectOfType<NGame2.NUI.NWindow.LearnTranscendSkillMessagePopup>();
                    var text=popup.Label_SkillDesc.text;
                    popup.Label_SkillDesc.text=text+"\n\n"+text+"\n\n"+text;
                    RestoredPortal.ReadablePerkDescription(popup);
                }
            }
            else if(cmd.action!="hero-growth-probe" && cmd.action!="account-form-probe" && cmd.action!="guild-attendance-probe" && cmd.action!="payment-probe" && cmd.action!="lobby-probe" && cmd.action!="snapshot" && cmd.action!="screenshot" && cmd.action!="quit" && cmd.action!="renderers" && cmd.action!="probe-renderer" && cmd.action!="soul-scene" && cmd.action!="dispatch-probe" && cmd.action!="party-ui" && cmd.action!="inn-scene" && cmd.action!="input-focus" && cmd.action!="perk-popup" && cmd.action!="skill-probe" && cmd.action!="combat-probe" && cmd.action!="combat-simulation" && cmd.action!="skill-table-probe" && cmd.action!="combat-table-probe") throw new InvalidOperationException("Unknown action");
            reply=Snapshot(cmd.id);
            if(cmd.action=="ui-navigation-probe") {
                var global=NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NUI.NManager.GlobalManagement>.instance;
                global.ForEachOpenedWindow(window=>reply.materials.Add(new Dictionary<string,object>{
                    {"kind","backStack"},{"type",window==null?null:window.GetType().FullName},
                    {"active",window!=null && window.gameObject.activeInHierarchy},
                    {"state",window==null?null:window.WindowState.ToString()},
                    {"opening",window!=null && window.ReservedOpenTask}}));
                var top=NGame2.NUI.NWindow.GlobalTopmostWindow.instance;
                reply.materials.Add(new Dictionary<string,object>{{"kind","inputBlock"},{"blocked",top.ExecuteBack()}});
                var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
                foreach(var holder in (System.Collections.IEnumerable)top.GetType().GetField("_touchDisablerList",flags).GetValue(top)) {
                    var disabler=holder.GetType().GetProperty("Disabler").GetValue(holder,null);
                    reply.materials.Add(new Dictionary<string,object>{{"kind","touchOwner"},{"type",disabler.GetType().FullName},{"detail",disabler.ToString()}});
                }
                foreach(var window in Resources.FindObjectsOfTypeAll<NGame2.NUI.NWindow.NBattle.AutoNextBattle>())
                    if(window.gameObject.scene.IsValid())reply.materials.Add(new Dictionary<string,object>{
                        {"kind","autoNextWindow"},{"active",window.gameObject.activeInHierarchy},
                        {"state",window.WindowState.ToString()},{"opening",window.ReservedOpenTask},
                        {"backType",window.BackButtonType.ToString()},{"backRegister",window.BackRegister}});
            }
            if(cmd.action=="jane-legend-probe" || cmd.action=="maria-legend-probe") ProbeJaneLegend(cmd,reply);
            if(cmd.action=="settings-probe")reply.materials.Add(new Dictionary<string,object>{
                {"kind","settings"},{"fps",(int)NVespa.SystemOptionData.Graphic_FrameControl},
                {"targetFps",Application.targetFrameRate},{"vSync",QualitySettings.vSyncCount},
                {"texture",(int)NVespa.SystemOptionData.Spec},{"bgm",NVespa.SystemOptionData.Sound_BgmVolume},
                {"outline",NVespa.SystemOptionData.Render_EnableOutline},
                {"prefsPath",Application.persistentDataPath}});
            if(cmd.action=="video-probe") {
                var manager=NVespa.SingletonUpdater<NGame2.NUI.NWindow.VideoManager>.instance;
                var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Public;
                var type=manager.GetType();
                var player=(UnityEngine.Video.VideoPlayer)type.GetField("VideoPlayer",flags).GetValue(manager);
                reply.materials.Add(new Dictionary<string,object>{{"kind","video"},
                    {"state",type.GetField("_state",flags).GetValue(manager).ToString()},
                    {"active",NGame2.NUI.NWindow.VideoManager.IsPlaying()},
                    {"prepared",player!=null && player.isPrepared},{"playing",player!=null && player.isPlaying},
                    {"frame",player==null?-1:player.frame},{"time",player==null?0:player.time},
                    {"url",player==null?"":player.url},{"width",player==null?0:player.width},
                    {"height",player==null?0:player.height}});
            }
            if(cmd.action=="story-auto-probe") {
                var helper=AppDomain.CurrentDomain.GetAssemblies().FirstOrDefault(a=>a.GetName().Name=="SprkStoryAuto");
                bool active=helper!=null && (bool)helper.GetType("SprkStoryAuto").GetMethod("Active").Invoke(null,null);
                var enter=NVespa.SingletonAuto<NGame2.NBattleContext.CampaignContext>.instance.EnterDungeon;
                var tutorial=NVespa.SingletonUpdater<NGame2.NTutorial.TutorialManager>.instance;
                reply.materials.Add(new Dictionary<string,object>{{"kind","storyAuto"},{"active",active},
                    {"chapter",enter==null?0:enter.Chapter},{"dungeon",enter==null?0:enter.OriginalDungeon},
                    {"difficulty",enter==null?"":enter.Difficulty.ToString()},{"scenario",enter!=null && enter.IsScenario},
                    {"tutorial",tutorial.IsProgressing},{"tutorialCommand",tutorial.CurrentTutorialData==null?"":tutorial.CurrentTutorialData.Command}});
                foreach(var party in UnityEngine.Object.FindObjectsOfType<NGame2.NUI.NWindow.BattlePartySetting>()) {
                    var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
                    var provider=(NGame2.NUI.NWindow.BattlePartySetting.IDataProvider)typeof(NGame2.NUI.NWindow.BattlePartySetting).GetField("_dataProvider",flags).GetValue(party);

    static Dictionary<string,object> TableProperties(object row) {
        var result=new Dictionary<string,object>();
        foreach(var property in row.GetType().GetProperties(System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public)) {
            if(!property.CanRead || property.GetIndexParameters().Length!=0)continue;
            var type=property.PropertyType;
            if(!(type.IsPrimitive || type.IsEnum || type==typeof(string) || type.IsArray))continue;
            var value=property.GetValue(row,null);
            result[property.Name]=type.IsEnum ? Convert.ToInt64(value) : value;
        }
        return result;
    }
