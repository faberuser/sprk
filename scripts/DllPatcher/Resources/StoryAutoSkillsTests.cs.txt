using System;
using NGame2.NUI.NWindow.NBattle;

namespace NVespa {
    public static class SceneAutoBehaviour<T> { public static NGame2.NUI.NManager.NBattle.BattleManagement instance; }
}
namespace NGame2.NUI.NWindow.NBattle {
    public class AutoControl { public bool Open, SkillReserveAuto; }
    public class SystemButtonAuto : AutoControl { }
    public class SystemButtonAutoFastContinue : AutoControl { }
    public class SystemButtonAutoSimple : AutoControl { }
}
namespace NGame2.NUI.NManager.NBattle {
    public class BattleManagement {
        public AutoControl Control;
        public bool IsOpened<T>() { return Control is T && Control.Open; }
        public T GetInstance<T>() { return (T)(object)Control; }
    }
}
public static partial class SprkStoryAuto {
    static bool active;
    static bool Active() { return active; }
    static void Check(bool value,string message) { if(!value)throw new Exception(message); }
    public static void RunTests() {
        var ui=new NGame2.NUI.NManager.NBattle.BattleManagement();
        NVespa.SceneAutoBehaviour<NGame2.NUI.NManager.NBattle.BattleManagement>.instance=ui;
        foreach(bool story in new[]{false,true}) {
            active=story;
            foreach(var control in new AutoControl[]{new SystemButtonAuto(),new SystemButtonAutoFastContinue(),new SystemButtonAutoSimple()}) {
                ui.Control=control;control.Open=true;
                foreach(bool enabled in new[]{false,true,false,true}) {
                    Check(AutoSkills(false)==false,"Continuous mode forced global Auto on");
                    Check(AutoSkills(true)==true,"Native editor Auto override changed");
                    control.SkillReserveAuto=enabled;
                    Check(ScriptedAutoSkills()==enabled,"Scripted skills ignored current AUTO button: "+control.GetType().Name+", story="+story);
                }
                control.Open=false;
                Check(!ScriptedAutoSkills(),"Closed control must not trigger scripted skills");
            }
        }
        NVespa.SceneAutoBehaviour<NGame2.NUI.NManager.NBattle.BattleManagement>.instance=null;
        Check(!ScriptedAutoSkills(),"Missing battle UI must not trigger scripted skills");
        Console.WriteLine("PASS: Continuous Battle/Story do not force skill Auto; all three button variants and scripted encounters respect on/off toggles, closed and absent UI.");
    }
}
