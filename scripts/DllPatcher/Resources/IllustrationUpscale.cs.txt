    string ModelDownloadUrl() {
        // The launcher captures this profile's exact update endpoint at launch.
        // Do not reread/guess a profile when the launcher supplied its URL.
        string manifest=Environment.GetEnvironmentVariable("SPRK_UPDATE_MANIFEST_URL");
        if(!String.IsNullOrEmpty(manifest))return SprkIllustrationUpscaleCore.ModelUrlFromManifest(manifest);
        string file=Path.Combine(Path.GetDirectoryName(Application.dataPath),"sprk-launcher.json");
        SprkIllustrationLauncherConfig config=null;
        if(File.Exists(file)) {
            if(new FileInfo(file).Length>64*1024)throw new InvalidDataException("Launcher server profile is too large");
            // JsonUtility cannot populate unregistered helper DTOs in this
            // player. Use the game's managed JSON parser for launcher profiles.
            var parsed=MiniJSON.Json.Deserialize(File.ReadAllText(file)) as IDictionary<string,object>;
            if(parsed==null)throw new InvalidDataException("Launcher server profile is invalid");
            object value;
            config=new SprkIllustrationLauncherConfig {
                SelectedProfile=parsed.TryGetValue("SelectedProfile",out value)?value as string:null,
                ManifestUrl=parsed.TryGetValue("ManifestUrl",out value)?value as string:null
            };
            if(parsed.TryGetValue("Profiles",out value) && value!=null) {
                var entries=value as IList;
                if(entries==null)throw new InvalidDataException("Launcher server profiles are invalid");
                var profiles=new List<SprkIllustrationLauncherProfile>();
                foreach(var entry in entries) {
                    var profile=entry as IDictionary<string,object>;
                    if(profile==null)throw new InvalidDataException("Launcher server profile is invalid");
                    profiles.Add(new SprkIllustrationLauncherProfile {
                        Id=profile.TryGetValue("Id",out value)?value as string:null,
                        HostUrl=profile.TryGetValue("HostUrl",out value)?value as string:null,
                        ManifestUrl=profile.TryGetValue("ManifestUrl",out value)?value as string:null
                    });
                }
                config.Profiles=profiles.ToArray();
            }
        }
        return SprkIllustrationUpscaleCore.ModelUrlFromProfile(config,Environment.GetEnvironmentVariable("SPRK_HOST_URL"));
    }
    void Cancel() {

        // Observe cancelled/faulted jobs even if their viewer has been destroyed.

        pending=null;progress=null;
    }

            window.HeroIllustTexture.uvRect=originalUv;

    }

    public void Select(int factor) {
        if(!Visible() || (factor!=0 && factor!=4) || factor==requested)return;
        ++generation;Cancel();requested=factor;status="";
        if(factor==0){Restore();lastSource=null;reuse=new SprkIllustrationReuse();Refresh();return;}
        if(!ready || source==null){status="Waiting for artwork";Refresh();return;}
        if(active!=null && active!=this)active.Select(0);
        active=this;status="Preparing x"+factor;Refresh();
        StartCoroutine(Prepare(generation,factor));
    }
    IEnumerator Prepare(int version,int factor) {
        yield return null;
        if(version!=generation || !Visible())yield break;
        try {
            if(SystemInfo.graphicsDeviceType==UnityEngine.Rendering.GraphicsDeviceType.Null || SystemInfo.maxTextureSize<4096)
                throw new InvalidOperationException("This graphics device does not support illustration enhancement");
            inputWidth=Mathf.RoundToInt(Mathf.Abs(originalUv.width)*source.width);
            inputHeight=Mathf.RoundToInt(Mathf.Abs(originalUv.height)*source.height);
            if(inputWidth<1 || inputHeight<1 || inputWidth>SprkIllustrationUpscaleCore.MaxInput || inputHeight>SprkIllustrationUpscaleCore.MaxInput)
                throw new InvalidOperationException("Artwork exceeds the 1024 pixel input limit");
            // Sampling the crop at native pixel centers keeps neighboring atlas
            // sprites out. NCNN pads the isolated crop, never the whole atlas.
            if(lastSource==null) {
                var crop=ReadCrop(source,inputWidth,inputHeight,originalUv);
                try{lastSource=crop.EncodeToPNG();}finally{Destroy(crop);}
            }
            var bytes=lastSource;int w=inputWidth,h=inputHeight;
            string model=SprkIllustrationUpscaleCore.ModelPath(storage),cache=Path.Combine(storage,"cache");
            // A usable replacement remains independent of profile files/network.
            string modelUrl=SprkIllustrationUpscaleCore.Nonempty(model)?null:ModelDownloadUrl();
            cancellation=new CancellationTokenSource();var token=cancellation.Token;
            progress=new SprkIllustrationProgress();var jobProgress=progress;var jobReuse=reuse;
            pending=Task.Run(delegate {return SprkIllustrationUpscaleCore.Run(bytes,w,h,factor,model,runtime,cache,token,modelUrl,jobProgress,jobReuse);},token);
            status="Processing x"+factor;Refresh();
        }catch(Exception error){Fail(error);}
    }
    static Texture2D ReadCrop(Texture texture,int width,int height,Rect uv) {
        var previous=RenderTexture.active;
        var target=RenderTexture.GetTemporary(width,height,0,RenderTextureFormat.ARGB32,RenderTextureReadWrite.sRGB);
        Texture2D read=null;
        try {
            bool srgb=GL.sRGBWrite;
            try {GL.sRGBWrite=QualitySettings.activeColorSpace==ColorSpace.Linear;Graphics.Blit(texture,target,new Vector2(uv.width,uv.height),new Vector2(uv.x,uv.y));}
            finally{GL.sRGBWrite=srgb;}
            RenderTexture.active=target;
            read=new Texture2D(width,height,TextureFormat.RGBA32,false,false);
            read.ReadPixels(new Rect(0,0,width,height),0,0,false);read.Apply(false,false);return read;
        }catch{if(read!=null)Destroy(read);throw;}
        finally{RenderTexture.active=previous;RenderTexture.ReleaseTemporary(target);}
    }
    void Fail(Exception error) {
        Restore();progress=null;reuse.Clear();status="Unavailable: "+error.GetBaseException().Message+". Original displayed.";
        Debug.LogWarning("SPRK illustration AI: "+error.GetBaseException().Message);Refresh();
    }
    void Update() {
        if(!Visible())return;
        if(button==null)CreateControls();
        if(button!=null && window.WidgetHideUI!=null) {
            bool show=window.WidgetHideUI.gameObject.activeInHierarchy && window.WidgetHideUI.alpha>0.01f;
            if(button.gameObject.activeSelf!=show){if(!show)HideHelp();button.gameObject.SetActive(show);}
        }
        // A subtle pulse communicates pending work without adding button text.
        if(label!=null)label.alpha=pending!=null || status=="Waiting for artwork" || status.StartsWith("Preparing")
            ? .65f+.35f*Mathf.Sin(Time.realtimeSinceStartup*5)*Mathf.Sin(Time.realtimeSinceStartup*5):1;
        UpdateLoading();
        // A selection made before the native assignment finishes waits; it never
        // starts against the previous artwork or schedules work while Off.
        if(requested!=0 && ready && pending==null && status=="Waiting for artwork") {
            status="Preparing x"+requested;Refresh();StartCoroutine(Prepare(generation,requested));
        }
        if(pending==null || !pending.IsCompleted)return;
        var completed=pending;pending=null;
        if(completed.IsCanceled)return;
        if(completed.IsFaulted){Fail(completed.Exception);return;}
        if(requested==0 || !ready || (window.HeroIllustTexture.mainTexture!=source && window.HeroIllustTexture.mainTexture!=enhanced))return;
        Texture2D result=null;
        try {
            result=new Texture2D(2,2,TextureFormat.RGBA32,false,false);
            if(!result.LoadImage(completed.Result,false) || result.width!=inputWidth*4 || result.height!=inputHeight*4)
                throw new InvalidDataException("Enhanced PNG could not be decoded");
            result.wrapMode=TextureWrapMode.Clamp;result.filterMode=FilterMode.Bilinear;result.Apply(false,true);
            result.name="SPRK Illustration x"+requested;
            Restore();
            enhanced=result;result=null;
            if(originalMaterial!=null) {
                enhancedMaterial=new Material(originalMaterial);enhancedMaterial.mainTexture=enhanced;
            }
            window.HeroIllustTexture.material=enhancedMaterial;
            window.HeroIllustTexture.mainTexture=enhanced;window.HeroIllustTexture.uvRect=new Rect(0,0,1,1);
            progress=null;reuse.Clear();lastSource=null;status="On (x4 ready)";Refresh();
        }catch(Exception error) {
            if(result!=null)Destroy(result);
            var bytes=lastSource;int scale=requested;
            if(bytes!=null)Task.Run(delegate {SprkIllustrationUpscaleCore.RemoveCache(bytes,SprkIllustrationUpscaleCore.ModelPath(storage),scale,Path.Combine(storage,"cache"));});
            Fail(error);
        }
    }
    void UpdateLoading() {
        if(loading==null)return;
        bool busy=requested!=0 && (pending!=null || status.StartsWith("Preparing") || status=="Waiting for artwork");
        bool show=busy && button!=null && button.gameObject.activeInHierarchy;
        if(loading.activeSelf!=show)loading.SetActive(show);
        if(!show)return;
        string stage=status=="Waiting for artwork"?"Waiting for artwork":"Preparing";float value=-1;
        if(progress!=null)progress.Read(out stage,out value);
        loadingLabel.text=stage+(value<0?"...":" "+Mathf.FloorToInt(Mathf.Clamp01(value)*100)+"%");
        // Indeterminate stages pulse; percentages come from bytes/NCNN tiles.
        loadingFill.width=Mathf.Max(2,Mathf.RoundToInt(240*(value<0?.35f:Mathf.Clamp01(value))));
        loadingFill.alpha=value<0?.55f+.45f*Mathf.Sin(Time.realtimeSinceStartup*4)*Mathf.Sin(Time.realtimeSinceStartup*4):1;
    }
