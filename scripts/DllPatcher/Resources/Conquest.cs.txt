// Only the separate service process receives SPRK_CONQUEST_WORKER_KEY.
// Ordinary game clients use this assembly solely for authenticated login and
// initial shared boss HP; they cannot submit authoritative combat results.
public sealed class SprkConquest : MonoBehaviour {
    static SprkConquest instance;
    int lastFrame=-1;
    static readonly Dictionary<WaveReadyBattleLog,ResultCreatureInfo[]> seeds=new Dictionary<WaveReadyBattleLog,ResultCreatureInfo[]>();
    static readonly bool trace=Environment.GetEnvironmentVariable("SPRK_BATTLE_TRACE")=="1";
    static readonly object traceGate=new object();
    static volatile SprkReplayClock replayClock;
    public static bool FilterClock(ref long time) {
        var clock=replayClock;
        return clock==null || clock.Accept(ref time);
    }
    public static void Trace(string message) {
        if(!trace)return;
        try {
            string path=Environment.GetEnvironmentVariable("SPRK_BATTLE_TRACE_FILE");
            if(String.IsNullOrEmpty(path))path=Path.Combine(Application.persistentDataPath,"sprk-battle.log");
            lock(traceGate)File.AppendAllText(path,DateTime.UtcNow.ToString("O")+" "+message+Environment.NewLine);
        } catch { }
    }
    public static bool ArenaAutoPlayable(BattleType type,ref bool auto,ref bool fast) {
        if(type!=BattleType.Arena && type!=BattleType.ArenaBanPick)return false;
        auto=true;fast=false;return true;
    }
    public static bool ArenaAuto(bool original,object control) {
        if(!SingletonAuto<BattleFacade>.isValidInstance)return original;
        var cache=SingletonAuto<BattleFacade>.instance.GetBattleCache();
        if(cache==null || cache.ChapterIndex!=1000 ||
            (cache.BattleType!=BattleType.Arena && cache.BattleType!=BattleType.ArenaBanPick))return original;
        // A native button can light up while DisableAutoBattle suppresses the
        // getter. Restore its displayed toggle only in an actual Arena battle.
        var simple=control as NGame2.NUI.NWindow.NBattle.SystemButtonAutoSimple;
        if(simple!=null)return simple.IsAuto();
        var normal=control as NGame2.NUI.NWindow.NBattle.SystemButtonAuto;
        return normal==null?original:normal.IsAuto();
    }
    public static void LeaveSoloConquestOnLobbyReturn() {
        // Online solo Conquest uses PartyManager too. Its setup window was
        // destroyed on battle entry, so OnClosing cannot leave on result exit.
        // The explicit Lobby button must release the room before scene change.
        if(!GlobalAutoBehaviour<NGame2.NParty.PartyManager>.isValidInstance)return;
        var party=GlobalAutoBehaviour<NGame2.NParty.PartyManager>.instance;
        var info=party.PartyInfo;
        if(IsSoloConquestParty(info))
            party.SendLeaveParty();
    }
    public static bool IsSoloConquestParty(NShared.NParty.PartyInfo info) {
        return info!=null && info.RaidIndex==90001 && info.RaidLevel>=101 && info.RaidLevel<=103;
    }
    public static Dictionary<string,object> Login(Dictionary<string,object> value) {
        int buffer;
        if(!Int32.TryParse(Environment.GetEnvironmentVariable("SPRK_ARENA_REPLAY_BUFFER_MS"),out buffer))buffer=100;
        bool arena=value.ContainsKey("MatchUid") && Convert.ToInt64(value["MatchUid"])>0;
        bool coop=value.ContainsKey("MasterId") && Convert.ToInt64(value["MasterId"])>0;
        // GlobalTimer is continuous across native waves. Reset only for login
        // (including reconnect), never on WaveReady or a team replacement.
        // Co-op filters stale clocks with zero added presentation delay.
        replayClock=arena || coop?new SprkReplayClock(arena?buffer:0):null;
        var web=SingletonUpdater<WebServiceManager>.instance;
        if(web!=null && web.Requester!=null)value["SessionKey"]=web.Requester.SessionKey;
        return value;
    }
    public static Dictionary<string,object> ReplayStatus(Dictionary<string,object> value) {
        // GetStoppedBattleInstance stops the replay thread. Heartbeats must only
        // observe the published cache; the native result handler stops it later.
        var cache=SingletonAuto<BattleFacade>.isValidInstance ? SingletonAuto<BattleFacade>.instance.GetBattleCache() : null;
        value["BattleStopped"]=cache!=null && cache.IsBattleEnded;
        value["ReplayWaveIndex"]=cache==null ? 0 : cache.WaveIndex;
        return value;
    }
    public static BattleLogNot ReadClock(BattleLogNot value,IDictionary data) {
        // Native BattleLogNot ignores this extension. Apply it on the battle
        // thread so sparse action logs cannot make replay timing drift behind.
        long time;
        if(value!=null && data.Contains("ServerTimeMs") && Int64.TryParse(Convert.ToString(data["ServerTimeMs"]),out time) && time>=0 &&
            SingletonAuto<BattleFacade>.isValidInstance && SingletonAuto<BattleFacade>.instance.IsValidBattle()) {
            if(trace) {
                var cache=SingletonAuto<BattleFacade>.instance.GetBattleCache();
                if(cache!=null)Trace("clock serverMs="+time+" replayMs="+cache.GlobalCurrentTimeMs+" leadMs="+(cache.GlobalCurrentTimeMs-time));
            }
            SingletonAuto<BattleFacade>.instance.SetServerTimeMs(time);
        }
        return value;
    }
    public static WaveReadyBattleLog ReadSeed(WaveReadyBattleLog value,IDictionary data) {
        if(value!=null && value.ChapterIndex==9900 && data.Contains("ConquestInitialCreatures")) {
            var list=data["ConquestInitialCreatures"] as IList;
            if(list!=null)seeds[value]=list.Cast<IDictionary>().Select(JM_NShared_ResultCreatureInfo.Parse).ToArray();
        }
        return value;
    }
    public static void ApplySeed(WaveReadyBattleLog value,BattleInstance battle) {
        ResultCreatureInfo[] initial;
        if(seeds.TryGetValue(value,out initial)) {battle.SetInitialCreatureInfo(initial);seeds.Remove(value);}
    }
    public static List<ResultCreatureInfo> ScoreBoss(List<ResultCreatureInfo> infos,BattleInstance battle) {
        if(battle==null || battle.ChapterIndex!=9900 || (battle.DungeonIndex!=3 && battle.DungeonIndex!=103))return infos;
        foreach(var info in infos) {
            if(info.TeamId!=1 || info.Index!=8801)continue;
            var creature=battle.CreatureContainer.GetCreatureByKey(info.Key);
            if(creature!=null)info.GivedDamage=battle.Snapshot.StatisticsManager.TotalDamagedMeter(creature.Uid);
        }
        return infos;
    }
    public static void Tick() {
        if(String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_CONQUEST_WORKER_KEY")))return;
        // The game's login flow disables its logger. Keep service failures visible.
        Debug.unityLogger.logEnabled=true;
        if(instance==null) {
            var go=new GameObject("SprkConquestService");DontDestroyOnLoad(go);instance=go.AddComponent<SprkConquest>();
            Application.runInBackground=true;Application.targetFrameRate=60;
            Debug.Log("SPRK conquest service initialized");
        }
        instance.Step();
    }
    void Update() {Step();}
    class Run : IBattleLogListener {
        public SprkConquest owner;
        public string id;
        public CampaignUpdater updater;
        public long hp;
        public int boss;
        public bool ended;
        public double budget;
        public Dictionary<int,long> heroes=new Dictionary<int,long>();
        public HashSet<long> players=new HashSet<long>();
        public bool arena;
        public bool dispatch;
        public HashSet<long> withdrawn=new HashSet<long>();
        public List<object> logs=new List<object>();
        public Dictionary<string,int> defeated=new Dictionary<string,int>();
        readonly HashSet<string> defeatedKeys=new HashSet<string>();
        public void CaptureDefeats() {
            if(arena)return;
            foreach(var info in BattleInstanceIntegrityChecker.GetCreatureInfoList(updater.BattleInstance,-1)) {
                if(info.TeamId!=1 || info.Hp>0 || info.MaxHp<=0)continue;
                string key=updater.WaveIndex+":"+info.Key;
                if(!defeatedKeys.Add(key))continue;
                string index=info.Index.ToString();int count;
                defeated[index]=(defeated.TryGetValue(index,out count)?count:0)+1;
            }
        }
        long lastClockMs=-1;
        public void OnProcessBattleLog(long time,BattleLog log) {
            if(log.Type==BattleLogType.InitialDungeonBattleLog && updater.BattleInstance.CheckEachWaveEachTeam) {
                // The offline initializer gives every replacement hero to one
                // player. Clear once, then use native per-player creation logs
                // so both the simulation and client replays retain ownership.
                var initial=JM_NShared_InitializeDungeonBattleLog.Parse(log.Data);
                foreach(var creature in initial.CreateInfos ?? new BattleCreatureCreateInfo[0]) {
                    long account;
                    updater.BattleInstance.BattleLogContainer.AddLog(new CreatureCreateBattleLog {
                        PlayerId=heroes.TryGetValue(creature.HeroInfo.HeroIndex,out account)?account:initial.PlayerId,CreateInfo=creature});
                }
                initial.CreateInfos=new BattleCreatureCreateInfo[0];
                log.BaseBattleLog.Data=JsonMarshaler.EncodeJsonString(initial);
            }
            if(log.Type==BattleLogType.WaveReady && dispatch) {
                foreach(long player in players)updater.BattleInstance.BattleLogContainer.AddLog(new CreatureSetAiBattleLog{PlayerId=player,SetAi=true});
            }
            if(log.Type==BattleLogType.WaveReady && !arena) {
                var data=MiniJSON.Json.Deserialize(log.Data) as IDictionary;
                var ready=JM_NShared_WaveReadyBattleLog.Parse(data);
                var initial=new List<object>();
                foreach(var monster in ready.MonsterCreateInfos ?? new BattleCreatureCreateInfo[0]) {
                    if(monster.HeroInfo.HeroIndex==boss)initial.Add(JM_NShared_ResultCreatureInfo.EncodeJsonObject(new ResultCreatureInfo {
                        Index=boss,TeamId=1,Key=ResultCreatureInfo.GetKey(boss,monster.WaveSlotIndex,1),Hp=hp,
                        MaxHp=monster.StatData.GetStat(StatType.MaxHp),Mp=0,MaxMp=(int)monster.StatData.GetStat(StatType.MaxMp)}));
                }
                data["ConquestInitialCreatures"]=initial;log.BaseBattleLog.Data=MiniJSON.Json.Serialize(data);
            }
            logs.Add(JM_NShared_BaseBattleLog.EncodeJsonObject(log.BaseBattleLog));
            if(log.Type==BattleLogType.WaveReady || log.Type==BattleLogType.WaveEnd) {
                if(log.Type==BattleLogType.WaveEnd)CaptureDefeats();
                var wave=new Dictionary<string,object>{{"WaveIndex",updater.WaveIndex},{"TotalWaveIndex",updater.TotalWaveIndex}};
                if(log.Type==BattleLogType.WaveEnd)wave["Winner"]=JM_NShared_WaveEndBattleLog.Parse(log.Data).winner.ToString();
                Flush();owner.Send("WorkerPacket",new Dictionary<string,object>{{"RunId",id},{"Name",log.Type==BattleLogType.WaveReady?"WaveReadyNot":"WaveEndNot"},
                    {"Body",wave}});
            }
        }
        public void Flush() {
            long clock=updater.BattleInstance.GlobalTimer.GetCurrentTimeMs();
            // Publish advancing time even when no action logs were generated.
            // Otherwise PingRes repeats an old time and slows the native replay.
            if(logs.Count==0 && clock==lastClockMs)return;
            lastClockMs=clock;
            owner.Send("WorkerPacket",new Dictionary<string,object>{{"RunId",id},{"Name","BattleLogNot"},
                {"Body",new Dictionary<string,object>{{"BattleLogs",logs.ToArray()},{"ServerTimeMs",clock}}}});logs.Clear();
        }
    }
    readonly Dictionary<string,Run> runs=new Dictionary<string,Run>();
    volatile SprkBattleLink link;
    float retryAt,lastHeartbeat;
    long lastStep;
    int firstRun;
    int connecting;
    void Connect() {
        if(Interlocked.CompareExchange(ref connecting,1,0)!=0)return;
        retryAt=Time.realtimeSinceStartup+5;
        // Data containers load through the ordinary client startup; no player
        // account or client-supplied stat snapshot is needed by this service.
        NShared.NJit.TableRoot.Instance.SetPatchDownLoadComplete(true);
        if(GameRoot.Instance.JsonLoader==null) {Debug.Log("SPRK native worker loading table assets");SingletonAuto<TableDataManager>.instance.Load(true);connecting=0;return;}
        var creatures=GameRoot.Instance.CreatureDataContainer;
        // Creature keys are table-defined; a loaded table need not contain 1.
        if(creatures==null || !creatures.GetAllData().Any()) {Debug.Log("SPRK native worker waiting for creature table");connecting=0;return;}
        Debug.Log("SPRK native worker connecting");
        new Thread(()=> {
            TcpClient connection=null;
            try {
                var socket=connection=new TcpClient();socket.NoDelay=true;
                int port; if(!Int32.TryParse(Environment.GetEnvironmentVariable("SPRK_CONQUEST_WORKER_PORT"),out port))port=9002;
                socket.Connect(Environment.GetEnvironmentVariable("SPRK_CONQUEST_WORKER_HOST") ?? "127.0.0.1",port);
                var session=new SprkBattleLink(socket,MiniJSON.Json.Serialize,message=>Debug.LogWarning("SPRK worker transport: "+message));
                session.Send("WorkerLogin",new Dictionary<string,object>{{"Key",Environment.GetEnvironmentVariable("SPRK_CONQUEST_WORKER_KEY")},{"ArenaVersion",1},{"ChallengeRaidVersion",1},{"DispatchVersion",1}});
                link=session;
                session.ReceiveLoop();
            } catch(Exception e) {Debug.LogWarning("SPRK conquest service connection: "+e.Message);}
            finally {if(connection!=null)connection.Close();Interlocked.Exchange(ref connecting,0);}
        }){IsBackground=true}.Start();
    }
    void Send(string name,object value) {
        var session=link;if(session!=null)session.Send(name,value);
    }
    void Step() {
        Debug.unityLogger.logEnabled=true;
        if(lastFrame==Time.frameCount)return;lastFrame=Time.frameCount;
        float now=Time.realtimeSinceStartup;
        long stamp=System.Diagnostics.Stopwatch.GetTimestamp();
        if(link==null || !link.Connected) {
            if(runs.Count>0) {runs.Clear();Debug.LogWarning("SPRK conquest service disconnected; active runs cancelled");}
            if(now>=retryAt)Connect();lastStep=stamp;return;
        }
        if(now-lastHeartbeat>=5) {lastHeartbeat=now;Send("WorkerPing",new Dictionary<string,object>());}
        try {
            for(int count=0;count<100;count++) {
                string line;if(!link.TryReceive(out line))break;
                var pair=line.Split(new[]{' '},2);var data=MiniJSON.Json.Deserialize(Encoding.UTF8.GetString(Convert.FromBase64String(pair[1]))) as IDictionary;
                if(pair[0]=="WorkerLoginRes") {Debug.Log("SPRK conquest service authenticated");continue;}
                string id=Convert.ToString(data["RunId"]);Run run;
                if(pair[0]=="WorkerPrepare")Prepare(id,data);
                else if(runs.TryGetValue(id,out run)) {
                    if(pair[0]=="WorkerStart")run.updater.StartBattle(true);
                    else if(pair[0]=="WorkerResume")run.updater.SetWavePauseFalse();
                    else if(pair[0]=="WorkerCancel")runs.Remove(id);
                    else if(pair[0]=="WorkerCommand")Command(run,data);
                    else if(pair[0]=="WorkerWithdraw") {
                        long account=Convert.ToInt64(data["AccountId"]);run.withdrawn.Add(account);
                        run.updater.BattleInstance.BattleLogContainer.AddLog(new CreatureSetAiBattleLog{PlayerId=account,SetAi=true});
                        if(run.players.All(run.withdrawn.Contains))run.updater.BattleInstance.BattleLogContainer.AddLog(new WaveWithdrawBattleLog());
                    }
                }
            }
            double elapsed=(stamp-lastStep)*1000.0/System.Diagnostics.Stopwatch.Frequency;lastStep=stamp;
            var active=runs.Values.ToArray();
            foreach(var run in active)run.budget+=elapsed;
            // Fair round-robin catch-up, at most five native ticks per run in a
            // frame. Preserve elapsed time as debt rather than discarding it.
            // Flush every tick so one busy run cannot batch away clock updates.
            for(int round=0;round<5;round++) {
                for(int i=0;i<active.Length;i++) {
                    var run=active[(firstRun+i)%active.Length];
                    if(run.ended || run.budget<100)continue;
                    run.budget-=100;run.updater.Update();run.updater.BattleInstance.UpdateTick(false);SnapshotObjectPoolContainer.FlushPendingRelease();
                    run.Flush();
                    if(run.updater.IsBattleEnd())Finish(run);
                }
            }
            if(active.Length>0)firstRun=(firstRun+1)%active.Length;
            if(trace && elapsed>150)Trace("worker frameGapMs="+elapsed+" runs="+active.Length+" maxDebtMs="+(active.Length==0?0:active.Max(r=>r.budget)));
        } catch(Exception e) {Debug.LogError("SPRK conquest service: "+e);if(link!=null)link.Close();}
    }
    void Prepare(string id,IDictionary data) {
        if(runs.ContainsKey(id))return;
        GameRoot.Instance.InitForClient();
        GameRoot.Instance.BattleLogFactory.SetSerializer(new BattleLogFactory.BattleLogSerializer(JsonMarshaler.EncodeJsonString<IBattleLogRuntime>));
        GameRoot.Instance.BattleLogFactory.SetDeserializer(new BattleLogFactory.BattleLogDeserializer(JsonMarshaler.Parse));
        GameRoot.Instance.OnDataContainerLoaded(false);
        var info=JM_NShared_BattleSimulationInfo.Parse(MiniJSON.Json.Serialize(data["Simulation"]));
        var builder=new BattleInstanceBuilder();builder.SetCampaignDungeon(info.ChapterIndex,info.DungeonIndex,info.DungeonDifficulty,false);builder.EnableSnapshot(false);
        var affixes=data.Contains("AffixIndices")?data["AffixIndices"] as IList:null;
        if(affixes!=null)builder.SetExtraData(new BattleInstanceExtraData{AffixIndices=affixes.Cast<object>().Select(Convert.ToInt32).ToArray()});
        var enter=new List<IBattleLogRuntime>();var run=new Run{owner=this,id=id,hp=Convert.ToInt64(data["BossHp"]),boss=Convert.ToInt32(data["BossIndex"]),arena=data.Contains("ArenaMatch") && Convert.ToBoolean(data["ArenaMatch"]),dispatch=data.Contains("Dispatch") && Convert.ToBoolean(data["Dispatch"])};
        var dungeon=GameRoot.Instance.CampaignDungeonDataContainer.GetData(info.ChapterIndex,info.DungeonIndex);
        bool alternating=dungeon!=null && (dungeon.BattleType==BattleType.ArcdimRaid || dungeon.BattleType==BattleType.ArcdimHardRaid);
        var waveTeams=new[]{new List<BattleCreatureCreateInfo>(),new List<BattleCreatureCreateInfo>()};
        var match=run.arena?JM_NShared_MatchInfo.Parse(MiniJSON.Json.Serialize(data["MatchInfo"])):null;
        foreach(var account in info.AccountInfos) {
            run.players.Add(account.UserInfo.AccountId);
            var matched=match==null?null:match.AccountInfos.First(a=>a.AccountId==account.UserInfo.AccountId);
            var creatures=new List<BattleCreatureCreateInfo>();
            var reserved=new List<BattleCreatureCreateInfo>();
            foreach(var h in account.HeroInfos.Values) {
                h.Validate();run.heroes[h.HeroIndex]=account.UserInfo.AccountId;
                bool second=account.GroupHeroInfos!=null && account.GroupHeroInfos.ContainsKey(h.HeroIndex);
                // Conquest's sub deck fights alongside the main deck as AI.
                // Group 1 is reserved for replacement teams and is not spawned
                // by this dungeon's single-team battle definition.
                bool aiSupport=second && info.ChapterIndex==9900 && info.DungeonIndex>=101 && info.DungeonIndex<=103;
                // Native BattleSimulationInstance creates every dispatch hero with
                // IsAi and IsSmartAi enabled. Enabling persistent AI afterwards
                // starts the ordinary tactic and does not restore smart AI.
                var creature=BattleCreatureCreateInfo.CreateMyCreatureCreateInfo(h,matched!=null && h.HeroIndex==matched.LeaderHeroIndex,aiSupport || run.dispatch,
                    new BattleCreatureCreateInfo.BuffSkillInfo{GuildSkillDataBases=account.GuildSkills,AccountBuffDataBases=account.AccountBuffs,ClassBuffDataBasess=account.ClassBuffDataBases},
                    new BattleCreatureCreateInfo.BuffStatInfo{ExtraStatDatas=account.ExtraStatDatas,PetStatDataBases=account.PetStatDataBases},
                    !alternating && second && !aiSupport?1:0);
                if(run.dispatch && (!creature.IsAi || !creature.IsSmartAi))
                    throw new InvalidOperationException("Dispatch heroes must use native simulation AI");
                if(alternating)waveTeams[second?1:0].Add(creature);
                if(alternating && second)reserved.Add(creature);else creatures.Add(creature);
            }
            enter.Add(new PlayerEnterBattleLog{PlayerCreateInfo=new BattlePlayerCreateInfo{Id=account.UserInfo.AccountId,TeamType=matched==null?TeamType.Left:matched.TeamType},CreatureCreateInfos=creatures.ToArray(),BattleCheckCreatureCreateInfos=reserved.ToArray()});
        }
        builder.SetEnterPlayerBattleLogList(enter);run.updater=builder.BuildUpdater<BattleInstance>(-1,Convert.ToInt32(data["Seed"]),null);
        if(alternating) {
            foreach(var team in waveTeams)run.updater.AddEachWaveEachTeamBattleCreatureCreateInfo(team);
            run.updater.SetWavePauseCallback(winner=> {
                run.Flush();Send("WorkerPacket",new Dictionary<string,object>{{"RunId",id},{"Name","SetNextTeamNot"},
                    {"Body",new Dictionary<string,object>{{"ChatperIndex",info.ChapterIndex},{"DungeonIndex",info.DungeonIndex},{"DeckIndex",run.updater.WaveIndex-1}}}});
            });
        }
        run.updater.BattleInstance.BattleLogContainer.SetListener(run);run.updater.BattleInstance.SetDeltaTimeReductionRatio(1);
        runs[id]=run;Debug.Log("SPRK conquest service prepared "+id+" with "+run.heroes.Count+" heroes");
    }
    void Command(Run run,IDictionary data) {
        var name=Convert.ToString(data["Name"]);var body=data["Body"] as IDictionary;
        int uid=Convert.ToInt32(body["CreatureUid"]);long account=Convert.ToInt64(data["AccountId"]);
        var creature=run.updater.BattleInstance.CreatureContainer.GetCreature(uid);
        if(creature==null || creature.PlayerId!=account)return;
        if(trace)Trace("command run="+run.id+" name="+name+" request="+Convert.ToString(body["RequestId"])+" simulationMs="+run.updater.BattleInstance.GlobalTimer.GetCurrentTimeMs()
            +" relayAndQueueMs="+(data.Contains("QueuedAtUnixMs")?(DateTime.UtcNow.Ticks-621355968000000000L)/10000-Convert.ToInt64(data["QueuedAtUnixMs"]):-1));
        IBattleLogRuntime command=null;
        if(name=="ReserveOnSkillReq")command=JM_NShared_CreatureReserveOnSkillBattleLog.Parse(body);
        if(name=="ReserveOffSkillReq")command=JM_NShared_CreatureReserveOffSkillBattleLog.Parse(body);
        if(name=="FireNextSkillReq")command=JM_NShared_CreatureFireNextSkillBattleLog.Parse(body);
        if(name=="SetManualTargetReq")command=JM_NShared_CreatureManualTargetBattleLog.Parse(body);
        if(name=="SetManualMoveReq")command=JM_NShared_CreatureManualMoveBattleLog.Parse(body);
        if(command!=null)run.updater.BattleInstance.BattleLogContainer.AddLog(command);
    }
    void Finish(Run run) {
        run.CaptureDefeats();
        run.ended=true;var battle=run.updater.BattleInstance;
        var infos=BattleInstanceIntegrityChecker.GetCreatureInfoList(battle,-1);
        var creatures=new List<object>();
        foreach(var info in infos) {
            var encoded=JM_NShared_ResultCreatureInfo.EncodeJsonObject(info);
            // Native GivedDamage means damage received by an immortal creature.
            // Keep it intact; contributions use the actual attacker meter.
            var creature=battle.CreatureContainer.GetCreatureByKey(info.Key);
            encoded["AccountId"]=creature==null?0:creature.PlayerId;
            encoded["DealtDamage"]=creature==null?0:battle.Snapshot.StatisticsManager.TotalDamageMeter(creature.Uid);
            creatures.Add(encoded);
        }
        var contributions=new Dictionary<string,object>();
        foreach(long account in run.players)contributions[account.ToString()]=battle.Snapshot.StatisticsManager.GetTotalDamageOfAccount(account);
        Send("WorkerResult",new Dictionary<string,object>{{"RunId",run.id},{"Win",run.updater.GetWinner()==TeamType.Left},
            {"ReplayLogs",run.arena?ReplayUtil.CreateBattleLogString(battle,value=>JsonMarshaler.EncodeJsonString(value),true):""},
            {"Winner",run.updater.GetWinner().ToString()},
            {"Creatures",creatures},{"DefeatedCreatures",run.defeated},{"AccountDamage",contributions},{"TimeMs",battle.MainTimer.GetCurrentTimeMs()},{"MaxWaveIndex",run.updater.MaxWaveIndex}});
        runs.Remove(run.id);
    }
    void OnApplicationQuit() {if(link!=null)link.Close();}
}
