using System;
using System.Linq;
using System.Reflection;
using NGame2.NAccount;
using NGame2.NUI.NComponent2;
using NGame2.NUI.NComponent2.NHeroPortraitSub;
using NGame2.NUI.NWindow;
using NGame2.NUtil.NWrappedInfo;
using NShared;
using NVespa;
using NVespa.NSingleton;
using UnityEngine;

public static class SprkDispatchSelectionProbe {
    static UserHeroListPanel roster;
    static HeroPortraitComponent busy,free;
    static UserDispatchBattleInfoManager manager;
    const BindingFlags Flags=BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic;
    static void Check(bool value,string reason) {if(!value)throw new Exception("Dispatch selection: "+reason);}
    static WCL_DispatchBattleInfo Job(DispatchState state) {
        return new WCL_DispatchBattleInfo().Copy(new DispatchBattleInfo {
            SlotIndex=1,ChapterIndex=1,DungeonIndex=1,HeroIndices="[1]",State=state,
            BeginTime=DateTime.UtcNow.ToString("yyyy-MM-dd HH:mm:ss"),CompleteTime=DateTime.UtcNow.AddHours(1).ToString("yyyy-MM-dd HH:mm:ss") });
    }
    public static string Run(string action) {
        if(action=="init") {
            Assembly.Load("SprkSkillDescriptionFormula").GetType("SprkSkillDescriptionFormulaProbe").GetMethod("Init").Invoke(null,null);
            var account=GlobalAutoBehaviour<AccountManager>.instance;
            var managers=account.GetType().GetProperty("AccountManagers",Flags).GetValue(account,null);
            manager=new UserDispatchBattleInfoManager();managers.GetType().GetField("DispatchBattleInfoManager").SetValue(managers,manager);
            var panel=NGUITools.CreateUI(false,30);
            var host=new GameObject("Native dispatch selection fixture");host.transform.SetParent(panel.transform,false);
            var window=host.AddComponent<BattlePartySetting>();window.enabled=false;
            roster=new UserHeroListPanel();window.UserHeroListPanel=roster;
            var prefab=VespaResource.Load<GameObject>("Resources/"+HeroPortraitComponent.PrefabPath);
            Check(prefab!=null,"native portrait prefab");
            HeroPortraitComponent Card(int id,float x) {
                var go=UnityEngine.Object.Instantiate(prefab,host.transform,false);go.name="Hero "+id;
                go.transform.localPosition=new Vector3(x,0,0);NGUITools.SetLayer(go,30);
                var p=go.GetComponent<HeroPortraitComponent>();
                p.Open(new CachedHeroInfo(new HeroInfo{HeroIndex=id,Level=100,Star=5,Transcended=5}),null,null,HeroPortraitComponent.EPortraitMode.Basic);
                var dict=roster.GetType().GetField("_dicUserHeroPortarit",Flags).GetValue(roster);
                var key=new UserHeroListPanel.HeroInfoKey(0,id);dict.GetType().GetMethod("Add").Invoke(dict,new object[]{key,p});
                return p;
            }
            busy=Card(1,-200);free=Card(2,200);
            manager.Set(new[]{Job(DispatchState.Battle)});
            return "Initialized native portraits and dispatch manager; let textures finish loading.";
        }
        Check(roster!=null,"initialize fixture first");
        if(action=="clear")manager.Set(new WCL_DispatchBattleInfo[0]);
        else if(action=="restore")manager.Set(new[]{Job(DispatchState.Battle)});
        else if(action=="verify") {
            Check(busy.IsDispatched && !free.IsDispatched,"only the dispatched Hero is marked");
            Check(!SprkDispatchSelection.CanClick(busy) && SprkDispatchSelection.CanClick(free),"native roster clicks locked");
            busy.Owner=new HeroPortraitComponent.OwnerInfo{AccountId=99999};roster.SetDispatch(true);
            Check(!busy.IsDispatched,"another player's borrowed Hero is not locked by my dispatch");
            busy.Owner=null;roster.SetDispatch(true);Check(busy.IsDispatched,"owned Hero reservation restored");
            Check(SprkDispatchSelection.AvailableHeroes(new[]{1,2}).SequenceEqual(new[]{2}),"Honor collection excludes dispatched Heroes");
            Check(SprkDispatchSelection.CanUseHeroes(new[]{2}),"available Heroes can start");
            roster.SetDispatch(false);
            Check(busy.IsDispatched,"non-dispatchable stages cannot turn the lock off");
            manager.Remove(1);Check(!busy.IsDispatched,"cancel/removal updates an open roster");
            manager.Add(Job(DispatchState.Prepare));Check(busy.IsDispatched,"new dispatch updates an open roster");
            manager.Set(new[]{Job(DispatchState.ReadyToComplete)});Check(busy.IsDispatched,"unclaimed dispatch remains reserved");
            var status=busy.GetSubComponent<CustomLabelSubComponent>();
            Check(status!=null && status.gameObject.activeInHierarchy,"native dispatched label visible");
            Check(status.SpriteDiabled!=null && status.SpriteDiabled.width>0 && status.SpriteDiabled.height>0,"native dimmer anchored to card");
            Check(status.GetComponentsInChildren<UILabel>(true).Any(l=>l.text.IndexOf("Dispatch",StringComparison.OrdinalIgnoreCase)>=0),"native localized Dispatched text");
            return "PASS: native dispatched label/dimmer, blocked/available clicks, stage-independent locking, Honor filtering, live add/remove and ready-to-collect reservation.";
        } else if(action=="guard") {
            Check(!SprkDispatchSelection.CanUseHeroes(new[]{1,2}),"saved dispatched party refused before submit");
            // Uninitialized managers have no networking/state machine. The real
            // entry points must reject the dispatched deck before touching it.
            var duel=(NGame2.NMatch.DuelManager)System.Runtime.Serialization.FormatterServices.GetUninitializedObject(typeof(NGame2.NMatch.DuelManager));
            duel.CompletePartySetting(1,new[]{1},1);
            var matchType=typeof(NGame2.NMatch.MatchManager);
            var match=System.Runtime.Serialization.FormatterServices.GetUninitializedObject(matchType);
            var wrapper=matchType.GetNestedType("RequestWrapper_Pvp",BindingFlags.Public|BindingFlags.NonPublic);
            var request=Activator.CreateInstance(wrapper,Flags,null,new object[]{MatchPvpType.Normal,new[]{1},1,0L,0},null);
            var register=matchType.GetMethod("RequestMatchInternal_Pvp",Flags);
            register.Invoke(match,new object[]{request,null});
            register.Invoke(match,new object[]{null,null});
            Check(SprkDispatchSelection.CanRegister(new NShared.RegisterMatch.Request{DuelType=MatchDuelType.Friendly,HeroIndices=new[]{1}}),"late friendly rejection must reach server to release the partner");
            return "PASS: saved dispatched parties refused with an explanatory popup; actual duel-ready and registration entry points return before networking; native null request remains safe.";
        }
        return "busy="+busy.IsDispatched+", free="+free.IsDispatched;
    }
}
