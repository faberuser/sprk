// Special Shop search and bookmarks. Two client-only shop groups are added to
// the native category table so the sidebar, selection and product list stay
// native: "Search" as the first tab (Hot & New is hidden) and "Bookmarks" as
// Package's first child, which the lobby shop button opens.
public static class SprkShopSearch {
    public const string SearchKey="SprkSearch",BookmarkKey="SprkBookmark",ParentKey="Package";
    public const string SearchName="Search",BookmarkName="Bookmarks";
    const string StorageKey="SprkShopBookmarks";
    const BindingFlags Any=BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic;

    static readonly Dictionary<string,string> queries=new Dictionary<string,string>{{SearchKey,""},{BookmarkKey,""}};
    static readonly Dictionary<int,string[]> textCache=new Dictionary<int,string[]>();
    static Dictionary<int,int> pendingRank;
    static NewPayShopGroupData searchGroup,bookmarkGroup;
    static HashSet<int> bookmarks;
    static List<int> bookmarkOrder;
    static long bookmarkAccount=long.MinValue;
    static LocalAccountString storage;
    public static int LastCount=-1;
    public static string LastCategory="";

    public static bool IsVirtual(string category) { return category==SearchKey || category==BookmarkKey; }
    public static string Query(string category) { string q;return category!=null && queries.TryGetValue(category,out q)?q:""; }
    public static void SetQuery(string category,string value) { if(IsVirtual(category))queries[category]=value??""; }

    // ---- Matching (same folding rules as the selector-ticket search) --------
    public static string Fold(string value) {
        if(String.IsNullOrEmpty(value))return "";
        value=NGUIText.StripSymbols(value).Normalize(NormalizationForm.FormD);
        var sb=new StringBuilder(value.Length);
        foreach(char ch in value) {
            var category=CharUnicodeInfo.GetUnicodeCategory(ch);
            if(category==UnicodeCategory.NonSpacingMark || category==UnicodeCategory.SpacingCombiningMark || category==UnicodeCategory.EnclosingMark)continue;
            if(Char.IsLetterOrDigit(ch))sb.Append(Char.ToLowerInvariant(ch));
        }
        return sb.ToString();
    }
    public static string[] Terms(string query) {
        if(String.IsNullOrEmpty(query))return new string[0];
        return query.Split((char[])null,StringSplitOptions.RemoveEmptyEntries).Select(Fold).Where(t=>t.Length>0).Distinct().ToArray();
    }
    public static int Tolerance(string term) { return term.Length<4?0:term.Length<=5?1:term.Length<=8?2:3; }
    // Smallest edit distance between the term and any substring of the text.
    public static int Distance(string term,string text) {
        int m=term.Length,n=text.Length;
        if(m==0)return 0;
        var d=new int[m+1,n+1];
        for(int i=0;i<=m;i++)d[i,0]=i;
        for(int i=1;i<=m;i++)for(int j=1;j<=n;j++) {
            int cost=term[i-1]==text[j-1]?0:1;
            int v=Math.Min(Math.Min(d[i-1,j]+1,d[i,j-1]+1),d[i-1,j-1]+cost);
            if(i>1 && j>1 && term[i-1]==text[j-2] && term[i-2]==text[j-1])v=Math.Min(v,d[i-2,j-2]+1);
            d[i,j]=v;
        }
        int best=m;
        for(int j=0;j<=n;j++)best=Math.Min(best,d[m,j]);
        return best;
    }
    // tiers: [0] product names, [1] names of the items/contents, [2] descriptions.
    // Rank 0-2 is the lowest tier set needed for every word; -1 means no match.
    public static int Rank(string[] terms,string[] tiers) {
        if(terms.Length==0)return 0;
        for(int tier=0;tier<tiers.Length;tier++) {
            bool all=true;
            foreach(var term in terms) {
                bool found=false;
                for(int t=0;t<=tier && !found;t++)found=tiers[t].IndexOf(term,StringComparison.Ordinal)>=0;
                if(!found){all=false;break;}
            }
            if(all)return tier;
        }
        return -1;
    }
    // Typo-tolerant fallback over names and contents only.
    public static bool Fuzzy(string[] terms,string[] tiers) {
        if(terms.Length==0)return true;
        var parts=tiers.Take(2).SelectMany(t=>t.Split(new[]{'|'},StringSplitOptions.RemoveEmptyEntries)).ToArray();
        foreach(var term in terms) {
            if(parts.Any(p=>p.IndexOf(term,StringComparison.Ordinal)>=0))continue;
            int k=Tolerance(term);
            if(k==0 || !parts.Any(p=>Distance(term,p)<=k))return false;
        }
        return true;
    }
    public static string Join(IEnumerable<string> values) {
        return "|"+String.Join("|",values.Select(Fold).Where(v=>v.Length>0).Distinct().ToArray())+"|";
    }

    static string Loc(string key) {
        if(String.IsNullOrEmpty(key))return "";
        try { return LocalizationManager.GetStringDirect(key)??key; } catch { return key; }
    }
    static void AddItem(ItemData item,List<string> names,List<string> descs,bool expand) {
        if(item==null)return;
        try { names.Add(item.Name); } catch {}
        try { names.Add(NGame2.NUtil.NStringProxy.ItemNameBuilder.GetItemName(item,null,true)); } catch {}
        try { descs.Add(Loc(item.Description)); } catch {}
        try { descs.Add(Loc(item.Description2)); } catch {}
        if(!expand)return;
        try {
            if(!PackageItemHelper.IsPackageItem(item))return;
            var fixedList=new List<RewardElementData>();var added=new List<RewardElementData>();
            PackageItemHelper.GetRewardElementList(item,fixedList,added);
            foreach(var element in fixedList.Concat(added)) {
                if(element==null)continue;
                var currency=element.GetCurrencyType();
                if(currency!=CurrencyType.None){names.Add(currency==CurrencyType.Gem?"Ruby":currency.ToString());continue;}
                AddItem(GameRoot.Instance.ItemDataContainer.GetDataByCode(element.ItemCode),names,descs,false);
            }
        } catch(Exception e) { Debug.LogWarning("[SprkShopSearch] package contents: "+e.Message); }
    }
    public static string[] Text(NewPayShopProductInfo info) {
        string[] text;
        if(textCache.TryGetValue(info.Index,out text))return text;
        var names=new List<string>{Loc(info.CardTitle),Loc(info.ProductName),info.ProductName};
        var items=new List<string>();var descs=new List<string>{Loc(info.ProductDesc),Loc(info.ProductDetailDesc)};
        var container=GameRoot.Instance.ItemDataContainer;
        var seen=new HashSet<int>();
        void Add(ItemData item) { if(item!=null && seen.Add(item.Index))AddItem(item,items,descs,true); }
        if(info.HeroItemIndex>0)Add(container.GetData(info.HeroItemIndex));
        if(info.UseItemCode!=null)foreach(var code in info.UseItemCode)if(!String.IsNullOrEmpty(code))Add(container.GetDataByCode(code));
        if(info.ItemInfos!=null)foreach(var item in info.ItemInfos)if(item!=null)Add(container.GetData(item.ItemIndex));
        if(info.Gem>0 || info.FirstGem>0 || info.BonusGem>0 || info.DailyGem>0)items.Add("Ruby");
        if(info.Gold>0)items.Add("Gold");
        if(info.Mileage>0)items.Add("Mileage");
        if(info.LuaPoint>0)items.Add("Lua");
        text=new[]{Join(names),Join(items),Join(descs)};
        textCache[info.Index]=text;
        return text;
    }

    // ---- Bookmarks (per account, this device) ---------------------------------
    static void LoadBookmarks() {
        long account=LocalAccountDefine.AccountId;
        if(bookmarks!=null && bookmarkAccount==account)return;
        bookmarkAccount=account;bookmarks=new HashSet<int>();bookmarkOrder=new List<int>();
        storage=new LocalAccountString(StorageKey);
        string raw="";
        try { raw=storage.Value??""; } catch(Exception e) { Debug.LogException(e); }
        foreach(var part in raw.Split(',')) {
            int index;
            if(int.TryParse(part,NumberStyles.Integer,CultureInfo.InvariantCulture,out index) && index!=0 && bookmarks.Add(index))bookmarkOrder.Add(index);
        }
    }
    public static string Serialize(IEnumerable<int> order) { return String.Join(",",order.Select(i=>i.ToString(CultureInfo.InvariantCulture)).ToArray()); }
    public static bool IsBookmarked(int index) { LoadBookmarks();return bookmarks.Contains(index); }
    public static bool Toggle(int index) {
        LoadBookmarks();
        bool on=!bookmarks.Contains(index);
        if(on){bookmarks.Add(index);bookmarkOrder.Add(index);} else {bookmarks.Remove(index);bookmarkOrder.Remove(index);}
        try { storage.Value=Serialize(bookmarkOrder); } catch(Exception e) { Debug.LogException(e); }
        return on;
    }
    public static int BookmarkCount { get { LoadBookmarks();return bookmarks.Count; } }

    // ---- Category table entries -------------------------------------------------
    static NewPayShopGroupData Group(string key,string name,string parent,int sort) {
        return new NewPayShopGroupData{PayShopCategoryType=key,CategoryName=name,ParentCategory=parent,IsCustomize=false,Sort=sort,IsEnable=true,
            LinkCurrency=CurrencyType.None,EnableCondition=EGroupEnableConditionType.None,EPayShopCategoryType=NewPayShopCategoryType.None,
            EParentCategory=parent==null?NewPayShopCategoryType.None:NewPayShopGroupDataContainer.StringToCategoryType(parent)};
    }
    public static void Inject(NewPayShopGroupDataContainer container) {
        var all=container.GetAllData();
        if(all==null)return;
        var native=all.Where(g=>g!=null && !IsVirtual(g.PayShopCategoryType)).ToList();
        // Hot & New is hidden; Search becomes the first top-level tab.
        string hotKey=NewPayShopCategoryType.HotNew.ToString();
        foreach(var g in native.Where(g=>g.PayShopCategoryType==hotKey))g.IsEnable=false;
        var hotCached=container.GetData(hotKey);
        if(hotCached!=null)hotCached.IsEnable=false;
        var roots=native.Where(g=>String.IsNullOrEmpty(g.ParentCategory) && g.PayShopCategoryType!=hotKey).ToList();
        int searchSort=roots.Count==0?0:roots.Min(g=>g.Sort)-1;
        var children=native.Where(g=>g.ParentCategory==ParentKey).ToList();
        int bookmarkSort=children.Count==0?0:children.Min(g=>g.Sort)-1;
        searchGroup=searchGroup??Group(SearchKey,SearchName,null,searchSort);searchGroup.Sort=searchSort;
        bookmarkGroup=bookmarkGroup??Group(BookmarkKey,BookmarkName,ParentKey,bookmarkSort);bookmarkGroup.Sort=bookmarkSort;
        bool hasParent=native.Any(g=>g.PayShopCategoryType==ParentKey && String.IsNullOrEmpty(g.ParentCategory));
        bookmarkGroup.ParentCategory=hasParent?ParentKey:null;
        if(!hasParent)bookmarkGroup.Sort=searchSort;
        var list=new List<NewPayShopGroupData>(native){searchGroup,bookmarkGroup};
        var baseList=(JitRef<List<NewPayShopGroupData>>)typeof(JitDataContainer<NewPayShopGroupData>).GetProperty("BaseList",Any).GetValue(container,null);
        baseList.JsonLoad_Set(list);
        var dic=(JitDictionaryWithKey<string,NewPayShopGroupData>)typeof(JitDataContainerDefault<string,NewPayShopGroupData>).GetProperty("BaseDic",Any).GetValue(container,null);
        var cache=dic.JsonLoad_Get();
        lock(cache){cache[SearchKey]=searchGroup;cache[BookmarkKey]=bookmarkGroup;}
    }

    // ---- IL hooks -----------------------------------------------------------------
    // Start of PayShop.Open: entries must exist before the sidebar is built.
    public static void BeforeOpen(ShopWindow shop) {
        try {
            textCache.Clear();
            Inject(GameRoot.Instance.NewPayShopGroupDataContainer);
            // Enabled while the native sidebar creates its buttons.
            if(bookmarkGroup!=null)bookmarkGroup.IsEnable=true;
        } catch(Exception e) { Debug.LogException(e); }
    }
    // End of PayShop.Open.
    public static void AfterOpen(ShopWindow shop) {
        try {
            // Disabled afterwards so clicking Package still opens its usual
            // first child and flag refreshes skip it. Selection ignores IsEnable.
            if(bookmarkGroup!=null)bookmarkGroup.IsEnable=false;
            var bar=shop.GetComponent<SprkShopSearchBar>();
            if(bar==null)bar=shop.gameObject.AddComponent<SprkShopSearchBar>();
            bar.Bind(shop);
        } catch(Exception e) { Debug.LogException(e); }
    }
    // End of PayShopCategoryComponent.Open: name and icon for the added tabs.
    public static void DecorateCategory(PayShopCategoryComponent component,NewPayShopGroupData data) {
        try {
            SprkShopIcons.ApplyCategoryIcon(component,data!=null && data.PayShopCategoryType==SearchKey);
            if(data==null || !IsVirtual(data.PayShopCategoryType))return;
            component.SetName(data.PayShopCategoryType==SearchKey?SearchName:BookmarkName);
        } catch(Exception e) { Debug.LogException(e); }
    }
    // End of PayShopProductList.GetRealDataList(category).
    public static List<NewPayShopProductInfo> Products(List<NewPayShopProductInfo> native,PayShopProductList list,string category) {
        LastCategory=category??"";
        pendingRank=null;LastCount=-1;
        if(!IsVirtual(category))return native;
        try {
            var manager=GlobalAutoBehaviour<AccountManager>.instance.NewPayShopProductManager;
            IEnumerable<NewPayShopProductInfo> pool;
            if(category==BookmarkKey) {
                LoadBookmarks();
                pool=bookmarkOrder.Select(i=>manager.Get(i)).Where(p=>p!=null);
            } else {
                var groups=GameRoot.Instance.NewPayShopGroupDataContainer;
                pool=manager.GetValues().Where(p=>p!=null && Visible(groups,p.Category));
            }
            var terms=Terms(Query(category));
            if(category==SearchKey && terms.Length==0){LastCount=0;return new List<NewPayShopProductInfo>();}
            var ranks=new Dictionary<int,int>();
            var result=new List<NewPayShopProductInfo>();
            var candidates=pool.ToList();
            foreach(var product in candidates) {
                int rank=Rank(terms,Text(product));
                if(rank<0)continue;
                ranks[product.Index]=rank;result.Add(product);
            }
            if(result.Count==0 && terms.Length>0)
                foreach(var product in candidates)
                    if(Fuzzy(terms,Text(product))){ranks[product.Index]=3;result.Add(product);}
            if(category==SearchKey && terms.Length>0)pendingRank=ranks;
            LastCount=0;
            return result;
        } catch(Exception e) { Debug.LogException(e);return native; }
    }
    static bool Visible(NewPayShopGroupDataContainer groups,string category) {
        if(String.IsNullOrEmpty(category) || IsVirtual(category))return false;
        var group=groups.GetData(category);
        if(group==null)return false;
        // Hot & New is hidden from the sidebar, but its offers stay searchable.
        if(!group.IsEnable && category!=NewPayShopCategoryType.HotNew.ToString())return false;
        if(!String.IsNullOrEmpty(group.ParentCategory)) {
            var parent=groups.GetData(group.ParentCategory);
            if(parent==null || !parent.IsEnable)return false;
        }
        return true;
    }
    // End of PayShopProductList.Sort: relevance tiers, native order within a tier.
    public static void AfterSort(PayShopProductList list,List<NewPayShopProductInfo> data) {
        var ranks=pendingRank;pendingRank=null;
        if(data==null)return;
        if(IsVirtual(LastCategory))LastCount=data.Count;
        if(ranks==null || data.Any(p=>!ranks.ContainsKey(p.Index)))return;
        var sorted=data.OrderBy(p=>ranks[p.Index]).ToList();
        data.Clear();data.AddRange(sorted);
    }
    // End of PayShopProductComponent.Init: bookmark toggle on every card.
    public static void Decorate(PayShopProductComponent card,NewPayShopProductInfo info) {
        try {
            var star=card.GetComponent<SprkShopStar>();
            if(star==null)star=card.gameObject.AddComponent<SprkShopStar>();
            star.Bind(card,card.RootWidget,info,42,34f);
        } catch(Exception e) { Debug.LogException(e); }
    }
    // End of PayShopProductInfoPopup.Init: the same toggle on the details popup.
    public static void DecoratePopup(NGame2.NUI.NWindow.NPayShop.PayShopProductInfoPopup popup,NewPayShopProductInfo info) {
        try {
            var star=popup.GetComponent<SprkShopStar>();
            if(star==null)star=popup.gameObject.AddComponent<SprkShopStar>();
            UIWidget anchor=popup.ProductTexture!=null?popup.ProductTexture.GetComponent<UIWidget>():null;
            if(anchor==null && popup.ProductTexture!=null)anchor=popup.ProductTexture.GetComponentInChildren<UIWidget>(true);
            star.Bind(popup,anchor,info,56,30f);
        } catch(Exception e) { Debug.LogException(e); }
    }
    // PayShopManagement.OpenNewPayShop() (the lobby button) opens Package > Bookmarks.
    public static string DefaultCategory(string native) {
        return BookmarkKey;
    }
}

public static class SprkShopIcons {
    static UIAtlas starAtlas;static string starSprite;
    static Texture2D magnifier;
    static readonly string[] StarNames={"Common_Star","Common_Star_2"};
    public static bool Find(string[] names,out UIAtlas atlas,out string sprite) {
        atlas=null;sprite=null;
        foreach(var name in names)foreach(var candidate in Resources.FindObjectsOfTypeAll<UIAtlas>())
            if(candidate!=null && candidate.GetSprite(name)!=null){atlas=candidate;sprite=name;return true;}
        return false;
    }
    public static bool Star(out UIAtlas atlas,out string sprite) {
        if(starAtlas==null)Find(StarNames,out starAtlas,out starSprite);
        atlas=starAtlas;sprite=starSprite;return atlas!=null;
    }
    // No magnifier exists in the shop atlases; draw an anti-aliased one in white
    // and tint it like the native menu icon.
    public static Texture2D Magnifier() {
        if(magnifier!=null)return magnifier;
        const int size=96;
        var pixels=new Color32[size*size];
        float cx=40f,cy=56f,outer=27f,inner=19f;
        for(int y=0;y<size;y++)for(int x=0;x<size;x++) {
            float px=x+.5f,py=y+.5f;
            float r=Mathf.Sqrt((px-cx)*(px-cx)+(py-cy)*(py-cy));
            float ring=Mathf.Clamp01(Mathf.Min(r-inner+.75f,outer-r+.75f));
            // Handle: thick segment towards the lower right.
            Vector2 a=new Vector2(cx+outer*.68f,cy-outer*.68f),b=new Vector2(84f,12f),p=new Vector2(px,py);
            float t=Mathf.Clamp01(Vector2.Dot(p-a,b-a)/(b-a).sqrMagnitude);
            float handle=Mathf.Clamp01(7.5f-Vector2.Distance(p,a+(b-a)*t)+.75f);
            byte alpha=(byte)(Mathf.Max(ring,handle)*255f);
            pixels[y*size+x]=new Color32(255,255,255,alpha);
        }
        magnifier=new Texture2D(size,size,TextureFormat.RGBA32,false);
        magnifier.wrapMode=TextureWrapMode.Clamp;magnifier.SetPixels32(pixels);magnifier.Apply(false,true);
        magnifier.name="SprkShopMagnifier";
        return magnifier;
    }
    public static void ApplyCategoryIcon(PayShopCategoryComponent component,bool search) {
        if(component.CategoryImageSelector==null)return;
        var target=typeof(NNguiExtension.UISpriteSelector).GetField("_targetSprite",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(component.CategoryImageSelector) as UISprite;
        if(target==null)target=component.CategoryImageSelector.GetComponent<UISprite>();
        if(target==null)target=component.CategoryImageSelector.GetComponentInChildren<UISprite>(true);
        if(target==null){Debug.LogWarning("[SprkShopSearch] category icon sprite not found");return;}
        var existing=target.transform.parent.Find("SprkSearchIcon");
        if(!search) {
            // Pooled sidebar buttons are reused for native categories.
            if(existing!=null)existing.gameObject.SetActive(false);
            target.enabled=true;
            return;
        }
        UITexture icon=existing!=null?existing.GetComponent<UITexture>():null;
        if(icon==null) {
            icon=NGUITools.AddWidget<UITexture>(target.transform.parent.gameObject);
            icon.name="SprkSearchIcon";
            icon.shader=Shader.Find("Unlit/Transparent Colored");
        }
        icon.gameObject.SetActive(true);
        icon.mainTexture=Magnifier();
        int side=Math.Min(target.width,target.height);
        icon.width=(int)(side*.82f);icon.height=icon.width;icon.depth=target.depth;
        icon.color=target.color;
        icon.transform.localPosition=target.transform.localPosition;icon.transform.localScale=target.transform.localScale;
        target.enabled=false;
    }
}

public sealed class SprkShopStar : MonoBehaviour {
    NewPayShopProductInfo info;
    GameObject root;UISprite sprite;UILabel glyph;bool shown;
    static readonly Color On=new Color(1f,.84f,.25f,1f),Off=new Color(.42f,.40f,.38f,.45f);
    public void Bind(Component owner,UIWidget anchor,NewPayShopProductInfo product,int size,float inset) {
        info=product;
        if(root==null) {
            if(anchor==null)anchor=owner.GetComponentInChildren<UIWidget>(true);
            Build(owner,anchor,size,inset);
        }
        Refresh();
    }
    void Build(Component owner,UIWidget anchor,int size,float inset) {
        int depth=0;
        var panel=anchor.panel??NGUITools.FindInParents<UIPanel>(anchor.gameObject);
        foreach(var w in owner.GetComponentsInChildren<UIWidget>(true))
            if(w.panel==null || w.panel==panel)depth=Math.Max(depth,w.depth);
        root=new GameObject("SprkBookmark");root.layer=anchor.gameObject.layer;
        root.transform.SetParent(anchor.transform,false);
        var corners=anchor.localCorners;
        root.transform.localPosition=new Vector3(corners[2].x-inset,corners[2].y-inset,0f);
        int hitSize=size+22;
        var hit=root.AddComponent<UIWidget>();hit.width=hitSize;hit.height=hitSize;hit.depth=depth+10;
        UIAtlas atlas;string name;
        if(SprkShopIcons.Star(out atlas,out name)) {
            sprite=NGUITools.AddWidget<UISprite>(root);sprite.atlas=atlas;sprite.spriteName=name;
            sprite.type=UIBasicSprite.Type.Simple;sprite.width=size;sprite.height=size;sprite.depth=depth+11;
        } else {
            glyph=NGUITools.AddWidget<UILabel>(root);
            var font=owner.GetComponentsInChildren<UILabel>(true).FirstOrDefault(l=>l.bitmapFont!=null || l.trueTypeFont!=null);
            if(font!=null){glyph.bitmapFont=font.bitmapFont;glyph.trueTypeFont=font.trueTypeFont;}
            glyph.fontSize=size;glyph.width=size+16;glyph.height=size+16;glyph.depth=depth+11;
            glyph.overflowMethod=UILabel.Overflow.ShrinkContent;glyph.alignment=NGUIText.Alignment.Center;
            glyph.effectStyle=UILabel.Effect.Outline;glyph.effectColor=new Color(.2f,.13f,.05f,1f);
        }
        var collider=root.AddComponent<BoxCollider>();collider.size=new Vector3(hitSize,hitSize,1f);
        var button=root.AddComponent<UIButton>();button.tweenTarget=null;
        EventDelegate.Add(button.onClick,OnClick);
        var scroll=NGUITools.FindInParents<UIScrollView>(anchor.gameObject);
        if(scroll!=null)root.AddComponent<UIDragScrollView>().scrollView=scroll;
    }
    void OnClick() {
        if(info==null)return;
        SprkShopSearch.Toggle(info.Index);
        Refresh();
    }
    void Refresh() {
        bool on=info!=null && SprkShopSearch.IsBookmarked(info.Index);
        shown=on;
        root.SetActive(info!=null);
        if(sprite!=null)sprite.color=on?On:Off;
        if(glyph!=null){glyph.text=on?"★":"☆";glyph.color=on?On:Off;}
    }
    // Keeps cards in sync after toggling the same product in its popup.
    void Update() {
        if(root!=null && info!=null && SprkShopSearch.IsBookmarked(info.Index)!=shown)Refresh();
    }
}

// The tag bar becomes a search field while Search or Bookmarks is selected.
public sealed class SprkShopSearchBar : MonoBehaviour {
    static readonly FieldInfo CurrentField=typeof(ShopWindow).GetField("_currentCategory",BindingFlags.Instance|BindingFlags.NonPublic);
    ShopWindow shop;
    Transform tagRoot;
    GameObject bar;UISprite field;UIInput input;UILabel inputLabel,title,count,hint;GameObject clear;
    string shown;bool silent;float pending=-1f;bool styled;

    public void Bind(ShopWindow window) { shop=window;shown=null; }
    string Current { get { return CurrentField.GetValue(shop) as string ?? ""; } }

    void Build() {
        var tagList=shop.PayShopTagList;
        tagRoot=tagList.transform.Find("Root");
        if(tagRoot==null)return;
        var frame=tagRoot.Find("Sprite_TagFrame")?.GetComponent<UISprite>();
        var tagLabel=tagRoot.GetComponentsInChildren<UILabel>(true).FirstOrDefault(l=>l.name.Contains("TagLabel"));
        if(frame==null || tagLabel==null)return;
        bar=new GameObject("SprkShopSearchBar");bar.layer=tagRoot.gameObject.layer;
        bar.transform.SetParent(tagRoot.parent,false);
        bar.transform.localPosition=tagRoot.localPosition;bar.transform.localScale=tagRoot.localScale;
        var back=NGUITools.AddWidget<UISprite>(bar);
        back.name="Frame";back.atlas=frame.atlas;back.spriteName=frame.spriteName;back.type=frame.type;back.pivot=frame.pivot;
        back.width=frame.width;back.height=frame.height;back.depth=frame.depth;back.color=frame.color;

    void OnChanged() {
        if(silent || input==null)return;
        var category=Current;
        if(!SprkShopSearch.IsVirtual(category))return;
        string value=input.value??"";
        if(value==SprkShopSearch.Query(category))return;
        SprkShopSearch.SetQuery(category,value);
        clear.SetActive(value.Length>0);
        pending=Time.unscaledTime+.25f;
    }
    void OnSubmitted() { OnChanged();if(pending>=0f)Apply(); }
    void Clear() {
        var category=Current;
        SprkShopSearch.SetQuery(category,"");
        SetSilently("");clear.SetActive(false);
        Apply();
    }
    void SetSilently(string value) { silent=true;try { input.value=value; } finally { silent=false; } }
    void Apply() {
        pending=-1f;
        var category=Current;
        if(!SprkShopSearch.IsVirtual(category))return;
        try { shop.PayShopProductList.CreateList(category); } catch(Exception e) { Debug.LogException(e); }
    }
    void Update() {
        if(shop==null)return;
        if(bar==null){Build();if(bar==null)return;}
        var category=Current;
        bool on=SprkShopSearch.IsVirtual(category);
        if(on && tagRoot.gameObject.activeSelf)tagRoot.gameObject.SetActive(false);
        if(category!=shown) {
            if(!on && SprkShopSearch.IsVirtual(shown))tagRoot.gameObject.SetActive(true);
            bar.SetActive(on);
            if(on) {
                bool search=category==SprkShopSearch.SearchKey;
                title.text=search?"SEARCH":"SAVED";
                input.defaultText=search?"Search packages, items and descriptions":"Filter bookmarked packages";
                string q=SprkShopSearch.Query(category);
                SetSilently(q);clear.SetActive(q.Length>0);
                if(input.isSelected)input.isSelected=false;
            }
            pending=-1f;shown=category;
        }
        if(!on){if(hint.gameObject.activeSelf)hint.gameObject.SetActive(false);return;}
        MatchStyle();
        if(pending>=0f && Time.unscaledTime>=pending)Apply();
        bool searchView=category==SprkShopSearch.SearchKey;
        bool hasQuery=SprkShopSearch.Query(category).Trim().Length>0;
        int n=SprkShopSearch.LastCategory==category?SprkShopSearch.LastCount:-1;
        count.text=hasQuery && n>=0?(n==1?"1 result":n+" results"):searchView?"":(SprkShopSearch.BookmarkCount+" saved");
        string message=null;
        if(searchView && !hasQuery)message="Type a package, item or description to search every shop category.";
        else if(n<=0 && hasQuery)message="No packages match \""+SprkShopSearch.Query(category).Trim()+"\".";
        else if(!searchView && n<=0)message="No bookmarks yet. Tap the star on any package to save it here.";
        hint.text=message??"";
        if(hint.gameObject.activeSelf!=(message!=null))hint.gameObject.SetActive(message!=null);
    }
