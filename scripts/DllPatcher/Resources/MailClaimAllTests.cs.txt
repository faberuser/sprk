public static class MailClaimAllTestRunner {
    static NVespa.NGlobal.Requester Web => NVespa.SingletonUpdater<NVespa.NGlobal.WebServiceManager>.instance.Requester;
    static NGame2.NAccount.AccountManager Account => NVespa.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.instance;
    static void Assert(bool condition,string label){if(!condition)throw new Exception(label);}
    static NShared.ReceiveAllMail.Response Reply(int count,string reason=null){
        var result=new NShared.ReceiveAllMail.Response {ReceivedMailCount=count,InternalErrorMessage=reason,MailReceiveResults=new NShared.MailReceiveResult[count]};
        for(int i=0;i<count;i++)result.MailReceiveResults[i]=new NShared.MailReceiveResult();return result;
    }
    static void Reset(){
        Web.Pending=null;Web.Calls=0;Web.SessionKey="test";Account.Applied=0;
        UnityEngine.Time.realtimeSinceStartup=0;
        NGame2.NUI.NWindow.MessagePopup.Calls=0;
        NGame2.NUI.NWindow.NBattle.BaseRewardPopup.Calls=0;
    }
    static void Deliver(NShared.ReceiveAllMail.Response response){var reply=Web.Pending;Assert(reply!=null,"Expected pending request");Web.Pending=null;reply(response);}
    static void Drain(int mails,int capacity){
        Reset();var owner=new NGame2.NUI.NManager.MailManagement();var popup=new NGame2.NUI.NWindow.MailListPopup();
        SprkMailClaimAll.Caption(popup);Assert(!popup.Notice.enabled&&!popup.Notice.Label.text.Contains("50"),"Updated receipt limit caption");
        SprkMailClaimAll.Start(owner,popup,50);
        int remaining=mails,space=capacity;
        for(int i=0;i<100;i++){
            SprkMailClaimAll.Pump();if(Web.Pending==null)break;
            int count=Math.Min(50,Math.Min(remaining,space));remaining-=count;space-=count;
            Deliver(Reply(count,remaining>0&&space==0?"Equipment bag is full. Free space and try claiming the mail again.":null));
        }
        Assert(Account.Applied==Math.Min(mails,capacity),"Claims exactly the available capacity");
        Assert(NGame2.NUI.NWindow.NBattle.BaseRewardPopup.Calls==(Account.Applied>0?1:0),"One combined reward popup");
        Assert(Web.Calls<=mails/50+1,"No empty or capacity loop");
        Assert(popup.ReceiveAllButton.isEnabled,"Restores claim button");
        SprkMailClaimAll.Pump();Assert(Web.Pending==null,"Stopped at empty inbox/capacity");
    }
    public static void Run(){
        Drain(1505,3000);Drain(1500,317);Drain(100,0);Drain(100,100);Drain(0,100);
        Reset();var owner=new NGame2.NUI.NManager.MailManagement();var popup=new NGame2.NUI.NWindow.MailListPopup();
        SprkMailClaimAll.Start(owner,popup,50);SprkMailClaimAll.Start(owner,popup,50);SprkMailClaimAll.Pump();SprkMailClaimAll.Pump();
        Assert(Web.Calls==1,"Duplicate click and in-flight protection");
        var duplicate=Web.Pending;Deliver(Reply(50));duplicate(Reply(50));Assert(Account.Applied==50,"Ignore duplicate callback");
        SprkMailClaimAll.Cancel(popup);SprkMailClaimAll.Pump();Assert(Web.Calls==1,"Closing cancels continuation");
        Reset();SprkMailClaimAll.Start(owner,popup,50);SprkMailClaimAll.Pump();SprkMailClaimAll.Cancel(popup);Deliver(Reply(50));
        SprkMailClaimAll.Pump();Assert(Account.Applied==50&&Web.Calls==1,"Keeps completed in-flight claim after close");
        Reset();SprkMailClaimAll.Start(owner,popup,50);SprkMailClaimAll.Pump();var late=Web.Pending;
        Web.SessionKey="other-account";Web.Pending=null;SprkMailClaimAll.Pump();late(Reply(50));Assert(Account.Applied==0,"Never apply old session rewards");
        Reset();SprkMailClaimAll.Start(owner,popup,50);SprkMailClaimAll.Pump();Deliver(Reply(50));SprkMailClaimAll.Pump();
        Deliver(new NShared.ReceiveAllMail.Response{Result=NShared.ReceiveAllMail.ResultType.Fail});SprkMailClaimAll.Pump();
        Assert(Account.Applied==50&&Web.Calls==2&&Web.Pending==null,"Stops on server failure, preserves previous rewards");
        Reset();SprkMailClaimAll.Start(owner,popup,50);SprkMailClaimAll.Pump();late=Web.Pending;
        UnityEngine.Time.realtimeSinceStartup=61;SprkMailClaimAll.Pump();late(Reply(50));
        Assert(Account.Applied==0&&NGame2.NUI.NWindow.MessagePopup.Calls==1,"Timeout stops without retrying or applying late response");
        Web.Pending=null;
        var fields=SprkMailClaimAll.Encode(new Dictionary<string,object>());Assert((string)fields["ClaimUntilFull"]=="true","Server partial-claim opt-in encoded");
    }
