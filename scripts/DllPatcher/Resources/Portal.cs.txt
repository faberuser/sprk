    public static bool IsOpenedDragonHard(RaidData raid)
    {
        if (raid == null || (raid.Type != RaidType.DragonHard && raid.Type != RaidType.DragonHardSingle)
            || raid.OpenCondition == null || raid.OpenCondition.Length < 2) return false;
        int index = raid.OpenCondition[0], required = raid.OpenCondition[1];
        if (index <= 0 || required <= 0) return false;
        if (NGame2.NUI.NWindow.RaidHelper.GetRaidMaxLevel(index) >= required) return true;
        // Older solo clears can be saved under the solo ID. Match the server's
        // compatibility check without granting progress or bypassing the prerequisite.
        foreach (var normal in GameRoot.Instance.RaidDataContainer.GetAllData())
            if (normal.Index == index && normal.SingleRaidIndex > 0
                && NGame2.NUI.NWindow.RaidHelper.GetRaidMaxLevel(normal.SingleRaidIndex) >= required)
                return true;
        return false;
    }

    public static string DragonHardRequirement(RaidData raid)
    {
        if (raid == null || (raid.Type != RaidType.DragonHard && raid.Type != RaidType.DragonHardSingle)
            || IsOpenedDragonHard(raid)) return null;
        RaidData prerequisite = null;
        if (raid.OpenCondition != null && raid.OpenCondition.Length >= 2)
            foreach (var normal in GameRoot.Instance.RaidDataContainer.GetAllData())
                // RaidLevel records the highest selectable stage, so reaching
                // the required level means clearing the preceding normal stage.
                if (normal.Index == raid.OpenCondition[0] && normal.Level < raid.OpenCondition[1]
                    && (prerequisite == null || normal.Level > prerequisite.Level)) prerequisite = normal;
        if (prerequisite == null)
            return "This Hard raid is locked. Its normal raid prerequisite is unavailable.";
        string dragon = NVespa.NGlobal.LocalizationManager.GetStringDirect(prerequisite.Name);
        string stage = NVespa.NGlobal.LocalizationManager.GetStringDirect(prerequisite.Step_Name);
        return "Clear " + dragon + " (Normal), " + stage + " once to unlock this Hard raid.";
    }

    public static bool CheckDragonHardEntry(RaidData raid)
    {
        string requirement = DragonHardRequirement(raid);
        if (requirement == null) return true;
        NGame2.NUI.NWindow.MessagePopup.Open(
            NVespa.NGlobal.LocalizationManager.GetString(NShared.NLocalization.Common.INFORM), requirement);
        return false;
    }

    public static bool CheckDragonHardEntryByIndex(int raidIndex, int raidLevel)
    {
        return CheckDragonHardEntry(GameRoot.Instance.RaidDataContainer.GetData(raidIndex, raidLevel));
    }

    public static void FinalizeCombatContainer(object container)
    {
        // Whole-table readers bypass GetData. Correct their rows at load time,
        // and also correct dictionary rows if JIT uses a separate object graph.
        var skills=container as SkillDataContainer;
        if(skills!=null) {
            var rows=skills.GetAllData();if(rows==null)return;
            foreach(var row in rows) {
                Dictionary<string,object> changes;
                if(PunishmentSkillChanges.TryGetValue(row.Index,out changes)) {
                    ApplyPunishmentChanges(row,changes);PunishmentSkill(skills,row.Index);
                }
            }
            // Serialized rage lists can hold their own row instances.
            var rage=skills.GetOnRageSkillList();
            if(rage!=null)foreach(var row in rage) {
                Dictionary<string,object> changes;
                if(PunishmentSkillChanges.TryGetValue(row.Index,out changes))ApplyPunishmentChanges(row,changes);
            }
            return;
        }
        var creatures=container as CreatureDataContainer;
        if(creatures!=null) {
            var rows=creatures.GetAllData();if(rows==null)return;
            foreach(var row in rows) {
                Dictionary<string,object> changes;
                if(CombatCreatureChanges.TryGetValue(row.Index,out changes)) {
                    ApplyPunishmentChanges(row,changes);CombatCreature(creatures,row.Index);
                }
            }
            return;
        }
        var states=container as StateDataContainer;
        if(states!=null) {
            var rows=states.GetAllData();if(rows==null)return;
            var present=new HashSet<string>();
            foreach(var row in rows) {
                present.Add(row.StateType);Dictionary<string,object> changes;
                if(CombatStateChanges.TryGetValue(row.StateType,out changes)) {
                    ApplyPunishmentChanges(row,changes);SynchronizeCombatStateFlags(row);CombatState(states,row.StateType);
                }
            }
            foreach(var pair in CombatStateChanges)
                if(!present.Contains(pair.Key))rows.Add(CombatState(states,pair.Key));
            return;
        }
        var projectiles=container as SkillProjectileDataContainer;
        if(projectiles!=null) {
            var rows=projectiles.GetAllData();if(rows==null)return;
            foreach(var row in rows) {
                var corrected=CombatProjectile(projectiles,row.Index);
                if(object.ReferenceEquals(row,corrected))continue;
                object table;
                if(CombatTables.TryGetValue("SkillProjectile",out table)) {
                    var changes=new Dictionary<string,object>();string prefix=row.Index+":";
                    foreach(var entry in (Dictionary<string,object>)table)
                        if(entry.Key.StartsWith(prefix))changes[entry.Key.Substring(prefix.Length)]=entry.Value;
                    ApplyPunishmentChanges(row,changes);
                }
            }
        }
    }
    public static SkillProjectileData CombatProjectile(NShared.NJit.JitDataContainerDefault<int,SkillProjectileData> container,int index)
    {
        var row=container.GetData(index);object table;
        if(row==null || !CombatTables.TryGetValue("SkillProjectile",out table))return row;
        lock(PunishmentSkillSync) {
            SkillProjectileData cached;
            if(!RestoredProjectileCache.TryGetValue(index,out cached) || !object.ReferenceEquals(cached,row)) {
                var changes=new Dictionary<string,object>();string prefix=index+":";
                foreach(var entry in (Dictionary<string,object>)table)
                    if(entry.Key.StartsWith(prefix))changes[entry.Key.Substring(prefix.Length)]=entry.Value;
                ApplyPunishmentChanges(row,changes);RestoredProjectileCache[index]=row;
            }
        }
        return row;
    }
    static Dictionary<string,Dictionary<string,object>> BuildCombatStateChanges()
    {
        var rows=new Dictionary<string,Dictionary<string,object>>();object table;
        if(CombatTables.TryGetValue("State",out table))
            foreach(var entry in (Dictionary<string,object>)table) {
                int split=entry.Key.LastIndexOf(':');string key=entry.Key.Substring(0,split);
                Dictionary<string,object> changes;
                if(!rows.TryGetValue(key,out changes))rows[key]=changes=new Dictionary<string,object>();
                changes[entry.Key.Substring(split+1)]=entry.Value;
            }
        object missing;
        if(CombatRestoration.TryGetValue("MissingStates",out missing))
            foreach(Dictionary<string,object> row in (System.Collections.IList)missing)
                rows[(string)row["Key"]]=(Dictionary<string,object>)row["Values"];
        return rows;
    }
    public static void SynchronizeCombatStateFlags(StateData row)
    {
        // StateDataContainer's original initializer derives this bit mask from
        // Boolean properties. Restore both forms to keep fast filters coherent.
        StateFlag flags=StateFlag.None;
        foreach(string name in System.Enum.GetNames(typeof(StateFlag))) {
            var property=typeof(StateData).GetProperty(name);
            if(property!=null && property.PropertyType==typeof(bool) && (bool)property.GetValue(row,null))
                flags|=(StateFlag)System.Enum.Parse(typeof(StateFlag),name);
        }
        row.StateFlag=flags;
    }
    public static StateData CombatState(NShared.NJit.JitDataContainerDefault<string,StateData> container,string key)
    {
        var row=container.GetData(key);Dictionary<string,object> changes;
        if(!CombatStateChanges.TryGetValue(key,out changes))return row;
        lock(PunishmentSkillSync) {
            StateData cached;
            if(row==null && RestoredCombatStateCache.TryGetValue(key,out cached))return cached;
            if(row==null)row=new StateData();
            if(!RestoredCombatStateCache.TryGetValue(key,out cached) || !object.ReferenceEquals(cached,row)) {
                ApplyPunishmentChanges(row,changes);SynchronizeCombatStateFlags(row);
                RestoredCombatStateCache[key]=row;
            }
            return row;
        }
    }
    static Dictionary<int,Dictionary<string,object>> BuildCombatCreatureChanges()
    {
        var rows=new Dictionary<int,Dictionary<string,object>>();
        foreach(var entry in (Dictionary<string,object>)CombatTables["Creature"]) {
            int split=entry.Key.IndexOf(':');int index=int.Parse(entry.Key.Substring(0,split));
            Dictionary<string,object> changes;
            if(!rows.TryGetValue(index,out changes))rows[index]=changes=new Dictionary<string,object>();
            changes[entry.Key.Substring(split+1)]=entry.Value;
        }
        return rows;
    }
    static Dictionary<long,Dictionary<string,object>> PunishmentChanges(string category)
    {
        var result = new Dictionary<long,Dictionary<string,object>>();
        var root = CombatRestoration;
        foreach(var value in (System.Collections.IList)root[category]) {
            var row = (Dictionary<string,object>)value;
            long key = category == "Skills" ? System.Convert.ToInt64(row["Index"])
                : System.Convert.ToInt64(row["SkillIndex"]) * 100 + System.Convert.ToInt64(row["OperationNo"]);
            result[key] = (Dictionary<string,object>)row["Changes"];
        }
        return result;
    }
    static object PunishmentPropertyValue(object value, System.Type type)
    {
        if(value == null) return null;
        if(type.IsArray) {
            var list = (System.Collections.IList)value;var element = type.GetElementType();
            var array = System.Array.CreateInstance(element,list.Count);
            for(int i=0;i<list.Count;i++)array.SetValue(PunishmentPropertyValue(list[i],element),i);
            return array;
        }
        if(type.IsEnum)return System.Enum.ToObject(type,System.Convert.ToInt64(value));
        return System.Convert.ChangeType(value,type,System.Globalization.CultureInfo.InvariantCulture);
    }
    static void ApplyPunishmentChanges(object row, Dictionary<string,object> changes)
    {
        foreach(var change in changes) {
            var property = row.GetType().GetProperty(change.Key);
            if(property == null || !property.CanWrite)throw new System.InvalidOperationException("Unknown restored skill field: "+change.Key);
            property.SetValue(row,PunishmentPropertyValue(change.Value,property.PropertyType),null);
        }
    }
    public static SkillData PunishmentSkill(NShared.NJit.JitDataContainerDefault<int,SkillData> container,int index)
    {
        var skill = container.GetData(index);Dictionary<string,object> changes;
        if(skill != null && PunishmentSkillChanges.TryGetValue(index,out changes))
            lock(PunishmentSkillSync) {
                SkillData cached;
                if(!RestoredCombatSkillCache.TryGetValue(index,out cached) || !object.ReferenceEquals(cached,skill)) {
                    ApplyPunishmentChanges(skill,changes);RestoredCombatSkillCache[index]=skill;
                }
            }
        return skill;
    }
    public static void RestorePunishmentOperation(OperationData.LoadData load)
    {
        Dictionary<string,object> changes;
        if(load != null && PunishmentOperationChanges.TryGetValue((long)load.SkillIndex*100+load.OperationNo,out changes))
            ApplyPunishmentChanges(load,changes);
    }
    public static bool IsRemovedPunishmentOperation(int skillIndex,int operationNo)
    {
        Dictionary<string,object> changes;object type;
        return PunishmentOperationChanges.TryGetValue((long)skillIndex*100+operationNo,out changes)
            && changes.TryGetValue("Type",out type) && System.Convert.ToInt32(type)==0;
    }
    public static OperationData RestoreMissingCombatOperation(OperationData actual,int skillIndex,int operationNo)
    {
        if(actual != null)return actual;
        long key=(long)skillIndex*100+operationNo;
        Dictionary<string,object> changes;
        if(!MissingCombatOperations.TryGetValue(key,out changes))return null;
        lock(PunishmentSkillSync) {
            OperationData result;
            if(MissingCombatOperationCache.TryGetValue(key,out result))return result;
            var load=new OperationData.LoadData { SkillIndex=skillIndex,OperationNo=operationNo };
            ApplyPunishmentChanges(load,changes);
            result=new OperationData(load);
            MissingCombatOperationCache[key]=result;
            return result;
        }
    }
    static long CombatNumber(long actual,string table,string key,string field)
    {
        object values,value;
        if(CombatTables.TryGetValue(table,out values) && ((Dictionary<string,object>)values).TryGetValue(key+":"+field,out value))
            return System.Convert.ToInt64(value);
        return actual;
    }
    public static long CombatRowNumber(long actual,object row,string table,string field)
    {
        string key;
        var factor=row as SkillLevelFactorData;
        var soul=row as SoulWeaponOptionData;
        var tier=row as MonsterTierData;
        var define=row as BattleDefineData;
        if(factor!=null)key=factor.Level.ToString();
        else if(soul!=null)key=((int)soul.TagType).ToString();
        else if(tier!=null)key=tier.Index.ToString();
        else if(define!=null)key=define.Code.ToString();
        else throw new System.InvalidOperationException("Unknown restored combat row: "+table);
        return CombatNumber(actual,table,key,field);
    }

    public static long LegacyPunishmentDamage(long attack, long defense, int attackerLevel, int damagedLevel)
    {
        // Build 1558 has the original exponential defense constants and no
        // DefenceLevelTable. The retained Season1 routine must use those
        // constants, not the later client's 1013959 level growth factor.
        attackerLevel = System.Math.Min(attackerLevel, 80);
        damagedLevel = System.Math.Min(damagedLevel, 80);
        long ratio = PunishmentDefenseRatios.GetAccRatio(1047294, attackerLevel * 3 - damagedLevel * 2 - 1);
        long divisor = defense * 2 * 1000000 / System.Math.Max(ratio, 1) + 1000;
        return attack * (20 + 980 * 1000 / System.Math.Max(divisor, 1)) / 1000;
    }

    static Dictionary<int,GuildSuppressSessionData> serverConquestSessions;
    static Dictionary<int,List<GuildSuppressSessionData>> serverConquestSeasons;
    public static void AcceptServerConquestSessions(System.Collections.IDictionary response)
    {
        var definitions=response==null ? null : response["SPRKConquestSessionDefinitions"] as System.Collections.IList;
        if(definitions==null)return;
        if(serverConquestSessions==null)serverConquestSessions=new Dictionary<int,GuildSuppressSessionData>();
        if(serverConquestSeasons==null)serverConquestSeasons=new Dictionary<int,List<GuildSuppressSessionData>>();
        foreach(System.Collections.IDictionary definition in definitions) {
            var session=JM_NShared_GuildSuppressSessionData.Parse(definition);
            if(session.SessionIndex<10000 || session.SeasonIndex<10000)continue;
            serverConquestSessions[session.SessionIndex]=session;
            List<GuildSuppressSessionData> season;
            if(!serverConquestSeasons.TryGetValue(session.SeasonIndex,out season)) {
                season=new List<GuildSuppressSessionData>();serverConquestSeasons.Add(session.SeasonIndex,season);
            }
            bool replaced=false;
            for(int i=0;i<season.Count;i++)if(season[i].SessionIndex==session.SessionIndex){season[i]=session;replaced=true;break;}
            if(!replaced)season.Add(session);
        }
    }
    public static bool CheckNativeStoryCondition(string[] condition)
    {
        int tutorial,chapter,dungeon;

            || !int.TryParse(condition[3],out chapter) || !int.TryParse(condition[4],out dungeon))return false;
        var account=NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.instance;

            || (account.IsTutorialSkip && account.CampaignManager.IsCompleted(chapter,dungeon,DungeonDifficultyType.Easy));
    }

    public static OptionSellData[] RestoreInventoryFilterOptions(OptionSellData[] original, OptionSellCategory category)
    {
        if (category != OptionSellCategory.Tier && category != OptionSellCategory.Grade) return original;
        var rows = new List<OptionSellData>(original ?? new OptionSellData[0]);
        int start = category == OptionSellCategory.Tier ? 1 : 6;
        int end = category == OptionSellCategory.Tier ? 10 : 7;
        for (int value = start; value <= end; value++)
        {
            bool found = false;
            for (int i = 0; i < rows.Count; i++) if (rows[i].Value == value) found = true;
            if (found) continue;
            int[] dependencies = new int[category == OptionSellCategory.Tier ? 7 : 0];
            for (int d = 0; d < dependencies.Length; d++) dependencies[d] = d + 1;
            rows.Add(new OptionSellData {
                Category = category, Value = value, BackGroundIndex = 0,
                GradeStringKey = value == 6 ? "GRADE_ORANGE2" : "GRADE_TECHNOMAGIC",
                ReplaceKey = "", SelectType = value == 7 && category == OptionSellCategory.Grade
                    ? OptionSellSelectType.OnlyOne : OptionSellSelectType.Multi,
                GradeDependency = dependencies
            });
        }
        return rows.ToArray();
    }

        if (!account.IsTutorialSkip) return condition;

            case "200000": chapter = 97; dungeon = 2; break;

            case "14000": chapter = 1; dungeon = 18; break;

    public static void CurveCardMesh(UIWidget widget, int offset, RenderingList<Vector3> verts,
        RenderingList<Vector2> uvs, RenderingList<Color32> colors)
    {
        if (offset != 0 || verts.size != 4) return;
        Vector3 bottom = verts[0], top = verts[2];
        Vector2 uvBottom = uvs[0], uvTop = uvs[2];
        Color32 color = colors[0];
        verts.Clear(); uvs.Clear(); colors.Clear();
        const int steps = 64;
        verts.ReserveCount(steps * 4); uvs.ReserveCount(steps * 4);
        for (int i = 0; i < steps; i++)
        {
            float low = i / (float)steps, high = (i + 1) / (float)steps;
            float rightLow = 0.12f + 0.88f * Mathf.Sqrt(1f - low * low);
            float rightHigh = 0.12f + 0.88f * Mathf.Sqrt(1f - high * high);
            verts.AddUnsafe(new Vector3(bottom.x, Mathf.Lerp(bottom.y, top.y, low), bottom.z));
            verts.AddUnsafe(new Vector3(bottom.x, Mathf.Lerp(bottom.y, top.y, high), bottom.z));
            verts.AddUnsafe(new Vector3(Mathf.Lerp(bottom.x, top.x, rightHigh), Mathf.Lerp(bottom.y, top.y, high), bottom.z));
            verts.AddUnsafe(new Vector3(Mathf.Lerp(bottom.x, top.x, rightLow), Mathf.Lerp(bottom.y, top.y, low), bottom.z));
            uvs.AddUnsafe(new Vector2(uvBottom.x, Mathf.Lerp(uvBottom.y, uvTop.y, low)));
            uvs.AddUnsafe(new Vector2(uvBottom.x, Mathf.Lerp(uvBottom.y, uvTop.y, high)));
            uvs.AddUnsafe(new Vector2(Mathf.Lerp(uvBottom.x, uvTop.x, rightHigh), Mathf.Lerp(uvBottom.y, uvTop.y, high)));
            uvs.AddUnsafe(new Vector2(Mathf.Lerp(uvBottom.x, uvTop.x, rightLow), Mathf.Lerp(uvBottom.y, uvTop.y, low)));
            colors.AddCount(color, 4);
        }
    }

    public static bool OpenShakmehBranch(string[] values)
    {
        if (values == null || values.Length == 0 || (values[0] != "Middle" && values[0] != "Final")) return false;
        if (!ShakmehAccess()) return true;
        if (!NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.instance.CampaignManager.IsCompleted(50000, 601, DungeonDifficultyType.Easy))
        {
            NGame2.NUI.NWindow.MessagePopup.Open("Content locked", "Complete The Shadow of Aegina story to unlock the Shakmeh boss shortcuts.");
            return true;
        }
        bool final = values[0] == "Final";
        if (!ShakmehGaugeAccess(final)) return true;
        // This manager is internal in the recovered DLL. Use its native singleton
        // and explicit chapter/node overload; the boss-type overload ignores its argument.
        var type = typeof(GameRoot).Assembly.GetType("NGame2.NUI.NManager.ShakmehDungeonManagement", true);
        var singleton = typeof(NVespa.NSingleton.SceneAutoBehaviour<>).MakeGenericType(type);
        var manager = singleton.GetProperty("instance").GetValue(null, null);
        NGame2.NAccount.UserCampaignManager.ChapterIndex = 50000;
        NGame2.NAccount.UserCampaignManager.DungeonIndex = final ? 11 : 10;
        type.GetMethod("OpenShakmehDungeonWindow", new System.Type[] { typeof(int), typeof(int) })
            .Invoke(manager, new object[] { 50000, final ? 11 : 10 });
        return true;
    }

    public static bool ChapterFiveSubStoryHidden(bool hidden, CampaignNodeData node)
    {
        if (node == null) return hidden;
        // Both territory NPCs are visible in the pre-Doomsday Guild map.
        if (node.ChapterIndex == 995 &&
            ((node.DungeonIndex == 5 && node.EntryNPCCode == "GuildSuppress") ||
             (node.DungeonIndex == 6 && node.EntryNPCCode == "GuildRankingBoard"))) return false;
        if (node.ChapterIndex == 99 && node.DungeonIndex == 41) return false;
        // Shipped branch 5-21 -> 133 -> 31 leads to hero substory 100017.
        // Keep the native graph progression and substory entry requirements.
        if (node.ChapterIndex == 5 && node.DungeonIndex == 31 &&
            node.Function == NodeFunctionType.MoveSubStory && node.Value != null &&
            node.Value.Length >= 2 && node.Value[0] == "100017" && node.Value[1] == "1") return false;
        return hidden;
    }

    static List<RaidData> ReconstructedRaids()
    {
        var rows = (System.Collections.IList)MiniJSON.Json.Deserialize("__RECONSTRUCTED_RAIDS__");
        var result = new List<RaidData>();
        foreach (System.Collections.IDictionary row in rows) result.Add(new RaidData {
            Index=System.Convert.ToInt32(row["Index"]), Level=System.Convert.ToInt32(row["Level"]),
            Type=(RaidType)System.Convert.ToInt32(row["Type"]), Name=(string)row["Name"],
            ChapterIndex=System.Convert.ToInt32(row["ChapterIndex"]), DungeonIndex=System.Convert.ToInt32(row["DungeonIndex"]),
            IsOpen=(bool)row["IsOpen"], PlayerCount=System.Convert.ToInt32(row["PlayerCount"]),
            DeckCount=System.Convert.ToInt32(row["DeckCount"]), DeckCountPerPlayer=System.Convert.ToInt32(row["DeckCountPerPlayer"]),
            MainPartyCount=System.Convert.ToInt32(row["MainPartyCount"]), SubPartyCount=System.Convert.ToInt32(row["SubPartyCount"]),
            ReqHeroLevel=System.Convert.ToInt32(row["ReqHeroLevel"]), IsOnlineSingle=(bool)row["IsOnlineSingle"],
            AliveDuration=System.Convert.ToInt32(row["AliveDuration"]), ClearRaidIndex=System.Convert.ToInt32(row["ClearRaidIndex"]),
            IndividualReward=System.Convert.ToInt32(row["IndividualReward"]), ResetGem=System.Convert.ToInt32(row["ResetGem"]),
            BanIndex=System.Convert.ToInt32(row["BanIndex"]),
            RaidMultiInfoGroupIndex=System.Convert.ToInt32(row["RaidMultiInfoGroupIndex"]),
            HideRaidInfoSkillTab=System.Convert.ToBoolean(row["HideRaidInfoSkillTab"]),
            // Older reconstructed rows use numeric zero for no custom animation.
            RaidInfoCustomIdle=row["RaidInfoCustomIdle"] as string,
            NodeTreePath=(string)row["NodeTreePath"], NodeTreeCode=(string)row["NodeTreeCode"],
            PreviewBg=(string)row["PreviewBg"], Tactic=(string)row["Tactic"], RaidDesc=(string)row["RaidDesc"],
            SingleRaidIndex=System.Convert.ToInt32(row["SingleRaidIndex"]),
            FilterValue=System.Convert.ToInt32(row["FilterValue"]),
            Step_Name=(string)row["Step_Name"] ?? "", Step_HelpDesc=(string)row["Step_HelpDesc"] ?? "",
            OccurChapterIndex=System.Convert.ToInt32(row["OccurChapterIndex"]),
            RaidPoint=System.Convert.ToInt32(row["RaidPoint"]),
            OpenCondition=IntArray(row["OpenCondition"]),
            RecommendedHeroIndex=IntArray(row["RecommendedHeroIndex"]), ShowInfoSkill=IntArray(row["ShowInfoSkill"])
        });
        return result;
    }
    static int[] IntArray(object value)
    {
        var list = value as System.Collections.IList;
        if (list == null) return new int[0];
        var result = new int[list.Count];
        for (int i = 0; i < list.Count; i++) result[i] = System.Convert.ToInt32(list[i]);
        return result;
    }

    public static IList<RaidData> RaidList(IList<RaidData> original, RaidType type)
    {
        var result = original == null ? new List<RaidData>() : new List<RaidData>(original);
        foreach (var row in ReconstructedRaids()) {
            if (row.Type != type) continue;
            bool exists = false;
            foreach (var old in result) if (old.Index == row.Index && old.Level == row.Level) exists = true;
            if (!exists) result.Add(row);
        }
        return result;
    }

                Add(rows, category, PortalRenewalSubCategory.PunishmentRaid, "Apocalypsion Raid", ClientActivityType.OpenPunishmentRaid, "11", "700"); break;

                Add(rows, category, PortalRenewalSubCategory.PunishmentRuneCraft, "Rune Crafting", ClientActivityType.UsingActionContainer, "OpenPunishmentRuneCraft"); break;
        }
        return rows;
    }

    static string[] GodkingCondition(PortalRenewalCategory category)
    {
        if (category == PortalRenewalCategory.TechnoRaid) return new string[] { "CompleteDungeon", "10", "9", "true" };
        if (category == PortalRenewalCategory.PunishmentRaid) return new string[] { "CompleteDungeon", "11", "10", "true" };
        if (category == PortalRenewalCategory.Shakmeh) return new string[] { "CompleteDungeon", "97", "2", "true" };
        return category == PortalRenewalCategory.GodKingTemple || category == PortalRenewalCategory.Eclipse
            ? new string[] { "CompleteDungeon", "65", "11", "true" } : null;
    }

    static void Add(List<PortalRenewalData> rows, PortalRenewalCategory category, PortalRenewalSubCategory sub,
        string title, ClientActivityType activity, params string[] args)
    {
        if ((category == PortalRenewalCategory.Guild || category == PortalRenewalCategory.GuildTerritory)
            && NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NGuild.GuildManager>.instance.MyGuildManager.GetGuildId() == 0)
        {
            activity = ClientActivityType.MoveToNode;
            args = new string[] { "99", "5", "true", "true" };
        }
        bool shakmehBoss = category == PortalRenewalCategory.Shakmeh && (sub == PortalRenewalSubCategory.ShakmehMiddle || sub == PortalRenewalSubCategory.ShakmehFinal);
        rows.Add(new PortalRenewalData { Category = category, SubCategory = sub, TitleLocalKey = title,
            MenuButtonLocalkey = title, MainImagePath = CardImage(sub), ContentsShowIndex = rows.Count, SubComponentShape = PortalRenewalCompType.Small,
            OpenCondition = shakmehBoss ? new string[] { "CompleteDungeon", "50000", "601", "true" } : GodkingCondition(category),
            OpenConditionLocalKey = shakmehBoss ? "Complete The Shadow of Aegina story to unlock Shakmeh boss battles." : UnlockMessage(category),
            MoveTaskType = activity, MoveTaskValue = args });
    }
