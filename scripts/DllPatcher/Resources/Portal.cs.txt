    public static void SetupCardCurve(NGame2.NUI.NComponent2.PortalContentsComponent component, PortalRenewalData data)
    {
        if (data == null || data.SubComponentShape != PortalRenewalCompType.Small) return;
        string code = data.MainImagePath;
        if (code != "Portal_WorldBoss_NotProgress_Image" && code != "Portal_ChallengeRaid_NotProgress_Image"
            && code != "Portal_PetHouse_Image" && code != "Portal_TreasureHouse_Image"
            && code != "Portal_Punishment_Image" && code != "Portal_MiddleBossShakmeh_Image") return;
        UITexture texture = component._basicData.ContentsTexture;
        if (texture != null) texture.onPostFill = new UIWidget.OnPostFillCallback(CurveCardMesh);
    }

    public static void CurveCardMesh(UIWidget widget, int offset, RenderingList<Vector3> verts,
        RenderingList<Vector2> uvs, RenderingList<Color32> colors)
    {
        if (offset != 0 || verts.size != 4) return;
        Vector3 bottom = verts[0], top = verts[2];
        Vector2 uvBottom = uvs[0], uvTop = uvs[2];
        Color32 color = colors[0];
        verts.Clear(); uvs.Clear(); colors.Clear();
        const int steps = 64;
        verts.ReserveCount(steps * 4); uvs.ReserveCount(steps * 4);
        for (int i = 0; i < steps; i++)
        {
            float low = i / (float)steps, high = (i + 1) / (float)steps;
            float rightLow = 0.12f + 0.88f * Mathf.Sqrt(1f - low * low);
            float rightHigh = 0.12f + 0.88f * Mathf.Sqrt(1f - high * high);
            verts.AddUnsafe(new Vector3(bottom.x, Mathf.Lerp(bottom.y, top.y, low), bottom.z));
            verts.AddUnsafe(new Vector3(bottom.x, Mathf.Lerp(bottom.y, top.y, high), bottom.z));
            verts.AddUnsafe(new Vector3(Mathf.Lerp(bottom.x, top.x, rightHigh), Mathf.Lerp(bottom.y, top.y, high), bottom.z));
            verts.AddUnsafe(new Vector3(Mathf.Lerp(bottom.x, top.x, rightLow), Mathf.Lerp(bottom.y, top.y, low), bottom.z));
            uvs.AddUnsafe(new Vector2(uvBottom.x, Mathf.Lerp(uvBottom.y, uvTop.y, low)));
            uvs.AddUnsafe(new Vector2(uvBottom.x, Mathf.Lerp(uvBottom.y, uvTop.y, high)));
            uvs.AddUnsafe(new Vector2(Mathf.Lerp(uvBottom.x, uvTop.x, rightHigh), Mathf.Lerp(uvBottom.y, uvTop.y, high)));
            uvs.AddUnsafe(new Vector2(Mathf.Lerp(uvBottom.x, uvTop.x, rightLow), Mathf.Lerp(uvBottom.y, uvTop.y, low)));
            colors.AddCount(color, 4);
        }
    }

    public static void RefreshHeroDeckDispatch(NGame2.NUI.NWindow.HeroDeck2 window)
    {
        foreach (var portrait in window.GetComponentsInChildren<NGame2.NUI.NComponent2.HeroPortraitComponent>(true))
            if (portrait.HeroInfo != null)
            {
                portrait.SetDispatched(NGame2.NDispatchBattle.DispatchHelper.IsDispatchedHero(portrait.HeroInfo),
                    portrait.PortraitMode == NGame2.NUI.NComponent2.HeroPortraitComponent.EPortraitMode.Icon, "");
                FitHeroDeckDispatchLabel(portrait);
            }
    }
    public static void FitHeroDeckDispatchLabel(NGame2.NUI.NComponent2.HeroPortraitComponent portrait)
    {
        if (!portrait.IsDispatched || portrait.PortraitMode == NGame2.NUI.NComponent2.HeroPortraitComponent.EPortraitMode.Icon) return;
        var status = portrait.GetSubComponent<NGame2.NUI.NComponent2.NHeroPortraitSub.CustomLabelSubComponent>();
        if (status == null || status.SpriteDiabled == null || portrait.WidgetThis == null) return;
        // Fit the native dimmer to the actual card, including cards currently
        // culled by the scroll view. Anchors follow resizing and list movement.
        UIWidget backdrop = status.SpriteDiabled;
        UIWidget bounds = portrait.HeroBundleTex != null ? portrait.HeroBundleTex.UiTex : null;
        if (bounds == null) bounds = portrait.FrameWidget != null ? portrait.FrameWidget : portrait.WidgetThis;
        backdrop.SetAnchor(bounds.gameObject, 0, 0, 0, 0);
        backdrop.UpdateAnchors();
    }

    public static void TempleMissingScene(NGame2.NCutScene.CutSceneManager scene)
    {
        // Only replace the known nine-byte placeholder's fallback, never valid scenes.
        if (scene.ScriptPath != "CutScene/CH65_Main_4") return;
        foreach (var group in scene.GroupList)
            foreach (var holder in group.ScriptList)
            {
                var text = holder.Value as NGame2.NCutScene.NScript.DisplayText;
                if (text == null) continue;
                text.TitleLink.Key = "Temple of the God King";
                text.TextLink.Key = "This story scene is unavailable in the recovered game files. Tap to continue to the next stage.";
            }
    }

    public static void RestoreGodkingClassRules()
    {
        foreach (var group in GameRoot.Instance.GodkingTrialGroupDataContainer.GetAllData())
        {
            // Recovered verbatim from the reference client's trial group table.
            if (group.ChapterIndex == 100000) group.BanRuleIndex = 303;
            else if (group.ChapterIndex == 100001) group.BanRuleIndex = 304;
            else if (group.ChapterIndex == 100002) group.BanRuleIndex = 305;
        }
    }

    public static bool ChapterFiveSubStoryHidden(bool hidden, CampaignNodeData node)
    {
        // Shipped branch 5-21 -> 133 -> 31 leads to hero substory 100017.
        // Keep the native graph progression and substory entry requirements.
        if (node.ChapterIndex == 5 && node.DungeonIndex == 31 &&
            node.Function == NodeFunctionType.MoveSubStory && node.Value != null &&
            node.Value.Length >= 2 && node.Value[0] == "100017" && node.Value[1] == "1") return false;
        return hidden;
    }

    public static bool WorldMapHidden(bool hidden, PortalData portal)
    {
        // The illustrated World Map uses PortalTable, independently of PortalRenewal.
        // Restore chapter landmarks and their centering records, retaining the
        // original clear-dungeon conditions and leaving special destinations alone.
        if (portal.ChapterIndex >= 5 && portal.ChapterIndex <= 11 &&
            (portal.DungeonIndex == -1 || portal.DungeonIndex == 1)) return false;
        return hidden;
    }

    public static bool RestoredCampaignChapterOpen(bool original, CampaignChapterData chapter)
    {
        // Restore shipped story chapters and the 9.5/epilogue prerequisites for 10.
        // Ice Crystal Cave uses 117 for the first visit and 127 for the return map.
        return original || (chapter.Index >= 4 && chapter.Index <= 11) || chapter.Index == 95 || chapter.Index == 97
            || chapter.Index == 117 || chapter.Index == 127;
    }

    public static bool ReserveContentsUnlockNotice(NShared.ContentsOpenConditionData data)
    {
        if (data == null || !NGame2.NAccount.AccountManager.IsLogined()) return false;
        // Full quest snapshots can report an unclaimed completed quest repeatedly.
        // Announcements are one-time per content and account, independent of claiming.
        var seen = new NVespa.LocalAccountBool(false, "Sprk.ContentsUnlockNotice." + data.Index);
        if (seen.Value) return false;
        seen.Value = true;
        return true;
    }

    public static int EquipmentBagMaxExtensions(int original, NShared.InventoryType inventoryType)
    {
        // 280 base slots + 90 extensions of 8 slots = 1000.
        return inventoryType == NShared.InventoryType.Inventory ? System.Math.Min(original, 90) : original;
    }

    // Boss HP gates require their story sequences even for skipped tutorials:
    // 2-18 event 20181; 4-21 sequence 40212 removes Malduk's skill 39.
    public static bool RequiredBattleStoryEnabled(bool enabled)
    {
        if (enabled) return true;
        if (UnityEngine.SceneManagement.SceneManager.GetActiveScene().name != "Battle2") return false;

            ((dungeon.Chapter == 2 && dungeon.OriginalDungeon == 18) ||
             (dungeon.Chapter == 4 && dungeon.OriginalDungeon == 21));
    }

    static List<RaidData> ReconstructedRaids()
    {
        var rows = (System.Collections.IList)MiniJSON.Json.Deserialize("__RECONSTRUCTED_RAIDS__");
        var result = new List<RaidData>();
        foreach (System.Collections.IDictionary row in rows) result.Add(new RaidData {
            Index=System.Convert.ToInt32(row["Index"]), Level=System.Convert.ToInt32(row["Level"]),
            Type=(RaidType)System.Convert.ToInt32(row["Type"]), Name=(string)row["Name"],
            ChapterIndex=System.Convert.ToInt32(row["ChapterIndex"]), DungeonIndex=System.Convert.ToInt32(row["DungeonIndex"]),
            IsOpen=(bool)row["IsOpen"], PlayerCount=System.Convert.ToInt32(row["PlayerCount"]),
            DeckCount=System.Convert.ToInt32(row["DeckCount"]), DeckCountPerPlayer=System.Convert.ToInt32(row["DeckCountPerPlayer"]),
            MainPartyCount=System.Convert.ToInt32(row["MainPartyCount"]), SubPartyCount=System.Convert.ToInt32(row["SubPartyCount"]),
            ReqHeroLevel=System.Convert.ToInt32(row["ReqHeroLevel"]), IsOnlineSingle=(bool)row["IsOnlineSingle"],
            AliveDuration=System.Convert.ToInt32(row["AliveDuration"]), ClearRaidIndex=System.Convert.ToInt32(row["ClearRaidIndex"]),
            IndividualReward=System.Convert.ToInt32(row["IndividualReward"]), ResetGem=System.Convert.ToInt32(row["ResetGem"]),
            NodeTreePath=(string)row["NodeTreePath"], NodeTreeCode=(string)row["NodeTreeCode"],
            PreviewBg=(string)row["PreviewBg"], Tactic=(string)row["Tactic"], RaidDesc=(string)row["RaidDesc"],
            SingleRaidIndex=System.Convert.ToInt32(row["SingleRaidIndex"]),
            FilterValue=System.Convert.ToInt32(row["FilterValue"]),
            Step_Name=(string)row["Step_Name"] ?? "", Step_HelpDesc=(string)row["Step_HelpDesc"] ?? "",
            OccurChapterIndex=System.Convert.ToInt32(row["OccurChapterIndex"]),
            RaidPoint=System.Convert.ToInt32(row["RaidPoint"]),
            OpenCondition=IntArray(row["OpenCondition"]),
            RecommendedHeroIndex=IntArray(row["RecommendedHeroIndex"]), ShowInfoSkill=IntArray(row["ShowInfoSkill"])
        });
        return result;
    }
    static int[] IntArray(object value)
    {
        var list = value as System.Collections.IList;
        if (list == null) return new int[0];
        var result = new int[list.Count];
        for (int i = 0; i < list.Count; i++) result[i] = System.Convert.ToInt32(list[i]);
        return result;
    }
    public static int MultiRaid(int original, int singleIndex)
    {
        if (original != 0) return original;
        foreach (var row in ReconstructedRaids()) if (row.SingleRaidIndex == singleIndex && singleIndex != 0) return row.Index;
        return 0;
    }
    public static RaidData Raid(RaidData original, int index, int level)
    {
        if (original != null) return original;
        foreach (var row in ReconstructedRaids()) if (row.Index == index && row.Level == level) return row;
        return null;
    }
    public static RaidData RaidDungeon(RaidData original, int chapter, int dungeon)
    {
        if (original != null) return original;
        foreach (var row in ReconstructedRaids()) if (row.ChapterIndex == chapter && row.DungeonIndex == dungeon) return row;
        return null;
    }
    public static ICollection<int> RaidIndices(ICollection<int> original, RaidType type)
    {
        var result = original == null ? new List<int>() : new List<int>(original);
        foreach (var row in ReconstructedRaids()) if (row.Type == type && !result.Contains(row.Index)) result.Add(row.Index);
        return result;
    }
    public static ICollection<int> RaidLevels(ICollection<int> original, int index)
    {
        var result = original == null ? new List<int>() : new List<int>(original);
        foreach (var row in ReconstructedRaids()) if (row.Index == index && !result.Contains(row.Level)) result.Add(row.Level);
        return result;
    }
    public static IList<RaidData> RaidList(IList<RaidData> original, RaidType type)
    {
        var result = original == null ? new List<RaidData>() : new List<RaidData>(original);
        foreach (var row in ReconstructedRaids()) {
            if (row.Type != type) continue;
            bool exists = false;
            foreach (var old in result) if (old.Index == row.Index && old.Level == row.Level) exists = true;
            if (!exists) result.Add(row);
        }
        return result;
    }

    static void Menu(List<PortalRenewalData> rows, PortalRenewalCategory category, string title)
    {
        foreach (var row in rows) if (row.Category == category) return;
        rows.Add(new PortalRenewalData { Category = category, SubCategory = PortalRenewalSubCategory.MENUBUTTON,

    }

    public static IList<PortalRenewalData> Contents(IList<PortalRenewalData> original, PortalRenewalCategory category)
    {
        var rows = original == null ? new List<PortalRenewalData>() : new List<PortalRenewalData>(original);
        if (category == PortalRenewalCategory.Chapter)
        {
            for (int chapter = 5; chapter <= 11; chapter++)
            {
                bool exists = false;
                foreach (var row in rows) if (row.ContentsShowIndex == chapter) exists = true;
                if (!exists) rows.Add(new PortalRenewalData {
                    Category = category, SubCategory = (PortalRenewalSubCategory)(100 + chapter),
                    ContentsShowIndex = chapter, SubComponentShape = (PortalRenewalCompType)5,
                    TitleLocalKey = "PORTAL_CHAPTER_" + chapter + "_NAME",
                    MainImagePath = "Portal_Chapter" + chapter + "_Image", VisualChapterIndex = chapter.ToString()
                });
            }
        }
        // Do not replace native content panels when another client table already provides them.
        foreach (var row in rows) if (row.SubCategory != PortalRenewalSubCategory.MENUBUTTON) return rows;
        foreach (var menu in Categories(rows))
            if (menu.Category == category && !rows.Contains(menu)) rows.Add(menu);
        switch (category)
        {
            case PortalRenewalCategory.Valance:
                Add(rows, category, PortalRenewalSubCategory.Identified, "Identification", ClientActivityType.UsingActionContainer, "OpenValanceIndentified");
                Add(rows, category, PortalRenewalSubCategory.Craft, "Craft", ClientActivityType.UsingActionContainer, "OpenValanceCraft");

    {

        if (category == PortalRenewalCategory.PunishmentRaid) return new string[] { "CompleteDungeon", "11", "10", "true" };

    }

    {
        switch (sub)
