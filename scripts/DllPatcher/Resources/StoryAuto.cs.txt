// Uses the native next-stage, entry, reward, and Skip paths. Never fabricates clears.
public static class SprkStoryAuto
{
    static BattlePartySetting waitingParty;
    static bool waitingForEvent;
    static bool continuousStory;
    static EnterDungeonData pendingNext;
    static bool movingToNext, openingNextParty;
    static float resumeAfter;
    static float lastProgress, nextTick, retryAfter;
    static float nextCrowSkill;
    static SprkStoryAutoRunner runner;
    static TutorialData lastTutorialStep;
    static bool caveCompletionRequested;
    static int[] storyMainParty, storySubParty;
    static readonly FieldInfo currentSquard = typeof(BattlePartySetting).GetField("_currentSquard", BindingFlags.Instance | BindingFlags.NonPublic);
    static readonly FieldInfo autoStarted = typeof(BattlePartySetting).GetField("_isAutoCampaign", BindingFlags.Instance | BindingFlags.NonPublic);
    static readonly FieldInfo cutsceneSkippable = typeof(CutSceneManager).GetField("_canSkip", BindingFlags.Instance | BindingFlags.NonPublic);
    static readonly FieldInfo narrationIndex = typeof(NGame2.NCutScene.NScript.ForegroundTextList).GetField("CurrentIndex", BindingFlags.Instance | BindingFlags.NonPublic);
    static readonly FieldInfo partyProvider = typeof(BattlePartySetting).GetField("_dataProvider", BindingFlags.Instance | BindingFlags.NonPublic);
    static readonly MethodInfo checkNext = typeof(EndCampaign).GetMethod("CheckNextDungeon", BindingFlags.Instance | BindingFlags.NonPublic);
    static readonly MethodInfo skillUseResult = typeof(BaseSkillComponent).GetMethod("GetSyncSkillUseResult", BindingFlags.Instance | BindingFlags.NonPublic);

    public static bool AutoSkills(bool enabled) { return enabled || Active(); }

    public static bool IsStory(EnterDungeonData enter)
    {
        // IsScenarioReplay is false for ordinary farming entries too. Read the
        // original map node's saved clear instead of relying on scenario mode.
        return IsStoryRoute(enter) && GlobalAutoBehaviour<AccountManager>.isValidInstance &&
            AccountManager.IsLogined() && !GlobalAutoBehaviour<AccountManager>.instance.CampaignManager
                .IsCompleted(enter.Chapter, enter.OriginalDungeon);
    }

    public static bool IsStoryRoute(EnterDungeonData enter)
    {
        if (enter != null && IsScarlet2(enter.Chapter, enter.OriginalDungeon))
            return (enter.Difficulty == DungeonDifficultyType.Easy || enter.Difficulty == DungeonDifficultyType.Normal) &&
                enter.LinkedDungeonData != null && enter.LinkedDungeonData.BattleType == BattleType.SubStory;
        return enter != null && (enter.Difficulty == DungeonDifficultyType.Normal ||
            (IsCaveBattle(enter.Chapter, enter.OriginalDungeon) && enter.Difficulty == DungeonDifficultyType.Easy)) &&
            enter.LinkedDungeonData != null && IsStoryType(enter.LinkedDungeonData.BattleType);
    }

    static bool IsCaveBattle(int chapter, int dungeon)
    {
        return chapter == 117 && (dungeon == 3 || dungeon == 4);
    }

    static bool IsScarlet2(int chapter, int dungeon)
    {
        // Scarlet 2 is the required substory between 9-13 and 9-14.
        return chapter == 9 && dungeon == 51;
    }

    public static DungeonDifficultyType NextDifficulty(DungeonDifficultyType difficulty, CampaignDungeonData next)
    {
        // Scripted detours use Easy. The surrounding story remains Normal; retain
        // native difficulty checks using the actual destination's mode.
        return Active() && next != null ?
            (IsCaveBattle(next.ChapterIndex, next.DungeonIndex) || IsScarlet2(next.ChapterIndex, next.DungeonIndex) ?
                DungeonDifficultyType.Easy : DungeonDifficultyType.Normal) : difficulty;
    }

    public static bool IsStoryType(BattleType type)
    {
        return type == BattleType.Campaign || type == BattleType.Boss;
    }

    public static bool Active()
    {
        // IsEnabled intentionally returns false during tutorials. Keep skipping alive then.
        // Native Continuous Battle also uses AutoNextBattle on Normal. Only the
        // explicit Continuous Story choice opts into our first-clear behavior.
        return continuousStory && GlobalAutoBehaviour<AccountManager>.isValidInstance && AccountManager.IsLogined() &&
            AutoNextBattle.IsValidInstance && IsStoryType(AutoNextBattle.instance.AutoNextBattleType) &&
            (AutoNextBattle.instance.DungeonDifficulty == DungeonDifficultyType.Normal ||
             AutoNextBattle.instance.DungeonDifficulty == DungeonDifficultyType.Easy);
    }

    public static bool ShowChoice(EnterDungeonData enter)
    {
        // Scarlet 2 is entered only by an existing main-story run.
        if (enter != null && IsScarlet2(enter.Chapter, enter.OriginalDungeon)) return false;
        if (!IsStory(enter)) return false;
        BattlePartySetting party = null;
        foreach (var candidate in UnityEngine.Object.FindObjectsOfType<BattlePartySetting>())
            if (candidate.IsOpen()) { party = candidate; break; }
        if (party == null) return false;
        MessagePopup3Button.Open("Story battle", "Play once, or advance through available story battles and unlocked chapters automatically. Continuous Story skips skippable dialogue and stops on defeat or when entry requires your attention.",
            delegate { party.TryStartBattle(null); }, delegate {
                if (party == null || !party.IsOpen()) return;
                // Starting a run must initialize its own native tutorial updates
                // and Unity update loop, including on an ordinary player launch.
                // Test instrumentation must never be needed to make it advance.
                var tutorial = SingletonUpdater<TutorialManager>.instance;
                if (runner == null) {
                    var host = new GameObject("SprkStoryAuto");
                    UnityEngine.Object.DontDestroyOnLoad(host);
                    runner = host.AddComponent<SprkStoryAutoRunner>();
                }
                runner.gameObject.SetActive(true);
                Application.runInBackground = true;
                CaptureParty(party, (BattlePartySetting.IDataProvider)partyProvider.GetValue(party));
                AutoNextBattle.OpenGlobal(enter.LinkedDungeonData.BattleType, DungeonDifficultyType.Normal, 0);
                continuousStory = true;
                pendingNext = null;
                lastProgress = Time.realtimeSinceStartup;
                waitingParty = party;
                waitingForEvent = true;
                retryAfter = lastProgress + 0.5f;
                // Leave the popup's click/close stack before native entry checks.
                // The same queue resumes an entry intercepted by a story event.
            }, "Battle Once", "Continuous Story");
        return true;
    }

    public static bool ChooseForParty(BattlePartySetting.IDataProvider provider, BattlePartySetting.Listener listener)
    {
        // First-time scenario entry bypasses the native OpenAutoNextBattleSelect
        // callback entirely. Offer the same choice before that special path.
        return !Active() && listener is NGame2.NUI.NManager.NLobby.LobbyCampaignDungeon &&
            provider != null && ShowChoice(provider.GetEnterDungeon());
    }

    public static void BeforeStart(BattlePartySetting party, BattlePartySetting.IDataProvider provider)
    {
        if (!Active()) return;
        if (provider == null || !IsStory(provider.GetEnterDungeon())) { Stop(); return; }
        CaptureParty(party, provider);
        if (waitingParty != party) lastProgress = Time.realtimeSinceStartup;
        waitingParty = party;
        waitingForEvent = false;
        var enter = provider.GetEnterDungeon();
        if (enter.Chapter == 117 && enter.OriginalDungeon == 4)
            QueueNext(GameRoot.Instance.CampaignDungeonDataContainer.GetData(7, 7));
        retryAfter = Time.realtimeSinceStartup + 1f;
        // The native timer must not submit a second request while this one is in flight.
        autoStarted.SetValue(party, true);
    }

    public static bool EventRaised(bool raised)
    {
        if (raised && Active()) waitingForEvent = true;
        return raised;
    }

    public static void StopForEntryEvent(AutoBattle battle, bool send, NShared.NMessage.NProtocol.PartyRoomRepeat.ResultType result)
    {
        if (Active() && waitingForEvent) return;
        battle.OnClickStop(send, result);
    }

    public static bool EntryChecked(bool allowed)
    {
        if (!allowed && Active()) Stop();
        return allowed;
    }

    public static void Stop()
    {
        bool active = Active();
        Cancelled();
        if (active) AutoNextBattle.instance.OnClickStopWithoutMessage();
    }

    // Called by native Stop/defeat too, so a queued lobby continuation cannot
    // restart the run after the player cancels it.
    public static void Cancelled()
    {
        continuousStory = false;
        waitingParty = null;
        waitingForEvent = false;
        pendingNext = null;
        movingToNext = openingNextParty = false;
        lastTutorialStep = null;
        caveCompletionRequested = false;
        nextCrowSkill = 0;
        storyMainParty = storySubParty = null;
        if (runner != null && runner.gameObject.activeSelf) runner.gameObject.SetActive(false);
    }

    public static bool CanNext()
    {
        return NextRoute(SingletonAuto<CampaignContext>.instance.EnterDungeon) != null;
    }

    static CampaignDungeonData NextRoute(EnterDungeonData enter)
    {
        // Completion changes IsScenarioReplay, so use the route at reward time.
        if (!IsStoryRoute(enter) || MainStoryManagement.IsPlayMainStoryMode) return null;
        // Finish Chapter 9 here even if Chapter X is already unlocked. Players
        // handle the epilogue and Chapter 9.5, then start a new run at 10-1.
        if (enter.Chapter == 9 && enter.OriginalDungeon == 23) return null;
        var current = enter.OriginalDungeonData;
        if (current == null || current.NextDungeonIndex == current.DungeonIndex) return null;
        CampaignDungeonData next = null;
        var manager = GlobalAutoBehaviour<AccountManager>.instance.CampaignManager;
        if (enter.Chapter == 9 && enter.OriginalDungeon == 13) {
            next = GameRoot.Instance.CampaignDungeonDataContainer.GetData(9, manager.IsCompleted(9, 51) ? 14 : 51);
        } else if (enter.Chapter == 9 && enter.OriginalDungeon == 51) {
            next = GameRoot.Instance.CampaignDungeonDataContainer.GetData(9, 14);
        } else if (enter.Chapter == 7 && enter.OriginalDungeon == 6) {
            // This boss ends the ordinary route but not Chapter 7. The first
            // cave visit has a real battle and then the native scripted battle.
            next = !manager.IsCompleted(117, 3) ? GameRoot.Instance.CampaignDungeonDataContainer.GetData(117, 3) :
                !manager.IsCompleted(117, 4) ? GameRoot.Instance.CampaignDungeonDataContainer.GetData(117, 4) :
                GameRoot.Instance.CampaignDungeonDataContainer.GetData(7, 7);
        } else if (IsCaveBattle(enter.Chapter, enter.OriginalDungeon)) {
            next = enter.OriginalDungeon == 3 && !manager.IsCompleted(117, 4) ?
                GameRoot.Instance.CampaignDungeonDataContainer.GetData(117, 4) :
                GameRoot.Instance.CampaignDungeonDataContainer.GetData(7, 7);
        } else if (current.NextDungeonIndex > 0)
            next = GameRoot.Instance.CampaignDungeonDataContainer.GetData(enter.Chapter, current.NextDungeonIndex);
        else {
            // Cross chapters only at a terminal WorldMap story node, using the
            // native chapter and node unlock checks. Never guess a side route.
            var chapter = GameRoot.Instance.CampaignChapterDataContainer.GetData(enter.Chapter + 1);
            if (enter.ChapterData == null || enter.ChapterData.Type != ChapterType.WorldMap ||
                chapter == null || chapter.Type != ChapterType.WorldMap) return null;
            next = GameRoot.Instance.CampaignDungeonDataContainer.GetData(enter.Chapter + 1, 1);
        }
        if (next == null || !(IsStoryType(next.BattleType) ||
            (IsScarlet2(next.ChapterIndex, next.DungeonIndex) && next.BattleType == BattleType.SubStory))) return null;
        return manager.CanGoToChapter(next.ChapterIndex) &&
            (manager.CanVisit(next.ChapterIndex, next.DungeonIndex) || IsCaveExit(next)) ? next : null;
    }

    static bool IsCaveExit(CampaignDungeonData next)
    {
        return next != null && next.ChapterIndex == 7 && next.DungeonIndex == 7 &&
            GlobalAutoBehaviour<AccountManager>.instance.CampaignManager.IsCompleted(117, 3);
    }

    static void QueueNext(CampaignDungeonData next)
    {
        var manager = GlobalAutoBehaviour<AccountManager>.instance.CampaignManager;
        pendingNext = new EnterDungeonData(next.ChapterIndex, next.DungeonIndex,
            manager.GetLinkedDungeonIndex(next.ChapterIndex, next.DungeonIndex),
            NextDifficulty(DungeonDifficultyType.Normal, next),
            manager.CanPlayScenario(next.ChapterIndex, next.DungeonIndex) && !manager.IsCompleted(next.ChapterIndex, next.DungeonIndex));
        movingToNext = openingNextParty = false;
        lastProgress = Time.realtimeSinceStartup;
    }

    public static bool Advance(EndCampaign end)
    {
        if (!Active()) return false;
        var next = NextRoute(SingletonAuto<CampaignContext>.instance.EnterDungeon);
        // 7-7 unlocks only after tutorial 70001 finishes its real cave story.
        // Return through native chapter flow, then wait for CanVisit in the lobby.
        var manager = GlobalAutoBehaviour<AccountManager>.instance.CampaignManager;
        bool waitingForCave = IsCaveExit(next) && !manager.CanVisit(7, 7);
        if (next == null || (!waitingForCave && !(bool)checkNext.Invoke(end, new object[] { next }))) { Stop(); return true; }
        QueueNext(next);
        // Keep the native return states: they apply chapter flow and story side
        // effects. Resume party entry after those events finish, rather than
        // relying on the overview's one-shot timer during a tutorial.
        LobbyScene2.ReturnState.OpenFlag &= ~LobbyScene2.ReturnStateOpenFlag.OpenNextDungeon;
        end.ReturnToLobby();
        return true;
    }

    static void ResumeNext()
    {
        if (pendingNext == null || UnityEngine.SceneManagement.SceneManager.GetActiveScene().name != "Lobby" ||
            GlobalAutoBehaviour<LoadingManager>.instance.IsSceneLoading ||
            !SceneAutoBehaviour<WorldMapManagement>.isValidInstance || !SceneAutoBehaviour<LobbyCampaignDungeon>.isValidInstance) return;
        var world = SceneAutoBehaviour<WorldMapManagement>.instance;
        if (world.IsMapLoading) return;
        var manager = GlobalAutoBehaviour<AccountManager>.instance.CampaignManager;
        // Native story events can complete 117-4 before its queued party opens.
        if (pendingNext.Chapter == 117 && pendingNext.OriginalDungeon == 4 && manager.IsCompleted(117, 4))
            QueueNext(GameRoot.Instance.CampaignDungeonDataContainer.GetData(7, 7));
        if (IsScarlet2(pendingNext.Chapter, pendingNext.OriginalDungeon) &&
            manager.IsCompleted(pendingNext.Chapter, pendingNext.OriginalDungeon))
            QueueNext(GameRoot.Instance.CampaignDungeonDataContainer.GetData(9, 14));
        if (!manager.CanGoToChapter(pendingNext.Chapter) || GlobalAutoBehaviour<AccountManager>.instance.IsMaxEquipItemCount()) { Stop(); return; }
        if (!manager.CanVisit(pendingNext.Chapter, pendingNext.OriginalDungeon)) {
            if (pendingNext.Chapter == 7 && pendingNext.OriginalDungeon == 7 && manager.IsCompleted(117, 3)) {
                CompleteSkippedCave();
                return;
            }
            Stop(); return;
        }
        if (!movingToNext) {
            movingToNext = true;
            resumeAfter = Time.realtimeSinceStartup + 2.5f;
            lastProgress = Time.realtimeSinceStartup;
            world.MovePlayerToDungeon(pendingNext.Chapter, pendingNext.OriginalDungeon, false, true);
            return;
        }
        if (NGame2.NAccount.UserCampaignManager.ChapterIndex != pendingNext.Chapter || Time.realtimeSinceStartup < resumeAfter) return;
        if (IsScarlet2(pendingNext.Chapter, pendingNext.OriginalDungeon)) {
            // Scarlet 2 starts directly from its map action.
            // Opening ordinary campaign party setup would lose its forced hero.
            if (!openingNextParty && NGame2.NAccount.UserCampaignManager.DungeonIndex == pendingNext.OriginalDungeon &&
                SceneAutoBehaviour<LobbyManagement>.isValidInstance) {
                openingNextParty = true;
                lastProgress = Time.realtimeSinceStartup;
                SceneAutoBehaviour<LobbyManagement>.instance.ExecuteCurrentNode(null);
            }
            return;
        }
        if (pendingNext.Chapter == 117 && pendingNext.OriginalDungeon == 4 &&
            NGame2.NAccount.UserCampaignManager.DungeonIndex == 4 &&
            NGame2.NAccount.Helper.HasEventTrigger(117, 4) && SceneAutoBehaviour<LobbyManagement>.isValidInstance) {
            // This node starts tutorial 70001 through NodeFunctionMenu, rather
            // than the ordinary party Start click. Let its native scripted
            // battle award the cave clear and unlock the main route.
            QueueNext(GameRoot.Instance.CampaignDungeonDataContainer.GetData(7, 7));
            SceneAutoBehaviour<LobbyManagement>.instance.ExecuteCurrentNode(null);
            return;
        }
        foreach (var party in UnityEngine.Object.FindObjectsOfType<BattlePartySetting>()) {
            if (!party.IsOpen()) continue;
            var provider = (BattlePartySetting.IDataProvider)partyProvider.GetValue(party);
            var enter = provider == null ? null : provider.GetEnterDungeon();
            if (enter == null || enter.Chapter != pendingNext.Chapter || enter.OriginalDungeon != pendingNext.OriginalDungeon) continue;
            if (!RestoreParty(party, provider)) {
                Stop();
                GlobalCenterMessage.Open("Continuous Story stopped: your party needs attention for this stage.");
                return;
            }
            if (SingletonUpdater<TutorialManager>.instance.IsProgressing ||
                GlobalAutoBehaviour<PopupManegement>.instance.AnyPopupExist()) return;
            pendingNext = null;
            autoStarted.SetValue(party, true);
            // Use the normal click path to retain party, stamina and entry checks.
            party.OnClickStartBattle();
            return;
        }
        if (!openingNextParty) {
            openingNextParty = true;
            lastProgress = Time.realtimeSinceStartup;
            SceneAutoBehaviour<LobbyCampaignDungeon>.instance.OpenPartySetting(pendingNext);
        }
    }

    static bool HasScriptedParty(BattlePartySetting.IDataProvider provider)
    {
        var enter = provider.GetEnterDungeon();
        if (IsScarlet2(enter.Chapter, enter.OriginalDungeon)) return true;
        var forced = provider.GetForcedPartyMemberHeroIndices();
        return (forced != null && forced.Length > 0) ||
            NGame2.NAccount.Helper.CheckFixedSetHero(enter.Chapter, enter.LinkedDungeon, enter.Difficulty, enter.IsScenario);
    }

    static void CaptureParty(BattlePartySetting party, BattlePartySetting.IDataProvider provider)
    {
        // Keep the player's run selection across chapter-specific saved decks.
        // A temporary story party must never replace the player's own team.
        if (provider == null || HasScriptedParty(provider)) return;
        var squad = (Squard)currentSquard.GetValue(party);
        if (squad == null || squad.GetMainSquardMyHeroIndexList().Count == 0) return;
        storyMainParty = squad.GetMainSquardMyHeroIndexList().ToArray();
        var sub = squad.GetSubSquardMyHeroIndexList();
        storySubParty = sub == null ? new int[0] : sub.ToArray();
    }

    static bool SameParty(System.Collections.Generic.List<int> current, int[] selected)
    {
        if (current == null) return selected.Length == 0;
        if (current.Count != selected.Length) return false;
        foreach (int hero in selected) if (!current.Contains(hero)) return false;
        return true;
    }

    static bool RestoreParty(BattlePartySetting party, BattlePartySetting.IDataProvider provider)
    {
        if (storyMainParty == null || HasScriptedParty(provider)) return true;
        var squad = (Squard)currentSquard.GetValue(party);
        if (squad == null || storyMainParty.Length > squad.GetMainSquardSlotCount() ||
            storySubParty.Length > squad.GetSubSquardSlotCount()) return false;
        if (SameParty(squad.GetMainSquardMyHeroIndexList(), storyMainParty) &&
            SameParty(squad.GetSubSquardMyHeroIndexList(), storySubParty)) return true;
        var listener = (NGame2.NUI.NComponent2.HeroPortraitComponent.Listener)party;
        squad.SwichingSquard(true);
        foreach (int hero in squad.GetMainSquardMyHeroIndexList().ToArray()) {
            var portrait = party.GetDeckHeroPortrait(hero);
            if (portrait == null) return false;
            listener.OnClick(portrait);
            if (squad.GetMainSquardMyHeroIndexList().Contains(hero)) return false;
        }
        var sub = squad.GetSubSquardMyHeroIndexList();
        if (sub != null && sub.Count > 0) {
            squad.SwichingSquard(false);
            foreach (int hero in sub.ToArray()) {
                var portrait = party.GetDeckHeroPortrait(hero);
                if (portrait == null) return false;
                listener.OnClick(portrait);
                if (squad.GetSubSquardMyHeroIndexList().Contains(hero)) return false;
            }
        }
        // Native clicks retain ownership, dispatch, bans, class restrictions,
        // duplicate and slot checks, and save the destination's normal deck.
        squad.SwichingSquard(true);
        foreach (int hero in storyMainParty) {
            var portrait = party.GetDeckHeroPortrait(hero);
            if (portrait == null) return false;
            listener.OnClick(portrait);
            if (!squad.GetMainSquardMyHeroIndexList().Contains(hero)) return false;
        }
        if (storySubParty.Length > 0) {
            squad.SwichingSquard(false);
            foreach (int hero in storySubParty) {
                var portrait = party.GetDeckHeroPortrait(hero);
                if (portrait == null) return false;
                listener.OnClick(portrait);
                if (!squad.GetSubSquardMyHeroIndexList().Contains(hero)) return false;
            }
            squad.SwichingSquard(true);
        }
        return SameParty(squad.GetMainSquardMyHeroIndexList(), storyMainParty) &&
            SameParty(squad.GetSubSquardMyHeroIndexList(), storySubParty);
    }

    static void CompleteSkippedCave()
    {
        var account = GlobalAutoBehaviour<AccountManager>.instance;
        // Tutorial skip replaces the scripted battle with two ordinary clears.
        // Submit its native completion only after both have actually been won;
        // use the server's response to unlock 7-7 without needing another login.
        if (caveCompletionRequested || !account.IsTutorialSkip ||
            !account.CampaignManager.IsCompleted(117, 3) || !account.CampaignManager.IsCompleted(117, 4) ||
            account.TutorialManager.IsComplete(70001)) return;
        caveCompletionRequested = true;
        var continuation = pendingNext;
        SingletonUpdater<WebServiceManager>.instance.Requester.Request<NShared.CompleteTutorial.Request, NShared.CompleteTutorial.Response>(
            new NShared.CompleteTutorial.Request { TutorialIndex = 70001 }, delegate(NShared.CompleteTutorial.Response response) {
                if (!GlobalAutoBehaviour<AccountManager>.isValidInstance ||
                    !object.ReferenceEquals(account, GlobalAutoBehaviour<AccountManager>.instance)) return;
                bool success = response != null && response.BaseResult == BaseResultType.Success &&
                    response.Result == NShared.CompleteTutorial.ResultType.Success && response.Info != null && response.Info.TutorialIndex == 70001;
                if (success) {
                    account.TutorialManager.Apply(response.Info);
                    account.CampaignManager.Apply(response.DungeonInfos);
                }
                if (!Active() || !object.ReferenceEquals(pendingNext, continuation)) return;
                if (success) lastProgress = Time.realtimeSinceStartup;
                else Stop();
            });
    }

    // Normal story is the entry difficulty; requiring an Easy clear prevents every first clear.
    // Restrict this exception to this one auto-next prerequisite call, not account progress.
    public static bool LowerDifficultyCleared(NGame2.NAccount.UserCampaignManager manager, int chapter, int dungeon, DungeonDifficultyType difficulty)
    {
        return (Active() && difficulty == DungeonDifficultyType.Easy) || manager.IsCompleted(chapter, dungeon, difficulty);
    }

    static bool AcknowledgeBattleTutorial(TutorialManager tutorial, TutorialWindow window)
    {
        // Informational battle tutorials (for example the boss warning) have
        // no Skip button. Complete only their native, one-shot WaitTouch action.
        // Widget/skill clicks and long presses keep their normal entry checks.
        if (UnityEngine.SceneManagement.SceneManager.GetActiveScene().name != "Battle2" ||
            !(tutorial.CurrentTutorialRuntime is NGame2.NTutorial.NRuntime.WaitTouch) ||
            window == null || !window.gameObject.activeInHierarchy) return false;
        var action = window.GetFirstTouchActionInfo();
        if (action == null || action.Action == null || !action.Oneshot || action.IsLongTouch ||
            action.ProcessTouch || action.ProcessBack || action.TargetGo != null) return false;
        window.ProcessTouchAction(action);
        return true;
    }

    static bool AcknowledgeStoryNarration(CutSceneManager cutscene, bool hideOnly = true)
    {
        // Illustrated narration (9-9's campfire, for example) hides Skip and
        // forwards each tap to ForegroundTextList rather than DisplayText.
        // Keep its native per-line effects and finalization, and never dispatch
        // another script's click callback or bypass a timed/interactive group.
        var group = cutscene.CurrentScriptGroup;
        var foreground = cutscene.CutSceneForeground;
        if (group == null || (hideOnly && group.SkipType != ScriptGroup.ESkipType.StopThisAndHide) ||
            group.TouchType != ScriptGroup.ETouchType.PassToScript ||
            foreground == null || !foreground.IsOpen() || !foreground.gameObject.activeInHierarchy) return false;
        NGame2.NCutScene.NScript.ForegroundTextList narration = null;
        foreach (var holder in group.ScriptList) {
            var script = holder.Value;
            if (script is NGame2.NCutScene.NScript.ForegroundTextList) {
                if (narration != null) return false;
                narration = (NGame2.NCutScene.NScript.ForegroundTextList)script;
            } else if (script != null && !script.IsEnd && script is IClickScript) return false;
        }
        if (narration == null) return false;
        // The text timer can finish the final line before our next tick. The
        // group still needs its last window tap to run native finalization.
        int before = (int)narrationIndex.GetValue(narration);
        cutscene.OnClickWindow();
        return group.IsEnd || before != (int)narrationIndex.GetValue(narration);
    }

    static bool AcknowledgeStoryDialogue(CutSceneManager cutscene)
    {
        // Some story lines hide Skip until a normal dialogue tap completes the
        // group (6-6, for example). Use its native touch/finalization path.
        var group = cutscene.CurrentScriptGroup;
        var window = cutscene.CutSceneWindow;
        if (group == null || group.SkipType != ScriptGroup.ESkipType.StopThisAndHide ||
            window == null || !window.IsOpen() || !window.gameObject.activeInHierarchy ||
            (group.TouchType != ScriptGroup.ETouchType.Touch &&
             group.TouchType != ScriptGroup.ETouchType.TouchAndTouch &&
             group.TouchType != ScriptGroup.ETouchType.TimeoutOrTouch)) return false;
        foreach (var holder in group.ScriptList) {
            if (!(holder.Value is NGame2.NCutScene.NScript.DisplayText)) continue;
            cutscene.OnClickWindow();
            return group.IsEnd;
        }
        return false;
    }

    static bool AcknowledgeForegroundCard(CutSceneManager cutscene)
    {
        // Single-screen narration (after 9-22) uses ForegroundText and a normal
        // group touch, unlike ForegroundTextList's per-line script callback.
        var group = cutscene.CurrentScriptGroup;
        var foreground = cutscene.CutSceneForeground;
        if (group == null || group.SkipType != ScriptGroup.ESkipType.StopThisAndHide ||
            group.TouchType != ScriptGroup.ETouchType.Touch ||
            foreground == null || !foreground.IsOpen() || !foreground.gameObject.activeInHierarchy) return false;
        bool hasCard = false;
        foreach (var holder in group.ScriptList) {
            var script = holder.Value;
            if (script is NGame2.NCutScene.NScript.ForegroundText) hasCard = true;
            else if (script != null && !script.IsEnd && script is IClickScript) return false;
        }
        if (!hasCard) return false;
        cutscene.OnClickWindow();
        return group.IsEnd;
    }

    static bool AcknowledgeCampaignTouch(CutSceneManager cutscene)
    {
        // Scarlet 2 and Chapter X have noninteractive cutscene pauses that
        // require native window touches when Skip is hidden or disabled.
        string scene = UnityEngine.SceneManagement.SceneManager.GetActiveScene().name;
        if (scene != "Lobby" && scene != "Battle2") return false;
        var route = pendingNext;
        if (route == null && scene == "Battle2") route = SingletonAuto<CampaignContext>.instance.EnterDungeon;
        if (route == null || !(IsScarlet2(route.Chapter, route.OriginalDungeon) || route.Chapter == 10)) return false;
        var group = cutscene.CurrentScriptGroup;
        if (group == null) return false;
        if (group.TouchType == ScriptGroup.ETouchType.PassToScript)
            return AcknowledgeStoryNarration(cutscene, false);
        if (group.TouchType != ScriptGroup.ETouchType.Touch && group.TouchType != ScriptGroup.ETouchType.TouchAndTouch &&
            group.TouchType != ScriptGroup.ETouchType.TimeoutOrTouch) return false;
        var window = cutscene.CutSceneWindow;
        var foreground = cutscene.CutSceneForeground;
        if (!(window != null && window.IsOpen() && window.gameObject.activeInHierarchy) &&
            !(foreground != null && foreground.IsOpen() && foreground.gameObject.activeInHierarchy)) return false;
        foreach (var holder in group.ScriptList)
            if (holder.Value != null && !holder.Value.IsEnd && holder.Value is IClickScript) return false;
        cutscene.OnClickWindow();
        return group.IsEnd;
    }

    static void PlayCrowScenario()
    {
        // 10-23 disables ordinary skill-auto and uses a fixed story party.
        // Press only Crow's S3 through the native button callback: the battle
        // thread still checks targets, mana, cooldowns and the scripted victory.
        var enter = SingletonAuto<CampaignContext>.instance.EnterDungeon;
        if (enter == null || enter.Chapter != 10 || enter.OriginalDungeon != 23 || !enter.IsScenario ||
            enter.LinkedDungeonData == null || enter.LinkedDungeonData.ScenarioBattleDefine != "Crow_Chapter10" ||
            !SingletonAuto<BattleFacade>.isValidInstance || Time.realtimeSinceStartup < nextCrowSkill) return;
        var facade = SingletonAuto<BattleFacade>.instance;
        var battle = facade.GetBattleCache();
        if (battle == null || !battle.IsBattleStarted || battle.IsBattleEnded || !battle.IsBattleEnabled) return;
        foreach (var slot in UnityEngine.Object.FindObjectsOfType<PartySlotComponent2>()) {
            var creature = slot.CreatureCache;
            if (!slot.gameObject.activeInHierarchy || creature == null || creature.Uid == 0 ||
                creature.IsDeadOrResting() || creature.CreateInfo == null || creature.CreateInfo.HeroInfo == null ||
                creature.CreateInfo.HeroInfo.HeroIndex != 52 || creature.Player == null || !creature.Player.IsMe ||
                creature.ReservedSkillSlotIndex == 3 || creature.UsingSkillSlotIndex == 3) continue;
            foreach (var skill in slot.GetComponentsInChildren<NormalSkillComponent>()) {
                if (!skill.gameObject.activeInHierarchy || skill.SkillSlotIndex != 3 || skill.SkillInfo.IsLocked) continue;
                var result = (SkillUseResult)skillUseResult.Invoke(skill, null);
                if (result != SkillUseResult.Success && result != SkillUseResult.EnhancedSuccess) continue;
                skill.OnClickSkill();
                nextCrowSkill = Time.realtimeSinceStartup + 1f;
                return;
            }
        }
    }

    public static void Tick()
    {
        if (Time.realtimeSinceStartup < nextTick) return;
        nextTick = Time.realtimeSinceStartup + 0.25f;
        if (!Active()) {
            // A newly opened global window can finish opening on the next frame.
            // Native cancellation clears continuousStory immediately instead.
            if (continuousStory && Time.realtimeSinceStartup - lastProgress < 3f) return;
            Cancelled();
            return;
        }
        try {
            bool storyPlaying = false;
            var tutorial = SingletonUpdater<TutorialManager>.instance;
            if (tutorial.IsProgressing) {
                storyPlaying = true;
                if (!object.ReferenceEquals(lastTutorialStep, tutorial.CurrentTutorialData)) {
                    lastTutorialStep = tutorial.CurrentTutorialData;
                    lastProgress = Time.realtimeSinceStartup;
                }
                var window = TutorialWindow.instance;
                if (window != null && window.NeedShowSkipButton()) window.OnClickSkip();
                else if (AcknowledgeBattleTutorial(tutorial, window)) lastProgress = Time.realtimeSinceStartup;
            }
            foreach (var cutscene in UnityEngine.Object.FindObjectsOfType<CutSceneManager>()) {
                if (!cutscene.IsPlaying && !cutscene.IsLoading) continue;
                storyPlaying = true;
                if (!cutscene.IsPlaying || !cutscene.IsShowing()) continue;
                if (!(bool)cutsceneSkippable.GetValue(cutscene)) {
                    if (AcknowledgeCampaignTouch(cutscene)) lastProgress = Time.realtimeSinceStartup;
                    continue;
                }
                var menu = cutscene.CutSceneMenu;
                // A visible Skip can advance part of a story even when a later
                // group must play. The native click stops at those boundaries
                // and finalizes script effects; CanSkipToEnd blocks too early.
                if (menu == null || !menu.IsOpen() || menu.SkipButton == null) continue;
                if (!menu.SkipButton.gameObject.activeInHierarchy) {
                    if (AcknowledgeStoryNarration(cutscene) || AcknowledgeStoryDialogue(cutscene) ||
                        AcknowledgeForegroundCard(cutscene) || AcknowledgeCampaignTouch(cutscene)) lastProgress = Time.realtimeSinceStartup;
                    continue;
                }
                if (!menu.SkipButton.isEnabled) continue;
                int script = cutscene.CurrentScript;
                menu.OnClickSkip();
                if (!cutscene.IsPlaying || cutscene.CurrentScript != script) lastProgress = Time.realtimeSinceStartup;
            }
            bool inBattle = UnityEngine.SceneManagement.SceneManager.GetActiveScene().name == "Battle2";
            bool rewarding = false;
            foreach (var reward in UnityEngine.Object.FindObjectsOfType<EndBattleReward>())
                if (reward.IsOpen()) { rewarding = true; break; }
            if (inBattle && !storyPlaying && !rewarding && !GlobalAutoBehaviour<LoadingManager>.instance.IsSceneLoading) {
                PlayCrowScenario();
                lastProgress = Time.realtimeSinceStartup;
                waitingParty = null;
                waitingForEvent = false;
            }
            if (waitingForEvent && Time.realtimeSinceStartup >= retryAfter && !storyPlaying && !GlobalAutoBehaviour<LoadingManager>.instance.IsSceneLoading &&
                waitingParty != null && waitingParty.IsOpen() &&
                !GlobalAutoBehaviour<NGame2.NUI.NManager.PopupManegement>.instance.AnyPopupExist()) {
                waitingForEvent = false;
                waitingParty.TryStartBattle(new BattlePartySetting.ExtraOptionData { AutoNext = true });
            }
            if (!storyPlaying && !GlobalAutoBehaviour<PopupManegement>.instance.AnyPopupExist()) ResumeNext();
            if (Time.realtimeSinceStartup - lastProgress > 90f) {
                Stop();
                GlobalCenterMessage.Open("Continuous Story stopped: an event or entry requires your attention.");
            }
        } catch (Exception error) {
            Stop();
            Debug.LogError("[SPRK Story Auto] Stopped: " + error);
        }
    }
}

public sealed class SprkStoryAutoRunner : MonoBehaviour
{
    void Update() { SprkStoryAuto.Tick(); }
}
