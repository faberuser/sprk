using System;
using System.Collections.Generic;
using System.IO;
using System.Net.Sockets;
using System.Text;
using System.Threading;

// A single native tick of presentation headroom absorbs ordinary delivery
// jitter. Reject stale/duplicate heartbeat clocks instead of moving playback
// backwards or repeatedly restarting its extrapolation from an old timestamp.
public sealed class SprkReplayClock {
    readonly int bufferMs;
    long latest=-1;
    public SprkReplayClock(int bufferMs) {this.bufferMs=Math.Max(0,Math.Min(200,bufferMs));}
    public bool Accept(ref long time) {
        lock(this) {
            if(time<0 || time<=latest)return false;
            latest=time;time=Math.Max(0,time-bufferMs);return true;
        }
    }
}

// Own one connection's queues and threads. No serialization or socket writes
// run on Unity's simulation thread. Overflow closes the connection rather than
// dropping ordered combat logs and continuing a corrupt replay.
public sealed class SprkBattleLink {
    sealed class Packet { public string Name; public object Body; }
    readonly object gate=new object();
    readonly Queue<Packet> outgoing=new Queue<Packet>();
    readonly Queue<string> incoming=new Queue<string>();
    readonly TcpClient socket;
    readonly Func<object,string> serialize;
    readonly Action<string> warning;
    readonly int capacity;
    bool closed;
    public bool Connected { get {lock(gate)return !closed;} }
    public SprkBattleLink(TcpClient socket,Func<object,string> serialize,Action<string> warning,int capacity=1024) {
        this.socket=socket;this.serialize=serialize;this.warning=warning;this.capacity=capacity;
        socket.NoDelay=true;socket.SendTimeout=5000;
        new Thread(WriteLoop){IsBackground=true}.Start();
    }
    public void Close() {
        lock(gate) {if(closed)return;closed=true;outgoing.Clear();incoming.Clear();Monitor.PulseAll(gate);}
        socket.Close();
    }
    public void Send(string name,object body) {
        lock(gate) {
            if(closed)return;
            if(outgoing.Count<capacity) {outgoing.Enqueue(new Packet{Name=name,Body=body});Monitor.PulseAll(gate);return;}
        }
        Fail("outgoing queue full");
    }
    public void ReceiveLoop() {
        try {
            using(var reader=new StreamReader(socket.GetStream())) {
                string line;
                while((line=reader.ReadLine())!=null) {
                    if(line.Length>8*1024*1024)throw new IOException("Oversized worker message");
                    lock(gate) {
                        if(closed)return;
                        if(incoming.Count>=capacity)throw new IOException("incoming queue full");
                        incoming.Enqueue(line);
                    }
                }
            }
        } catch(Exception e) {if(Connected)warning("receive: "+e.Message);}
        finally {Close();}
    }
    public bool TryReceive(out string line) {
        lock(gate) {line=incoming.Count==0?null:incoming.Dequeue();return line!=null;}
    }
    void WriteLoop() {
        try {
            using(var writer=new StreamWriter(socket.GetStream(),new UTF8Encoding(false)){AutoFlush=true,NewLine="\r\n"}) {
                while(true) {
                    Packet packet;
                    lock(gate) {
                        while(!closed && outgoing.Count==0)Monitor.Wait(gate);
                        if(closed)return;
                        packet=outgoing.Dequeue();
                    }
                    writer.WriteLine(packet.Name+" "+Convert.ToBase64String(Encoding.UTF8.GetBytes(serialize(packet.Body))));
                }
            }
        } catch(Exception e) {if(Connected)warning("send: "+e.Message);}
        finally {Close();}
    }
    void Fail(string message) {warning(message+"; reconnect required");Close();}
}
