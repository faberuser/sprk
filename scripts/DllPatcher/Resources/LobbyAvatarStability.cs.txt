public sealed class SprkLobbyAccessoryLease {
    object owner;
    public bool Pending { get; private set; }
    public void Bind(object value) { owner=value;Pending=false; }
    public void Cancel() { Pending=false; }
    public bool Matches(object value) { return value!=null && System.Object.ReferenceEquals(owner,value); }
    public bool Queue(object value) { if(!Matches(value))return false;Pending=true;return true; }
    public bool Take(object value,bool ready) { if(!Matches(value)||!Pending||!ready)return false;Pending=false;return true; }
}
public sealed class SprkLobbyAppearance {
    readonly CachedHeroInfo saved=new CachedHeroInfo();
    bool initialized;
    static bool SamePosition(PositionInfo a,PositionInfo b) {
        a=a??PositionInfo.Create();b=b??PositionInfo.Create();
        return a.PositionX==b.PositionX && a.PositionY==b.PositionY && a.PositionZ==b.PositionZ &&
            a.RotationX==b.RotationX && a.RotationY==b.RotationY && a.RotationZ==b.RotationZ && a.Scale==b.Scale;
    }
    // 1 means the body/hair/weapon needs rebuilding; 2 changes accessories only.
    public int Observe(CachedHeroInfo hero) {
        if(hero==null)return 0;
        int changed=initialized && (saved.HeroIndex!=hero.HeroIndex || saved.CostumeIndex!=hero.CostumeIndex ||
            saved.HairCostumeIndex!=hero.HairCostumeIndex || saved.WeaponCostumeIndex!=hero.WeaponCostumeIndex ||
            saved.HideUniqueWeapon!=hero.HideUniqueWeapon)?1:0;
        foreach(var part in AccessoryPartTypeHelper.PartTypes()) {
            var position=hero.GetAccessoryPositionInfo(part);
            if(initialized && (saved.GetAccessoryIndex(part)!=hero.GetAccessoryIndex(part) ||
                !SamePosition(saved.GetAccessoryPositionInfo(part),position)))changed|=2;
            saved.SetAccessoryCostume(part,hero.GetAccessoryIndex(part),position==null?null:position.Clone());
        }
        saved.HeroIndex=hero.HeroIndex;saved.CostumeIndex=hero.CostumeIndex;
        saved.HairCostumeIndex=hero.HairCostumeIndex;saved.WeaponCostumeIndex=hero.WeaponCostumeIndex;
        saved.HideUniqueWeapon=hero.HideUniqueWeapon;initialized=true;return changed;
    }
}
public sealed class SprkLobbyAccessoryState : MonoBehaviour {
    public readonly SprkLobbyAccessoryLease Lease=new SprkLobbyAccessoryLease();
    public readonly SprkLobbyAppearance Appearance=new SprkLobbyAppearance();
    public Avatar Target;
    public ICreatureLoad ObservedLoad;
    public ICreature ObservedCreature;
    public float Deadline;
    public float NextTry;
    // Retain the observed appearance while hidden so edits in the dressing
    // room are detected on return, even if its notifications were missed.
    void OnDisable() { Lease.Cancel();ObservedLoad=null;ObservedCreature=null;NextTry=0; }
    void Update() {
        if(Target==null)return;
        var owner=SprkLobbyAvatar.CurrentOwner();
        if(!Lease.Matches(owner)) { Lease.Bind(null);Target=null;return; }
        if(Time.realtimeSinceStartup<NextTry)return;
        NextTry=Time.realtimeSinceStartup+0.1f;
        int changed=Appearance.Observe(SprkLobbyAvatar.CurrentHero());
        if((changed&1)!=0) {
            SprkLobbyAvatar.Reload(GetComponent<MapLayout>(),Target);return;
        }
        if(changed!=0) {Lease.Queue(owner);Deadline=Time.realtimeSinceStartup+10f;}
        var facade=Target.RenderFacade;
        var load=facade==null?null:facade.AssetCreatureAsync;
        var creature=load==null?null:load.Value;
        if(!System.Object.ReferenceEquals(load,ObservedLoad) || !System.Object.ReferenceEquals(creature,ObservedCreature)) {
            ObservedLoad=load;ObservedCreature=creature;Lease.Queue(owner);Deadline=Time.realtimeSinceStartup+10f;
        }
        if(!Lease.Pending)return;
        if(SprkLobbyAvatar.Apply(Target)) { Lease.Take(owner,true); }
        else if(Time.realtimeSinceStartup>=Deadline) {
            Lease.Take(owner,true);
            Debug.LogWarning("SPRK lobby accessories: avatar not ready; kept current hero.");
        }
    }
}
public sealed class SprkProfilePortraitState : MonoBehaviour {
    public UIBundleLoader Target;
    public bool FullPortrait;
    float nextPoll;
    void OnEnable() {nextPoll=0;}
    public void Refresh() {
        if(Target==null || !GlobalAutoBehaviour<AccountManager>.isValidInstance)return;
        var account=GlobalAutoBehaviour<AccountManager>.instance;
        if(account.UserManager==null || account.HeroManager==null)return;
        int selected=account.UserManager.AvatarHeroIndex;
        string path=FullPortrait?SprkLobbyAvatar.GetProfilePortraitPath(selected):SprkLobbyAvatar.GetProfileIconPath(selected);
        if(!String.IsNullOrEmpty(path) && Target.AssetPath!=path)
            Target.SetAssetPath(path,FullPortrait?UIHelper.EAssetTextureAssign.ClipCenter:UIHelper.EAssetTextureAssign.__Ignore__);
    }
    void Update() {
        if(Time.realtimeSinceStartup<nextPoll)return;
        nextPoll=Time.realtimeSinceStartup+0.1f;
        Refresh();
    }
}
public static class SprkLobbyAvatar {
    public static int ProfileAvatarIndex(CachedHeroInfo hero,int fallback) {
        if(hero==null)return fallback;
        return hero.CostumeIndex>0?10000+hero.CostumeIndex:hero.HeroIndex*10+hero.Star;
    }
    public static string GetProfileIconPath(int selected) {
        return UIAvatarHelper.GetIconPath(ProfileAvatarIndex(CurrentHero(),selected));
    }
    public static string GetProfilePortraitPath(int selected) {
        return UIAvatarHelper.GetPortraitPath(ProfileAvatarIndex(CurrentHero(),selected));
    }
    static void WatchProfilePortrait(UIBundleLoader loader,bool full) {
        if(loader==null)return;
        var state=loader.GetComponent<SprkProfilePortraitState>();
        if(state==null)state=loader.gameObject.AddComponent<SprkProfilePortraitState>();
        state.Target=loader;state.FullPortrait=full;state.Refresh();
    }
    public static void WatchProfileIcon(LobbyTopmostMenu view) {WatchProfilePortrait(view.ProfileUi.BundleTexMyHero,false);}
    public static void WatchMyInfoPortrait(MyInfoWindow view) {WatchProfilePortrait(view.MyInfo.BundleTex_Avatar,true);}
    static readonly System.Reflection.MethodInfo rebuild=typeof(MapLayout).GetMethod("ChangePlayerAvatarRenderFacadeOnly",
        System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic,null,new[]{typeof(Avatar),typeof(int)},null);
    public static CachedHeroInfo CurrentHero() {
        if(CurrentOwner()==null)return null;
        var account=GlobalAutoBehaviour<AccountManager>.instance;
        if(account.HeroManager==null)return null;
        return account.HeroManager.Get(new UIAvatarHelper.AvatarInfo(account.UserManager.AvatarHeroIndex).HeroIndex);
    }
    public static bool IsAppearanceChange(UserHeroManager.ChangeHeroInfo change,int hero) {
        if(change==null || change.HeroIndex!=hero)return false;
        return change.ChangeType==UserHeroManager.HeroChangeType.Costume ||
            change.ChangeType==UserHeroManager.HeroChangeType.HairCostume ||
            change.ChangeType==UserHeroManager.HeroChangeType.WeaponCostume ||
            change.ChangeType==UserHeroManager.HeroChangeType.AccessoryCostume ||
            change.ChangeType==UserHeroManager.HeroChangeType.HideUniqueWeapon;
    }
    public static void Notify(MapLayout layout,UserHeroManager.ChangeHeroInfo change,bool forced) {
        if(forced || layout==null || layout.Mediator==null || CurrentOwner()==null)return;
        var hero=CurrentHero();
        if(hero!=null && IsAppearanceChange(change,hero.HeroIndex))Refresh(layout,layout.Mediator.Avatar,0);
    }
    public static void Reload(MapLayout layout,Avatar target) {
        if(layout==null || target==null || CurrentOwner()==null || !layout.gameObject.activeInHierarchy)return;
        // Use the native replacement so the same lobby avatar keeps its world
        // position/camera. Track runs after replacement and restores accessories.
        rebuild.Invoke(layout,new object[]{target,GlobalAutoBehaviour<AccountManager>.instance.UserManager.AvatarHeroIndex});
    }
    public static object CurrentOwner() {
        return GlobalAutoBehaviour<AccountManager>.isValidInstance?GlobalAutoBehaviour<AccountManager>.instance.UserManager:null;
    }
    public static void Bind(MapLayout layout) {
        var state=layout.GetComponent<SprkLobbyAccessoryState>();
        if(state==null)state=layout.gameObject.AddComponent<SprkLobbyAccessoryState>();
        state.Lease.Bind(CurrentOwner());state.Target=null;state.ObservedLoad=null;state.ObservedCreature=null;
    }
    public static void Track(MapLayout layout,bool forced) {
        if(layout==null)return;
        var state=layout.GetComponent<SprkLobbyAccessoryState>();
        if(forced) { if(state!=null){state.Lease.Cancel();state.Target=null;}return; }
        if(state!=null)state.Appearance.Observe(CurrentHero());
        if(layout.Mediator!=null)Refresh(layout,layout.Mediator.Avatar,0);
    }
    public static CachedHeroInfo Create(CachedHeroInfo hero,PlayerAvatarHeroInfo avatar) {
        if(hero==null)return null;
        var account=GlobalAutoBehaviour<AccountManager>.instance;
        var current=account.HeroManager.Get(hero.HeroIndex);
        if(current==null)return null;
        // The selected hero follows its live appearance, rather than the
        // snapshot saved by the last "set as main hero" request.
        var copy=current.CreatePlayerAvatarHeroInfo(null);
        foreach(var part in AccessoryPartTypeHelper.PartTypes()) {
            int index=current.GetAccessoryIndex(part);
            if(index!=0 && account.AccessoryCostumeManager.Get(new PlayerAccessoryCostumeInfoKey(current.HeroIndex,index))==null)index=0;
            var position=current.GetAccessoryPositionInfo(part);
            copy.SetAccessoryCostume(part,index,position==null?PositionInfo.Create():position.Clone());
        }
        return copy;
    }
    public static void Refresh(MapLayout layout,Avatar avatar,int avatarIndex) {
        if(layout==null || avatar==null || !layout.gameObject.activeInHierarchy)return;
        var state=layout.GetComponent<SprkLobbyAccessoryState>();
        if(state==null || !state.Lease.Queue(CurrentOwner()))return;
        state.Target=avatar;state.Deadline=Time.realtimeSinceStartup+10f;state.NextTry=0;
    }
    public static bool Apply(Avatar avatar) {
        if(avatar==null || CurrentOwner()==null)return false;
        var facade=avatar.RenderFacade;
        if(facade==null || facade.AssetCreatureAsync==null || !facade.AssetCreatureAsync.IsDone || !facade.AssetCreatureAsync.IsValid)return false;
        var account=GlobalAutoBehaviour<AccountManager>.instance;
        int heroIndex=new UIAvatarHelper.AvatarInfo(account.UserManager.AvatarHeroIndex).HeroIndex;
        var hero=account.HeroManager.Get(heroIndex);
        var data=GameRoot.Instance.CreatureDataContainer.GetData(heroIndex);
        if(hero==null || data==null || facade.AssetCreatureAsync.Stock.Code!=(String.IsNullOrEmpty(data.ModelCode)?data.CodeName:data.ModelCode))return false;
        var load=facade.AssetCreatureAsync;
        var desired=new List<CustomizeStock>();
        foreach(var part in AccessoryPartTypeHelper.PartTypes()) {
            int index=hero.GetAccessoryIndex(part);
            var stock=CustomizeStock.CreateEmptyPart(part);
            if(index!=0) {
                // Login can create the lobby before accessory ownership finishes
                // loading. Wait rather than interpreting that as an unequip.
                if(account.AccessoryCostumeManager.Get(new PlayerAccessoryCostumeInfoKey(heroIndex,index))==null)return false;
                var accessory=GameRoot.Instance.AccessoryCostumeDataContainer.GetData(index);
                var position=hero.GetAccessoryPositionInfo(part);
                if(accessory==null || !RenderFacadeBuilder.TryAssignCustomizeStock(ref stock,heroIndex,accessory,position==null?PositionInfo.Create():position.Clone()))return false;
            }
            desired.Add(stock);
        }
        return Reconcile(load,desired.ToArray());
    }
    public static bool Reconcile(ICreatureLoad load,CustomizeStock[] desired) {
        if(load==null || !load.IsDone || !load.IsValid || load.Value==null || desired==null)return false;
        var creature=load.Value;
        var open=creature.GetCreatureOpenData();
        if(open==null)return false;
        foreach(var stock in desired) {
            bool found=false;
            // CreatureLoad.Stock is the initial request; ChangeCustomizeStock
            // updates OpenData.Stock. Compare against the live stock instead.
            var values=open.GetStock().CustomizeStockValues();
            if(values!=null)foreach(var old in values)if(old.Part==stock.Part) {
                found=true;
                if(!old.IsEqualItem(stock) || !old.IsEqualTransform(stock) ||
                   (!stock.IsEmptyPart() && creature.GetCutomizeItemInstance(stock)==null))load.ChangeCustomizeStock(stock);
                break;
            }
            if(!found && !stock.IsEmptyPart())load.ChangeCustomizeStock(stock);
        }
        // A part's builder can still be unavailable on the first ready body
        // frame. Keep the queued retry until the live accessory instance exists.
        foreach(var stock in desired)
            if(!stock.IsEmptyPart() && creature.GetCutomizeItemInstance(stock)==null)return false;
        return true;
    }
}
