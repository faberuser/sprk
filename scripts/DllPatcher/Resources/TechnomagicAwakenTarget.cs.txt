using System;
using System.Collections.Generic;
using System.Reflection;
using NGame2.NAccount;
using NGame2.NUI.NComponent2;
using NGame2.NUI.NWindow;
using NShared;
using NVespa.NSingleton;
using UnityEngine;

// Lets the Technomagic awakening window select a target star. The native
// window prices one star; this helper replaces its single-star data with the
// summed cost of every step up to the chosen target, so native material/gold
// labels, colors, Enhance button state and stat preview all show the total.
public sealed class SprkTechnomagicAwaken : MonoBehaviour {
    const BindingFlags Any = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
    // Fits between the material icon and the NPC model at the window's scale.
    const int SliderWidth = 380;
    static readonly MethodInfo SetAfterStar = typeof(ValanceAwakenWindow).GetProperty("AfterStar").GetSetMethod(true);
    static readonly MethodInfo SetAfterStat = typeof(ValanceAwakenWindow).GetProperty("AfterStat").GetSetMethod(true);
    static readonly MethodInfo SetData = typeof(ValanceAwakenWindow).GetProperty("ValanceAwakenData").GetSetMethod(true);
    static readonly MethodInfo SetupMaterialIcon = typeof(ValanceAwakenWindow).GetMethod("SetupMaterialIcon", Any);
    static readonly FieldInfo ResultComp = typeof(ValanceAwakenWindow).GetField("_resultComp", Any);

    // Slot/target captured when Enhance is pressed and added by the request
    // encoder. A new selection or cleared target discards it.
    static int pendingSlot = -1;
    static int pendingTarget;

    ValanceAwakenWindow window;
    GameObject root;
    UISlider slider;
    UILabel valueLabel;
    readonly List<ValanceAwakenData> steps = new List<ValanceAwakenData>();
    int baseStar;
    int target;
    bool updating;

    // Called after native SetupResultIcon has priced BeforeStar + 1.
    public static void Setup(ValanceAwakenWindow window) {
        pendingSlot = -1;
        var state = window.GetComponent<SprkTechnomagicAwaken>();
        if(state == null) state = window.gameObject.AddComponent<SprkTechnomagicAwaken>();
        state.window = window;
        state.Load();
    }

    // Called before native ClearSeletedResult releases the target.
    public static void Clear(ValanceAwakenWindow window) {
        pendingSlot = -1;
        var state = window.GetComponent<SprkTechnomagicAwaken>();
        if(state == null) return;
        state.steps.Clear();
        state.Show(false);
    }

    // Called at the start of native OnClickAwaken.
    public static void Remember(ValanceAwakenWindow window) {
        pendingSlot = -1;
        var info = window.EquipItemInfo;
        if(info == null || window.ValanceAwakenData == null) return;
        if(window.AfterStar > window.BeforeStar + 1) {
            pendingSlot = info.SlotIndex;
            pendingTarget = window.AfterStar;
        }
    }

    // Called by JM_NShared_ValanceAwaken_Request.EncodeJsonObject.
    public static void Encode(Dictionary<string, object> json, NShared.ValanceAwaken.Request request) {
        if(json == null || request == null || pendingSlot < 0 || request.EquipItemSlotIndex != pendingSlot) return;
        json["TargetStar"] = pendingTarget.ToString();
    }

    void Load() {
        steps.Clear();
        var info = window.EquipItemInfo;
        var data = window.EquipItemData;
        var first = window.ValanceAwakenData;
        if(info == null || data == null || first == null) {Show(false);return;}
        baseStar = window.BeforeStar;
        int max = __Extension_Method_ContentsInfo_Item_Equip_MaxStar__.CLI_GetMaxStar_for_Awaken(info);
        var tag = data.TagType == null || data.TagType.Length == 0 ? default(CreatureTagType) : data.TagType[0];
        for(int star = baseStar + 1; star <= max; star++) {
            var row = GameRoot.Instance.ValanceAwakenDataContainer.GetData(
                new ValanceAwakenDataKey((byte)data.Tier, data.DetailIndex, tag, star));
            if(row == null || row.ItemCode1 != first.ItemCode1) break;
            steps.Add(row);
        }
        if(steps.Count < 2) {Show(false);return;}
        // Default to the highest star the player can currently afford.
        long gold = GlobalAutoBehaviour<AccountManager>.instance.UserInfo.Gold;
        long owned = GlobalAutoBehaviour<AccountManager>.instance.ItemManager.GetItemCount(first.ItemCode1);
        int best = baseStar + 1;
        long needGold = 0, needItems = 0;
        for(int i = 0; i < steps.Count; i++) {
            needGold += steps[i].ReqGold;
            needItems += steps[i].ItemCount1;
            if(needGold <= gold && needItems <= owned) best = baseStar + 1 + i;
        }
        EnsureSlider();
        Show(true);
        updating = true;
        slider.numberOfSteps = steps.Count;
        slider.value = (float)(best - baseStar - 1) / (steps.Count - 1);
        updating = false;
        // Native SelectedTargetItem refreshes labels, button and stats next.
        Apply(best, false);
    }

    void Apply(int star, bool refresh) {
        target = Mathf.Clamp(star, baseStar + 1, baseStar + steps.Count);
        long gold = 0, count = 0;
        for(int i = 0; i < target - baseStar; i++) {gold += steps[i].ReqGold;count += steps[i].ItemCount1;}
        var first = steps[0];
        var total = new ValanceAwakenData {
            Tier = first.Tier, DetailIndex = first.DetailIndex, CreatureTagType = first.CreatureTagType, Star = target,
            ReqGold = (int)Math.Min(gold, int.MaxValue), ItemCode1 = first.ItemCode1, ItemCount1 = (int)Math.Min(count, int.MaxValue)
        };
        var info = window.EquipItemInfo;
        SetData.Invoke(window, new object[] {total});
        SetAfterStar.Invoke(window, new object[] {target});
        SetAfterStat.Invoke(window, new object[] {StatCalculator.GetEquipItemStatData(info.ItemIndex, target, info.Level)});
        var icon = ResultComp.GetValue(window) as ItemIconComponent;
        if(icon != null) {
            var preview = info.Clone();
            preview.Star = target;
            preview.SoulWeaponInfo = info.SoulWeaponInfo;
            preview.Identified = 1;
            preview.ExtraOptionIndex1 = info.ExtraOptionIndex1;
            preview.ExtraOptionIndex2 = info.ExtraOptionIndex2;
            preview.ExtraOptionStep1 = info.ExtraOptionStep1;
            preview.ExtraOptionStep2 = info.ExtraOptionStep2;
            icon.SetStarBlink(false);
            icon.Open(window.ItemData, preview, (ItemIconComponent.Listener)null);
            icon.SetStarBlink();
        }
        if(valueLabel != null) valueLabel.text = "★" + target;
        if(!refresh) return;
        SetupMaterialIcon.Invoke(window, null);
        window.SetReqGoldLabel();
        window.SetButton();
        window.SetItemStat();
    }

    void OnSliderChanged() {
        if(updating || steps.Count < 2 || window == null) return;
        int star = baseStar + 1 + Mathf.RoundToInt(slider.value * (steps.Count - 1));
        if(star != target) Apply(star, true);
    }

    void Show(bool visible) {
        if(root != null && root.activeSelf != visible) root.SetActive(visible);
    }

    // Reuse the native Settings slider artwork (track, fill and thumb) and its
    // value label, placed beside the required-material icon.
    void EnsureSlider() {
        if(root != null) return;
        var group = window.LabelReqMaterialCount.transform.parent.parent;
        root = new GameObject("SprkTechnomagicTarget");
        root.transform.SetParent(group, false);
        root.transform.localPosition = new Vector3(5, 17, 0);
        var host = new GameObject("SprkTechnomagicSliderSource");
        host.SetActive(false);
        try {
            var source = Instantiate(GameOptionSliderComponent.PrefabAuto.Value.LoadGo, host.transform, false);
            slider = source.GetComponent<GameOptionSliderComponent>().Slider;
            slider.transform.SetParent(root.transform, false);
        } finally {
            Destroy(host);
        }
        // Native prefabs store sprite atlases by name; their root controller
        // normally assigns them on Awake. The lifted widgets left that root.
        foreach(var assigner in root.GetComponentsInChildren<NGame2.NUI.NAtlasAssignComponent.AtlasAssignerBase>(true))
            assigner.Assign();
        slider.name = "SprkTargetStarSlider";
        slider.onChange.Clear();
        EventDelegate.Add(slider.onChange, OnSliderChanged);
        // Keep the Settings slider's own proportions and draw order; only
        // detach anchors from its discarded row and place it in this window.
        int depth = window.LabelReqMaterialCount.depth + 1, lowest = int.MaxValue;
        var widgets = slider.GetComponentsInChildren<UIWidget>(true);
        foreach(var widget in widgets) lowest = Math.Min(lowest, widget.depth);
        foreach(var widget in widgets) {
            widget.leftAnchor.target = null; widget.rightAnchor.target = null;
            widget.topAnchor.target = null; widget.bottomAnchor.target = null;
            widget.depth = depth + widget.depth - lowest;
        }
        slider.transform.localPosition = Vector3.zero;
        slider.transform.localScale = Vector3.one;
        var bounds = NGUIMath.CalculateRelativeWidgetBounds(root.transform, slider.transform, true);
        float scale = bounds.size.x > SliderWidth ? SliderWidth / bounds.size.x : 1f;
        slider.transform.localScale = new Vector3(scale, scale, 1);
        slider.transform.localPosition = new Vector3(-bounds.center.x * scale, -bounds.center.y * scale, 0);
        // Star readout reuses this window's material-count label styling.
        var label = Instantiate(window.LabelReqMaterialCount.gameObject, root.transform, false);
        label.name = "SprkTargetStarLabel";
        valueLabel = label.GetComponent<UILabel>();
        valueLabel.color = window.ColorGoldEnough;
        valueLabel.pivot = UIWidget.Pivot.Left;
        valueLabel.overflowMethod = UILabel.Overflow.ResizeFreely;
        valueLabel.depth = depth;
        label.transform.localScale = Vector3.one;
        label.transform.localPosition = new Vector3(SliderWidth * .5f + 24, 0, 0);
        NGUITools.SetLayer(root, window.gameObject.layer);
        NGUITools.MarkParentAsChanged(root);
    }
}
