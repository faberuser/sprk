using System;
using System.Reflection;
using NShared;
using NShared.NMessage.NProtocol;
using NGame2.NAccount;
using NGame2.NBattleContext;
using NGame2.NBattleField.NManager.NPlay;
using NGame2.NParty;
using NGame2.NSocket;
using NGame2.NUI;
using NGame2.NUI.NManager;
using NGame2.NUI.NManager.NBattle;
using NGame2.NUI.NWindow;
using NGame2.NUI.NWindow.NBattle;
using NGame2;
using NVespa;
using NVespa.NGlobal;
using NVespa.NSingleton;
using NVespa.NSocket;

// Party Exit / vote Retry for cooperative PvE raids. Uses the native party
// exit button, battle popup, center notices and PartyBattleStart restart path.
public static class SprkPartyRetry
{
    static MessagePopup3Button votePopup;
    static int openVote;

    static CampaignContext Context()
    {
        if (!SingletonAuto<BattleContext>.isValidInstance) return null;
        return SingletonAuto<BattleContext>.instance.DataProvider as CampaignContext;
    }

    static bool Eligible(CampaignContext context)
    {
        if (context == null) return false;
        try { return SprkPartyRetryPolicy.Eligible(context.IsMultiplay(), (int)((IDataProvider)context).GetBattleType()); }
        catch (Exception) { return false; }
    }

    // CampaignContext.GetBattleMenuType return hook: add the native party exit
    // button to every eligible room raid (story party dungeons already have it).
    public static EBattleMenu Menu(EBattleMenu menu, CampaignContext context)
    {
        return Eligible(context) ? menu | EBattleMenu.PartyExitButton : menu;
    }

    static string Text(string key, string fallback)
    {
        string value;
        try { if (LocalizationManager.GetStringDirect(key, out value) && !string.IsNullOrEmpty(value)) return value; }
        catch (Exception) { }
        return fallback;
    }

    static string Failure(string code)
    {
        NShared.BeginCampaign.ResultType result;
        try { if (Enum.TryParse(code, out result)) return LocalizationManager.GetString(result); }
        catch (Exception) { }
        return code;
    }

    static void Notice(string message)
    {
        try { CenterMessage.Open(message); } catch (Exception) { }
    }

    // Same battle popup prefab as the native party exit confirmation, with
    // its own labels and the native countdown line.
    static MessagePopup3Button Popup(string title, string message, Action first, Action second, string firstLabel, string secondLabel, int seconds)
    {
        var prefab = new WindowPrefab<MessagePopup3Button>(MessagePopup3Button.PrefabPath_Battle, GameUIRoot.ELayerType.UI);
        var popup = prefab.instance;
        typeof(MessagePopup3Button).GetMethod("OpenInternal", BindingFlags.Instance | BindingFlags.NonPublic, null,
            new[] { typeof(string), typeof(string), typeof(Action), typeof(Action), typeof(string), typeof(string), typeof(bool) }, null)
            .Invoke(popup, new object[] { title, message, first, second, firstLabel, secondLabel, true });
        GlobalAutoBehaviour<PopupManegement>.instance.Add(prefab);
        if (seconds > 0)
            typeof(MessagePopup3Button).GetField("_remainedTime", BindingFlags.Instance | BindingFlags.NonPublic).SetValue(popup, (float)seconds);
        return popup;
    }

    static void Send(string message)
    {
        var connector = Connector();
        if (connector == null || connector.State != ConnectState.Connected)
        {
            Notice("Retry is not available right now.");
            return;
        }
        connector.Request(new BattleMessageReq { Message = message }, delegate(BattleMessageRes res) { });
    }

    static BattleSocketConnector Connector()
    {
        return GlobalAutoBehaviour<SocketServiceManager>.instance.BattleConnector;
    }

    // SystemButtonPartyExit.Listener.OnExit prefix. Returns true when handled.
    public static bool OnExit()
    {
        if (!Eligible(Context())) return false;
        Popup(Text("COMMON_NOTIFICATION", "Notice"),
            "Retry this battle with your party, or exit to the party room?\nRetry starts when every player agrees.",
            delegate { Send(SprkPartyRetryPolicy.Propose()); },
            delegate
            {
                MessagePopup3Button.Open_BattlePopup(Text("COMMON_NOTIFICATION", "Notice"),
                    "Leave this battle and return to the party room?\nYour heroes keep fighting under AI control.",
                    ExitToRoom, null, NShared.NLocalization.Common.COMMON_CONFIRM, NShared.NLocalization.Common.COMMON_CANCEL);
            },
            "Retry", Text("COMMON_EXIT", "Exit"), 0);
        return true;
    }

    // Withdraw from the battle but keep the party. The return state recorded at
    // battle start reopens the raid / party room, as after a finished battle.
    static void ExitToRoom()
    {
        bool left = false;
        Action leave = delegate
        {
            if (left) return;
            left = true;
            CloseVote();
            var context = Context();
            if (context != null) context.SetOnlineBattleCallbackIsBattleEnd(true);
            var current = Connector();
            if (current != null && current.State == ConnectState.Connected) current.Close();
            SingletonAuto<BattleContext>.instance.MoveSceneToLobby(null, EBoolCheck.DontCare);
        };
        var connector = Connector();
        if (connector == null || connector.State != ConnectState.Connected) { leave(); return; }
        connector.Request(new WithdrawBattleReq(), delegate(WithdrawBattleRes res) { leave(); });
        // Never strand the player if the battle server does not answer.
        GlobalAutoBehaviour<PartyManager>.instance.StartCoroutine(After(3f, leave));
    }

    static System.Collections.IEnumerator After(float seconds, Action action)
    {
        yield return new UnityEngine.WaitForSecondsRealtime(seconds);
        action();
    }

    // MessagePopup3Button.OnClickClose prefix: the Retry vote can only be
    // answered with Yes or No, never dismissed by Back.
    public static bool KeepOpen(MessagePopup3Button popup)
    {
        return popup != null && votePopup != null && ReferenceEquals(popup, votePopup);
    }

    // The battle popup prefab has a full-screen `Outside_ClickClose` layer
    // (dim sprite, collider and a close trigger). For the vote only, hide that
    // layer: outside taps then reach the battle UI underneath, so a player can
    // keep using skills while the vote is shown, and only Yes/No answer it.
    // The layer is shown again when the popup closes, for a reused instance.
    static void PassThroughOutside(MessagePopup3Button popup)
    {
        var hidden = new System.Collections.Generic.List<UnityEngine.GameObject>();
        foreach (var node in popup.GetComponentsInChildren<UnityEngine.Transform>(true))
            if (node.name == "Outside_ClickClose" && node.gameObject.activeSelf) hidden.Add(node.gameObject);
        foreach (var layer in hidden) layer.SetActive(false);
        if (hidden.Count > 0) GlobalAutoBehaviour<PartyManager>.instance.StartCoroutine(ShowOnClose(popup, hidden));
    }

    static System.Collections.IEnumerator ShowOnClose(MessagePopup3Button popup, System.Collections.Generic.List<UnityEngine.GameObject> hidden)
    {
        while (popup != null && popup.IsOpen()) yield return null;
        foreach (var layer in hidden)
            if (layer != null) layer.SetActive(true);
    }

    static void CloseVote()
    {
        try { if (votePopup != null && votePopup.IsOpen()) votePopup.Close(); } catch (Exception) { }
        votePopup = null;
        openVote = 0;
    }

    // OnlineBattleCallback.OnPacket(BattleMessageNot) prefix. Returns true when handled.
    public static bool OnMessage(BattleMessageNot data)
    {
        if (data == null) return false;
        var notice = SprkPartyRetryPolicy.Parse(data.Type, data.Content);
        if (notice.Kind == SprkPartyRetryPolicy.Kind.None) return data.Type == SprkPartyRetryPolicy.MessageType;
        switch (notice.Kind)
        {
            case SprkPartyRetryPolicy.Kind.Vote:
                CloseVote();
                if (notice.Proposer == GlobalAutoBehaviour<AccountManager>.instance.UserInfo.AccountId)
                {
                    Notice("Retry requested. Waiting for the other players to agree.");
                    break;
                }
                int vote = notice.Vote;
                openVote = vote;
                votePopup = Popup(Text("COMMON_NOTIFICATION", "Notice"),
                    notice.Who + " wants to retry this battle.\nDo you want to retry?",
                    delegate { openVote = 0; votePopup = null; Send(SprkPartyRetryPolicy.Answer(vote, true)); },
                    delegate { openVote = 0; votePopup = null; Send(SprkPartyRetryPolicy.Answer(vote, false)); },
                    LocalizationManager.GetString(NShared.NLocalization.Common.COMMON_YES),
                    LocalizationManager.GetString(NShared.NLocalization.Common.COMMON_NO), notice.Seconds);
                try { PassThroughOutside(votePopup); } catch (Exception) { }
                break;
            case SprkPartyRetryPolicy.Kind.Cancel:
                if (openVote == 0 || notice.Vote == 0 || notice.Vote == openVote) CloseVote();
                Notice(SprkPartyRetryPolicy.CancelMessage(notice, Text, Failure));
                break;
            case SprkPartyRetryPolicy.Kind.Restart:
                CloseVote();
                Restart(notice.Start);
                break;
        }
        return true;
    }

    // Every player, master included, follows the native PartyBattleStart path.
    // The server already created the new run, so each begin returns its entry.
    static void Restart(string json)
    {
        var start = JsonMarshaler.Parse<PartyBattleStart>(json);
        Notice("All players agreed. Restarting the battle.");
        var context = Context();
        if (context != null) context.SetOnlineBattleCallbackIsBattleEnd(true);
        var connector = Connector();
        if (connector != null && connector.State == ConnectState.Connected) connector.Close();
        typeof(PartyManager).GetMethod("Handle", BindingFlags.Instance | BindingFlags.NonPublic, null,
            new[] { typeof(long), typeof(PartyBattleStart) }, null)
            .Invoke(GlobalAutoBehaviour<PartyManager>.instance, new object[] { 0L, start });
    }
}
