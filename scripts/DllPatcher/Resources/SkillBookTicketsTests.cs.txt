public static class SkillBookTicketTests {
 sealed class Native:NGame2.NUI.NWindow.HeroSkillDesc.Listener {
  public List<string> Calls=new List<string>();
  public void ExtendSkill(int slot,int extend,bool enough){Calls.Add(slot+":"+extend+":"+enough);}
 }
 static void Check(bool b,string s){if(!b)throw new Exception("skill book tickets: "+s);}
 static NGame2.NUI.NWindow.SelectionItemUsePopup Prefilled(){
  var popup=new NGame2.NUI.NWindow.SelectionItemUsePopup();
  var choices=NGame2.NUI.NWindow.SelectionItemUsePopup.Rewards.Select(r=>new NGame2.NUI.NComponent2.ItemIconComponent{ItemIndex=r.Data.Index}).ToList();
  SprkHeroMaterialTickets.Prefill(popup,NGame2.NUI.NWindow.SelectionItemUsePopup.Listener,choices);
  return popup;
 }
 static NShared.UsePackageSelectItem.Response Ok(int used,int usedLeft,int got,int gotNow)=>new NShared.UsePackageSelectItem.Response{
  BaseResult=NShared.BaseResultType.Success,Result=NShared.UsePackageSelectItem.ResultType.Success,
  ItemResult=new NShared.ItemResultInfo{Index=used,Count=usedLeft},RewardItemResult=new NShared.ItemResultInfo{Index=got,Count=gotNow}};
 public static void Run(){
  var root=NShared.GameRoot.Instance;
  var items=NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.instance.ItemManager;
  var requester=NVespa.SingletonUpdater<NVespa.NGlobal.WebServiceManager>.instance.Requester;requester.SessionKey="s";
  NShared.ItemData Item(int index,string code,bool selector){var d=new NShared.ItemData{Index=index,Code=code,Name=code,Type=selector?NShared.ItemType.PackageSelectItem:NShared.ItemType.Other};root.ItemDataContainer.rows.Add(d);return d;}
  void Select(NShared.ItemData d,params int[] picks){root.SelectItemDataContainer.rows[d.Code]=new NShared.SelectItemData{ItemIndices=picks};}
  // Knight slot-2 book, its class ticket, slot ticket and the general ticket (live table shape).
  var book=Item(30002,"SKILLBOOK_KNIGHT_2",false);var other=Item(30007,"SKILLBOOK_WARRIOR_2",false);
  var classTicket=Item(30036,"SKILLBOOK_SELECT_KNIGHT",true);Select(classTicket,30001,30002,30003,30004);
  var slotTicket=Item(30044,"SKILLBOOK_SELECT_2",true);Select(slotTicket,30002,30007);
  var slot1=Item(30043,"SKILLBOOK_SELECT_1",true);Select(slot1,30001,30006);
  var general=Item(93538,"PACKAGE_SKILLBOOK_ALL",true);Select(general,30043,30044,30045,30046);
  var rune=Item(99901,"RUNE_PICK",true);root.SelectItemDataContainer.rows["RUNE_PICK"]=new NShared.SelectItemData{ItemIndices=new[]{30002},SelectType=NShared.SelectItemPopupType.RuneSelectItem};
  items.rows[rune.Index]=new NShared.ItemInfo{Count=50};
  NGame2.NAccount.Helper.Book=book;
  var desc=new NGame2.NUI.NWindow.HeroSkillDesc{ExtendSkills=new[]{
   new NGame2.NUI.NWindow.HeroSkillDesc.ExtendSkillUiGroup{ExtendInfo=new NGame2.NAccount.Helper.SkillExtendInfo{ExtendItemCount=10}},
   new NGame2.NUI.NWindow.HeroSkillDesc.ExtendSkillUiGroup{ExtendInfo=new NGame2.NAccount.Helper.SkillExtendInfo{ExtendItemCount=25}}}};
  var native=new Native();
  items.rows[book.Index]=new NShared.ItemInfo{Count=4};

  // Affordable tiers and players without usable tickets keep the native path.
  items.rows[book.Index].Count=12;SprkSkillBookTickets.Extend(native,2,1,true,desc);
  Check(native.Calls.Last()=="2:1:True","affordable tier changed");
  items.rows[book.Index].Count=4;
  SprkSkillBookTickets.Extend(native,2,1,false,desc);
  Check(native.Calls.Last()=="2:1:False","rune/no-ticket shortage did not use native prison prompt");
  // A tier lit by tickets that are gone since the last refresh must not send a native request.
  SprkSkillBookTickets.Extend(native,2,1,true,desc);
  Check(native.Calls.Last()=="2:1:False","ticket-lit tier without books or tickets sent the native request");
  Check(SprkSkillBookTickets.Usable(new NGame2.NAccount.Helper.SkillBookInfo{BookItemData=book,CurItemCount=4})==4,"tier status counted rune or missing tickets");
  items.rows[book.Index].Count=12;SprkSkillBookTickets.Extend(native,2,1,false,desc);
  Check(native.Calls.Last()=="2:1:True","stale shortage label blocked an affordable tier");
  items.rows[book.Index].Count=4;
  items.rows[general.Index]=new NShared.ItemInfo{Count=9,Locked=1};
  int calls=native.Calls.Count;SprkSkillBookTickets.Extend(native,2,1,false,desc);
  Check(native.Calls.Count==calls+1 && native.Calls.Last()=="2:1:False","locked general ticket offered");

  // General ticket alone: two native conversions, 1:1:1, then the native confirmation.
  items.rows[general.Index]=new NShared.ItemInfo{Count=9};
  NGame2.NUI.NWindow.SelectionItemUsePopup.Listener=null;
  Check(SprkSkillBookTickets.Usable(new NGame2.NAccount.Helper.SkillBookInfo{BookItemData=book,CurItemCount=4})==13 &&
   SprkSkillBookTickets.Usable(new NGame2.NAccount.Helper.SkillBookInfo{CurItemCount=4})==4,"tier status must count tickets as books");
  // The tier now shows as upgradable (enough=true) but real books are short: the ticket popup opens instead.
  calls=native.Calls.Count;SprkSkillBookTickets.Extend(native,2,1,true,desc);
  Check(native.Calls.Count==calls,"ticket-lit tier sent a native request without books");
  calls=native.Calls.Count;SprkSkillBookTickets.Extend(native,2,1,false,desc);
  Check(native.Calls.Count==calls,"general ticket shortage fell back to prison prompt");
  Check(NGame2.NUI.NWindow.SelectionItemUsePopup.Ticket.Data.Index==general.Index && NGame2.NUI.NWindow.SelectionItemUsePopup.Rewards.Single().Data.Index==book.Index,"general ticket popup must offer the needed book");
  var popup=Prefilled();
  Check(popup.InputSelectedCount.value=="6" && popup.Selected==book.Index,"shortage prefill/preselection");
  var listener=NGame2.NUI.NWindow.SelectionItemUsePopup.Listener;
  int requests=requester.Requests;
  listener.OnClickOk(popup,general.Index,other.Index,6);listener.OnClickOk(popup,slotTicket.Index,book.Index,6);listener.OnClickOk(popup,general.Index,book.Index,1001);
  Check(requester.Requests==requests,"mismatched or excessive conversion accepted");
  listener.OnClickOk(popup,general.Index,book.Index,6);listener.OnClickOk(popup,general.Index,book.Index,6);
  Check(requester.Requests==requests+1 && requester.Last.ItemIndex==general.Index && requester.Last.SelectItemIndex==slotTicket.Index && requester.Last.ItemCount==6 && !popup.ButtonOk.enabled,"general first step/duplicate click");
  int refreshes=desc.Refreshes;
  requester.Callback(Ok(general.Index,3,slotTicket.Index,6));
  Check(requester.Requests==requests+2 && requester.Last.ItemIndex==slotTicket.Index && requester.Last.SelectItemIndex==book.Index && requester.Last.ItemCount==6 && !popup.Closed,"general second step");
  var stale=requester.Callback;
  requester.Callback(Ok(slotTicket.Index,0,book.Index,10));stale(Ok(slotTicket.Index,0,book.Index,99));
  Check(items.Get(book.Index).Count==10 && items.Get(general.Index).Count==3 && items.Get(slotTicket.Index).Count==0,"conversion balances");
  Check(popup.Closed && popup.ButtonOk.enabled && desc.Refreshes==refreshes+1 && native.Calls.Last()=="2:1:True" && native.Calls.Count==calls+1,"converted tier did not reach the native confirmation");

  // Direct tickets come first; the chooser can still pick the general ticket.
  items.rows[classTicket.Index]=new NShared.ItemInfo{Count=2};items.rows[slotTicket.Index]=new NShared.ItemInfo{Count=1};items.rows[slot1.Index]=new NShared.ItemInfo{Count=5};
  var routes=SprkSkillBookTickets.Routes(book.Index);
  Check(routes.Select(r=>r.Ticket.Index).SequenceEqual(new[]{classTicket.Index,slotTicket.Index,general.Index}) && routes[2].Via==slotTicket.Index && routes[0].Via==0,"route order/membership");
  items.rows[slotTicket.Index].Locked=1;
  Check(SprkSkillBookTickets.Routes(book.Index).Select(r=>r.Ticket.Index).SequenceEqual(new[]{classTicket.Index}),"locked slot-ticket stack must block the chained route");
  items.rows[slotTicket.Index].Locked=0;
  NGame2.NUI.NWindow.SelectItemMessagePopup.Choose=null;
  SprkSkillBookTickets.Extend(native,2,2,false,desc);
  Check(NGame2.NUI.NWindow.SelectItemMessagePopup.Choose!=null,"multiple ticket chooser missing");
  NGame2.NUI.NWindow.SelectItemMessagePopup.Choose(null,classTicket.Index,1);
  Check(NGame2.NUI.NWindow.SelectionItemUsePopup.Ticket.Data.Index==classTicket.Index,"chooser ignored class ticket");
  popup=Prefilled();
  Check(popup.InputSelectedCount.value=="2","prefill must cap at owned tickets");
  listener=NGame2.NUI.NWindow.SelectionItemUsePopup.Listener;requests=requester.Requests;
  listener.OnClickOk(popup,classTicket.Index,book.Index,2);
  Check(requester.Requests==requests+1 && requester.Last.ItemIndex==classTicket.Index && requester.Last.SelectItemIndex==book.Index,"direct conversion");
  int errors=NGame2.NUI.NWindow.MessagePopup.Errors;
  requester.Callback(new NShared.UsePackageSelectItem.Response{BaseResult=NShared.BaseResultType.Success,Result=NShared.UsePackageSelectItem.ResultType.Failure});
  Check(!popup.Closed && popup.ButtonOk.enabled && NGame2.NUI.NWindow.MessagePopup.Errors==errors+1 && items.Get(classTicket.Index).Count==2,"first-step failure must stay retryable");
  calls=native.Calls.Count;
  listener.OnClickOk(popup,classTicket.Index,book.Index,2);requester.Callback(Ok(classTicket.Index,0,book.Index,12));
  Check(popup.Closed && native.Calls.Count==calls,"partial conversion must not open the upgrade confirmation");

  // Second-step failure closes (slot tickets were already granted); stale sessions are ignored.
  SprkSkillBookTickets.Extend(native,2,2,false,desc);
  NGame2.NUI.NWindow.SelectItemMessagePopup.Choose(null,general.Index,1);
  popup=Prefilled();listener=NGame2.NUI.NWindow.SelectionItemUsePopup.Listener;
  Check(popup.InputSelectedCount.value=="3","general prefill must cap at owned general tickets");
  listener.OnClickOk(popup,general.Index,book.Index,3);requester.Callback(Ok(general.Index,0,slotTicket.Index,4));
  errors=NGame2.NUI.NWindow.MessagePopup.Errors;
  requester.Callback(new NShared.UsePackageSelectItem.Response{BaseResult=NShared.BaseResultType.Success,Result=NShared.UsePackageSelectItem.ResultType.Failure});
  Check(popup.Closed && NGame2.NUI.NWindow.MessagePopup.Errors==errors+1 && items.Get(slotTicket.Index).Count==4,"second-step failure");
  items.rows[general.Index]=new NShared.ItemInfo{Count=5};
  SprkSkillBookTickets.Extend(native,2,2,false,desc);
  NGame2.NUI.NWindow.SelectItemMessagePopup.Choose(null,general.Index,1);
  popup=Prefilled();listener=NGame2.NUI.NWindow.SelectionItemUsePopup.Listener;
  requester.SessionKey="changed";requests=requester.Requests;listener.OnClickOk(popup,general.Index,book.Index,1);
  Check(requester.Requests==requests && popup.Closed,"stale session accepted");
  requester.SessionKey="s";

  // Shared pool for Upgrade All: one balance across slots, direct tickets first, general via its slot ticket.
  void Own(NShared.ItemData d,int n){items.rows[d.Index]=new NShared.ItemInfo{Count=n};}
  Own(book,0);Own(other,0);Own(classTicket,3);Own(slotTicket,4);Own(slot1,5);Own(general,10);
  var pool=new SprkSkillBookTickets.Pool();
  Check(pool.Available(book.Index)==17 && pool.Available(other.Index)==14,"pool availability");
  Check(pool.Take(book.Index,5)==5 && pool.Available(other.Index)==12 && pool.Available(book.Index)==12,"shared tickets counted twice");
  Check(pool.Take(other.Index,20)==12 && pool.Available(book.Index)==0,"pool exhaustion");
  Check(pool.Conversions.Select(c=>c.Ticket+">"+c.Via+">"+c.Book+"x"+c.Count).SequenceEqual(new[]{
   "30036>0>30002x3","30044>0>30002x2","30044>0>30007x2","93538>30044>30007x10"}),"pool conversions");

  // Counters show books plus every convertible ticket; ticket-only refreshes update changed counters only.
  var counter=new NGame2.NUI.NComponent2.UpgradeBookItemIconComponent{ItemData=book};
  counter.SetItemCount(0);
  Check(counter.Shown==17,"counter must include tickets");
  SprkSkillBookTickets.RefreshCounters(new[]{counter});
  Check(counter.Sets==1,"unchanged counter was reset");
  Own(general,11);SprkSkillBookTickets.RefreshCounters(new[]{counter});
  Check(counter.Sets==2 && counter.Shown==18,"ticket change did not refresh counter");
  Own(general,10);

  // Redemption: sequential native requests, each applied, two steps for a general ticket.
  bool? done=null;requests=requester.Requests;
  SprkSkillBookTickets.Redeem(pool.Conversions,ok=>done=ok);
  Check(requester.Last.ItemIndex==classTicket.Index && requester.Last.SelectItemIndex==book.Index && requester.Last.ItemCount==3,"redeem step 1");
  requester.Callback(Ok(classTicket.Index,0,book.Index,3));
  Check(requester.Last.ItemIndex==slotTicket.Index && requester.Last.SelectItemIndex==book.Index && requester.Last.ItemCount==2,"redeem step 2");
  requester.Callback(Ok(slotTicket.Index,2,book.Index,5));
  requester.Callback(Ok(slotTicket.Index,0,other.Index,2));
  Check(requester.Last.ItemIndex==general.Index && requester.Last.SelectItemIndex==slotTicket.Index && requester.Last.ItemCount==10,"redeem general first step");
  requester.Callback(Ok(general.Index,0,slotTicket.Index,10));
  Check(requester.Last.ItemIndex==slotTicket.Index && requester.Last.SelectItemIndex==other.Index && requester.Last.ItemCount==10 && done==null,"redeem general second step");
  requester.Callback(Ok(slotTicket.Index,0,other.Index,12));
  Check(done==true && requester.Requests==requests+5 && items.Get(book.Index).Count==5 && items.Get(other.Index).Count==12,"redeem completion");

  // Failure stops the chain and keeps earlier conversions; batches split at 1,000.
  Own(classTicket,2500);done=null;requests=requester.Requests;errors=NGame2.NUI.NWindow.MessagePopup.Errors;
  SprkSkillBookTickets.Redeem(new List<SprkSkillBookTickets.Conversion>{new SprkSkillBookTickets.Conversion{Ticket=classTicket.Index,Book=book.Index,Count=2500}},ok=>done=ok);
  Check(requester.Last.ItemCount==1000,"redeem batch size");
  requester.Callback(Ok(classTicket.Index,1500,book.Index,1005));
  Check(requester.Last.ItemCount==1000 && requester.Requests==requests+2,"redeem second batch");
  requester.Callback(new NShared.UsePackageSelectItem.Response{BaseResult=NShared.BaseResultType.Success,Result=NShared.UsePackageSelectItem.ResultType.Failure});
  Check(done==false && requester.Requests==requests+2 && items.Get(book.Index).Count==1005 && NGame2.NUI.NWindow.MessagePopup.Errors==errors+1,"redeem failure");
  done=null;Own(classTicket,0);
  SprkSkillBookTickets.Redeem(new List<SprkSkillBookTickets.Conversion>{new SprkSkillBookTickets.Conversion{Ticket=classTicket.Index,Book=book.Index,Count=1}},ok=>done=ok);
  Check(done==false,"redeem with missing tickets must stop before requesting");
  Console.WriteLine("Skill book ticket probe passed: native fallbacks, class/slot/general routes, locked stacks, prefill, ticket-lit tiers, 1:1:1 chaining, shared pool, counters, batch redemption, failures and sessions.");
 }
