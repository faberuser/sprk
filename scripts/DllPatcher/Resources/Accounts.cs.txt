    public static string[] GetHostTokens(string host) {

    public static string SecureHost(string host) {
        // SPRK exposes login and gameplay on the same reverse-proxy origin.
        // A schemeless public host uses HTTPS's default port, not HTTP port + 1.
        var uri=new Uri(host.Contains("://")?host:"https://"+host);
        if(uri.Scheme=="https")return uri.AbsoluteUri.TrimEnd('/');
        var secure=new UriBuilder(uri){Scheme="https",Port=uri.IsDefaultPort?443:uri.Port};
        return secure.Uri.AbsoluteUri.TrimEnd('/');
    }

    static SprkAccountStore Store {
        get {
            if(store==null) {
                string directory=Application.persistentDataPath;
                string profile=Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE");
                if(!String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR")) && !String.IsNullOrEmpty(profile) && System.Text.RegularExpressions.Regex.IsMatch(profile,@"\A[A-Za-z0-9_-]{1,64}\z"))
                    directory=Path.Combine(directory,"SprkAutomation",profile);
                string host=Environment.GetEnvironmentVariable("SPRK_HOST_URL");
                if(!String.IsNullOrEmpty(host)) {
                    // Namespace by server origin, not editable profile display name.
                    var origin=new Uri(ResolveHostUrl(host)).GetLeftPart(UriPartial.Authority);
                    string key;
                    using(var hash=SHA256.Create())key=BitConverter.ToString(hash.ComputeHash(Encoding.UTF8.GetBytes(origin))).Replace("-","");
                    string legacy=Path.Combine(directory,"sprk-accounts.dat");
                    directory=Path.Combine(directory,"SprkServers",key);
                    string file=Path.Combine(directory,"sprk-accounts.dat");
                    bool migrate=!File.Exists(file);
                    store=new SprkAccountStore(file);
                    if(migrate && File.Exists(legacy)) {
                        foreach(var account in new SprkAccountStore(legacy).Accounts)
                            if(String.Equals(SprkAccountStore.Endpoint(account.Endpoint),origin,StringComparison.OrdinalIgnoreCase))store.Upsert(account);
                    }
                } else store=new SprkAccountStore(Path.Combine(directory,"sprk-accounts.dat"));
            }
            return store;
        }
    }

    public static string ResolveHostUrl(string original) {
        string selected=Environment.GetEnvironmentVariable("SPRK_HOST_URL");
        if(String.IsNullOrEmpty(selected))return original;
        Uri uri;
        if(!Uri.TryCreate(selected,UriKind.Absolute,out uri) ||
            !(uri.Scheme=="https" || uri.Scheme=="http" && uri.IsLoopback) ||
            !String.IsNullOrEmpty(uri.UserInfo) || !String.IsNullOrEmpty(uri.Fragment))
            throw new InvalidOperationException("Invalid SPRK server profile URL");
        return uri.AbsoluteUri;
    }

    public static void SyncRefreshToken() {
        string token=LoginData.LastRefreshToken;
        if(activeAccount==null || String.IsNullOrEmpty(token) || activeAccount.RefreshToken==token)return;
        var updated=new SprkSavedAccount {Username=activeAccount.Username,ServerName=activeAccount.ServerName,Endpoint=activeAccount.Endpoint,RefreshToken=token};
        try {Store.Upsert(updated);activeAccount=updated;}catch(Exception){Debug.LogWarning("SPRK could not remember the renewed login.");}
    }

    }

    static void SwitchAccount() {

        switchingAccount=true;
        // Keep the encrypted saved login; only discard the currently loaded game session.
        SyncRefreshToken();activeAccount=null;
        SingletonUpdater<NVespa.NGlobal.WebServiceManager>.instance.Requester.Reset();
        ClearSession();LoginData.ClearLastLoginData();
        SingletonUpdater<LoginManager>.instance.LastLoginVerificationInfo.Value=String.Empty;
        if(GlobalAutoBehaviour<NGame2.NChatting.ChattingManager>.isValidInstance)
            GlobalAutoBehaviour<NGame2.NChatting.ChattingManager>.instance.ClearChatList(ChatType.None,0L);
        AccountManager.Logout();
        if(GlobalAutoBehaviour<NGame2.NGuild.GuildManager>.isValidInstance)
            GlobalAutoBehaviour<NGame2.NGuild.GuildManager>.instance.RemoveMyGuild();
        if(GlobalAutoBehaviour<NGame2.NOrdeal.OrdealArenaManager>.isValidInstance)
            GlobalAutoBehaviour<NGame2.NOrdeal.OrdealArenaManager>.instance.ResetAll();
        if(SceneAutoBehaviour<NGame2.NUI.NManager.MailManagement>.isValidInstance)
            SceneAutoBehaviour<NGame2.NUI.NManager.MailManagement>.instance.Reset();
        NGame2.NLogin.StateBase.Reset();
        SingletonUpdater<LoginManager>.instance.ReturnToLogin(NGame2.NScene.LoadingManager.ELoadingType.Default);
    }

    // Keep the existing game state machine, but its old guest transition now requires

    public static void Complete(MasangLoginManager.LoginResultCallback callback) {
        var result=pending;var tokens=pendingTokens;
        pending=null;pendingTokens=null;
        bool authenticated=result!=null && tokens!=null && !String.IsNullOrEmpty(tokens.access_token);
        if(authenticated) {
            // MasangLoginManager is created lazily and its Awake clears old tokens.
            // Restore this authenticated session before native progress managers initialize:
            // AccountManager.IsLogined() depends on LastAccessToken being populated.
            LoginData.LastAccessToken=tokens.access_token;
            LoginData.LastRefreshToken=tokens.refresh_token;
        }
        callback(authenticated,authenticated?result:null);
    }
    public static void Reject(MasangLoginManager.LoginResultCallback callback) { callback(false,null); }

    IEnumerator Authenticate(SprkSavedAccount saved=null) {
        var server=servers[serverIndex];
        SingletonUpdater<LoginManager>.instance.SetServerName(server.Name);
        string endpoint=server.LoginServer.TrimEnd('/')+"/api/auth/";
        string json=saved==null?JsonUtility.ToJson(new Request {username=username.value.Trim(),password=password.value}):"{}";
        using(var request=new UnityWebRequest(endpoint+(saved!=null?"refresh-token":registering?"register":"login"),"POST")) {
            request.uploadHandler=new UploadHandlerRaw(Encoding.UTF8.GetBytes(json));json=null;
            request.downloadHandler=new DownloadHandlerBuffer();request.timeout=25;
            request.SetRequestHeader("Content-Type","application/json");
            if(saved!=null)request.SetRequestHeader("Authorization","Bearer "+saved.RefreshToken);
            yield return request.SendWebRequest();
            TokenReply response=null;
            try {response=JsonUtility.FromJson<TokenReply>(request.downloadHandler.text);}catch(Exception){}
            if(request.result!=UnityWebRequest.Result.Success || response==null || response.data==null || response.status!="success" || String.IsNullOrEmpty(response.data.access_token) || String.IsNullOrEmpty(response.data.refresh_token)) {
                if(saved!=null && (request.responseCode==401 || request.responseCode==403)) {
                    try {Store.Remove(saved);}catch(Exception){Fail("Unable to update saved accounts. Please check local storage.");yield break;}
                    choosing=false;registering=false;username.value=saved.Username;password.value="";confirm.value="";
                    Fail("Your saved login has expired. Please enter your password.");yield break;
                }
                Debug.LogWarning("SPRK account request failed: HTTP "+request.responseCode+", "+request.error+", "+endpoint);
                Fail(response!=null && !String.IsNullOrEmpty(response.InternalErrorMessage)?response.InternalErrorMessage:"Unable to sign in. Check the server connection and try again.");yield break;
            }
            // Refresh tokens rotate: persist the replacement before verifying the game connection.
            var remembered=new SprkSavedAccount {Username=saved==null?username.value.Trim().ToLowerInvariant():saved.Username,
                ServerName=server.Name,Endpoint=SprkAccountStore.Endpoint(server.LoginServer),RefreshToken=response.data.refresh_token};
            try {Store.Upsert(remembered);}catch(InvalidOperationException error){Fail(error.Message);yield break;}catch(Exception){Fail("Unable to remember this login. Please check local storage and try again.");yield break;}
            using(var verify=new UnityWebRequest(endpoint+"token/verify","POST")) {
                verify.uploadHandler=new UploadHandlerRaw(Encoding.UTF8.GetBytes("{}"));verify.downloadHandler=new DownloadHandlerBuffer();verify.timeout=25;
                verify.SetRequestHeader("Content-Type","application/json");verify.SetRequestHeader("Authorization","Bearer "+response.data.access_token);
                yield return verify.SendWebRequest();SessionReply verified=null;
                try {verified=JsonUtility.FromJson<SessionReply>(verify.downloadHandler.text);}catch(Exception){}
                if(verify.result!=UnityWebRequest.Result.Success || verified==null || verified.status!="success" || verified.data==null || String.IsNullOrEmpty(verified.data.session)) {
                    Fail("Account authenticated, but the game connection failed. Please log in again.");yield break;
                }
                pending=new LoginSuccData {id=verified.data.id,sessionid=verified.data.session};
                pendingTokens=response.data;activeAccount=remembered;
            }
        }
        busy=false;
        password.value="";confirm.value="";
        listener.OnSelect(window,NShared.LoginMethodType.Guest,server.Name,true);
    }

        // Let the submitting Enter event finish before restoring focus. Keep
        // the value, selecting it so the next keystroke can replace it.

        field.selectionStart=0;field.selectionEnd=field.value.Length;
        // NGUI initializes the caret in its next Update; retain select-all
        // until that happens, then restore normal mouse/Tab caret behavior.
        yield return null;
        if(field!=null)field.selectAllTextOnFocus=false;
    }

// Separate from Unity serialization so the store also works with the prebuilt player.
public sealed class SprkSavedAccount {
    public string Username, ServerName, Endpoint, RefreshToken;
}
public sealed class SprkAccountStore {
    public const int MaxAccounts=4;
    readonly string path;
    public readonly List<SprkSavedAccount> Accounts=new List<SprkSavedAccount>();
    public static string Endpoint(string value) {return value.TrimEnd('/');}
    public SprkAccountStore(string file) {path=file;Reload();}
    List<SprkSavedAccount> Read() {
        var entries=new List<SprkSavedAccount>();
        if(!File.Exists(path))return entries;
        try {
            var bytes=ProtectedData.Unprotect(File.ReadAllBytes(path),null,DataProtectionScope.CurrentUser);
            using(var reader=new BinaryReader(new MemoryStream(bytes),Encoding.UTF8)) {
                if(reader.ReadInt32()!=1)throw new InvalidDataException();
                int count=reader.ReadInt32();if(count<0 || count>10000)throw new InvalidDataException();
                for(int i=0;i<count;i++) {
                    var entry=new SprkSavedAccount {Username=reader.ReadString(),ServerName=reader.ReadString(),Endpoint=reader.ReadString(),RefreshToken=reader.ReadString()};
                    if(String.IsNullOrEmpty(entry.Username) || String.IsNullOrEmpty(entry.Endpoint) || String.IsNullOrEmpty(entry.RefreshToken))throw new InvalidDataException();
                    entries.Add(entry);
                }
            }
        }catch(CryptographicException){entries.Clear();}
         catch(InvalidDataException){entries.Clear();}
         catch(EndOfStreamException){entries.Clear();}
        return entries;
    }
    public bool Reload() {
        // Atomic file replacement means readers see either complete version.
        var next=Read();bool changed=next.Count!=Accounts.Count;
        if(!changed)for(int i=0;i<next.Count;i++)if(!Same(next[i],Accounts[i]) || next[i].RefreshToken!=Accounts[i].RefreshToken){changed=true;break;}
        if(changed){Accounts.Clear();Accounts.AddRange(next);}return changed;
    }
    public static bool Same(SprkSavedAccount a,SprkSavedAccount b) {return a.Username==b.Username && a.Endpoint==b.Endpoint && a.ServerName==b.ServerName;}
    public bool CanSave(SprkSavedAccount account) {return Accounts.Count<MaxAccounts || Accounts.Exists(delegate(SprkSavedAccount old){return Same(old,account);});}
    FileStream LockStore() {
        Directory.CreateDirectory(Path.GetDirectoryName(path));
        for(int i=0;;i++) {
            try {return new FileStream(path+".lock",FileMode.OpenOrCreate,FileAccess.ReadWrite,FileShare.None);}
            catch(IOException){if(i>=100)throw;System.Threading.Thread.Sleep(10);}
        }
    }
    public IDisposable TryAcquire(SprkSavedAccount account) {
        string key;
        using(var hash=SHA256.Create())key=BitConverter.ToString(hash.ComputeHash(Encoding.UTF8.GetBytes(account.Endpoint+"\n"+account.ServerName+"\n"+account.Username))).Replace("-","");
        string directory=path+".sessions";Directory.CreateDirectory(directory);
        try {return new FileStream(Path.Combine(directory,key+".lock"),FileMode.OpenOrCreate,FileAccess.ReadWrite,FileShare.None);}
        catch(IOException){return null;}
    }
    public bool InUse(SprkSavedAccount account) {using(var lease=TryAcquire(account))return lease==null;}
    public void Upsert(SprkSavedAccount account) {
        using(LockStore()) {
            var next=Read();bool exists=next.Exists(delegate(SprkSavedAccount old){return Same(old,account);});
            if(!exists && next.Count>=MaxAccounts)throw new InvalidOperationException("Four accounts are already saved. Log out of an account before adding another.");
            next.RemoveAll(delegate(SprkSavedAccount old){return Same(old,account);});next.Insert(0,account);Save(next);
        }
    }
    public void Remove(SprkSavedAccount account) {
        using(LockStore()) {
            var next=Read();next.RemoveAll(delegate(SprkSavedAccount old){return Same(old,account);});Save(next);
        }
    }
    void Save(List<SprkSavedAccount> next) {
        byte[] clear;
        using(var buffer=new MemoryStream()) {
            using(var writer=new BinaryWriter(buffer,Encoding.UTF8,true)) {
                writer.Write(1);writer.Write(next.Count);
                foreach(var entry in next){writer.Write(entry.Username);writer.Write(entry.ServerName);writer.Write(entry.Endpoint);writer.Write(entry.RefreshToken);}
            }
            clear=buffer.ToArray();
        }
        byte[] encrypted=ProtectedData.Protect(clear,null,DataProtectionScope.CurrentUser);
        Array.Clear(clear,0,clear.Length);
        Directory.CreateDirectory(Path.GetDirectoryName(path));
        string temp=path+".tmp";
        try {
            File.WriteAllBytes(temp,encrypted);
            if(File.Exists(path))File.Replace(temp,path,null);else File.Move(temp,path);
            Accounts.Clear();Accounts.AddRange(next);
        }finally {if(File.Exists(temp))File.Delete(temp);}
    }
