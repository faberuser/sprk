    public static void Open(LoginSelect view, LoginSelect.Listener callback, NShared.LoginMethodType ignored, AppServerInfo[] choices, string lastServer) {
        view.Open();

        if (form == null) form = view.gameObject.AddComponent<SprkAccounts>();
        form.window = view; form.listener = callback; form.servers = choices;

    public static void CleanLoginBackground(LoginBackground view) {
        HidePatchVersion(view);
        foreach(var label in view.GetComponentsInChildren<UILabel>(true))
            if(label.name=="Label_Company")label.gameObject.SetActive(false);
        var config=NVespa.SingletonAuto<NGame2.NConfig.Config>.instance.ClientConfig;
        if(config!=null && view.Label_Version!=null) {
            // Assign the current version even if the native label contains an older
            // config/prefab value; exact string replacement can silently miss it.
            string caption="Ver. "+DisplayVersion(config.Version);
            int remaining;
            if(config.GetRemainExpireSecond(out remaining))caption+="."+(remaining/3600);
            if(!String.IsNullOrEmpty(SingletonUpdater<LoginManager>.instance.BinaryVersionNumber.Value))caption="[FF0000]"+caption+"[-]";
            view.Label_Version.text=caption;
        }
    }
    public static void HidePatchVersion(LoginBackground view) {

    static string DisplayVersion(string version) {
        int major,minor,patch;
        ClientConfig.ParseVersion(version,out major,out minor,out patch);
        return String.Format("{0}.{1:D2}.{2:D2}",major,minor,patch);
    }

    // Keep the existing game state machine, but its old guest transition now requires
    // a password-authenticated, short-lived ticket from this form.
    public static void Complete(MasangLoginManager.LoginResultCallback callback) {
        var result=pending;var tokens=pendingTokens;
        pending=null;pendingTokens=null;
        bool authenticated=result!=null && tokens!=null && !String.IsNullOrEmpty(tokens.access_token);
        if(authenticated) {
            // MasangLoginManager is created lazily and its Awake clears old tokens.
            // Restore this authenticated session before native progress managers initialize:
            // AccountManager.IsLogined() depends on LastAccessToken being populated.
            LoginData.LastAccessToken=tokens.access_token;
            LoginData.LastRefreshToken=tokens.refresh_token;
        }
        callback(authenticated,authenticated?result:null);
    }
    public static void Reject(MasangLoginManager.LoginResultCallback callback) { callback(false,null); }

    void RefreshLayout() {
        // Fit errors to their rendered text with compact, equal padding.
        // Keep the normal panel size unless wrapping needs additional room.
        float messageHeight=Mathf.Max(message.fontSize,Mathf.Ceil(message.printedSize.y));
        float messageTop=(registering?-245:-137)-26-26;
        message.transform.localPosition=new Vector3(0,messageTop-messageHeight*.5f,0);
        background.height=registering?786:678;
        if(!String.IsNullOrEmpty(message.text))
            background.height=Mathf.Max(background.height,Mathf.CeilToInt(440-(messageTop-messageHeight-26)));
        float centerY=440-background.height*.5f;
        root.transform.localPosition=new Vector3(0,-centerY,0);
        background.transform.localPosition=new Vector3(0,centerY,0);
        border.height=background.height+6;border.transform.localPosition=background.transform.localPosition;
    }
    void SetMessage(string text) {message.text=text;RefreshLayout();}
    void Update() {
        // The title screen's native camera does not consistently dispatch Tab.
        // Handle it once per form, leaving arrow keys to the text caret.
        if(busy || root==null || !root.activeInHierarchy || !UnityEngine.Input.GetKeyDown(KeyCode.Tab))return;
        bool reverse=UnityEngine.Input.GetKey(KeyCode.LeftShift) || UnityEngine.Input.GetKey(KeyCode.RightShift);
        var selected=UIInput.selection;
        if(selected==username)(reverse?(registering?confirm:password):password).isSelected=true;
        else if(selected==password)(reverse?username:(registering?confirm:username)).isSelected=true;
        else if(selected==confirm && registering)(reverse?password:username).isSelected=true;
    }
    void SetPasswordVisibility(UIInput input,UILabel label,bool visible) {
        input.inputType=visible?UIInput.InputType.Standard:UIInput.InputType.Password;
        input.UpdateLabel();label.text=visible?"Hide":"Show";
    }
    void HidePasswords() {SetPasswordVisibility(password,passwordVisibilityLabel,false);SetPasswordVisibility(confirm,confirmVisibilityLabel,false);}

    void ChangeServer() { if(!busy && servers!=null && servers.Length>0) {serverIndex=(serverIndex+1)%servers.Length;Refresh();} }
    void SwitchMode() { if(busy)return;registering=!registering;password.value="";confirm.value="";HidePasswords();SetMessage("");Refresh();username.isSelected=true; }
    void OnDisable() { if(password!=null){password.value="";confirm.value="";HidePasswords();} }
    void InputSubmitted() {
        if(busy)return;
        Submit();
    }
    void Submit() {
        if(busy)return;
        if(servers==null || servers.Length==0){Fail("No server available.");return;}
        if(username.value.Trim().Length<3 || password.value.Length<8){Fail("Enter a username (3-32 characters) and password (at least 8 characters).",username.value.Trim().Length<3?username:password);return;}
        if(registering && password.value!=confirm.value){Fail("Passwords do not match.",confirm);return;}
        busy=true;SetMessage("");HidePasswords();UICamera.selectedObject=null;Refresh();StartCoroutine(Authenticate());
    }
    IEnumerator Authenticate() {
        var server=servers[serverIndex];
        SingletonUpdater<LoginManager>.instance.SetServerName(server.Name);
        string endpoint=server.LoginServer.TrimEnd('/')+"/api/auth/";
        string json=JsonUtility.ToJson(new Request {username=username.value.Trim(),password=password.value});
        using(var request=new UnityWebRequest(endpoint+(registering?"register":"login"),"POST")) {
            request.uploadHandler=new UploadHandlerRaw(Encoding.UTF8.GetBytes(json));json=null;
            request.downloadHandler=new DownloadHandlerBuffer();request.timeout=25;
            request.SetRequestHeader("Content-Type","application/json");yield return request.SendWebRequest();
            TokenReply response=null;
            try {response=JsonUtility.FromJson<TokenReply>(request.downloadHandler.text);}catch(Exception){}
            if(request.result!=UnityWebRequest.Result.Success || response==null || response.data==null || response.status!="success") {
                Debug.LogWarning("SPRK account request failed: HTTP "+request.responseCode+", "+request.error+", "+endpoint);
                Fail(response!=null && !String.IsNullOrEmpty(response.InternalErrorMessage)?response.InternalErrorMessage:"Unable to sign in. Check the server connection and try again.");yield break;
            }
            using(var verify=new UnityWebRequest(endpoint+"token/verify","POST")) {
                verify.uploadHandler=new UploadHandlerRaw(Encoding.UTF8.GetBytes("{}"));verify.downloadHandler=new DownloadHandlerBuffer();verify.timeout=25;
                verify.SetRequestHeader("Content-Type","application/json");verify.SetRequestHeader("Authorization","Bearer "+response.data.access_token);
                yield return verify.SendWebRequest();SessionReply verified=null;
                try {verified=JsonUtility.FromJson<SessionReply>(verify.downloadHandler.text);}catch(Exception){}
                if(verify.result!=UnityWebRequest.Result.Success || verified==null || verified.data==null || String.IsNullOrEmpty(verified.data.session)) {
                    Fail("Account authenticated, but the game connection failed. Please log in again.");yield break;
                }
                pending=new LoginSuccData {id=verified.data.id,sessionid=verified.data.session};
                pendingTokens=response.data;
            }
        }
        busy=false;
        password.value="";confirm.value="";
        listener.OnSelect(window,NShared.LoginMethodType.Guest,server.Name,true);
    }
    void Fail(string error,UIInput field=null) {
        busy=false;SetMessage(error);Refresh();
        StartCoroutine(FocusForRetry(field==null?password:field,registering));
    }
    IEnumerator FocusForRetry(UIInput field,bool mode) {
        // Let the submitting Enter event finish before restoring focus. Keep
        // the value, selecting it so the next keystroke can replace it.
        yield return null;
        if(busy || root==null || !root.activeInHierarchy || registering!=mode || field==null || !field.isActiveAndEnabled)yield break;
        field.selectAllTextOnFocus=true;field.isSelected=true;
        field.selectionStart=0;field.selectionEnd=field.value.Length;
        // NGUI initializes the caret in its next Update; retain select-all
        // until that happens, then restore normal mouse/Tab caret behavior.
        yield return null;
        if(field!=null)field.selectAllTextOnFocus=false;
    }
