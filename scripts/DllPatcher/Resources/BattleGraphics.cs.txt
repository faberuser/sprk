// Sampled at 4 Hz. Six overloaded GPU samples lower scale by 25 points;
// twelve samples with headroom recover by 25. Never reacts to CPU-only stalls.
public sealed class SprkBattleResolution {
    int maximum=100,target,over,under;
    public int Current { get; private set; } =100;
    public void Configure(int scale,int fps) {
        if(maximum==scale && target==fps)return;
        maximum=scale;target=fps;Restore();
    }
    public void Restore() { Current=maximum;over=under=0; }
    public void Submit(double gpu,double cpu) {
        if(target==0 || double.IsNaN(gpu) || double.IsInfinity(gpu) || gpu<=0 || gpu>=250 || double.IsNaN(cpu) || double.IsInfinity(cpu)){over=under=0;return;}
        double budget=1000.0/target;
        over=gpu>budget*1.1 && gpu>=cpu*.9?over+1:0;
        under=gpu<budget*.7?under+1:0;
        if(over>=6){Current=Math.Max(75,Current-25);over=under=0;}
        else if(under>=12){Current=Math.Min(maximum,Current+25);over=under=0;}
    }
}
