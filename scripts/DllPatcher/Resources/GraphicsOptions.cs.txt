    void BeforeUI(Camera camera) {
        if(world==null || !world.IsCreated() || lastWorld==null || camera.depth<=lastWorld.depth || camera.targetTexture!=null || !IsOverlayCamera(camera) || presentationFrame==Time.frameCount)return;
        presentationFrame=Time.frameCount;presentationCamera=camera;
        // Execute after Unity binds/clears this camera's actual render target.
        // Graphics.Blit(..., null) in a camera callback can be overwritten by
        // Unity's subsequent target binding in the automatic camera loop.
        UnityEngine.Rendering.CommandBuffer command;
        if(!presentationCommands.TryGetValue(camera,out command)) {
            command=new UnityEngine.Rendering.CommandBuffer{name="SPRK world before UI"};
            presentationCommands[camera]=command;
            camera.AddCommandBuffer(UnityEngine.Rendering.CameraEvent.BeforeForwardOpaque,command);
        }
        command.Clear();command.Blit(world,UnityEngine.Rendering.BuiltinRenderTextureType.CameraTarget);
    }

    static void EnsurePreset() {
        if(preferencesReady)return;
        preferencesReady=true;
        // Preserve Off; every previous enabled preset migrates to TAA On.
        int enabled=1;
        if(SprkTestPrefs.HasKey("Sprk.Graphics.Preset"))enabled=SprkTestPrefs.GetInt("Sprk.Graphics.Preset",3)>0?1:0;
        else if(SprkTestPrefs.HasKey("Sprk.Graphics.PostAA"))enabled=SprkTestPrefs.GetInt("Sprk.Graphics.PostAA",0)>0 || SprkTestPrefs.GetInt("Sprk.Graphics.MSAA",0)>0?1:0;
        else if(SprkTestPrefs.HasKey("Sprk.Graphics.MSAA"))enabled=SprkTestPrefs.GetInt("Sprk.Graphics.MSAA",0)>0?1:0;
        ApplyPreset(enabled);
    }
    static void ApplyPreset(int preset) {
        // Keep legacy values 0/3 on disk so older clients still read Off/TAA.
        int mode=preset>0?3:0;
        SprkTestPrefs.SetInt("Sprk.Graphics.Preset",mode);
        SprkTestPrefs.SetInt("Sprk.Graphics.PostAA",mode);
        SprkTestPrefs.SetInt("Sprk.Graphics.MSAA",0);
        SprkTestPrefs.SetInt("Sprk.Graphics.Quality",2);
        SprkTestPrefs.SetInt("Sprk.Graphics.Jitter",2);

        SprkTestPrefs.SetInt("Sprk.Graphics.Stationary",3);
        SprkTestPrefs.SetInt("Sprk.Graphics.Motion",3);
        SprkTestPrefs.Save();
    }

    static bool ConfigureTarget(RenderTexture rt,TargetSize size,int samples) {
        int limit=Mathf.Min(SystemInfo.maxTextureSize,8192);
        float factor=Mathf.Min(Scale/100f,Mathf.Min(limit/(float)Mathf.Max(1,size.width),limit/(float)Mathf.Max(1,size.height)));
        int width=Mathf.Max(1,Mathf.RoundToInt(size.width*factor)),height=Mathf.Max(1,Mathf.RoundToInt(size.height*factor));
        var descriptor=rt.descriptor;descriptor.width=width;descriptor.height=height;descriptor.msaaSamples=samples;
        int supported=SystemInfo.GetRenderTextureSupportedMSAASampleCount(descriptor);
        if(rt.width==width && rt.height==height && rt.antiAliasing==supported)return false;
        bool created=rt.IsCreated();
        if(created)rt.Release();
        rt.width=width;rt.height=height;rt.antiAliasing=supported;
        if(created)rt.Create();
        return true;
    }
    public static RenderTexture CreateTarget(int width,int height,int depth) {
        var rt=new RenderTexture(Mathf.Max(1,width),Mathf.Max(1,height),depth);
        rt.filterMode=FilterMode.Bilinear;
        var size=new TargetSize(width,height);ownedTargets[rt]=size;
        ConfigureTarget(rt,size,Samples);
        return rt;
    }

        if(instance!=null)return;

    }

    static void PlaceBelowScreen(GameOptionWindow.TabUIInfo tab) {
        var rows=tab.GameOptionGrid.GetComponentsInChildren<GameOptionBaseComponent>(true);
        var screen=rows.FirstOrDefault(r=>r.gameOptionData!=null && r.gameOptionData.FieldName=="Graphic_WindowControl");
        var resolution=rows.FirstOrDefault(r=>r.gameOptionData!=null && r.gameOptionData.OptionName=="OPTION_GAME_RESOULUTION");
        var aa=rows.FirstOrDefault(r=>r.name=="SPRK Anti-aliasing");
        var scale=rows.FirstOrDefault(r=>r.name=="SPRK Render scale");
        var grid=tab.GameOptionGrid.Grid;
        if(screen==null || aa==null || scale==null || grid==null)return;
        bool hide=grid.hideInactive;grid.hideInactive=false;
        var order=grid.GetChildList();grid.hideInactive=hide;
        order.Remove(aa.transform);order.Remove(scale.transform);
        int index=order.IndexOf(screen.transform);
        if(index<0)return;
        // Resolution is Screen's native child option. Keep that pair adjacent
        // before inserting the additional graphics settings.
        if(resolution!=null && order.Remove(resolution.transform)) {
            index=order.IndexOf(screen.transform);
            order.Insert(index+1,resolution.transform);
            index++;
        }
        order.Insert(index+1,aa.transform);order.Insert(index+2,scale.transform);
        // Preserve the native visual order, with the two inserted rows, across
        // position/name sorting and subsequent native refreshes.
        grid.sorting=UIGrid.Sorting.None;
        typeof(UIGrid).GetField("sorted",BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic).SetValue(grid,false);
        for(int i=0;i<order.Count;i++)order[i].SetSiblingIndex(i);
    }

    static void ConfigureChoices(GameOptionToggle4Component row,string key,string[] choices) {
        var toggles=new List<UIToggle>{row.toggleInfo.Toggle1,row.toggleInfo.Toggle2,row.toggleInfo.Toggle3,row.toggleInfo.Toggle4};
        if(choices.Length==5) {
            var fifth=toggles[3].transform.parent.Find("Toggle5");
            if(fifth==null) {
                var clone=NGUITools.AddChild(toggles[3].transform.parent.gameObject,toggles[3].gameObject);clone.name="Toggle5";fifth=clone.transform;
            }
            toggles.Add(fifth.GetComponent<UIToggle>());
        }
        // Native prefab has four choices; unused choices must lose both
        // their visuals and callbacks when reducing an existing row to two.
        for(int i=0;i<toggles.Count;i++) {
            toggles[i].onChange.Clear();
            if(i>=choices.Length)toggles[i].Set(false);
            toggles[i].gameObject.SetActive(i<choices.Length);
        }
        toggles=toggles.Take(choices.Length).ToList();
        var optionGrid=toggles[0].transform.parent.GetComponent<UIGrid>();
        float spacing=choices.Length==2?280f:840f/(choices.Length-1);
        if(optionGrid!=null){optionGrid.cellWidth=spacing;optionGrid.maxPerLine=0;}
        bool syncing=false;
        Action refresh=delegate {
            syncing=true;int selected=ReadIndex(key);
            for(int i=0;i<toggles.Count;i++){toggles[i].startsActive=i==selected;toggles[i].Set(i==selected);}
            syncing=false;
        };
        for(int i=0;i<toggles.Count;i++) {
            int choice=i;var toggle=toggles[i];toggle.group=0;toggle.onChange.Clear();
            var label=toggle.GetComponentInChildren<UILabel>(true);label.text=choices[i];
            label.width=choices.Length==5?130:180;label.overflowMethod=UILabel.Overflow.ShrinkContent;
            toggle.transform.localPosition=new Vector3((choices.Length==2?i+1:i)*spacing,0,0);
            EventDelegate.Add(toggle.onChange,delegate {
                if(syncing)return;
                if(toggle.value) {
                    if(key=="Preset")ApplyPreset(choice);
                    else if(key=="Frame"){NVespa.SystemOptionData.Graphic_FrameControl=(EFrameControl)frameValues[choice];SprkTestPrefs.Save();}
                    else {SprkTestPrefs.SetInt("Sprk.Graphics.Scale",100+25*choice);SprkTestPrefs.Save();}
                    instance.ResetHistories();
                }
                refresh();
            });
        }
        refresh();
        if(key=="Preset")CenterChoices(new[]{row.toggleInfo.Toggle1,row.toggleInfo.Toggle2,row.toggleInfo.Toggle3,row.toggleInfo.Toggle4});
        else if(optionGrid!=null)optionGrid.Reposition();
    }
    static void CenterChoices(IEnumerable<UIToggle> toggles,Transform row=null,float center=420f) {
        var choices=toggles.Where(t=>t.gameObject.activeSelf).ToArray();
        if(choices.Length==0)return;
        var grid=choices[0].transform.parent.GetComponent<UIGrid>();
        if(grid!=null) {
            // Horizontal Reposition starts at zero and overwrites manual offsets.
            // Keep the explicit native Screen-style positions through Start,
            // delayed layout requests and subsequent settings refreshes.
            grid.arrangement=UIGrid.Arrangement.CellSnap;grid.cellWidth=0;grid.cellHeight=0;
            grid.pivot=UIWidget.Pivot.TopLeft;grid.animateSmoothly=false;
        }
        for(int i=0;i<choices.Length;i++) {
            var choice=choices[i].transform;
            if(row==null) {
                var position=choice.localPosition;
                position.x=center+(i-(choices.Length-1)*0.5f)*280f;
                choice.localPosition=position;
            } else {
                // Different native prefabs have different option-parent origins.
                // Align in row coordinates, not in each grid's local coordinates.
                var position=row.InverseTransformPoint(choice.position);
                position.x=center+(i-(choices.Length-1)*0.5f)*280f;
                choice.position=row.TransformPoint(position);
            }
        }
        if(grid!=null)grid.Reposition();
    }
    public static void CenterTextureChoices(GameOptionToggle3Component row) {
        if(row.gameOptionData==null || row.gameOptionData.FieldName!="Spec")return;
        var window=row.GetComponentInParent<GameOptionWindow>(true);
        if(window==null)return;
        var aa=window.GetComponentsInChildren<GameOptionToggle4Component>(true).FirstOrDefault(r=>r.name=="SPRK Anti-aliasing");
        if(aa==null)return; // Setup applies the alignment once AA exists.
        float center=(aa.transform.InverseTransformPoint(aa.toggleInfo.Toggle1.transform.position).x+aa.transform.InverseTransformPoint(aa.toggleInfo.Toggle2.transform.position).x)*0.5f;
        CenterChoices(new[]{row.toggleInfo.Toggle1,row.toggleInfo.Toggle2,row.toggleInfo.Toggle3},row.transform,center);
    }
    static int ReadIndex(string key) {
        if(key=="Frame")return Mathf.Max(0,Array.IndexOf(frameValues,(int)NVespa.SystemOptionData.Graphic_FrameControl));
        if(key=="Preset"){EnsurePreset();return SprkTestPrefs.GetInt("Sprk.Graphics.Preset",3)>0?1:0;}
        if(key=="PostAA"){EnsurePreset();return SprkTestPrefs.GetInt("Sprk.Graphics.PostAA",3)>0?3:0;}
        int n=SprkTestPrefs.GetInt("Sprk.Graphics."+key,key=="MSAA"?2:key=="Scale"?100:key=="Jitter"?2:key=="Sharpness"?1:key=="Stationary"||key=="Motion"?3:0);
        if(key=="MSAA")return n==2?1:n==4?2:n==8?3:0;
        if(key=="Scale")return Mathf.Clamp((n-100)/25,0,4);
        int maximum=key=="Quality"?2:key=="Sharpness"?6:key=="Stationary"||key=="Motion"?4:3;
        return Mathf.Clamp(n,0,maximum);
    }

    // Lobby models use orthographic NGUI cameras. Classify their actual
    // rendering layers, rather than assuming every UICamera only draws UI.
    static bool OnlyLayers(Camera camera,params string[] names) {
        int mask=0;foreach(var name in names){int layer=LayerMask.NameToLayer(name);if(layer>=0)mask|=1<<layer;}
        return camera.cullingMask!=0 && (camera.cullingMask & ~mask)==0;
    }
    static bool IsLobbyBackground(Camera camera) {
        return OnlyLayers(camera,"BottomUI","BottomFixedUI","Bottom3D","Bottom3DUI","Pet","BattleBackgroundUI");
    }
    static bool IsModelCamera(Camera camera) {
        return (camera.GetComponent<UICamera>()==null && !camera.orthographic && !OnlyLayers(camera,"BattleUI")) || OnlyLayers(camera,"Bottom3D","Pet");
    }
    static bool IsOverlayCamera(Camera camera) {
        return camera.GetComponent<UICamera>()!=null || OnlyLayers(camera,"BattleUI");
    }
    void LateUpdate() {
        RestoreTargets();
        int mode=ReadIndex("PostAA"),quality=ReadIndex("Quality");
        bool configured=SprkTestPrefs.HasKey("Sprk.Graphics.PostAA")||SprkTestPrefs.HasKey("Sprk.Graphics.MSAA")||SprkTestPrefs.HasKey("Sprk.Graphics.Scale");
        if(!configured)return;
        QualitySettings.antiAliasing=Samples==1?0:Samples;
        var cameras=Camera.allCameras.Where(c=>c!=null&&c.isActiveAndEnabled).OrderBy(c=>c.depth).ToArray();
        var ui=cameras.Where(c=>c.targetTexture==null && c.GetComponent<UICamera>()!=null && !IsLobbyBackground(c)).ToArray();
        float firstUi=ui.Length==0?float.PositiveInfinity:ui.Min(c=>c.depth);
        var onscreen=cameras.Where(c=>c.targetTexture==null && c.depth<firstUi && (c.GetComponent<UICamera>()==null || IsLobbyBackground(c))).ToArray();
        var worldCameras=onscreen.Where(IsModelCamera).ToArray();
        // Native Bottom3DUI contains map labels after the final model camera.
        // Keep that overlay on the display, outside the post-processed target.
        if(worldCameras.Length>0)onscreen=onscreen.Where(c=>c.depth<=worldCameras.Last().depth).ToArray();
        bool fullRect=onscreen.All(c=>c.rect==new Rect(0,0,1,1));
        if(worldCameras.Length>0 && fullRect) {
            int worldSamples=TargetSamples(worldCameras.First());
            int signature=Screen.width*397^Screen.height*17^Scale^worldSamples*31;
            if(world==null || signature!=worldSignature) {
                ReleaseWorld();world=CreateTarget(Screen.width,Screen.height,24);world.name="SPRK World";
                ConfigureTarget(world,new TargetSize(Screen.width,Screen.height),worldSamples);
                world.Create();worldSignature=signature;ResetHistories();
            }
            foreach(var c in onscreen){redirected[c]=null;c.targetTexture=world;c.allowMSAA=world.antiAliasing>1;}
            lastWorld=onscreen.Last();
            presentationCamera=cameras.FirstOrDefault(c=>c.depth>lastWorld.depth && c.targetTexture==null && !onscreen.Contains(c));
            if(lastLogged!=signature+mode*101) {
                lastLogged=signature+mode*101;
                Debug.Log("SPRK graphics: mode="+mode+" target="+world.width+"x"+world.height+" samples="+world.antiAliasing+" worldCameras="+onscreen.Length);
            }
        } else { ReleaseWorld(); }
        bool available=mode==0||FindResources();
        status=available?"Ready":"Native post-processing resources unavailable";
        foreach(var camera in cameras) {
            bool screen=onscreen.Contains(camera);
            if(camera.GetComponent<UICamera>()!=null && !screen)continue;
            var previewTarget=camera.targetTexture;
            TargetSize nativeSize;
            if(!screen && previewTarget!=null && ownedTargets.TryGetValue(previewTarget,out nativeSize))
                if(ConfigureTarget(previewTarget,nativeSize,TargetSamples(camera)))ResetHistories();
            camera.allowMSAA=camera.targetTexture!=null?camera.targetTexture.antiAliasing>1:Samples>1;
            if(!screen && camera.targetTexture==null)continue;
            // Spatial AA runs once after the world stack; preview cameras have independent targets.
            bool temporalWorld=mode==3 && worldCameras.Length>0;
            bool apply=!screen || (temporalWorld?camera==worldCameras.FirstOrDefault():camera==lastWorld);
            var layer=camera.GetComponent<PostProcessLayer>();
            if(layer==null && mode!=0 && apply && available) {
                layer=camera.gameObject.AddComponent<PostProcessLayer>();
                layer.Init(resources);layer.volumeTrigger=camera.transform;layer.volumeLayer=0;
                layer.fog.enabled=false;
            }
            if(layer==null)continue;
            layers[camera]=layer;
            // Preserve the image-effects output in the world texture for the
            // subsequent UI-camera presentation, including native PP layers.
            if(screen)layer.finalBlitToCameraTarget=false;
            // Enum order in Unity is None, FXAA, SMAA, TAA; validated by the patch verifier.
            int effective=apply&&available?mode:0;
            var target=camera.targetTexture;
            if(effective==3 && (!SystemInfo.supportsMotionVectors || !layer.temporalAntialiasing.IsSupported() || (target!=null && target.antiAliasing>1))){effective=0;status="TAA unavailable for camera "+camera.name;}
            int signature=effective*100000+quality*1000+(target==null?Screen.width:target.width)*31+(target==null?Screen.height:target.height);
            bool cut=positions.ContainsKey(camera) && (Vector3.Distance(positions[camera],camera.transform.position)>3 || Quaternion.Angle(rotations[camera],camera.transform.rotation)>35);
            if(!signatures.ContainsKey(camera)||signatures[camera]!=signature||cut)layer.ResetHistory();
            signatures[camera]=signature;positions[camera]=camera.transform.position;rotations[camera]=camera.transform.rotation;
            layer.antialiasingMode=(PostProcessLayer.Antialiasing)effective;
            layer.temporalAntialiasing.jitterSpread=(ReadIndex("Jitter")+1)*0.25f;
            layer.temporalAntialiasing.sharpness=ReadIndex("Sharpness")*0.5f;
            layer.temporalAntialiasing.stationaryBlending=new[]{.6f,.7f,.8f,.9f,.95f}[ReadIndex("Stationary")];
            layer.temporalAntialiasing.motionBlending=new[]{.5f,.6f,.7f,.8f,.85f}[ReadIndex("Motion")];
        }
        foreach(var dead in presentationCommands.Keys.Where(c=>c==null).ToArray()){presentationCommands[dead].Release();presentationCommands.Remove(dead);}
        foreach(var dead in ownedTargets.Keys.Where(t=>t==null).ToArray())ownedTargets.Remove(dead);
        foreach(var dead in layers.Keys.Where(c=>c==null).ToArray()){layers.Remove(dead);signatures.Remove(dead);positions.Remove(dead);rotations.Remove(dead);}
    }
