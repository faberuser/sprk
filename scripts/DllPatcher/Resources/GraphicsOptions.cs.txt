    void UpdateTemporalResources(Camera camera,PostProcessLayer layer,int effective) {
        TemporalResources state;
        if(!temporalResources.TryGetValue(camera,out state))temporalResources[camera]=state=new TemporalResources();
        if(effective==3) {
            if(!state.active)state.owned=(DepthTextureMode.Depth|DepthTextureMode.MotionVectors)&~camera.depthTextureMode;
            state.active=true;
        } else if(state.active) {
            releaseTemporalHistory.Invoke(layer.temporalAntialiasing,null);
            camera.depthTextureMode &= ~state.owned;
            state.active=false;state.owned=DepthTextureMode.None;
        }
    }

    void BeforeUI(Camera camera) {
        if(world==null || !world.IsCreated() || lastWorld==null || camera.depth<=lastWorld.depth || camera.targetTexture!=null || !IsOverlayCamera(camera) || presentationFrame==Time.frameCount)return;
        presentationFrame=Time.frameCount;
        if(presentationCamera!=camera) {
            foreach(var existing in presentationCommands.Values)existing.Clear();
            presentationCamera=camera;
        }
        // Execute after Unity binds/clears this camera's actual render target.
        // Graphics.Blit(..., null) in a camera callback can be overwritten by
        // Unity's subsequent target binding in the automatic camera loop.
        UnityEngine.Rendering.CommandBuffer command;
        if(!presentationCommands.TryGetValue(camera,out command)) {
            command=new UnityEngine.Rendering.CommandBuffer{name="SPRK world before UI"};
            presentationCommands[camera]=command;
            camera.AddCommandBuffer(UnityEngine.Rendering.CameraEvent.BeforeForwardOpaque,command);
        }
        if(command.sizeInBytes==0)command.Blit(world,UnityEngine.Rendering.BuiltinRenderTextureType.CameraTarget);
    }

    static void EnsurePreset() {
        if(preferencesReady)return;

        if(!SprkTestPrefs.HasKey("Sprk.Graphics.Method")) {
            int enabled=1;
            if(SprkTestPrefs.HasKey("Sprk.Graphics.Preset"))enabled=SprkTestPrefs.GetInt("Sprk.Graphics.Preset",3)>0?1:0;
            else if(SprkTestPrefs.HasKey("Sprk.Graphics.PostAA"))enabled=SprkTestPrefs.GetInt("Sprk.Graphics.PostAA",0)>0 || SprkTestPrefs.GetInt("Sprk.Graphics.MSAA",0)>0?1:0;

            SprkTestPrefs.SetInt("Sprk.Graphics.Method",enabled==0?0:4);
            SprkTestPrefs.Save();
        }
    }

        EnsurePreset();

        SprkTestPrefs.SetInt("Sprk.Graphics.Method",method);

        SprkTestPrefs.SetInt("Sprk.Graphics.PostAA",postModes[method]);
        SprkTestPrefs.SetInt("Sprk.Graphics.MSAA",method==3?Samples:0);

    static bool ConfigureTarget(RenderTexture rt,TargetSize size,int samples) {
        int scale=size.world?Scale:PreviewScale;
        if(size.scale==scale && size.samples==samples && rt.width==size.actualWidth && rt.height==size.actualHeight && rt.antiAliasing==size.actualSamples)return false;
        int limit=Mathf.Min(SystemInfo.maxTextureSize,8192);
        float factor=Mathf.Min(scale/100f,Mathf.Min(limit/(float)Mathf.Max(1,size.width),limit/(float)Mathf.Max(1,size.height)));
        int width=Mathf.Max(1,Mathf.RoundToInt(size.width*factor)),height=Mathf.Max(1,Mathf.RoundToInt(size.height*factor));
        var descriptor=rt.descriptor;descriptor.width=width;descriptor.height=height;descriptor.msaaSamples=samples;
        int supported=SystemInfo.GetRenderTextureSupportedMSAASampleCount(descriptor);
        size.scale=scale;size.samples=samples;size.actualWidth=width;size.actualHeight=height;size.actualSamples=supported;
        if(rt.width==width && rt.height==height && rt.antiAliasing==supported)return false;
        bool created=rt.IsCreated();
        if(created)rt.Release();
        rt.width=width;rt.height=height;rt.antiAliasing=supported;
        if(created)rt.Create();
        return true;
    }

    public static bool ApplyCustomResolution() {
        // Native settings callbacks can reapply a window preset while Full is
        // selected. Borderless fullscreen must retain the desktop dimensions.
        if(Application.platform==RuntimePlatform.WindowsPlayer && !NVespa.SystemOptionData.Graphic_WindowControl)return true;
        if(!CustomWindow)return false;
        Ensure();
        if(NVespa.SystemOptionData.Graphic_WindowControl) {
            int width=Mathf.Max(320,SprkTestPrefs.GetInt("Sprk.Window.Width",Screen.width));
            int height=Mathf.Max(240,SprkTestPrefs.GetInt("Sprk.Window.Height",Screen.height));
            Screen.SetResolution(width,height,FullScreenMode.Windowed);
        }
        return true;
    }
    IntPtr FindGameWindow() {
        if(gameWindow!=IntPtr.Zero && IsWindow(gameWindow))return gameWindow;
        gameWindow=IntPtr.Zero;resizeStyleCaptured=false;
        uint ownPid=(uint)System.Diagnostics.Process.GetCurrentProcess().Id;
        EnumWindows(delegate(IntPtr window,IntPtr state) {
            uint pid;GetWindowThreadProcessId(window,out pid);
            if(pid!=ownPid || !IsWindowVisible(window))return true;
            var name=new System.Text.StringBuilder(64);GetClassName(window,name,name.Capacity);
            if(name.ToString()!="UnityWndClass")return true;
            gameWindow=window;return false;
        },IntPtr.Zero);
        return gameWindow;
    }
    void UpdateCustomWindow() {
        if(Application.platform!=RuntimePlatform.WindowsPlayer || Screen.fullScreen)return;
        bool custom=CustomWindow;
        bool checkStyle=!resizeStyleCaptured || custom!=lastCustomWindow || Time.realtimeSinceStartup>=nextWindowStyleCheck;
        if(!checkStyle && !custom)return;
        var window=FindGameWindow();if(window==IntPtr.Zero)return;
        if(checkStyle || !resizeStyleCaptured) {
            nextWindowStyleCheck=Time.realtimeSinceStartup+.25f;lastCustomWindow=custom;
            int style=GetWindowLong(window,-16);
            if(!resizeStyleCaptured){originalResizeStyle=style&ResizeStyle;resizeStyleCaptured=true;}
            int desired=(style&~ResizeStyle)|(custom?ResizeStyle:originalResizeStyle);
            if(style!=desired) {
                WindowRect client,outer;
                if(GetClientRect(window,out client) && GetWindowRect(window,out outer)) {
                    SetWindowLong(window,-16,desired);
                    AdjustWindowRectEx(ref client,desired,false,GetWindowLong(window,-20));
                    SetWindowPos(window,IntPtr.Zero,outer.Left,outer.Top,client.Right-client.Left,client.Bottom-client.Top,0x20|0x4|0x10);
                }
            }
        }
        if(!custom || IsIconic(window) || IsZoomed(window))return;
        if(Screen.width<320 || Screen.height<240)return;
        if(savedWidth!=Screen.width || savedHeight!=Screen.height) {
            savedWidth=Screen.width;savedHeight=Screen.height;saveSizeAt=Time.realtimeSinceStartup+.5f;
        } else if(saveSizeAt>0 && Time.realtimeSinceStartup>=saveSizeAt) {
            SaveCustomSize();saveSizeAt=0;
        }
    }
    void SaveCustomSize() {
        if(savedWidth<320 || savedHeight<240)return;
        SprkTestPrefs.SetInt("Sprk.Window.Width",savedWidth);SprkTestPrefs.SetInt("Sprk.Window.Height",savedHeight);SprkTestPrefs.Save();
    }

    static void PlaceBelowScreen(GameOptionWindow.TabUIInfo tab) {
        var rows=tab.GameOptionGrid.GetComponentsInChildren<GameOptionBaseComponent>(true);
        var screen=rows.FirstOrDefault(r=>r.gameOptionData!=null && r.gameOptionData.FieldName=="Graphic_WindowControl");
        var resolution=rows.FirstOrDefault(r=>r.gameOptionData!=null && r.gameOptionData.OptionName=="OPTION_GAME_RESOULUTION");
        var aa=rows.FirstOrDefault(r=>r.name=="SPRK Anti-Aliasing");
        var scale=rows.FirstOrDefault(r=>r.name=="SPRK Render Scale");
        var grid=tab.GameOptionGrid.Grid;
        if(screen==null || aa==null || scale==null || grid==null)return;
        bool hide=grid.hideInactive;grid.hideInactive=false;
        var order=grid.GetChildList();grid.hideInactive=hide;
        var inserted=new List<Transform>{aa.transform};
        foreach(var title in adjustmentNames) {
            var row=rows.FirstOrDefault(r=>r.name=="SPRK "+title);if(row!=null)inserted.Add(row.transform);
        }
        inserted.Add(scale.transform);
        foreach(var title in new[]{"Preview Scale","Particle Density"}) {
            var extra=rows.FirstOrDefault(r=>r.name=="SPRK "+title);if(extra!=null)inserted.Add(extra.transform);
        }
        foreach(var title in new[]{"Lobby Camera Sensitivity","Model Camera Sensitivity"}) {
            var sensitivity=rows.FirstOrDefault(r=>r.name=="SPRK "+title);if(sensitivity!=null)inserted.Add(sensitivity.transform);
        }
        foreach(var row in inserted)order.Remove(row);
        int index=order.IndexOf(screen.transform);

            index++;
        }

        // position/name sorting and subsequent native refreshes.
        grid.sorting=UIGrid.Sorting.None;
        typeof(UIGrid).GetField("sorted",BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic).SetValue(grid,false);

    static void ConfigureChoices(GameOptionToggle4Component row,string key,string[] choices) {
        var toggles=new List<UIToggle>{row.toggleInfo.Toggle1,row.toggleInfo.Toggle2,row.toggleInfo.Toggle3,row.toggleInfo.Toggle4};
        if(choices.Length==5) {
            var fifth=toggles[3].transform.parent.Find("Toggle5");
            if(fifth==null) {
                var clone=NGUITools.AddChild(toggles[3].transform.parent.gameObject,toggles[3].gameObject);clone.name="Toggle5";fifth=clone.transform;
            }
            toggles.Add(fifth.GetComponent<UIToggle>());
        } else {
            var fifth=toggles[3].transform.parent.Find("Toggle5");
            if(fifth!=null){var extra=fifth.GetComponent<UIToggle>();extra.onChange.Clear();extra.Set(false);fifth.gameObject.SetActive(false);}
        }
        // Native prefab has four choices; unused choices must lose both
        // their visuals and callbacks when reducing an existing row to two.
        for(int i=0;i<toggles.Count;i++) {
            toggles[i].onChange.Clear();
            if(i>=choices.Length)toggles[i].Set(false);
            toggles[i].gameObject.SetActive(i<choices.Length);
        }
        toggles=toggles.Take(choices.Length).ToList();
        var optionGrid=toggles[0].transform.parent.GetComponent<UIGrid>();
        float spacing=choices.Length<=3?280f:840f/(choices.Length-1);
        if(optionGrid!=null){optionGrid.arrangement=UIGrid.Arrangement.Horizontal;optionGrid.pivot=UIWidget.Pivot.TopLeft;optionGrid.cellWidth=spacing;optionGrid.maxPerLine=0;optionGrid.hideInactive=true;}
        bool syncing=false;
        Action refresh=delegate {
            syncing=true;int selected=ReadIndex(key);
            for(int i=0;i<toggles.Count;i++){toggles[i].startsActive=i==selected;toggles[i].Set(i==selected);}
            syncing=false;
        };
        for(int i=0;i<toggles.Count;i++) {
            int choice=i;var toggle=toggles[i];toggle.group=0;toggle.onChange.Clear();
            var label=toggle.GetComponentInChildren<UILabel>(true);label.text=choices[i];
            label.width=choices.Length==5?130:180;label.overflowMethod=UILabel.Overflow.ShrinkContent;
            toggle.transform.localPosition=new Vector3((choices.Length==2?i+1:i)*spacing,0,0);
            EventDelegate.Add(toggle.onChange,delegate {
                if(syncing)return;
                if(toggle.value) {
                    if(key=="Preset")ApplyPreset(choice);
                    else if(key=="Frame"){NVespa.SystemOptionData.Graphic_FrameControl=(EFrameControl)frameValues[choice];SprkTestPrefs.Save();}
                    else if(key=="Resolution") {
                        if(CustomWindow)instance.SaveCustomSize();
                        NVespa.SystemOptionData.Graphic_ResolutionControl=(EResolutionControl)resolutionValues[choice];SprkTestPrefs.Save();
                    }
                    else {
                        int value=key=="Scale"?100+25*choice:key=="PreviewScale"?previewScales[choice]:IsSensitivityKey(key)?50+25*choice:key=="MSAASamples"?(SupportedSamples().Length==0?1:SupportedSamples()[choice]):choice;
                        SprkTestPrefs.SetInt("Sprk.Graphics."+key,value);SprkTestPrefs.Save();
                    }
                    if(!IsSensitivityKey(key))instance.ResetHistories();
                }
                refresh();
                if(key!="Frame" && key!="Resolution" && !IsSensitivityKey(key))RefreshRows(row.GetComponentInParent<GameOptionWindow>(true));
            });
        }
        refresh();
        if(choices.Length<=3)CenterChoices(toggles);
        else if(optionGrid!=null)optionGrid.Reposition();
    }
    static void CenterChoices(IEnumerable<UIToggle> toggles,Transform row=null,float center=420f) {
        var choices=toggles.Where(t=>t.gameObject.activeSelf).ToArray();
        if(choices.Length==0)return;
        var grid=choices[0].transform.parent.GetComponent<UIGrid>();
        if(grid!=null) {
            // Horizontal Reposition starts at zero and overwrites manual offsets.
            // Keep the explicit native Screen-style positions through Start,
            // delayed layout requests and subsequent settings refreshes.
            grid.arrangement=UIGrid.Arrangement.CellSnap;grid.cellWidth=0;grid.cellHeight=0;
            grid.pivot=UIWidget.Pivot.TopLeft;grid.animateSmoothly=false;
        }
        for(int i=0;i<choices.Length;i++) {
            var choice=choices[i].transform;
            if(row==null) {
                var position=choice.localPosition;
                position.x=center+(i-(choices.Length-1)*0.5f)*280f;
                choice.localPosition=position;
            } else {
                // Different native prefabs have different option-parent origins.
                // Align in row coordinates, not in each grid's local coordinates.
                var position=row.InverseTransformPoint(choice.position);
                position.x=center+(i-(choices.Length-1)*0.5f)*280f;
                choice.position=row.TransformPoint(position);
            }
        }
        if(grid!=null)grid.Reposition();
    }

    static bool IsSensitivityKey(string key) { return key=="LobbyCameraSensitivity" || key=="ModelCameraSensitivity"; }
    static int ReadIndex(string key) {
        if(key=="Resolution")return Mathf.Max(0,Array.IndexOf(resolutionValues,(int)NVespa.SystemOptionData.Graphic_ResolutionControl));
        if(key=="Frame")return Mathf.Max(0,Array.IndexOf(frameValues,(int)NVespa.SystemOptionData.Graphic_FrameControl));
        if(key=="Preset")return Method;
        if(key=="PostAA")return postModes[Method];
        if(IsSensitivityKey(key))return Mathf.Clamp((SprkTestPrefs.GetInt("Sprk.Graphics."+key,SprkTestPrefs.GetInt("Sprk.Graphics.CameraSensitivity",100))-50)/25,0,4);
        if(key=="Scale")return Mathf.Clamp((Scale-100)/25,0,4);
        if(key=="PreviewScale")return Mathf.Max(0,Array.IndexOf(previewScales,SprkTestPrefs.GetInt("Sprk.Graphics.PreviewScale",0)));
        if(key=="ParticleDensity")return Mathf.Clamp(SprkTestPrefs.GetInt("Sprk.Graphics.ParticleDensity",0),0,2);
        if(key=="MSAASamples")return Mathf.Max(0,Array.IndexOf(SupportedSamples(),SelectedSamples()));
        if(key=="FXAAQuality")return Mathf.Clamp(SprkTestPrefs.GetInt("Sprk.Graphics.FXAAQuality",1),0,1);
        if(key=="SMAAQuality")return Mathf.Clamp(SprkTestPrefs.GetInt("Sprk.Graphics.SMAAQuality",2),0,2);
        if(key=="TAASharpness")return Mathf.Clamp(SprkTestPrefs.GetInt("Sprk.Graphics.TAASharpness",1),0,2);
        return 0;
    }

    void RecordFrameCost(List<string> report,string label) {
        // Measures helper CPU overhead only, not GPU time or gameplay FPS.
        for(int i=0;i<20;i++)LateUpdate();
        var method=typeof(GC).GetMethod("GetAllocatedBytesForCurrentThread",BindingFlags.Public|BindingFlags.Static);
        var allocated=method==null?null:(Func<long>)Delegate.CreateDelegate(typeof(Func<long>),method);
        var timer=new System.Diagnostics.Stopwatch();
        long before=allocated==null?0:allocated();timer.Start();
        for(int i=0;i<500;i++)LateUpdate();
        timer.Stop();long bytes=allocated==null?-1:(allocated()-before)/500;
        report.Add("graphicsTick="+label+" cpuUs="+(timer.Elapsed.TotalMilliseconds*1000/500).ToString("F2",System.Globalization.CultureInfo.InvariantCulture)+" allocatedBytes="+bytes);
    }
    // Lobby models use orthographic NGUI cameras. Classify their actual
    // rendering layers, rather than assuming every UICamera only draws UI.
    static readonly int lobbyMask=LayerMask.GetMask("BottomUI","BottomFixedUI","Bottom3D","Bottom3DUI","Pet","BattleBackgroundUI");
    static readonly int modelMask=LayerMask.GetMask("Bottom3D","Pet");
    static readonly int battleUiMask=LayerMask.GetMask("BattleUI");
    static bool OnlyLayers(Camera camera,int mask) {
        return camera.cullingMask!=0 && (camera.cullingMask & ~mask)==0;
    }
    static bool IsLobbyBackground(Camera camera) {
        return OnlyLayers(camera,lobbyMask);
    }
    static bool IsModelCamera(Camera camera) {
        return (camera.GetComponent<UICamera>()==null && !camera.orthographic && !OnlyLayers(camera,battleUiMask)) || OnlyLayers(camera,modelMask);
    }
    static bool IsOverlayCamera(Camera camera) {
        return camera.GetComponent<UICamera>()!=null || OnlyLayers(camera,battleUiMask);
    }
    bool IsUI(Camera camera) {
        bool value;
        if(!uiCameras.TryGetValue(camera,out value) || !value)uiCameras[camera]=value=camera.GetComponent<UICamera>()!=null;

    RenderTexture NativeTarget(Camera camera) {

    void CollectCameras() {
        int count=Camera.allCamerasCount;
        if(cameraBuffer.Length<count)Array.Resize(ref cameraBuffer,Mathf.NextPowerOfTwo(count));
        count=Camera.GetAllCameras(cameraBuffer);
        cameras.Clear();onscreen.Clear();worldCameras.Clear();
        for(int i=0;i<count;i++)if(cameraBuffer[i]!=null && cameraBuffer[i].isActiveAndEnabled)cameras.Add(cameraBuffer[i]);
        Array.Clear(cameraBuffer,0,cameraBuffer.Length);
        // Camera stacks are small. Stable insertion sorting also preserves
        // Unity's order for equal-depth cameras without allocating a comparer.
        for(int i=1;i<cameras.Count;i++) {
            var current=cameras[i];int j=i-1;
            while(j>=0 && cameras[j].depth>current.depth){cameras[j+1]=cameras[j];j--;}
            cameras[j+1]=current;
        }
        float firstUi=float.PositiveInfinity;
        foreach(var camera in cameras)if(NativeTarget(camera)==null && IsUI(camera) && !IsLobbyBackground(camera)){firstUi=camera.depth;break;}
        foreach(var camera in cameras) {
            if(NativeTarget(camera)!=null || camera.depth>=firstUi || IsUI(camera) && !IsLobbyBackground(camera))continue;
            onscreen.Add(camera);
            if(IsModelCamera(camera))worldCameras.Add(camera);
        }
        if(worldCameras.Count>0) {
            float lastDepth=worldCameras[worldCameras.Count-1].depth;
            while(onscreen.Count>0 && onscreen[onscreen.Count-1].depth>lastDepth)onscreen.RemoveAt(onscreen.Count-1);
        }
    }

        foreach(var target in removeTargets)ownedTargets.Remove(target);

        foreach(var camera in removeCameras){layers.Remove(camera);signatures.Remove(camera);positions.Remove(camera);rotations.Remove(camera);temporalResources.Remove(camera);}
        removeCameras.Clear();foreach(var camera in uiCameras.Keys)if(camera==null)removeCameras.Add(camera);
        foreach(var camera in removeCameras)uiCameras.Remove(camera);

    void LateUpdate() {

        presentationFrame=-1;
        int mode=ReadIndex("PostAA"),quality=ReadIndex("SMAAQuality"),fxaa=ReadIndex("FXAAQuality"),sharpness=ReadIndex("TAASharpness");
        int scale=Scale,samples=Samples;
        bool configured=SprkTestPrefs.HasKey("Sprk.Graphics.PostAA")||SprkTestPrefs.HasKey("Sprk.Graphics.MSAA")||SprkTestPrefs.HasKey("Sprk.Graphics.Scale");
        if(!configured)return;
        // This Unity player requires global MSAA enabled for multisampled
        // camera rendering, even with an explicit RenderTexture sample count.
        // UI cameras opt out individually below.
        int globalSamples=samples==1?0:samples;
        if(QualitySettings.antiAliasing!=globalSamples)QualitySettings.antiAliasing=globalSamples;
        CollectCameras();

        // Keep that overlay on the display, outside the post-processed target.

        bool useWorld=worldCameras.Count>0 && fullRect && (scale!=100 || mode!=0 || samples>1);
        // Restore only cameras leaving the stack. Stable frames retain their
        // targets instead of binding null and the world target again.
        removeCameras.Clear();
        foreach(var pair in redirected)if(!useWorld || !onscreen.Contains(pair.Key)) {
            if(pair.Key!=null && pair.Key.targetTexture==world)pair.Key.targetTexture=pair.Value;

        foreach(var camera in removeCameras)redirected.Remove(camera);

            TargetSize worldSize;
            if(world==null || !world.IsCreated() || !ownedTargets.TryGetValue(world,out worldSize) || worldSize.width!=Screen.width || worldSize.height!=Screen.height || worldSize.scale!=scale || worldSize.samples!=worldSamples) {
                RestoreTargets();

                world.Create();ResetHistories();

            foreach(var c in onscreen){if(!redirected.ContainsKey(c))redirected[c]=c.targetTexture;if(c.targetTexture!=world)c.targetTexture=world;c.allowMSAA=world.antiAliasing>1;}
            lastWorld=onscreen[onscreen.Count-1];
            Camera nextPresenter=null;
            foreach(var camera in cameras)if(camera.depth>lastWorld.depth && camera.targetTexture==null && IsOverlayCamera(camera)){nextPresenter=camera;break;}
            if(presentationCamera!=nextPresenter){foreach(var command in presentationCommands.Values)command.Clear();presentationCamera=nextPresenter;}
            if(lastLogged!=signature+mode*101) {

            }
        } else if(world!=null || presentationCamera!=null) { RestoreTargets();ReleaseWorld(); }
        RestoreFinalBlits();
        bool available=mode==0||FindResources();
        status=available?"Ready":"Native post-processing resources unavailable";
        foreach(var camera in cameras) {
            bool screen=onscreen.Contains(camera);
            if((IsUI(camera) || OnlyLayers(camera,battleUiMask)) && !screen){if(camera.targetTexture==null)camera.allowMSAA=false;continue;}
            var previewTarget=camera.targetTexture;
            TargetSize nativeSize;
            if(!screen && previewTarget!=null && ownedTargets.TryGetValue(previewTarget,out nativeSize))
                if(ConfigureTarget(previewTarget,nativeSize,samples))ResetHistories();
            camera.allowMSAA=camera.targetTexture!=null && camera.targetTexture.antiAliasing>1;
            if(!screen && (camera.targetTexture==null || !ownedTargets.ContainsKey(camera.targetTexture)))continue;
            // Spatial AA runs once after the world stack; preview cameras have independent targets.
            bool temporalWorld=mode==3 && worldCameras.Count>0;
            bool apply=!screen || (temporalWorld?camera==worldCameras[0]:camera==lastWorld);
            PostProcessLayer layer;
            if(!layers.TryGetValue(camera,out layer) || layer==null)layer=camera.GetComponent<PostProcessLayer>();
            if(layer==null && mode!=0 && apply && available) {
                layer=camera.gameObject.AddComponent<PostProcessLayer>();
                layer.Init(resources);layer.volumeTrigger=camera.transform;layer.volumeLayer=0;
                layer.fog.enabled=false;
                helperLayers.Add(layer);
            }
            if(layer==null)continue;
            layers[camera]=layer;
            // Preserve the image-effects output in the world texture for the
            // subsequent UI-camera presentation, including native PP layers.
            if(screen && useWorld){finalBlitDefaults[layer]=layer.finalBlitToCameraTarget;layer.finalBlitToCameraTarget=false;}
            // Enum order in Unity is None, FXAA, SMAA, TAA; validated by the patch verifier.
            int effective=apply&&available?mode:0;
            var target=camera.targetTexture;
            if(effective==3 && (!SystemInfo.supportsMotionVectors || !layer.temporalAntialiasing.IsSupported() || (target!=null && target.antiAliasing>1))){effective=0;status="TAA unavailable for camera "+camera.name;}
            UpdateTemporalResources(camera,layer,effective);
            bool helper=helperLayers.Contains(layer);
            if(helper && layer.enabled!=(effective!=0))layer.enabled=effective!=0;
            int signature=effective*100000+quality*1000+fxaa*101+sharpness*103+(target==null?Screen.width:target.width)*31+(target==null?Screen.height:target.height);
            bool cut=effective==3 && positions.ContainsKey(camera) && (Vector3.Distance(positions[camera],camera.transform.position)>3 || Quaternion.Angle(rotations[camera],camera.transform.rotation)>35);
            bool changed=!signatures.ContainsKey(camera)||signatures[camera]!=signature;
            if(changed||cut)layer.ResetHistory();
            signatures[camera]=signature;positions[camera]=camera.transform.position;rotations[camera]=camera.transform.rotation;
            if(!changed)continue;
            layer.antialiasingMode=(PostProcessLayer.Antialiasing)effective;
            layer.fastApproximateAntialiasing.fastMode=fxaa==0;
            layer.fastApproximateAntialiasing.keepAlpha=true;
            layer.subpixelMorphologicalAntialiasing.quality=(SubpixelMorphologicalAntialiasing.Quality)quality;
            layer.temporalAntialiasing.jitterSpread=.75f;
            layer.temporalAntialiasing.sharpness=sharpness*.5f;
            layer.temporalAntialiasing.stationaryBlending=.9f;
            layer.temporalAntialiasing.motionBlending=.8f;
        }
        foreach(var pair in layers)if(pair.Key!=null && pair.Value!=null && (!cameras.Contains(pair.Key) || !onscreen.Contains(pair.Key) && (pair.Key.targetTexture==null || !ownedTargets.ContainsKey(pair.Key.targetTexture)))) {
            UpdateTemporalResources(pair.Key,pair.Value,0);
            if(helperLayers.Contains(pair.Value))pair.Value.enabled=false;
        }
        if(Time.frameCount%120==0)PruneCaches();
    }

// Only dense continuous billboard emitters are eligible. Bursts, manual Emit,
// trails, mesh particles, low-count systems and indicator-like nodes retain
// their native settings. Effect spawning, callbacks and lifetimes are untouched.
public sealed class SprkParticleDensity : MonoBehaviour {
    sealed class Emitter { public ParticleSystem system;public float rate; }
    readonly List<Emitter> emitters=new List<Emitter>();
    static readonly string[] protectedNames={"warning","telegraph","indicator","target","range","circle","decal","marker","area","aoe"};
    static bool Protected(Transform node,Transform root) {
        for(var current=node;current!=null;current=current.parent) {
            string name=current.name;
            foreach(var word in protectedNames)if(name.IndexOf(word,StringComparison.OrdinalIgnoreCase)>=0)return true;
            if(current==root)break;
        }
        return false;
    }
    public void Capture() {
        Clear();
        foreach(var system in GetComponentsInChildren<ParticleSystem>(true)) {
            var emission=system.emission;var main=system.main;var renderer=system.GetComponent<ParticleSystemRenderer>();
            if(renderer==null || renderer.renderMode!=ParticleSystemRenderMode.Billboard || !emission.enabled || emission.burstCount!=0 || main.maxParticles<128 ||
               emission.rateOverTime.mode!=ParticleSystemCurveMode.Constant || emission.rateOverTimeMultiplier<32 || system.trails.enabled || system.collision.enabled || system.trigger.enabled || system.subEmitters.enabled || system.lights.enabled || Protected(system.transform,transform))continue;
            emitters.Add(new Emitter{system=system,rate=emission.rateOverTimeMultiplier});
        }
    }
    public void Apply(float density) {
        foreach(var entry in emitters)if(entry.system!=null) {
            var emission=entry.system.emission;
            emission.rateOverTimeMultiplier=entry.rate*(gameObject.activeInHierarchy?density:1);
        }
    }
    public void Clear() { Apply(1);emitters.Clear(); }
    void OnDisable() { Apply(1); }
    void OnEnable() { Apply(SprkGraphicsOptions.ParticleDensity); }
}
