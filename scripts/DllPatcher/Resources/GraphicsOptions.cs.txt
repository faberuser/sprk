// Native UI controls and rendering configuration. Preferences use the isolated test adapter.
public sealed class SprkGraphicsOptions : MonoBehaviour {
    static SprkGraphicsOptions instance;
    static readonly FieldInfo resourceField=typeof(PostProcessLayer).GetField("m_Resources",BindingFlags.Instance|BindingFlags.NonPublic);
    PostProcessResources resources;
    readonly Dictionary<Camera,PostProcessLayer> layers=new Dictionary<Camera,PostProcessLayer>();
    readonly Dictionary<Camera,RenderTexture> redirected=new Dictionary<Camera,RenderTexture>();
    readonly Dictionary<Camera,int> signatures=new Dictionary<Camera,int>();
    readonly Dictionary<Camera,Vector3> positions=new Dictionary<Camera,Vector3>();
    readonly Dictionary<Camera,Quaternion> rotations=new Dictionary<Camera,Quaternion>();
    RenderTexture world;
    Camera lastWorld;
    int worldSignature;
    int lastLogged;
    string status="";
    public static string Status { get { return instance==null?"Not initialized":instance.status; } }
    void OnEnable() { Camera.onPostRender+=AfterCamera;SceneManager.activeSceneChanged+=SceneChanged; }
    void OnDisable() { Camera.onPostRender-=AfterCamera;SceneManager.activeSceneChanged-=SceneChanged;RestoreTargets();ReleaseWorld(); }
    void SceneChanged(Scene a,Scene b) { ResetHistories();RestoreTargets();ReleaseWorld(); }
    void ResetHistories() { foreach(var layer in layers.Values)if(layer!=null)layer.ResetHistory(); }
    void RestoreTargets() { foreach(var pair in redirected)if(pair.Key!=null && pair.Key.targetTexture==world)pair.Key.targetTexture=pair.Value;redirected.Clear();lastWorld=null; }
    void ReleaseWorld() { if(world!=null){world.Release();Destroy(world);world=null;} }
    void AfterCamera(Camera camera) {
        if(camera!=lastWorld||world==null)return;
        var previous=RenderTexture.active;
        // Blit(null) follows Camera.main.targetTexture. Detach our stack temporarily
        // so the resolved image is presented to the display rather than itself.
        var detached=redirected.Keys.Where(c=>c!=null && c.targetTexture==world).ToArray();
        foreach(var c in detached)c.targetTexture=null;
        try { Graphics.Blit(world,(RenderTexture)null); }
        finally { foreach(var c in detached)if(c!=null)c.targetTexture=world;RenderTexture.active=previous; }
    }
    bool FindResources() {
        if(resources!=null)return true;
        var prefab=Resources.Load<GameObject>("Prefab2/Render3D/RenderTargetCinema");
        if(prefab!=null) {
            var source=prefab.GetComponentInChildren<PostProcessLayer>(true);
            if(source!=null)resources=(PostProcessResources)resourceField.GetValue(source);
        }
        if(resources==null)resources=Resources.FindObjectsOfTypeAll<PostProcessResources>().FirstOrDefault();
        return resources!=null && resources.shaders!=null && resources.smaaLuts!=null && resources.smaaLuts.area!=null && resources.smaaLuts.search!=null;
    }
    // Compatibility entry point for previously patched clients; temporary AA
    // comparison input has been removed.
    public static void PollComparisonKeys(HeroFullSceneViewBase window) { }
    static bool preferencesReady;
    static void EnsurePreset() {
        if(preferencesReady)return;
        preferencesReady=true;
        if(SprkTestPrefs.HasKey("Sprk.Graphics.Preset"))return;
        int preset=2;
        if(SprkTestPrefs.HasKey("Sprk.Graphics.PostAA")) {
            int old=SprkTestPrefs.GetInt("Sprk.Graphics.PostAA",0);
            preset=old==3?3:old==2?2:old==1?1:
                SprkTestPrefs.HasKey("Sprk.Graphics.MSAA")&&SprkTestPrefs.GetInt("Sprk.Graphics.MSAA",0)>0?2:0;
        } else if(SprkTestPrefs.HasKey("Sprk.Graphics.MSAA"))preset=SprkTestPrefs.GetInt("Sprk.Graphics.MSAA",0)>0?2:0;
        ApplyPreset(preset);
    }
    static void ApplyPreset(int preset) {
        preset=Mathf.Clamp(preset,0,3);
        SprkTestPrefs.SetInt("Sprk.Graphics.Preset",preset);
        SprkTestPrefs.SetInt("Sprk.Graphics.PostAA",new[]{0,1,2,3}[preset]);
        SprkTestPrefs.SetInt("Sprk.Graphics.MSAA",preset==2?2:0);
        SprkTestPrefs.SetInt("Sprk.Graphics.Quality",preset<2?0:preset==2?1:2);
        SprkTestPrefs.SetInt("Sprk.Graphics.Jitter",2);
        SprkTestPrefs.SetInt("Sprk.Graphics.Sharpness",1);
        SprkTestPrefs.SetInt("Sprk.Graphics.Stationary",3);
        SprkTestPrefs.SetInt("Sprk.Graphics.Motion",3);
        SprkTestPrefs.Save();
    }
    public static int Samples { get { EnsurePreset(); if(SprkTestPrefs.GetInt("Sprk.Graphics.PostAA",0)==3)return 1; int n=SprkTestPrefs.GetInt("Sprk.Graphics.MSAA",2); return n==2||n==4||n==8?n:1; } }
    public static int Scale { get { return Mathf.Clamp(SprkTestPrefs.GetInt("Sprk.Graphics.Scale",100),100,200); } }
    public static RenderTexture CreateTarget(int width,int height,int depth) {
        int limit=Mathf.Min(SystemInfo.maxTextureSize,8192);
        float factor=Mathf.Min(Scale/100f,Mathf.Min(limit/(float)Mathf.Max(1,width),limit/(float)Mathf.Max(1,height)));
        var rt=new RenderTexture(Mathf.Max(1,Mathf.RoundToInt(width*factor)),Mathf.Max(1,Mathf.RoundToInt(height*factor)),depth);
        rt.filterMode=FilterMode.Bilinear;
        var descriptor=rt.descriptor; descriptor.msaaSamples=Samples;
        rt.antiAliasing=SystemInfo.GetRenderTextureSupportedMSAASampleCount(descriptor);
        return rt;
    }
    public static void Ensure() {
        if(instance!=null)return;
        EnsurePreset();
        var go=new GameObject("SprkGraphicsOptions"); DontDestroyOnLoad(go);
        instance=go.AddComponent<SprkGraphicsOptions>();
        if(!String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_GRAPHICS_TEST")) && !String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE")) && !String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR")))instance.StartCoroutine(instance.Probe());
    }
    public static void Setup(GameOptionWindow window) {
        Ensure();
        foreach(var tab in window.MainTabs) {
            if(tab.GameOptionTab!=GameOptionTab.Game)continue;
            Add(tab,"Anti-aliasing", "Preset",new[]{"Off","Performance","Balanced","Quality"});
            Add(tab,"Render scale", "Scale",new[]{"100%","125%","150%","175%","200%"});
            PlaceBelowScreen(tab);
            tab.GameOptionGrid.RepositionGrid();
        }
    }
    static void PlaceBelowScreen(GameOptionWindow.TabUIInfo tab) {
        var rows=tab.GameOptionGrid.GetComponentsInChildren<GameOptionBaseComponent>(true);
        var screen=rows.FirstOrDefault(r=>r.gameOptionData!=null && r.gameOptionData.FieldName=="Graphic_WindowControl");
        var resolution=rows.FirstOrDefault(r=>r.gameOptionData!=null && r.gameOptionData.OptionName=="OPTION_GAME_RESOULUTION");
        var aa=rows.FirstOrDefault(r=>r.name=="SPRK Anti-aliasing");
        var scale=rows.FirstOrDefault(r=>r.name=="SPRK Render scale");
        var grid=tab.GameOptionGrid.Grid;
        if(screen==null || aa==null || scale==null || grid==null)return;
        bool hide=grid.hideInactive;grid.hideInactive=false;
        var order=grid.GetChildList();grid.hideInactive=hide;
        order.Remove(aa.transform);order.Remove(scale.transform);
        int index=order.IndexOf(screen.transform);
        if(index<0)return;
        // Resolution is Screen's native child option. Keep that pair adjacent
        // before inserting the additional graphics settings.
        if(resolution!=null && order.Remove(resolution.transform)) {
            index=order.IndexOf(screen.transform);
            order.Insert(index+1,resolution.transform);
            index++;
        }
        order.Insert(index+1,aa.transform);order.Insert(index+2,scale.transform);
        // Preserve the native visual order, with the two inserted rows, across
        // position/name sorting and subsequent native refreshes.
        grid.sorting=UIGrid.Sorting.None;
        typeof(UIGrid).GetField("sorted",BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic).SetValue(grid,false);
        for(int i=0;i<order.Count;i++)order[i].SetSiblingIndex(i);
    }
    static void Add(GameOptionWindow.TabUIInfo tab,string title,string key,string[] choices) {
        string rowName="SPRK "+title;
        var existing=tab.GameOptionGrid.GetComponentsInChildren<GameOptionBaseComponent>(true).Where(r=>r.name==rowName).ToArray();
        var row=existing.OfType<GameOptionToggle4Component>().FirstOrDefault();
        foreach(var duplicate in existing.Where(r=>r!=row)) {
            duplicate.gameObject.SetActive(false);duplicate.transform.SetParent(null,false);Destroy(duplicate.gameObject);
        }
        if(row==null)row=UIHelper.AddGridItem<GameOptionToggle4Component>(tab.GameOptionGrid,GameOptionToggle4Component.PrefabAuto);
        row.name=rowName;row.enabled=false;
        row.ParentRoot.Root.SetActive(true);row.ChildRoot.Root.SetActive(false);
        row.ParentRoot.Label_OptionName.text=title;
        if(row.ParentRoot.ToolTipImage!=null)row.ParentRoot.ToolTipImage.SetActive(false);
        var toggles=new List<UIToggle>{row.toggleInfo.Toggle1,row.toggleInfo.Toggle2,row.toggleInfo.Toggle3,row.toggleInfo.Toggle4};
        if(choices.Length==5) {
            var fifth=toggles[3].transform.parent.Find("Toggle5");
            if(fifth==null) {
                var clone=NGUITools.AddChild(toggles[3].transform.parent.gameObject,toggles[3].gameObject);clone.name="Toggle5";fifth=clone.transform;
            }
            toggles.Add(fifth.GetComponent<UIToggle>());
        }
        var optionGrid=toggles[0].transform.parent.GetComponent<UIGrid>();
        float spacing=840f/(choices.Length-1);
        if(optionGrid!=null){optionGrid.cellWidth=spacing;optionGrid.maxPerLine=0;}
        bool syncing=false;
        Action refresh=delegate {
            syncing=true;int selected=ReadIndex(key);
            for(int i=0;i<toggles.Count;i++){toggles[i].startsActive=i==selected;toggles[i].Set(i==selected);}
            syncing=false;
        };
        for(int i=0;i<toggles.Count;i++) {
            int choice=i;var toggle=toggles[i];toggle.group=0;toggle.onChange.Clear();
            var label=toggle.GetComponentInChildren<UILabel>(true);label.text=choices[i];
            label.width=choices.Length==5?130:180;label.overflowMethod=UILabel.Overflow.ShrinkContent;
            toggle.transform.localPosition=new Vector3(i*spacing,0,0);
            EventDelegate.Add(toggle.onChange,delegate {
                if(syncing)return;
                if(toggle.value) {
                    if(key=="Preset")ApplyPreset(choice);
                    else {SprkTestPrefs.SetInt("Sprk.Graphics.Scale",100+25*choice);SprkTestPrefs.Save();}
                    instance.ResetHistories();
                }
                refresh();
            });
        }
        refresh();
        if(optionGrid!=null)optionGrid.Reposition();
    }
    static int ReadIndex(string key) {
        if(key=="Preset"){EnsurePreset();return Mathf.Clamp(SprkTestPrefs.GetInt("Sprk.Graphics.Preset",2),0,3);}
        int n=SprkTestPrefs.GetInt("Sprk.Graphics."+key,key=="MSAA"?2:key=="Scale"?100:key=="Jitter"?2:key=="Sharpness"?1:key=="Stationary"||key=="Motion"?3:0);
        if(key=="MSAA")return n==2?1:n==4?2:n==8?3:0;
        if(key=="Scale")return Mathf.Clamp((n-100)/25,0,4);
        int maximum=key=="Quality"?2:key=="Sharpness"?6:key=="Stationary"||key=="Motion"?4:3;
        return Mathf.Clamp(n,0,maximum);
    }

        var report=new List<string>();
        bool resourcesOk=FindResources();report.Add("resources="+resourcesOk);

        var wrapper=uiHost.AddComponent<NVespa.NNGUIExtension.UISimpleGrid>();
        var gridHost=new GameObject("Grid");gridHost.transform.SetParent(uiHost.transform,false);wrapper.Grid=gridHost.AddComponent<UIGrid>();

        // toggles started so Set exercises native event dispatch and visuals.
        foreach(var toggle in wrapper.GetComponentsInChildren<UIToggle>(true))typeof(UIToggle).GetField("mStarted",BindingFlags.Instance|BindingFlags.NonPublic).SetValue(toggle,true);
        ApplyPreset(1);Setup(window);
        report.Add("settingsRefreshed="+(presetRow.toggleInfo.Toggle2.value?"Performance":"wrong"));
        presetRow.toggleInfo.Toggle3.Set(true);
        report.Add("settingsClickPreset="+ReadIndex("Preset"));
        var duplicate=UIHelper.AddGridItem<GameOptionButtonComponent>(wrapper,GameOptionButtonComponent.PrefabAuto);duplicate.name="SPRK Anti-aliasing";
        Setup(window);
        report.Add("settingsAfterRepair="+wrapper.GetComponentsInChildren<GameOptionBaseComponent>(true).Count(r=>r.name.StartsWith("SPRK ")));
        report.Add("settingsAASelected="+presetRow.GetComponentsInChildren<UIToggle>(true).Count(t=>t.value));
        var scaleRow=rows.Single(r=>r.name=="SPRK Render scale");
        var fifth=scaleRow.toggleInfo.Toggle4.transform.parent.Find("Toggle5").GetComponent<UIToggle>();fifth.Set(true);
        report.Add("settingsScale="+Scale+" selected="+scaleRow.GetComponentsInChildren<UIToggle>(true).Count(t=>t.value));
        var aaChoices=new[]{presetRow.toggleInfo.Toggle1,presetRow.toggleInfo.Toggle2,presetRow.toggleInfo.Toggle3,presetRow.toggleInfo.Toggle4};

        for(int i=0;i<scaleChoices.Length;i++){scaleChoices[i].Set(true);allChoices &= Scale==100+25*i && scaleChoices.Count(t=>t.value)==1;}
        report.Add("settingsAllChoices="+allChoices);

        var afterRow=UIHelper.AddGridItem<GameOptionButtonComponent>(wrapper,GameOptionButtonComponent.PrefabAuto);afterRow.name="ZZAfter";
        var resolutionRow=UIHelper.AddGridItem<GameOptionButtonComponent>(wrapper,GameOptionButtonComponent.PrefabAuto);
        resolutionRow.name="Resolution";resolutionRow.gameOptionData=new GameOptionData{OptionName="OPTION_GAME_RESOULUTION"};
        wrapper.Grid.sorting=UIGrid.Sorting.Alphabetic;

        report.Add("settingsOrder="+String.Join(",",wrapper.Grid.GetChildList().Select(t=>t.name).ToArray()));

        Destroy(uiHost);
        SprkTestPrefs.SetInt("Sprk.Graphics.Scale",125);
        for(int preset=0;preset<4;preset++) {
            ApplyPreset(preset);
            report.Add("preset="+preset+" mode="+ReadIndex("PostAA")+" samples="+Samples+" quality="+ReadIndex("Quality")+" scale="+Scale);
        }
        for(int old=0;old<4;old++) {
            SprkTestPrefs.DeleteKey("Sprk.Graphics.Preset");SprkTestPrefs.SetInt("Sprk.Graphics.PostAA",old);SprkTestPrefs.SetInt("Sprk.Graphics.MSAA",0);

        var host=new GameObject("GraphicsProbeCamera");var camera=host.AddComponent<Camera>();
        camera.cullingMask=1<<30;camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=Color.gray;

    void LateUpdate() {
        RestoreTargets();
        int mode=ReadIndex("PostAA"),quality=ReadIndex("Quality");
        bool configured=SprkTestPrefs.HasKey("Sprk.Graphics.PostAA")||SprkTestPrefs.HasKey("Sprk.Graphics.MSAA")||SprkTestPrefs.HasKey("Sprk.Graphics.Scale");
        if(!configured)return;
        QualitySettings.antiAliasing=Samples==1?0:Samples;
        var cameras=Camera.allCameras.Where(c=>c!=null&&c.isActiveAndEnabled).OrderBy(c=>c.depth).ToArray();
        var ui=cameras.Where(c=>c.GetComponent<UICamera>()!=null).ToArray();
        float firstUi=ui.Length==0?float.PositiveInfinity:ui.Min(c=>c.depth);
        var onscreen=cameras.Where(c=>c.targetTexture==null && c.depth<firstUi && c.GetComponent<UICamera>()==null).ToArray();
        var worldCameras=onscreen.Where(c=>!c.orthographic).ToArray();
        bool fullRect=onscreen.All(c=>c.rect==new Rect(0,0,1,1));
        if(worldCameras.Length>0 && fullRect) {
            int signature=Screen.width*397^Screen.height*17^Scale^Samples*31;
            if(world==null || signature!=worldSignature) {
                ReleaseWorld();world=CreateTarget(Screen.width,Screen.height,24);world.name="SPRK World";
                world.Create();worldSignature=signature;ResetHistories();
            }
            foreach(var c in onscreen){redirected[c]=null;c.targetTexture=world;c.allowMSAA=Samples>1;}
            lastWorld=onscreen.Last();
            if(lastLogged!=signature+mode*101) {
                lastLogged=signature+mode*101;
                Debug.Log("SPRK graphics: mode="+mode+" target="+world.width+"x"+world.height+" samples="+world.antiAliasing+" worldCameras="+onscreen.Length);
            }
        } else { ReleaseWorld(); }
        bool available=mode==0||FindResources();
        status=available?"Ready":"Native post-processing resources unavailable";
        foreach(var camera in cameras) {
            if(camera.GetComponent<UICamera>()!=null)continue;
            camera.allowMSAA=Samples>1;
            bool screen=onscreen.Contains(camera);
            if(!screen && camera.targetTexture==null)continue;
            // Spatial AA runs once after the world stack; preview cameras have independent targets.
            bool apply=!screen || (mode==3?camera==worldCameras.FirstOrDefault():camera==lastWorld);
            var layer=camera.GetComponent<PostProcessLayer>();
            if(layer==null && mode!=0 && apply && available) {
                layer=camera.gameObject.AddComponent<PostProcessLayer>();
                layer.Init(resources);layer.volumeTrigger=camera.transform;layer.volumeLayer=0;
                layer.fog.enabled=false;
            }
            if(layer==null)continue;
            layers[camera]=layer;
            // Enum order in Unity is None, FXAA, SMAA, TAA; validated by the patch verifier.
            int effective=apply&&available?mode:0;
            var target=camera.targetTexture;
            if(effective==3 && (!SystemInfo.supportsMotionVectors || !layer.temporalAntialiasing.IsSupported() || (target!=null && target.antiAliasing>1)))effective=2;
            int signature=effective*100000+quality*1000+(target==null?Screen.width:target.width)*31+(target==null?Screen.height:target.height);
            bool cut=positions.ContainsKey(camera) && (Vector3.Distance(positions[camera],camera.transform.position)>3 || Quaternion.Angle(rotations[camera],camera.transform.rotation)>35);
            if(!signatures.ContainsKey(camera)||signatures[camera]!=signature||cut)layer.ResetHistory();
            signatures[camera]=signature;positions[camera]=camera.transform.position;rotations[camera]=camera.transform.rotation;
            layer.antialiasingMode=(PostProcessLayer.Antialiasing)effective;
            layer.fastApproximateAntialiasing.fastMode=quality==0;layer.fastApproximateAntialiasing.keepAlpha=true;
            layer.subpixelMorphologicalAntialiasing.quality=(SubpixelMorphologicalAntialiasing.Quality)quality;
            layer.temporalAntialiasing.jitterSpread=(ReadIndex("Jitter")+1)*0.25f;
            layer.temporalAntialiasing.sharpness=ReadIndex("Sharpness")*0.5f;
            layer.temporalAntialiasing.stationaryBlending=new[]{.6f,.7f,.8f,.9f,.95f}[ReadIndex("Stationary")];
            layer.temporalAntialiasing.motionBlending=new[]{.5f,.6f,.7f,.8f,.85f}[ReadIndex("Motion")];
        }
        foreach(var dead in layers.Keys.Where(c=>c==null).ToArray()){layers.Remove(dead);signatures.Remove(dead);positions.Remove(dead);rotations.Remove(dead);}
    }
