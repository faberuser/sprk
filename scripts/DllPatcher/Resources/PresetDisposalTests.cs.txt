using System;
using System.Collections.Generic;
public static class PresetDisposalTests {
    static void Check(bool condition) { if (!condition) throw new Exception("Preset disposal regression"); }
    public static void Run() {
        var account=new NGame2.NAccount.AccountManager();
        NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.instance=account;
        NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.isValidInstance=true;
        var gear=new NShared.EquipPresetInfo {EquipItemSlotIndex=new[]{7,8,7},EquipItemUid=new[]{"seven","eight","seven"}};
        var hero=new NShared.HeroPresetInfo {EquipItemSlotIndex=new[]{7,8},EquipItemUid=new[]{"seven","eight"},EquipItemRunePage=new[]{2,1}};
        account.EquipPresetSotrageManger.Add(gear);
        account.HeroPresetStorageManager.Add(new NShared.HeroPresetStorageInfo{HeroPresetInfos=new[]{hero,null}});
        Check(!SprkPresetDisposal.AfterRemove(false,7) && gear.EquipItemSlotIndex[0]==7);
        Check(SprkPresetDisposal.AfterRemove(true,7));
        Check(gear.EquipItemSlotIndex[0]==0 && gear.EquipItemSlotIndex[2]==0 && gear.EquipItemSlotIndex[1]==8);
        Check(gear.EquipItemUid[0]=="" && gear.EquipItemUid[1]=="eight");
        Check(hero.EquipItemSlotIndex[0]==0 && hero.EquipItemUid[0]=="" && hero.EquipItemRunePage[0]==0);
        Check(hero.EquipItemSlotIndex[1]==8 && hero.EquipItemRunePage[1]==1);
        Check(account.EquipPresetSotrageManger.Notifications==1 && account.HeroPresetStorageManager.Notifications==1);
        SprkPresetDisposal.AfterRemove(true,7);
        Check(account.EquipPresetSotrageManger.Notifications==1);
        Check(!SprkPresetDisposal.Clear(null,null,null,7));
        Check(SprkPresetDisposal.Clear(new[]{7},new string[0],new int[0],7));
        NVespa.NSingleton.GlobalAutoBehaviour<NGame2.NAccount.AccountManager>.isValidInstance=false;
        Check(SprkPresetDisposal.AfterRemove(true,8) && gear.EquipItemSlotIndex[1]==8);
    }
}
namespace NVespa.NSingleton { public static class GlobalAutoBehaviour<T> { public static bool isValidInstance; public static T instance; } }
namespace NShared {
 public class EquipPresetInfo {public int[] EquipItemSlotIndex;public string[] EquipItemUid;}
 public class HeroPresetInfo : EquipPresetInfo {public int[] EquipItemRunePage;}
 public class HeroPresetStorageInfo {public HeroPresetInfo[] HeroPresetInfos;}
}
namespace NGame2.NAccount {
 public class Manager<T>:List<T> {public int Notifications;public void Apply(T value){Notifications++;}}
 public class AccountManager {
  public Manager<NShared.EquipPresetInfo> EquipPresetSotrageManger=new Manager<NShared.EquipPresetInfo>();
  public Manager<NShared.HeroPresetStorageInfo> HeroPresetStorageManager=new Manager<NShared.HeroPresetStorageInfo>();
 }
}
