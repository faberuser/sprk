public static class DispatchTestRunner {
    static void Check(bool ok,string label) {if(!ok)throw new Exception(label);}
    static SprkDispatch Worker() {
        var worker=new SprkDispatch();typeof(SprkDispatch).GetField("owner",BindingFlags.Instance|BindingFlags.NonPublic).SetValue(worker,1L);return worker;
    }
    static IEnumerator Invoke(SprkDispatch w,string method,params object[] args) {
        return (IEnumerator)typeof(SprkDispatch).GetMethod(method,BindingFlags.Instance|BindingFlags.NonPublic).Invoke(w,args);
    }
    static void RunRoutine(IEnumerator routine) {
        int count=0;
        while(routine.MoveNext()) {
            if(++count>500)throw new Exception("Unbounded dispatch coroutine");
            if(routine.Current is IEnumerator)RunRoutine((IEnumerator)routine.Current);
            UnityEngine.Time.realtimeSinceStartup+=1;
        }
    }
    static SprkDispatch.Job Job(string id="job",int slot=1) {return new SprkDispatch.Job {SlotIndex=slot,ChapterIndex=1,DungeonIndex=1,Difficulty=1,RepeatCount=200,SimulationVersion=1,State="Simulation",HeroIndices="[1]",SimulationJobId=id};}
    static void Reset() {FakeNetwork.Requests.Clear();FakeNetwork.Replies.Clear();FakeSimulator.Uids.Clear();FakeSimulator.Timeout=false;FakeSimulator.ChangeSession=false;UnityEngine.Time.realtimeSinceStartup=0;NVespa.NSingleton.SingletonUpdater<NVespa.WebServiceManager>.instance.Requester.SessionKey="session";}
    public static void Run() {
        Reset();var worker=Worker();var job=Job();
        RunRoutine(Invoke(worker,"Prepare",job));
        Check(FakeSimulator.Uids.Count==1,"First result waited for all 200 simulations");
        var first=FakeNetwork.Requests.Single();
        Check(first["Results"]=="[1]"&&first["TimesMs"]=="[1000]"&&first["Progressive"]=="true","First prefix payload");
        // Fresh snapshot after a lost reply has no saved prefix. Retry cached combat.
        RunRoutine(Invoke(worker,"Prepare",Job()));
        Check(FakeSimulator.Uids.Count==1 && FakeNetwork.Requests.Count==2,"Lost-response retry reran combat");
        job=Job();job.State="Battle";job.Saved.Outcomes.Add(1);job.Saved.Times.Add(1000);
        RunRoutine(Invoke(worker,"Prepare",job));
        Check(FakeSimulator.Uids.Last()=="sprk-dispatch:job:1" && FakeNetwork.Requests.Last()["Results"]=="[1,1]","Next prefix and deterministic run index");
        Reset();worker=Worker();FakeNetwork.Replies.Enqueue("{\"Result\":\"Fail\"}");
        RunRoutine(Invoke(worker,"Prepare",Job()));
        job=Job();job.State="Battle";job.Saved.Outcomes.Add(0);job.Saved.Times.Add(1000);
        RunRoutine(Invoke(worker,"Prepare",job));
        Check(FakeNetwork.Requests.Last()["Results"]=="[0,1]","Rejected upload did not reload authoritative prefix");
        Reset();worker=Worker();job=Job();job.State="Battle";job.Saved.Outcomes.Add(0);job.Saved.Times.Add(2000);
        Check(SprkDispatch.NeedsPreparation(job),"Partial Battle state was abandoned");
        RunRoutine(Invoke(worker,"Prepare",job));
        Check(FakeNetwork.Requests.Single()["Results"]=="[0,1]"&&FakeSimulator.Uids.Single().EndsWith(":1"),"Resume from persisted server prefix");
        job.State="Cancel";Check(!SprkDispatch.NeedsPreparation(job),"Cancelled job still pending");
        job.State="Complete";Check(!SprkDispatch.NeedsPreparation(job),"Complete job still pending");
        job.State="Battle";job.RepeatCount=job.Saved.Outcomes.Count;Check(!SprkDispatch.NeedsPreparation(job),"Prepared job still pending");
        Reset();worker=Worker();FakeSimulator.Timeout=true;
        RunRoutine(Invoke(worker,"Prepare",Job()));
        Check(FakeNetwork.Requests.Count==0,"Timed out combat was uploaded");
        Reset();worker=Worker();FakeSimulator.ChangeSession=true;
        RunRoutine(Invoke(worker,"Prepare",Job()));
        Check(FakeNetwork.Requests.Count==0,"Old-session combat was uploaded");
        Reset();worker=Worker();
        FakeNetwork.Replies.Enqueue("{\"Result\":\"Success\",\"DispatchBattleInfos\":["+JsonJob("one",1)+","+JsonJob("two",2)+"]}");
        RunRoutine(Invoke(worker,"Poll"));
        Check(FakeSimulator.Uids.SequenceEqual(new[]{"sprk-dispatch:one:0","sprk-dispatch:two:0"}),"A slot monopolized preparation");
        Check(FakeNetwork.Requests.Count(r=>r.ContainsKey("Progressive"))==2,"Both slots did not upload a first prefix");
    }
    static string JsonJob(string id,int slot) {return "{\"SlotIndex\":"+slot+",\"ChapterIndex\":1,\"DungeonIndex\":1,\"Difficulty\":1,\"RepeatCount\":200,\"SimulationVersion\":1,\"State\":\"Simulation\",\"HeroIndices\":\"[1]\",\"SimulationJobId\":\""+id+"\"}";}
}
