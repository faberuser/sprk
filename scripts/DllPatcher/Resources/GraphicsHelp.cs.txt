    static GraphicsHelpInfo HelpFor(GameOptionBaseComponent row) {
        if(row==null)return null;
        string key=row.name.StartsWith("SPRK ",StringComparison.Ordinal)?row.name.Substring(5):row.gameOptionData==null?null:row.gameOptionData.FieldName;
        GraphicsHelpInfo help;return key!=null && graphicsHelp.TryGetValue(key,out help)?help:null;
    }
    public static void ConfigureGraphicsHelp(GameOptionBaseComponent row) {
        var help=HelpFor(row);if(help==null)return;
        // Keep native option metadata and shared localization tables untouched.
        foreach(var root in new[]{row.ParentRoot,row.ChildRoot})if(root!=null && root.ToolTipImage!=null)root.ToolTipImage.SetActive(true);
        if(row.Collider_Tooltip!=null)foreach(var collider in row.Collider_Tooltip)if(collider!=null)collider.enabled=true;
    }
    public static bool ShowGraphicsHelp(GameOptionBaseComponent row) {
        var help=HelpFor(row);if(help==null)return false;
        var anchor=row.GetComponent<UIWidget>();if(anchor==null)return true;
        GameOptionTooltip.Open(anchor,new TooltipTextInfo {
            Name=new NGame2.NUtil.StringProxy{Value=help.title},
            Desc=new NGame2.NUtil.StringProxy{Value=help.description}
        });return true;
    }
    public static bool HideGraphicsHelp(GameOptionBaseComponent row) {
        if(HelpFor(row)==null)return false;
        var anchor=row.GetComponent<UIWidget>();if(anchor!=null)GameOptionTooltip.Hide(anchor);return true;
    }
    System.Collections.IEnumerator GraphicsHelpProbe(List<string> report,string output) {
        var manager=NVespa.NSingleton.SceneAutoBehaviour<NGame2.NUI.NManager.LoginManagement>.instance;
        var login=manager.GetInstance<LoginAccountInfo>();SprkLoginSettings.Configure(login);
        login.GetComponent<SprkLoginSettings>().OpenSettings();yield return new WaitForSecondsRealtime(.5f);
        var settings=manager.GetInstance<GameOptionWindow>();
        foreach(var camera in Camera.allCameras)if(camera.name=="BI UI Camera")camera.gameObject.SetActive(false);
        var rows=settings.GetComponentsInChildren<GameOptionBaseComponent>(true);
        bool helpOk=true,isolated=true,events=true;int checkedRows=0;
        var onShow=typeof(GameOptionBaseComponent).GetMethod("OnShowTooltip",BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic);
        var onHide=typeof(GameOptionBaseComponent).GetMethod("OnHideTooltip",BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic);
        var labelFlags=BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic;
        var records=new List<string>();
        foreach(var row in rows) {
            var help=HelpFor(row);if(help==null)continue;checkedRows++;
            if(row.name=="SPRK FXAA Quality")ApplyPreset(1);
            else if(row.name=="SPRK SMAA Quality")ApplyPreset(2);
            else if(row.name=="SPRK MSAA Samples")ApplyPreset(3);
            else if(row.name=="SPRK TAA Sharpness")ApplyPreset(4);
            Setup(settings);ConfigureGraphicsHelp(row);
            helpOk &= row.ParentRoot.ToolTipImage!=null && row.ParentRoot.ToolTipImage.activeSelf;
            if(row.gameOptionData!=null && row.gameOptionData.Index!=0) {
                var original=NShared.GameRoot.Instance.GameOptionDataContainer.GetData(row.gameOptionData.Index);
                isolated &= ReferenceEquals(original,row.gameOptionData) && original.ToolTipDesc!=help.description;
            }
            var scroll=row.GetComponentInParent<UIScrollView>();
            if(scroll!=null) {
                var panel=scroll.GetComponent<UIPanel>();var clip=panel.finalClipRegion;var position=panel.transform.InverseTransformPoint(row.transform.position);
                scroll.MoveRelative(new Vector3(0,clip.y-position.y,0));scroll.RestrictWithinBounds(true);
            }
            onShow.Invoke(row,null);yield return null;
            var owner=row.GetComponent<UIWidget>();
            var tooltip=Resources.FindObjectsOfTypeAll<GameOptionTooltip>().FirstOrDefault(t=>t.gameObject.activeInHierarchy);
            events &= owner!=null && GameOptionTooltip.IsVisible(owner) && tooltip!=null;
            if(tooltip!=null) {
                var desc=(UILabel)tooltip.GetType().BaseType.GetField("DescLabel",labelFlags).GetValue(tooltip);
                var title=(UILabel)tooltip.GetType().BaseType.GetField("NameLabel",labelFlags).GetValue(tooltip);
                helpOk &= desc.text==help.description && title.text==help.title;
                records.Add(help.title+"\n"+desc.text+"\n");
                if(help.title=="Max Frame" || help.title=="Adaptive Battle Scale" || help.title=="Render Scale") {
                    yield return new WaitForEndOfFrame();
                    var image=ScreenCapture.CaptureScreenshotAsTexture();System.IO.File.WriteAllBytes(System.IO.Path.Combine(output,"help-"+help.title.Replace(" ","-")+".png"),image.EncodeToPNG());Destroy(image);
                }
            }
            onHide.Invoke(row,null);yield return null;events &= !GameOptionTooltip.IsVisible(owner);
            if(help.title=="Render Scale") {
                int savedPreset=RenderScalePreset;
                var presets=row.GetComponentsInChildren<UIToggle>(true).Where(t=>t.gameObject.activeSelf).ToArray();
                var world=rows.Single(r=>r.name=="SPRK World Scale");var preview=rows.Single(r=>r.name=="SPRK Preview Scale");
                bool layout=true;
                foreach(int choice in new[]{2,5}) {
                    presets[choice].Set(true);yield return null;
                    bool custom=choice==5;
                    foreach(var child in new[]{world,preview}) {
                        layout &= child.ChildRoot.Root.activeSelf && !child.ParentRoot.Root.activeSelf && child.GetComponent<UIWidget>().isGrayscale==!custom;
                        foreach(var toggle in child.GetComponentsInChildren<UIToggle>(true).Where(t=>t.gameObject.activeSelf)) {
                            var collider=toggle.GetComponent<Collider>();layout &= collider!=null && collider.enabled==custom;
                        }
                    }
                    yield return new WaitForEndOfFrame();
                    var image=ScreenCapture.CaptureScreenshotAsTexture();System.IO.File.WriteAllBytes(System.IO.Path.Combine(output,custom?"scale-hierarchy-custom.png":"scale-hierarchy-preset.png"),image.EncodeToPNG());Destroy(image);
                }
                report.Add("renderScaleNativeHierarchy="+layout);
                SprkTestPrefs.SetInt("Sprk.Graphics.RenderScale",savedPreset);Setup(settings);
            }
        }
        report.Add("graphicsHelpRows="+checkedRows+" correctText="+helpOk+" nativeShowHide="+events+" tableIsolation="+isolated);
        System.IO.File.WriteAllLines(System.IO.Path.Combine(output,"help-text.txt"),records.ToArray());
        settings.ReOpen();yield return null;
        helpOk &= settings.GetComponentsInChildren<GameOptionBaseComponent>(true).Where(r=>HelpFor(r)!=null).All(r=>r.ParentRoot.ToolTipImage!=null && r.ParentRoot.ToolTipImage.activeSelf);
        report.Add("graphicsHelpAndPooling="+(checkedRows==graphicsHelp.Count && helpOk && isolated && events));
        settings.Close();
    }
