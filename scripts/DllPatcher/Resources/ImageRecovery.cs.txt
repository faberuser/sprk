    static void RemovePortraitFade(PortraitLayout saved) {
        if(saved.Fade!=null) {
            saved.Fade.gameObject.SetActive(false);
            UnityEngine.Object.Destroy(saved.Fade.gameObject);
            saved.Fade=null;
        }
        if(saved.FadeTexture!=null) {UnityEngine.Object.Destroy(saved.FadeTexture);saved.FadeTexture=null;}
        saved.FadeHeight=0;
    }
    static void UpdatePortraitFade(UITexture ui,PortraitLayout saved,int fittedHeight) {
        int gap=saved.Height-fittedHeight;
        if(gap<=0) {RemovePortraitFade(saved);return;}
        if(saved.Fade==null) {
            var host=new GameObject("SPRK Light portrait edge fade");
            host.transform.SetParent(ui.transform,false);
            saved.Fade=host.AddComponent<UITexture>();
            // NGUI trims one pixel from odd-sized textures. A one-pixel-wide
            // texture therefore produces a zero-width quad and cannot render.
            saved.FadeTexture=new Texture2D(2,256,TextureFormat.RGBA32,false);
            saved.FadeTexture.name="SPRK Light portrait edge fade";
            saved.FadeTexture.wrapMode=TextureWrapMode.Clamp;
            saved.FadeTexture.filterMode=FilterMode.Bilinear;
            saved.Fade.mainTexture=saved.FadeTexture;
        }
        if(saved.FadeHeight!=fittedHeight || saved.FadeCardHeight!=saved.Height) {
            var pixels=new Color32[2*256];
            float solid=gap*0.5f;
            float fade=Mathf.Max(1f,Mathf.Min(fittedHeight*0.15f,saved.Height*0.06f));
            for(int y=0;y<256;y++) {
                float pos=(y+0.5f)*saved.Height/256f;
                float distance=Mathf.Min(pos,saved.Height-pos);
                byte alpha=(byte)Mathf.RoundToInt(255f*Mathf.Clamp01((solid+fade-distance)/fade));
                pixels[y*2]=pixels[y*2+1]=new Color32(0,0,0,alpha);
            }
            saved.FadeTexture.SetPixels32(pixels);saved.FadeTexture.Apply();saved.FadeHeight=fittedHeight;saved.FadeCardHeight=saved.Height;
        }
        var fadeWidget=saved.Fade;
        fadeWidget.gameObject.layer=ui.gameObject.layer;
        fadeWidget.keepAspectRatio=UIWidget.AspectRatioSource.Free;
        fadeWidget.pivot=ui.pivot;fadeWidget.width=saved.Width;fadeWidget.height=saved.Height;
        fadeWidget.depth=ui.depth+1;
        fadeWidget.transform.localPosition=Vector3.zero;
    }
    static void RestorePortrait(UIBundleLoader loader) {
        PortraitLayout saved;
        if(loader==null || !portraitLayouts.TryGetValue(loader,out saved))return;
        var ui=loader.UiTex;
        if(ui!=null)ui.drawRegion=saved.DrawRegion;
        RemovePortraitFade(saved);
        portraitLayouts.Remove(loader);
    }
    public static void BeforePortraitPathChange(UIBundleLoader loader,string next) {
        if(loader!=null && loader.AssetPath!=next)RestorePortrait(loader);
    }
    public static void BeforePortraitChange(UIBundleLoader loader,string next,UIHelper.EAssetTextureAssign assign) {
        PortraitLayout saved;
        if(loader==null || !portraitLayouts.TryGetValue(loader,out saved))return;
        var effective=assign==UIHelper.EAssetTextureAssign.__Ignore__?(UIHelper.EAssetTextureAssign)defaultAssign.GetValue(loader):assign;
        if(loader.AssetPath!=next || (UIHelper.EAssetTextureAssign)currentAssign.GetValue(loader,null)!=effective)RestorePortrait(loader);
    }
    public static void FitHeroPortrait(UIBundleLoader loader,AssetTexture.TextureInfo info) {
        if(loader==null || info==null || info.Width<=0 || info.Height<=0 || !lightFallbackPaths.Contains(loader.AssetPath))return;
        var portrait=loader.GetComponentInParent<HeroPortraitComponent>();
        if(portrait==null || portrait.HeroBundleTex!=loader || portrait.PortraitMode==HeroPortraitComponent.EPortraitMode.Icon)return;
        int costume=portrait.HeroInfo==null?portrait.HeroCostumeIndex:portrait.HeroInfo.CostumeIndex;
        string expected;
        if(!lightFallbackPortraits.TryGetValue(costume,out expected) || !String.Equals(loader.AssetPath,expected,StringComparison.OrdinalIgnoreCase))return;
        var ui=loader.UiTex;if(ui==null || ui.width<=0 || ui.height<=0)return;
        PortraitLayout saved;
        if(!portraitLayouts.TryGetValue(loader,out saved)) {
            if(ui.width<=0 || ui.height<=0)return;
            saved=new PortraitLayout {DrawRegion=ui.drawRegion};
            portraitLayouts.Add(loader,saved);
        }
        saved.Width=ui.width;saved.Height=ui.height;
        float scale=Mathf.Min((float)saved.Width/info.Width,(float)saved.Height/info.Height);
        int width=Mathf.Max(1,Mathf.RoundToInt(info.Width*scale));
        int height=Mathf.Max(1,Mathf.RoundToInt(info.Height*scale));
        // NGUI width/height setters rewrite anchor offsets. Fit only the drawn
        // quad, as native UITexture.fixedAspect does, preserving the full UVs
        // and the anchored card rectangle across pooled hero/costume changes.
        float x=(saved.Width-width)*0.5f/saved.Width,y=(saved.Height-height)*0.5f/saved.Height;
        ui.drawRegion=new Vector4(x,y,1f-x,1f-y);
        // Undo native ClipBottom for these fallback images: keep the entire
        // atlas sprite, including its transparent pixels, inside the card.
        ui.uvRect=info.UvRect;
        UpdatePortraitFade(ui,saved,height);
    }
    public static void TickPortrait(UIBundleLoader loader) {
        // Cached cards can load before HeroInfo is reassigned, or have their
        // dimensions reset by the party layout without loading a new texture.
        // Reconcile after those updates, including the first frame after reuse.
        if(loader==null || !loader.isActiveAndEnabled || !lightFallbackPaths.Contains(loader.AssetPath))return;
        var ui=loader.UiTex;if(ui==null || !ui.enabled || ui.mainTexture==null)return;
        var asset=(AssetTexture)texture.GetValue(loader);
        var info=asset==null?null:asset.GetTextureInfo();
        if(info!=null && info.Tex==ui.mainTexture)FitHeroPortrait(loader,info);
    }
    static readonly Dictionary<int, Texture2D> tpIcons = new Dictionary<int, Texture2D>();

    // These two valid legacy consumables have no native atlas entries. Keep the
    // icons in the helper so every quality setting uses the same numbered art.
    public static bool TryTpIcon(UIBundleLoader loader)
    {
        if (loader == null) return false;
        string path = loader.AssetPath;
        int value = String.Equals(path, "Icon/Item/TRANSCEND_10POINT", StringComparison.OrdinalIgnoreCase) ? 10
            : String.Equals(path, "Icon/Item/TRANSCEND_15POINT", StringComparison.OrdinalIgnoreCase) ? 15 : 0;
        if (value == 0) return false;
        Texture2D icon;
        if (!tpIcons.TryGetValue(value, out icon) || icon == null) {
            using (var stream = typeof(SprkImageRecovery).Assembly.GetManifestResourceStream("SprkTpIcons." + value + ".png")) {
                if (stream == null) throw new InvalidOperationException("Missing TP icon: " + value);
                var bytes = new byte[(int)stream.Length];
                int offset = 0;
                while (offset < bytes.Length) {
                    int read = stream.Read(bytes, offset, bytes.Length - offset);
                    if (read == 0) throw new System.IO.EndOfStreamException();
                    offset += read;
                }
                icon = new Texture2D(2, 2, TextureFormat.RGBA32, false);
                if (!ImageConversion.LoadImage(icon, bytes, true)) throw new InvalidOperationException("Invalid TP icon: " + value);
                icon.name = "SPRK " + value + " TP";
                icon.hideFlags = HideFlags.HideAndDontSave;
                tpIcons[value] = icon;
            }
        }
        var ui = loader.UiTex;
        if (ui != null) {
            ui.material = null;
            ui.mainTexture = icon;
            ui.uvRect = new Rect(0, 0, 1, 1);
        }
        return true;
    }

    static void RemoveTime(object owner, FieldInfo times, FieldInfo order, string key)
    {
        ((IDictionary)times.GetValue(owner)).Remove(key);
        var sorted = (SortedList)order.GetValue(owner);
        for (int i = sorted.Count - 1; i >= 0; i--)
            if (Equals(sorted.GetByIndex(i), key)) sorted.RemoveAt(i);
    }

    public static void BeforeCachedBundle(AssetCacheService cache, string key)
    {
        var dict = (IDictionary)bundleCache.GetValue(cache);
        if (!dict.Contains(key) || (UnityEngine.Object)dict[key] != null) return;
        dict.Remove(key);
        ((IDictionary)assetCache.GetValue(cache)).Remove(key);
        RemoveTime(cache, bundleTimes, bundleOrder, key);
    }

    public static void BeforeCachedAsset(AssetCacheService cache, string bundle, string key)
    {
        var bundles = (IDictionary)assetCache.GetValue(cache);
        var dict = bundles[bundle] as IDictionary;
        if (dict != null && dict.Contains(key) && (UnityEngine.Object)dict[key] == null) dict.Remove(key);
    }

    public static void BeforeCachedResource(AssetCacheService cache, string key)
    {
        var dict = (IDictionary)resourceCache.GetValue(cache);
        if (!dict.Contains(key) || (UnityEngine.Object)dict[key] != null) return;
        dict.Remove(key);
        RemoveTime(cache, resourceTimes, resourceOrder, key);
    }

    internal static void PrepareAtlas(string path)
    {
        var link = SingletonAuto<AtlasDataContainer>.instance.GetAtlasLink(path);
        if (link == null || atlases.ContainsKey(link.AtlasCode)) return;
        var atlas = SingletonAuto<NguiAtlasContainer>.instance.Lock(link.AtlasCode) as AtlasLoad;
        if (atlas == null) return;
        bool retained = false;
        try {
            var async = (IAsyncLoad<UIAtlas>)atlas;
            if (!async.IsDone || (async.IsValid && async.Value != null && async.Value.texture != null) || atlas.AssetAsync == null) return;
            // Retain one native atlas lease while its failed underlying request is
            // released. Give native bundle cleanup a frame before reopening it.
            atlas.AssetAsync.Close(true);
            atlas.AssetAsync = null;
            atlases.Add(link.AtlasCode, new AtlasRetry { Atlas = atlas, At = Time.realtimeSinceStartup + 0.25f });
            retained = true;
        } finally {
            if (!retained) NguiAtlasContainer.ReleaseAtlas(atlas, false);
        }
    }

    public static void TickGlobal()
    {
        ProbePortraitRendering();
        ProbeLightPortraitFit();
        ProbeTpIcons();
        if (atlases.Count == 0) return;
        var due = new List<string>();
        foreach (var item in atlases) if (Time.realtimeSinceStartup >= item.Value.At) due.Add(item.Key);
        foreach (string code in due) {
            var atlas = atlases[code].Atlas;
            atlases.Remove(code);
            try { atlas.Open(code); }
            catch (Exception error) { Debug.LogWarning("[SPRK Images] Atlas retry failed: " + error.Message); }
            finally { NguiAtlasContainer.ReleaseAtlas(atlas, false); }
        }
    }

    static void ProbeLightPortraitFit() {
        string report=Environment.GetEnvironmentVariable("SPRK_LIGHT_PORTRAIT_PROBE");
        if(portraitProbeStarted || String.IsNullOrEmpty(report) || String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE")) ||
           String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR")))return;
        portraitProbeStarted=true;GameObject root=null;
        try {
            root=new GameObject("SPRK Light portrait fixture");
            var portrait=root.AddComponent<HeroPortraitComponent>();
            portrait.HeroInfo=new NShared.CachedHeroInfo {CostumeIndex=7280};
            var child=new GameObject("Portrait image");child.transform.SetParent(root.transform,false);
            var ui=child.AddComponent<UITexture>();var loader=child.AddComponent<UIBundleLoader>();
            typeof(UIBundleLoader).GetField("_uiTex",flags).SetValue(loader,ui);portrait.HeroBundleTex=loader;
            var pathField=typeof(UIBundleLoader).GetField("_assetPath",flags);
            string path=lightFallbackPortraits[7280];
            var info=new AssetTexture.TextureInfo {Width=260,Height=490,UvRect=new Rect(0.1f,0.2f,0.3f,0.4f)};
            foreach(var pivot in new[]{UIWidget.Pivot.Center,UIWidget.Pivot.TopLeft,UIWidget.Pivot.BottomRight}) {
                ui.pivot=pivot;ui.keepAspectRatio=UIWidget.AspectRatioSource.Free;ui.width=200;ui.height=450;
                ui.transform.localPosition=new Vector3(13,27,0);pathField.SetValue(loader,path);

                FitHeroPortrait(loader,info);
                var drawn=ui.drawingDimensions;
                if(ui.width!=200 || ui.height!=450 || Mathf.Abs(drawn.w-drawn.y-377)>0.01f || ui.uvRect!=info.UvRect)throw new Exception("Tall monster portrait was not fitted with original UVs and intact widget bounds");
                PortraitLayout fitted;
                if(!portraitLayouts.TryGetValue(loader,out fitted) || fitted.Fade==null || fitted.Fade.transform.parent!=ui.transform ||
                   fitted.Fade.width!=200 || fitted.Fade.height!=450 || fitted.Fade.depth!=ui.depth+1)
                    throw new Exception("Missing or incorrectly layered card-edge fade");

                   fadePixels[50].a==0 || fadePixels[50].a==255 || fadePixels[460].a!=fadePixels[50].a)
                    throw new Exception("Black edges must cover the gaps and fade smoothly into the portrait");
                foreach(var pixel in fadePixels)if(pixel.r!=0 || pixel.g!=0 || pixel.b!=0)throw new Exception("Portrait fade is not black");
                var center=ui.transform.localPosition+new Vector3((drawn.x+drawn.z)*0.5f,(drawn.y+drawn.w)*0.5f,0);
                if((center-originalCenter).sqrMagnitude>0.001f)throw new Exception("Portrait pivot is not centered");
                var overlayCenter=fitted.Fade.transform.localPosition+new Vector3((0.5f-offset.x)*fitted.Fade.width,(0.5f-offset.y)*fitted.Fade.height,0);

                    throw new Exception("Edge fade is not centered over the original card");
                FitHeroPortrait(loader,info);
                if(ui.width!=200 || ui.height!=450 || fitted.FadeTexture!=fitted.Fade.mainTexture)throw new Exception("Repeated loads changed the card dimensions or fade");
                // Exercise the actual patched native setter when the pooled card changes.
                var oldFade=fitted.Fade;
                loader.SetAssetPath("",UIHelper.EAssetTextureAssign.ClipBottom);
                if(ui.width!=200 || ui.height!=450 || ui.transform.localPosition!=new Vector3(13,27,0))throw new Exception("Reused portrait failed to restore its layout");
                if(oldFade!=null && oldFade.gameObject.activeSelf)throw new Exception("Fade leaked onto the next card");

                if(field.FieldType==typeof(HeroPortraitComponent.EPortraitMode)) {modeField=field;break;}

    static bool loadingProbeStarted;

    public static void ProbeLoadingBackground()
    {
        string report = Environment.GetEnvironmentVariable("SPRK_LOADING_BACKGROUND_PROBE");
        if (loadingProbeStarted || String.IsNullOrEmpty(report)) return;
        if (!loadingProbeAssets) {
            loadingProbeAssets = true;
            string root = System.IO.Path.Combine(Application.dataPath, "Documents/Patch/StandaloneWindows");
            var files = new Dictionary<string, NShared.AssetInfo>();
            foreach (string file in System.IO.Directory.GetFiles(System.IO.Path.Combine(root, "Assetbundle"), "*", System.IO.SearchOption.AllDirectories)) {
                string entry = file.Substring(root.Length + 1).Replace('\\', '/');
                files[entry] = new NShared.AssetInfo { EntryName = entry };
            }
            SingletonAuto<AssetFileManager>.instance.Setup(files);
        }
        // This isolated, account-free probe reads the already installed assets
        // before login would normally enable the native bundle loader.
        var bundles = SingletonAuto<BundleManager>.instance;
        NGameAsset.Define.PatchDownloaded = true;
        if (!bundles.IsReady) return;
        if (SingletonAuto<AtlasDataContainer>.instance.GetAtlasLink("LoadingTip/GamePlayTip/OrvelTown") == null) return;
        loadingProbeStarted = true;
        new GameObject("SPRK loading background probe").AddComponent<SprkLoadingBackgroundProbe>().Run(report);
    }

    public static void Tick(UIBundleLoader loader)
    {
        if (!Application.isPlaying || loader == null || !loader.isActiveAndEnabled) return;
        var state = states.GetOrCreateValue(loader);
        float now = Time.realtimeSinceStartup;
        if (now < state.Poll) return;
        state.Poll = now + 0.25f;
        string path = loader.AssetPath;
        if (path != state.Path) {
            state.Path = path;
            state.Attempts = 0;
            state.MissingSince = now;
            state.Released = state.Disabled = false;
        }
        if (state.Disabled || String.IsNullOrEmpty(path)) return;
        var ui = loader.UiTex;
        if (ui == null || !ui.enabled || !ui.gameObject.activeInHierarchy || ui.finalAlpha <= 0f) return;
        try {
            if (state.Released) {
                if (now < state.ReloadAfter) return;
                state.Released = false;
                state.MissingSince = now;
                load.Invoke(loader, null);
                return;
            }
            if (ui.mainTexture != null) {
                state.Attempts = 0;
                state.MissingSince = now;
                return;
            }
            var asset = (AssetTexture)texture.GetValue(loader);
            var info = asset == null ? null : asset.GetTextureInfo();
            if (info != null && info.Tex != null) {
                // A lost widget binding does not require reloading the atlas.
                complete.Invoke(loader, null);
                return;
            }
            if (now - state.MissingSince < (asset != null && !asset.IsDone ? 8f : 0.5f)) return;
            if (state.Attempts >= 3) return;
            state.Attempts++;
            PrepareAtlas(path);
            bool old = (bool)releaseOnClose.GetValue(loader);
            try {
                releaseOnClose.SetValue(loader, true);
                release.Invoke(loader, null);
            } finally { releaseOnClose.SetValue(loader, old); }
            state.Released = true;
            state.ReloadAfter = now + (state.Attempts == 1 ? 0.5f : state.Attempts == 2 ? 1f : 2f);
            if (state.Attempts == 1) Debug.Log("[SPRK Images] Retrying thumbnail: " + path);
        } catch (Exception error) {
            state.Disabled = true;
            Debug.LogWarning("[SPRK Images] Thumbnail recovery stopped: " + error.Message);
        }
    }
}

// Loading images use PlainAssetHolder, not UIBundleLoader. Keep their native
// ownership and completion checks, and repair only the loading image binding.
public sealed class SprkLoadingBackground : MonoBehaviour
{
    Component owner;
    UITexture target;
    FieldInfo assetField;
    AssetTexture asset;
    string path;
    float missingSince, nextPoll;
    int attempts;
    bool watching;

    public void Open(Component view, UITexture texture, FieldInfo field)
    {
        owner = view; target = texture; assetField = field;
        asset = (AssetTexture)field.GetValue(view);
        path = asset == null ? "" : asset.Path;
        missingSince = Time.realtimeSinceStartup;
        nextPoll = 0; attempts = 0; watching = true;
        BindOrCover();
    }

    bool BindOrCover()
    {
        var info = asset == null ? null : asset.GetTextureInfo();
        if (info != null && info.Tex != null) {
            if (target.mainTexture != info.Tex || target.material != info.Mtrl)
                UIHelper.SetTextureInfo(target, info, UIHelper.EAssetTextureAssign.None);
            return true;
        }
        // Native async completion can clear the widget on failure. Cover it
        // again in LateUpdate before rendering, without delaying scene entry.
        target.material = null;
        target.mainTexture = Texture2D.blackTexture;
        target.uvRect = new Rect(0, 0, 1, 1);
        return false;
    }

    void LateUpdate()
    {
        if (!watching || owner == null || target == null) return;
        try {
            // Native release owns the asset and cancels its pending callback.
            if (!ReferenceEquals(assetField.GetValue(owner), asset)) { watching = false; return; }
            if (BindOrCover()) { missingSince = Time.realtimeSinceStartup; return; }
            float now = Time.realtimeSinceStartup;
            if (now < nextPoll || String.IsNullOrEmpty(path) || attempts >= 3) return;
            if (now - missingSince < (asset != null && !asset.IsDone ? 8f : 0.5f)) return;
            attempts++;
            var holder = target.GetComponent<PlainAssetHolder>();
            if (holder != null) holder.Close();
            if (asset != null) asset.Close();
            SprkImageRecovery.PrepareAtlas(path);
            asset = UIHelper.GetPlainTexture(path);
            assetField.SetValue(owner, asset);
            holder = UIHelper.ResetPlaneAssetTextureAsync(owner, target, asset, UIHelper.EAssetTextureAssign.None, null);
            if (holder != null) holder.ReleaseAssetOnClose = true;
            missingSince = now;
            nextPoll = now + attempts * 0.5f;
            BindOrCover();
            if (attempts == 1) Debug.LogWarning("[SPRK Images] Retrying loading background: " + path);
        } catch (Exception error) {
            watching = false;
            Debug.LogWarning("[SPRK Images] Loading background recovery stopped: " + error.Message);
        }
    }

    void OnDisable() { watching = false; }
}
