using System.Collections;
using System.Collections.Generic;
using System.Reflection;

public static class SprkImageRecovery
{
    static readonly Dictionary<int, Texture2D> tpIcons = new Dictionary<int, Texture2D>();

    // These two valid legacy consumables have no native atlas entries. Keep the
    // icons in the helper so every quality setting uses the same numbered art.
    public static bool TryTpIcon(UIBundleLoader loader)
    {
        if (loader == null) return false;
        string path = loader.AssetPath;
        int value = String.Equals(path, "Icon/Item/TRANSCEND_10POINT", StringComparison.OrdinalIgnoreCase) ? 10
            : String.Equals(path, "Icon/Item/TRANSCEND_15POINT", StringComparison.OrdinalIgnoreCase) ? 15 : 0;
        if (value == 0) return false;
        Texture2D icon;
        if (!tpIcons.TryGetValue(value, out icon) || icon == null) {
            using (var stream = typeof(SprkImageRecovery).Assembly.GetManifestResourceStream("SprkTpIcons." + value + ".png")) {
                if (stream == null) throw new InvalidOperationException("Missing TP icon: " + value);
                var bytes = new byte[(int)stream.Length];
                int offset = 0;
                while (offset < bytes.Length) {
                    int read = stream.Read(bytes, offset, bytes.Length - offset);
                    if (read == 0) throw new System.IO.EndOfStreamException();
                    offset += read;
                }
                icon = new Texture2D(2, 2, TextureFormat.RGBA32, false);
                if (!ImageConversion.LoadImage(icon, bytes, true)) throw new InvalidOperationException("Invalid TP icon: " + value);
                icon.name = "SPRK " + value + " TP";
                icon.hideFlags = HideFlags.HideAndDontSave;
                tpIcons[value] = icon;
            }
        }
        var ui = loader.UiTex;
        if (ui != null) {
            ui.material = null;
            ui.mainTexture = icon;
            ui.uvRect = new Rect(0, 0, 1, 1);
        }
        return true;
    }
    sealed class State {
        public string Path;
        public int Attempts;
        public float Poll, MissingSince, ReloadAfter;
        public bool Released, Disabled;
    }
    sealed class AtlasRetry {
        public AtlasLoad Atlas;
        public float At;
    }
    static readonly ConditionalWeakTable<UIBundleLoader, State> states = new ConditionalWeakTable<UIBundleLoader, State>();
    static readonly Dictionary<string, AtlasRetry> atlases = new Dictionary<string, AtlasRetry>();
    const BindingFlags flags = BindingFlags.Instance | BindingFlags.NonPublic;
    static readonly FieldInfo texture = typeof(UIBundleLoader).GetField("_assetTex", flags);
    static readonly FieldInfo releaseOnClose = typeof(UIBundleLoader).GetField("_releaseAssetOnClose", flags);
    static readonly MethodInfo release = typeof(UIBundleLoader).GetMethod("ReleaseAssetTexture", flags);
    static readonly MethodInfo load = typeof(UIBundleLoader).GetMethod("LoadAssetTexture", flags);
    static readonly MethodInfo complete = typeof(UIBundleLoader).GetMethod("OnComplete", flags);
    static readonly FieldInfo bundleCache = typeof(AssetCacheService).GetField("_assetBundleDict", flags);
    static readonly FieldInfo assetCache = typeof(AssetCacheService).GetField("_assetDict", flags);
    static readonly FieldInfo bundleTimes = typeof(AssetCacheService).GetField("_cacheDict", flags);
    static readonly FieldInfo bundleOrder = typeof(AssetCacheService).GetField("_sortedCachedTime", flags);
    static readonly FieldInfo resourceCache = typeof(AssetCacheService).GetField("_resourceDict", flags);
    static readonly FieldInfo resourceTimes = typeof(AssetCacheService).GetField("_resourceCacheDict", flags);
    static readonly FieldInfo resourceOrder = typeof(AssetCacheService).GetField("_sortedResourceCachedTime", flags);

    static void RemoveTime(object owner, FieldInfo times, FieldInfo order, string key)
    {
        ((IDictionary)times.GetValue(owner)).Remove(key);
        var sorted = (SortedList)order.GetValue(owner);
        for (int i = sorted.Count - 1; i >= 0; i--)
            if (Equals(sorted.GetByIndex(i), key)) sorted.RemoveAt(i);
    }

    public static void BeforeCachedBundle(AssetCacheService cache, string key)
    {
        var dict = (IDictionary)bundleCache.GetValue(cache);
        if (!dict.Contains(key) || (UnityEngine.Object)dict[key] != null) return;
        dict.Remove(key);
        ((IDictionary)assetCache.GetValue(cache)).Remove(key);
        RemoveTime(cache, bundleTimes, bundleOrder, key);
    }

    public static void BeforeCachedAsset(AssetCacheService cache, string bundle, string key)
    {
        var bundles = (IDictionary)assetCache.GetValue(cache);
        var dict = bundles[bundle] as IDictionary;
        if (dict != null && dict.Contains(key) && (UnityEngine.Object)dict[key] == null) dict.Remove(key);
    }

    public static void BeforeCachedResource(AssetCacheService cache, string key)
    {
        var dict = (IDictionary)resourceCache.GetValue(cache);
        if (!dict.Contains(key) || (UnityEngine.Object)dict[key] != null) return;
        dict.Remove(key);
        RemoveTime(cache, resourceTimes, resourceOrder, key);
    }

    public static void Reset(UIBundleLoader loader) { states.Remove(loader); }

    internal static void PrepareAtlas(string path)
    {
        var link = SingletonAuto<AtlasDataContainer>.instance.GetAtlasLink(path);
        if (link == null || atlases.ContainsKey(link.AtlasCode)) return;
        var atlas = SingletonAuto<NguiAtlasContainer>.instance.Lock(link.AtlasCode) as AtlasLoad;
        if (atlas == null) return;
        bool retained = false;
        try {
            var async = (IAsyncLoad<UIAtlas>)atlas;
            if (!async.IsDone || (async.IsValid && async.Value != null && async.Value.texture != null) || atlas.AssetAsync == null) return;
            // Retain one native atlas lease while its failed underlying request is
            // released. Give native bundle cleanup a frame before reopening it.
            atlas.AssetAsync.Close(true);
            atlas.AssetAsync = null;
            atlases.Add(link.AtlasCode, new AtlasRetry { Atlas = atlas, At = Time.realtimeSinceStartup + 0.25f });
            retained = true;
        } finally {
            if (!retained) NguiAtlasContainer.ReleaseAtlas(atlas, false);
        }
    }

    public static void TickGlobal()
    {
        ProbeTpIcons();
        if (atlases.Count == 0) return;
        var due = new List<string>();
        foreach (var item in atlases) if (Time.realtimeSinceStartup >= item.Value.At) due.Add(item.Key);
        foreach (string code in due) {
            var atlas = atlases[code].Atlas;
            atlases.Remove(code);
            try { atlas.Open(code); }
            catch (Exception error) { Debug.LogWarning("[SPRK Images] Atlas retry failed: " + error.Message); }
            finally { NguiAtlasContainer.ReleaseAtlas(atlas, false); }
        }
    }

    static bool loadingProbeStarted;
    static bool loadingProbeAssets;
    public static void ProbeLoadingBackground()
    {
        string report = Environment.GetEnvironmentVariable("SPRK_LOADING_BACKGROUND_PROBE");
        if (loadingProbeStarted || String.IsNullOrEmpty(report)) return;
        if (!loadingProbeAssets) {
            loadingProbeAssets = true;
            string root = System.IO.Path.Combine(Application.dataPath, "Documents/Patch/StandaloneWindows");
            var files = new Dictionary<string, NShared.AssetInfo>();
            foreach (string file in System.IO.Directory.GetFiles(System.IO.Path.Combine(root, "Assetbundle"), "*", System.IO.SearchOption.AllDirectories)) {
                string entry = file.Substring(root.Length + 1).Replace('\\', '/');
                files[entry] = new NShared.AssetInfo { EntryName = entry };
            }
            SingletonAuto<AssetFileManager>.instance.Setup(files);
        }
        // This isolated, account-free probe reads the already installed assets
        // before login would normally enable the native bundle loader.
        var bundles = SingletonAuto<BundleManager>.instance;
        NGameAsset.Define.PatchDownloaded = true;
        if (!bundles.IsReady) return;
        if (SingletonAuto<AtlasDataContainer>.instance.GetAtlasLink("LoadingTip/GamePlayTip/OrvelTown") == null) return;
        loadingProbeStarted = true;
        new GameObject("SPRK loading background probe").AddComponent<SprkLoadingBackgroundProbe>().Run(report);
    }

    public static void WatchLoadingTip(LoadingGamePlayTipComponent view)
    {
        WatchLoading(view, view.BackgroundImageTexture, "_backgroundImageAsset");
    }

    public static void WatchLoadingHero(LoadingHeroIntroComponent view)
    {
        WatchLoading(view, view.HeroBackgroundImageTexture, "_imageAsset");
    }

    static void WatchLoading(Component view, UITexture target, string field)
    {
        if (target == null) return;
        var guard = view.GetComponent<SprkLoadingBackground>();
        if (guard == null) guard = view.gameObject.AddComponent<SprkLoadingBackground>();
        guard.Open(view, target, view.GetType().GetField(field, flags));
    }

    static bool probedTpIcons;
    static void ProbeTpIcons()
    {
        string report = Environment.GetEnvironmentVariable("SPRK_TP_ICON_PROBE");
        if (probedTpIcons || String.IsNullOrEmpty(report)) return;
        probedTpIcons = true;
        GameObject go = null;
        try {
            go = new GameObject("SPRK TP icon probe");
            go.AddComponent<UITexture>();
            var loader = go.AddComponent<UIBundleLoader>();
            typeof(UIBundleLoader).GetField("_uiTex", flags).SetValue(loader, go.GetComponent<UITexture>());
            loader.SetAssetPath(UIHelper.GetItemIconPath("TRANSCEND_10POINT"), (UIHelper.EAssetTextureAssign)4);
            var ten = loader.UiTex.mainTexture;
            if (ten == null || ten.name != "SPRK 10 TP" || ten.width != 124 || ten.height != 124)
                throw new InvalidOperationException("10 TP did not bind through the native loader");
            loader.SetAssetPath(UIHelper.GetItemIconPath("TRANSCEND_15POINT"), (UIHelper.EAssetTextureAssign)4);
            var fifteen = loader.UiTex.mainTexture;
            if (fifteen == null || fifteen.name != "SPRK 15 TP" || fifteen == ten)
                throw new InvalidOperationException("15 TP did not bind a distinct icon");
            loader.SetAssetPath("", (UIHelper.EAssetTextureAssign)4);
            if (loader.UiTex.mainTexture != null) throw new InvalidOperationException("Pooled icon retained stale TP art");
            loader.SetAssetPath(UIHelper.GetItemIconPath("TRANSCEND_10POINT"), (UIHelper.EAssetTextureAssign)4);
            if (loader.UiTex.mainTexture != ten) throw new InvalidOperationException("10 TP texture was not cached");
            go.SetActive(false); go.SetActive(true);
            if (loader.UiTex.mainTexture != ten) throw new InvalidOperationException("TP icon did not survive re-enable");
            System.IO.File.WriteAllText(report, "{\"success\":true,\"distinctIcons\":true,\"poolReset\":true,\"cacheReuse\":true,\"reenable\":true}");
        } catch (Exception error) {
            System.IO.File.WriteAllText(report, "{\"success\":false,\"error\":\"" + error.ToString().Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\r", "").Replace("\n", "\\n") + "\"}");
        } finally {
            if (go != null) UnityEngine.Object.Destroy(go);
            Application.Quit();
        }
    }

    public static void Tick(UIBundleLoader loader)
    {
        if (!Application.isPlaying || loader == null || !loader.isActiveAndEnabled) return;
        var state = states.GetOrCreateValue(loader);
        float now = Time.realtimeSinceStartup;
        if (now < state.Poll) return;
        state.Poll = now + 0.25f;
        string path = loader.AssetPath;
        if (path != state.Path) {
            state.Path = path;
            state.Attempts = 0;
            state.MissingSince = now;
            state.Released = state.Disabled = false;
        }
        if (state.Disabled || String.IsNullOrEmpty(path)) return;
        var ui = loader.UiTex;
        if (ui == null || !ui.enabled || !ui.gameObject.activeInHierarchy || ui.finalAlpha <= 0f) return;
        try {
            if (state.Released) {
                if (now < state.ReloadAfter) return;
                state.Released = false;
                state.MissingSince = now;
                load.Invoke(loader, null);
                return;
            }
            if (ui.mainTexture != null) {
                state.Attempts = 0;
                state.MissingSince = now;
                return;
            }
            var asset = (AssetTexture)texture.GetValue(loader);
            var info = asset == null ? null : asset.GetTextureInfo();
            if (info != null && info.Tex != null) {
                // A lost widget binding does not require reloading the atlas.
                complete.Invoke(loader, null);
                return;
            }
            if (now - state.MissingSince < (asset != null && !asset.IsDone ? 8f : 0.5f)) return;
            if (state.Attempts >= 3) return;
            state.Attempts++;
            PrepareAtlas(path);
            bool old = (bool)releaseOnClose.GetValue(loader);
            try {
                releaseOnClose.SetValue(loader, true);
                release.Invoke(loader, null);
            } finally { releaseOnClose.SetValue(loader, old); }
            state.Released = true;
            state.ReloadAfter = now + (state.Attempts == 1 ? 0.5f : state.Attempts == 2 ? 1f : 2f);
            if (state.Attempts == 1) Debug.Log("[SPRK Images] Retrying thumbnail: " + path);
        } catch (Exception error) {
            state.Disabled = true;
            Debug.LogWarning("[SPRK Images] Thumbnail recovery stopped: " + error.Message);
        }
    }
}

// Loading images use PlainAssetHolder, not UIBundleLoader. Keep their native
// ownership and completion checks, and repair only the loading image binding.
public sealed class SprkLoadingBackground : MonoBehaviour
{
    Component owner;
    UITexture target;
    FieldInfo assetField;
    AssetTexture asset;
    string path;
    float missingSince, nextPoll;
    int attempts;
    bool watching;

    public void Open(Component view, UITexture texture, FieldInfo field)
    {
        owner = view; target = texture; assetField = field;
        asset = (AssetTexture)field.GetValue(view);
        path = asset == null ? "" : asset.Path;
        missingSince = Time.realtimeSinceStartup;
        nextPoll = 0; attempts = 0; watching = true;
        BindOrCover();
    }

    bool BindOrCover()
    {
        var info = asset == null ? null : asset.GetTextureInfo();
        if (info != null && info.Tex != null) {
            if (target.mainTexture != info.Tex || target.material != info.Mtrl)
                UIHelper.SetTextureInfo(target, info, UIHelper.EAssetTextureAssign.None);
            return true;
        }
        // Native async completion can clear the widget on failure. Cover it
        // again in LateUpdate before rendering, without delaying scene entry.
        target.material = null;
        target.mainTexture = Texture2D.blackTexture;
        target.uvRect = new Rect(0, 0, 1, 1);
        return false;
    }

    void LateUpdate()
    {
        if (!watching || owner == null || target == null) return;
        try {
            // Native release owns the asset and cancels its pending callback.
            if (!ReferenceEquals(assetField.GetValue(owner), asset)) { watching = false; return; }
            if (BindOrCover()) { missingSince = Time.realtimeSinceStartup; return; }
            float now = Time.realtimeSinceStartup;
            if (now < nextPoll || String.IsNullOrEmpty(path) || attempts >= 3) return;
            if (now - missingSince < (asset != null && !asset.IsDone ? 8f : 0.5f)) return;
            attempts++;
            var holder = target.GetComponent<PlainAssetHolder>();
            if (holder != null) holder.Close();
            if (asset != null) asset.Close();
            SprkImageRecovery.PrepareAtlas(path);
            asset = UIHelper.GetPlainTexture(path);
            assetField.SetValue(owner, asset);
            holder = UIHelper.ResetPlaneAssetTextureAsync(owner, target, asset, UIHelper.EAssetTextureAssign.None, null);
            if (holder != null) holder.ReleaseAssetOnClose = true;
            missingSince = now;
            nextPoll = now + attempts * 0.5f;
            BindOrCover();
            if (attempts == 1) Debug.LogWarning("[SPRK Images] Retrying loading background: " + path);
        } catch (Exception error) {
            watching = false;
            Debug.LogWarning("[SPRK Images] Loading background recovery stopped: " + error.Message);
        }
    }

    void OnDisable() { watching = false; }
}

// Opt-in regression probe runs in a separate client process, through the patched
// native initializer and real atlas/PlainAssetHolder asynchronous lifecycle.
public sealed class SprkLoadingBackgroundProbe : MonoBehaviour
{
    string report;
    GameObject viewObject;
    readonly List<string> passed = new List<string>();
    const BindingFlags flags = BindingFlags.Instance | BindingFlags.NonPublic;
    static readonly MethodInfo initialize = typeof(LoadingGamePlayTipComponent).GetMethod("InitBackgroundImage", flags);
    static readonly FieldInfo nativeAsset = typeof(LoadingGamePlayTipComponent).GetField("_backgroundImageAsset", flags);

    public void Run(string path) { report = path; StartCoroutine(Observe(Check())); }
    IEnumerator Observe(IEnumerator checks)
    {
        while (true) {
            object current = null;
            bool more = false;
            Exception failure = null;
            try { more = checks.MoveNext(); if (more) current = checks.Current; }
            catch (Exception error) { failure = error; }
            if (failure != null || !more) {
                string error = failure == null ? "" : failure.ToString().Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\r", "").Replace("\n", "\\n");
                System.IO.File.WriteAllText(report, "{\"success\":" + (failure == null ? "true" : "false") + ",\"checks\":[\"" + String.Join("\",\"", passed.ToArray()) + "\"],\"error\":\"" + error + "\"}");
                if (viewObject != null) Destroy(viewObject);
                Application.Quit();
                yield break;
            }
            yield return current;
        }
    }
    void Require(bool value, string name)
    {
        if (!value) throw new InvalidOperationException(name);
        passed.Add(name);
    }
    IEnumerator Check()
    {
        viewObject = new GameObject("Loading tip test");
        var imageObject = new GameObject("Background");
        imageObject.transform.SetParent(viewObject.transform, false);
        var texture = imageObject.AddComponent<UITexture>();
        var view = viewObject.AddComponent<LoadingGamePlayTipComponent>();
        view.BackgroundImageTexture = texture;
        initialize.Invoke(view, new object[] { "OrvelTown" });
        Require(texture.mainTexture != null, "opaque before asynchronous completion");
        float deadline = Time.realtimeSinceStartup + 15f;
        while ((texture.mainTexture == null || texture.mainTexture == Texture2D.blackTexture) && Time.realtimeSinceStartup < deadline) yield return null;
        Require(texture.mainTexture != null && texture.mainTexture != Texture2D.blackTexture, "native background atlas loads");
        var original = texture.mainTexture;
        texture.mainTexture = null;
        yield return null; yield return null;
        Require(texture.mainTexture == original, "lost binding repaired without reloading");

        var holder = texture.GetComponent<PlainAssetHolder>();
        holder.Close();
        var invalid = new AssetTexture(); invalid.Open("LoadingTip/GamePlayTip/SPRK_MISSING_TEST");
        nativeAsset.SetValue(view, invalid);
        SprkImageRecovery.WatchLoadingTip(view);
        Require(texture.mainTexture == Texture2D.blackTexture, "failed request has opaque fallback");
        deadline = Time.realtimeSinceStartup + 4f;
        while (Time.realtimeSinceStartup < deadline) yield return null;
        var guard = view.GetComponent<SprkLoadingBackground>();
        Require((int)typeof(SprkLoadingBackground).GetField("attempts", flags).GetValue(guard) == 3, "failed requests retry exactly three times");
        var exhausted = nativeAsset.GetValue(view);
        deadline = Time.realtimeSinceStartup + 1f;
        while (Time.realtimeSinceStartup < deadline) yield return null;
        Require(ReferenceEquals(exhausted, nativeAsset.GetValue(view)), "exhausted request does not loop");

        initialize.Invoke(view, new object[] { "OrvelTown" });
        deadline = Time.realtimeSinceStartup + 15f;
        while ((texture.mainTexture == null || texture.mainTexture == Texture2D.blackTexture) && Time.realtimeSinceStartup < deadline) yield return null;
        Require(texture.mainTexture != null && texture.mainTexture != Texture2D.blackTexture, "next loading screen recovers original art");
        viewObject.SetActive(false);
        typeof(LoadingGamePlayTipComponent).GetMethod("NVespa.NPrefab2.InitOnCreate.OnRelease", flags).Invoke(view, null);
        Require(nativeAsset.GetValue(view) == null && texture.mainTexture == null, "native pool release clears asset and binding");
        viewObject.SetActive(true);
        initialize.Invoke(view, new object[] { "OrvelTown" });
        deadline = Time.realtimeSinceStartup + 15f;
        while ((texture.mainTexture == null || texture.mainTexture == Texture2D.blackTexture) && Time.realtimeSinceStartup < deadline) yield return null;
        Require(texture.mainTexture != null && texture.mainTexture != Texture2D.blackTexture, "pooled loading screen reloads art");
    }
}
