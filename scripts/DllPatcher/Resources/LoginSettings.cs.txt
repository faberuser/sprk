public sealed class SprkLoginSettings : MonoBehaviour,GameOptionWindow.Listener,GameOptionWindow.IDataProvider {
    LoginAccountInfo login;
    GameOptionWindow settings;
    public static void Configure(LoginAccountInfo view) {
        if(view==null)return;
        if(view.PatchValidationRoot!=null)view.PatchValidationRoot.SetActive(false);
        if(view.VoiceButtonRoot!=null)view.VoiceButtonRoot.SetActive(false);
        if(view.LanguageButtonRoot==null)return;
        var bridge=view.GetComponent<SprkLoginSettings>();
        if(bridge==null)bridge=view.gameObject.AddComponent<SprkLoginSettings>();
        bridge.login=view;
        foreach(var loc in view.LanguageButtonRoot.GetComponentsInChildren<UILocalize>(true))loc.enabled=false;
        foreach(var loc in view.LanguageButtonRoot.GetComponentsInChildren<NVespa.NNGUIExtension.UILocalizeEx>(true))loc.enabled=false;
        foreach(var label in view.LanguageButtonRoot.GetComponentsInChildren<UILabel>(true))label.text="Settings";
        var button=view.LanguageButtonRoot.GetComponentInChildren<UIButton>(true);
        if(button!=null){button.onClick.Clear();EventDelegate.Add(button.onClick,bridge.OpenSettings);}
        // TAP TO PLAY already opens the account form, so the Log In button is redundant.
        if(view.LoginButtonRoot!=null)view.LoginButtonRoot.SetActive(false);
        // The server name is chosen in the account form; hide the bottom-left row.
        var server=view.Label_ServerName!=null?view.Label_ServerName.transform.parent:null;
        if(server!=null && server.name=="ServerInfo"){server.gameObject.SetActive(false);if(view.AccountInfoGrid!=null)view.AccountInfoGrid.Reposition();}
        var intro=view.GameIntroduceButtonRoot;
        if(intro!=null && view.ButtonGrid!=null) {
            foreach(var loc in intro.GetComponentsInChildren<UILocalize>(true))loc.enabled=false;
            foreach(var loc in intro.GetComponentsInChildren<NVespa.NNGUIExtension.UILocalizeEx>(true))loc.enabled=false;
            foreach(var label in intro.GetComponentsInChildren<UILabel>(true))label.text="Introduction";
            // Introduction sits on the Settings row, to its left, outside the vertical grid.
            if(intro.transform.parent==view.ButtonGrid.transform)intro.transform.SetParent(view.ButtonGrid.transform.parent,false);
        }
        if(view.ButtonGrid!=null){view.ButtonGrid.hideInactive=true;view.ButtonGrid.Reposition();}
        bridge.PlaceIntroduction();
    }
    public static void ConfigureTouch(LoginTouchWait view) {
        if(view!=null && view.GetComponent<SprkTapToPlay>()==null)view.gameObject.AddComponent<SprkTapToPlay>();
    }
    void LateUpdate(){PlaceIntroduction();}
    void PlaceIntroduction() {
        if(login==null || login.GameIntroduceButtonRoot==null || login.LanguageButtonRoot==null || login.ButtonGrid==null)return;
        var intro=login.GameIntroduceButtonRoot.transform;var root=login.transform.Find("Panel/Root");
        if(intro.parent!=login.ButtonGrid.transform.parent || root==null)return;
        // Bottom-left, where the Server row was: mirror Settings across the screen
        // so both share one line and the same edge margin.
        var settings=root.InverseTransformPoint(login.LanguageButtonRoot.transform.position);
        intro.position=root.TransformPoint(new Vector3(-settings.x,settings.y,settings.z));
    }
    static void PrepareOptions() {
        // Load only the settings table from the installed client. Do not mark
        // patch download complete or unlock unrelated gameplay tables at login.
        var cache=(IDictionary)typeof(TableRoot).GetField("_dic",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(TableRoot.Instance);
        lock(cache) {
            if(cache.Contains(typeof(GameOptionDataContainer)))return;
            var table=new GameOptionDataContainer();table.Load();
            if(!table.IsLoaded)throw new InvalidOperationException("The installed settings table could not be loaded.");
            cache.Add(typeof(GameOptionDataContainer),table);
        }
    }
    public void OpenSettings() {
        PrepareOptions();
        // Login normally has only the small embedded localization dictionary.
        // Settings uses the installed full dictionary and enum redirections.
        SingletonAuto<NClientData.LocalizationLoader>.instance.SetEnableLoadFromPatch(true);
        var manager=SceneAutoBehaviour<LoginManagement>.instance;
        settings=manager.GetInstance<GameOptionWindow>();
        if(settings==null){manager.AddInstance<GameOptionWindow>(GameOptionWindow.PrefabPath);settings=manager.GetInstance<GameOptionWindow>();}
        settings.Open(this,this,GameOptionTab.Game);
    }
    public sealed class LoginContext : MonoBehaviour {
        public readonly Dictionary<GameObject,bool> Hidden=new Dictionary<GameObject,bool>();
        public void Hide(GameObject go){if(go==null)return;if(!Hidden.ContainsKey(go))Hidden.Add(go,go.activeSelf);go.SetActive(false);}
        public void Restore(){foreach(var item in Hidden)if(item.Key!=null)item.Key.SetActive(item.Value);Hidden.Clear();}
    }
    public static void PrepareSettings(GameOptionWindow view,GameOptionWindow.IDataProvider provider) {
        var context=view.GetComponent<LoginContext>();
        if(context!=null)context.Restore();
        if(provider is SprkLoginSettings){if(context==null)view.gameObject.AddComponent<LoginContext>();}
        else if(context!=null)UnityEngine.Object.DestroyImmediate(context);
    }
    public static void CleanSettings(GameOptionWindow view) {
        var context=view.GetComponent<LoginContext>();if(context==null || !AtLogin)return;
        var info=view.EtcInfo;
        foreach(var item in new[]{info.Button_Coupon,info.Button_Logout,info.Button_MemberLeave})if(item!=null)context.Hide(item.gameObject);
        foreach(var name in new[]{"Button_SwitchAccount","Button_ChangePassword","Button_DeleteAccount"})
            foreach(var tm in view.GetComponentsInChildren<Transform>(true))if(tm.name==name)context.Hide(tm.gameObject);
        foreach(var row in view.GetComponentsInChildren<GameOptionBaseComponent>(true))
            if(row.gameOptionData!=null && (row.gameOptionData.PostProcessFunction=="OpenSkipRubyCountPopup" || row.gameOptionData.PostProcessFunction=="ManageBlockedUser"))context.Hide(row.gameObject);
    }
    static readonly Dictionary<string,UIAtlas> loginAtlases=new Dictionary<string,UIAtlas>();
    public static UIAtlas TryLoginAtlas(string name) {
        if((name!="GameOption" && name!="PayShop") || !AtLogin)return null;
        return LoadLoginAtlas(name);
    }
    static UIAtlas LoadLoginAtlas(string name) {
        string folder=SystemOptionData.Spec==ESpecType.Low?"LowLargeSpec":"HighAndMidSpec";
        string key=folder+"/"+name;
        UIAtlas atlas;if(loginAtlases.TryGetValue(key,out atlas) && atlas!=null)return atlas;
        string path=System.IO.Path.Combine(Application.dataPath,"Documents/Patch/StandaloneWindows/Assetbundle",folder,"NguiAtlas_UI_Prefab_"+name+".unity3d");
        var bundle=AssetBundle.LoadFromFile(path);
        if(bundle==null)throw new InvalidOperationException("The installed settings artwork could not be loaded.");
        atlas=bundle.mainAsset as UIAtlas;
        if(atlas==null)throw new InvalidOperationException("The installed settings artwork has no atlas.");
        // Bundle materials depend on the game's later shader preload. Use the
        // already-shipped NGUI shader while opening these atlases before login.
        var shader=Shader.Find("Unlit/Transparent Colored");
        if(shader==null)throw new InvalidOperationException("The login UI shader is unavailable.");
        var material=new Material(shader);material.mainTexture=atlas.texture;
        if(material.mainTexture==null)throw new InvalidOperationException("The installed settings artwork has no texture.");
        atlas.spriteMaterial=material;bundle.Unload(false);
        loginAtlases[key]=atlas;return atlas;
    }
    static bool AtLogin { get { return SceneAutoBehaviour<LoginManagement>.isValidInstance && !SceneAutoBehaviour<NGame2.NUI.NManager.NLobby.LobbyManagement>.isValidInstance; } }
    public static bool SetTextureQuality(GameOptionBaseComponent row,object value) {
        if(!AtLogin)return false;
        SystemOptionData.Spec=(ESpecType)Enum.Parse(typeof(ESpecType),value.ToString());SprkTestPrefs.Save();
        return true;
    }
    public static bool SetPlayBGM(GameOptionBaseComponent row,object value) { if(!AtLogin)return false;SprkLoginPresentation.RefreshMusic();return true; }
    public LoginMethodType GetLastLoginMethod(){return LoginData.LastLoginMethod;}
    public CountryData GetCountryData(){return null;}
    public CountryServerData GetCountryServerData(){return null;}
    public void OnChangeLanguage(){login.OnClickChangeLanguage();}
    public void OnLogout(){login.OnClickLogout();}
    public void OnReset(){}
    public void OnUseCoupon(){}
    public void OnLeaveParty(){}
    public void ChangeLoginMethod(LoginMethodType type){}
    public void OnCsMailPopup(){}
    public void OnOqupieWebView(){}
    static bool probeStarted;
    public static void EnsureProbe() {
        SprkLoginPresentation.EnsureProbe();
        if(probeStarted || String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_LOGIN_SETTINGS_TEST")) || String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR")) || String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE")))return;
        probeStarted=true;
        var host=new GameObject("LoginSettingsProbe");UnityEngine.Object.DontDestroyOnLoad(host);host.AddComponent<SprkLoginSettingsProbe>();
    }
}
// TAP TO PLAY sits lower and smaller, centered on the Introduction/Settings row.
public sealed class SprkTapToPlay : MonoBehaviour {
    public const float Scale=.8f;
    // Root-space center of the Settings caption on the 1920x1080 login layout.
    public const float FallbackY=-464f;
    // Visible text center: the caption box is taller than its bottom-aligned text.
    public static float CaptionCenterY(UILabel caption,Transform space){var bottom=(caption.worldCorners[0]+caption.worldCorners[3])*.5f;return space.InverseTransformPoint(bottom+caption.transform.TransformVector(new Vector3(0f,caption.printedSize.y*.5f,0f))).y;}
    UIWidget art;Transform root;LoginAccountInfo menu;float nextSearch;
    // World Y of the TAP TO PLAY line from the last frame that placed it from the
    // Settings caption; the status message reuses it after this window closes.
    static float lineWorldY;static bool hasLine;
    public static bool TryGetLineWorldY(out float y){y=lineWorldY;return hasLine;}
    void Start() {
        root=transform.Find("Panel/Root");
        var tm=root!=null?root.Find("Sprite_backGround"):null;
        art=tm!=null?tm.GetComponent<UIWidget>():null;
        if(art==null)return;
        art.SetAnchor((Transform)null);
        art.transform.localScale=new Vector3(Scale,Scale,1f);
        LateUpdate();
    }
    void LateUpdate() {
        if(art==null)return;
        if(menu==null && Time.unscaledTime>=nextSearch){nextSearch=Time.unscaledTime+.5f;menu=FindObjectOfType<LoginAccountInfo>();}
        float y=FallbackY;
        UILabel caption=menu!=null && menu.LanguageButtonRoot!=null?menu.LanguageButtonRoot.GetComponentInChildren<UILabel>(true):null;
        if(caption!=null && caption.gameObject.activeInHierarchy) {
            y=CaptionCenterY(caption,root);
        }
        var position=art.transform.localPosition;
        if(Mathf.Abs(position.x)>.01f || Mathf.Abs(position.y-y)>.01f)art.transform.localPosition=new Vector3(0f,y,position.z);
        lineWorldY=root.TransformPoint(new Vector3(0f,y,0f)).y;hasLine=true;
    }
}
public sealed class SprkLoginSettingsProbe : MonoBehaviour {
    IEnumerator Start() {
        yield return new WaitForSecondsRealtime(4);
        var manager=SceneAutoBehaviour<LoginManagement>.instance;
        if(manager==null)throw new Exception("No native login management");
        var login=manager.GetInstance<LoginAccountInfo>();
        for(int i=0;i<8;i++)SprkLoginSettings.Configure(login);
        if(login.PatchValidationRoot.activeSelf || login.VoiceButtonRoot.activeSelf)throw new Exception("Legacy buttons visible");
        var button=login.LanguageButtonRoot.GetComponentInChildren<UIButton>(true);
        if(button.onClick.Count!=1)throw new Exception("Duplicate Settings callbacks");
        foreach(var label in login.LanguageButtonRoot.GetComponentsInChildren<UILabel>(true))if(label.text!="Settings")throw new Exception("Settings caption");
        bool patchReady=TableRoot.PatchDownloadComplete;
        EventDelegate.Execute(button.onClick);
        yield return new WaitForSecondsRealtime(1);
        var settings=manager.GetInstance<GameOptionWindow>();
        if(settings==null || !settings.IsOpen())throw new Exception("Native settings did not open");
        foreach(var label in settings.GetComponentsInChildren<UILabel>(true)) {
            var value=label.text??"";
            if(value.StartsWith("@") || value.StartsWith("OPTION_") || value=="CONTENT_GAMECONFIG" || value=="COMMON_ETC")throw new Exception("Untranslated settings label: "+value);
        }
        var rows=settings.GetComponentsInChildren<GameOptionBaseComponent>(true);
        if(rows.Length<20)throw new Exception("Native settings rows missing: "+rows.Length);
        int artwork=0;
        foreach(var assigner in settings.GetComponentsInChildren<NGame2.NUI.NAtlasAssignComponent.UISpriteAtlasAssigner>(true)) {
            if((assigner.AtlasName!="GameOption" && assigner.AtlasName!="PayShop") || assigner.Target==null)continue;
            if(assigner.Target.atlas==null || assigner.Target.atlas.GetSprite(assigner.SpriteName)==null)throw new Exception("Missing settings artwork: "+assigner.SpriteName);
            artwork++;
        }
        if(artwork==0)throw new Exception("Settings artwork was not checked");
        if(TableRoot.PatchDownloadComplete!=patchReady)throw new Exception("Settings changed download state");
        if(settings.EtcInfo.Button_Coupon.gameObject.activeSelf || settings.EtcInfo.Button_Logout.gameObject.activeSelf)throw new Exception("Lobby actions visible at login");
        var texture=Array.Find(rows,r=>r.gameOptionData!=null && r.gameOptionData.PostProcessFunction=="SetTextureQuality");
        if(texture==null)throw new Exception("Native texture row missing");
        var originalSpec=SystemOptionData.Spec;
        var otherSpec=originalSpec==ESpecType.Low?ESpecType.Mid:ESpecType.Low;
        texture.SetTextureQuality(otherSpec);if(SystemOptionData.Spec!=otherSpec)throw new Exception("Login texture change failed");
        texture.SetTextureQuality(originalSpec);
        foreach(var tab in settings.MainTabs){tab.Toggle.Set(true);yield return null;}
        settings.ReOpen();yield return null;
        settings.Close();yield return null;
        EventDelegate.Execute(button.onClick);yield return null;
        if(!settings.IsOpen() || settings.GetComponents<SprkLoginSettings.LoginContext>().Length!=1)throw new Exception("Reopen or duplicate context");
        var path=Environment.GetEnvironmentVariable("SPRK_LOGIN_SETTINGS_TEST");System.IO.Directory.CreateDirectory(path);
        // The standalone test has no host server to advance beyond the BI
        // intro. Remove that test-only overlay so the native panel is visible.
        foreach(var camera in Camera.allCameras)if(camera.name=="BI UI Camera")camera.gameObject.SetActive(false);
        yield return new WaitForSecondsRealtime(1);
        yield return new WaitForEndOfFrame();
        var diagnostics=new List<string>();
        for(var tm=settings.transform;tm!=null;tm=tm.parent)diagnostics.Add(tm.name+" active="+tm.gameObject.activeInHierarchy+" pos="+tm.localPosition+" scale="+tm.localScale);
        foreach(var panel in settings.GetComponentsInChildren<UIPanel>(true))diagnostics.Add("Panel "+panel.name+" alpha="+panel.alpha+" depth="+panel.depth+" active="+panel.gameObject.activeInHierarchy);
        foreach(var camera in Camera.allCameras)diagnostics.Add("Camera "+camera.name+" mask="+camera.cullingMask+" depth="+camera.depth+" rect="+camera.rect);
        System.IO.File.WriteAllLines(System.IO.Path.Combine(path,"diagnostics.txt"),diagnostics.ToArray());
        var image=new Texture2D(Screen.width,Screen.height,TextureFormat.RGB24,false);image.ReadPixels(new Rect(0,0,Screen.width,Screen.height),0,0);image.Apply();
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(path,"settings.png"),image.EncodeToPNG());Destroy(image);
        var context=settings.GetComponent<SprkLoginSettings.LoginContext>();
        SprkLoginSettings.PrepareSettings(settings,null);
        if(settings.GetComponent<SprkLoginSettings.LoginContext>()!=null || !settings.EtcInfo.Button_Coupon.gameObject.activeSelf)throw new Exception("Login cleanup leaked into a reused in-game panel");
        // Exercise the actual native Others prefab after restoring its lobby actions.
        foreach(var tab in settings.MainTabs)if(tab.GameOptionTab==GameOptionTab.Etc)tab.Toggle.Set(true);
        yield return null;
        for(int i=0;i<8;i++)SprkAccounts.CleanSettings(settings);
        var accountParent=settings.EtcInfo.Button_Logout.parent;
        var language=accountParent.Find("Button_ChangeLanguage");
        var coupon=settings.EtcInfo.Button_Coupon;
        var change=accountParent.Find("Button_ChangePassword");
        var delete=accountParent.Find("Button_DeleteAccount");
        var account=accountParent.Find("Button_SwitchAccount");
        var logout=settings.EtcInfo.Button_Logout;
        if(language==null || change==null || delete==null || account==null)throw new Exception("Others controls missing");
        if(!(language.localPosition.x<account.localPosition.x && account.localPosition.x<logout.localPosition.x) ||
           coupon.localPosition.x!=language.localPosition.x || coupon.localPosition.y<=language.localPosition.y)throw new Exception("Coupon must be above Language");
        if(change.localPosition.x!=account.localPosition.x || delete.localPosition.x!=logout.localPosition.x || change.localPosition.y<=account.localPosition.y || delete.localPosition.y!=change.localPosition.y || coupon.localPosition.y!=change.localPosition.y)throw new Exception("Others upper row alignment");
        var deleteButton=delete.GetComponent<UIButton>();
        foreach(var state in new[]{UIButtonColor.State.Normal,UIButtonColor.State.Hover,UIButtonColor.State.Pressed}) {
            deleteButton.SetState(state,true);
            var target=deleteButton.tweenTarget.GetComponent<UISprite>();
            if(target==null || target.spriteName!="Common_Button7_Cancel" || target.atlas==null || target.atlas.GetSprite(target.spriteName)==null)throw new Exception("Delete background is not red: "+state+" target="+deleteButton.tweenTarget.name+" sprite="+(target==null?"null":target.spriteName)+" atlas="+(target==null || target.atlas==null?"null":target.atlas.name));
        }
        deleteButton.SetState(UIButtonColor.State.Normal,true);
        foreach(var tm in new[]{change,delete,account})if(tm.GetComponent<UIButton>().onClick.Count!=1)throw new Exception("Duplicate account callbacks");
        yield return new WaitForEndOfFrame();
        image=new Texture2D(Screen.width,Screen.height,TextureFormat.RGB24,false);image.ReadPixels(new Rect(0,0,Screen.width,Screen.height),0,0);image.Apply();
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(path,"others.png"),image.EncodeToPNG());Destroy(image);
        System.IO.File.WriteAllText(System.IO.Path.Combine(path,"others-result.txt"),"PASS: native Others prefab Coupon above Language, upper alignment, eight refreshes, single callbacks and red Normal/Hover/Pressed backgrounds.");
        System.IO.File.WriteAllText(System.IO.Path.Combine(path,"result.txt"),"PASS: native login buttons cleaned, one Settings callback, native settings opened before login with "+rows.Length+" rows; translated labels, GameOption and PayShop artwork, all tabs, login texture changes, refresh, close/reopen, pool cleanup and download-state preservation; visible panel screenshot exported.");
        Application.Quit();
    }
}
