namespace UnityEngine { public static class Time { public static float realtimeSinceStartup = 10; } }
namespace NShared {
    public enum BaseResultType { Failure, Success }
    public sealed class TutorialInfo { public int TutorialIndex; }
    namespace CompleteTutorial {
        public enum ResultType { Failure, Success }
        public sealed class Request { public int TutorialIndex; }
        public sealed class Response {
            public BaseResultType BaseResult; public ResultType Result; public TutorialInfo Info;
            public object CurrencyResults, DungeonInfos, ItemResults, EquipItemResults, HeroInfos, TeamExpResultInfos, StaminaResultInfos;
        }
    }
}
namespace NVespa.NSingleton {
    public static class GlobalAutoBehaviour<T> where T:new() { public static bool isValidInstance=true; public static T instance=new T(); }
}
namespace NVespa {
    public static class SingletonUpdater<T> where T:new() { public static T instance=new T(); }
}
namespace NGame2.NAccount {
    public sealed class Manager {
        public int Applies; public void Apply(object value) { Applies++; }
    }
    public sealed class TutorialManager {
        public HashSet<int> Completed=new HashSet<int>();
        public bool IsComplete(int index) { return Completed.Contains(index); }
        public void Apply(TutorialInfo info) { Completed.Add(info.TutorialIndex); }
    }
    public sealed class CampaignManager {
        public bool Cleared; public int Applies;
        public bool IsCompleted(int chapter,int dungeon) { return chapter==1 && dungeon==20 && Cleared; }
        public void Apply(object value) { Applies++; }
    }
    public sealed class AccountManager {
        public bool IsTutorialSkip; public int HeroApplications, Announcements;
        public TutorialManager TutorialManager=new TutorialManager();
        public CampaignManager CampaignManager=new CampaignManager();
        public Manager UserManager=new Manager(), ItemManager=new Manager(), EquipItemManager=new Manager();
        public void ApplyAddHeroForTutorial(object heroes,object exp,bool reserve) { HeroApplications++; if(!reserve)Announcements++; }
    }
}
public sealed class WebServiceManager { public Requester Requester=new Requester(); }
public sealed class Requester {
    public int Count, Index; public Action<NShared.CompleteTutorial.Response> Callback;
    public void Request<TRequest,TResponse>(TRequest request,Action<TResponse> callback) {
        Count++; Index=((NShared.CompleteTutorial.Request)(object)request).TutorialIndex;
        Callback=response=>callback((TResponse)(object)response);
    }
}
public static partial class SprkStoryAuto {
    static bool clauseCompletionRequested, active=true; static float lastProgress, clauseResumeAfter; static int stops;
    static bool Active() { return active; }
    static void Stop() { active=false; stops++; }
    static void Check(bool value,string message) { if(!value)throw new Exception(message); }
    static AccountManager Reset() {
        clauseCompletionRequested=false;active=true;stops=0;lastProgress=0;clauseResumeAfter=0;UnityEngine.Time.realtimeSinceStartup=10;
        GlobalAutoBehaviour<AccountManager>.isValidInstance=true;
        var account=new AccountManager { IsTutorialSkip=true };account.CampaignManager.Cleared=true;
        GlobalAutoBehaviour<AccountManager>.instance=account;
        SingletonUpdater<WebServiceManager>.instance=new WebServiceManager();return account;
    }
    static Requester Requests { get { return SingletonUpdater<WebServiceManager>.instance.Requester; } }
    static NShared.CompleteTutorial.Response Success(int index) {
        return new NShared.CompleteTutorial.Response { BaseResult=BaseResultType.Success,
            Result=NShared.CompleteTutorial.ResultType.Success,Info=new TutorialInfo {TutorialIndex=index} };
    }
    public static void RunTests() {
        var account=Reset();account.IsTutorialSkip=false;
        Check(!RecoverSkippedClause() && Requests.Count==0,"Native tutorial playback was intercepted");
        account=Reset();account.CampaignManager.Cleared=false;
        Check(!RecoverSkippedClause() && Requests.Count==0,"Reward requested before 1-20 clear");
        account=Reset();account.TutorialManager.Completed.Add(20002);
        Check(!RecoverSkippedClause() && Requests.Count==0,"Completed recruitment was replayed");
        account=Reset();
        Check(RecoverSkippedClause() && Requests.Index==20001,"Missing first recovery receipt");
        Check(RecoverSkippedClause() && Requests.Count==1,"Duplicate in-flight request");
        Requests.Callback(Success(20001));
        Check(account.TutorialManager.IsComplete(20001) && lastProgress==10,"First receipt not applied");
        Check(RecoverSkippedClause() && Requests.Count==1,"Recruitment opening grace period was skipped");
        UnityEngine.Time.realtimeSinceStartup++;
        Check(RecoverSkippedClause() && Requests.Index==20002,"Recruitment receipt not chained");
        Requests.Callback(Success(20002));
        UnityEngine.Time.realtimeSinceStartup++;
        Check(!RecoverSkippedClause() && Requests.Count==2,"Recruitment did not unblock next entry");
        Check(account.HeroApplications==2 && account.Announcements==2 && account.UserManager.Applies==4 &&
            account.ItemManager.Applies==2 && account.EquipItemManager.Applies==2 && account.CampaignManager.Applies==2,
            "Reward managers or native recruitment announcement were skipped");
        account=Reset();account.TutorialManager.Completed.Add(20001);RecoverSkippedClause();active=false;
        Requests.Callback(Success(20002));
        Check(account.TutorialManager.IsComplete(20002) && account.HeroApplications==1 && !active,"Stop discarded an awarded hero or restarted the run");
        foreach(var bad in new[] { (NShared.CompleteTutorial.Response)null, new NShared.CompleteTutorial.Response(), Success(999) }) {
            account=Reset();RecoverSkippedClause();Requests.Callback(bad);
            Check(stops==1 && account.HeroApplications==0 && !clauseCompletionRequested,"Invalid response continued or applied rewards");
        }
        account=Reset();RecoverSkippedClause();var callback=Requests.Callback;
        var replacement=new AccountManager();GlobalAutoBehaviour<AccountManager>.instance=replacement;callback(Success(20001));
        Check(account.HeroApplications==0 && replacement.HeroApplications==0,"Response crossed account sessions");
        Console.WriteLine("Continuous Story recruitment: eligibility, receipt ordering, in-flight deduplication, reward/announcement application, cancellation, failures and account isolation passed.");
    }
}
