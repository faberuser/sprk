using System;
using System.Reflection;
using UnityEngine;
using NVespa;
using NVespa.NSingleton;
using NGame2.NAccount;
using NGame2.NMatch;
using NGame2.NUI.NWindow;
using NGame2.NUI.NManager;

// Friendly duels in League of Honor. The native friend menu only offered a
// League of Victory duel. GroupId (the inviter's account ID) travels with every
// native duel message and the arena registration, so an Honor duel uses
// accountId + GroupOffset; the server applies the same rule (duel.rs).
public static class SprkHonorDuel {
    public const long GroupOffset = 1L << 40;
    public const string ButtonName = "SprkHonorDuelButton";
    public static bool IsHonor(long groupId) { return groupId > GroupOffset; }
    // MatchPvpType: 0 League of Victory, 1 League of Honor (ban/pick).
    public static int PvpType(long groupId) { return IsHonor(groupId) ? 1 : 0; }

    static readonly FieldInfo InspectUserField = typeof(InspectUserMenuPopup).GetField("_inspectUser", BindingFlags.Instance | BindingFlags.NonPublic);
    static readonly MethodInfo BanPickFail = typeof(MatchManagement).GetMethod("OnFailCheckBanPickCondition", BindingFlags.Static | BindingFlags.NonPublic);

    static readonly MethodInfo RefreshButton = typeof(InspectUserMenuPopup).GetMethod("RefreshButton", BindingFlags.Instance | BindingFlags.NonPublic);
    // Replaces the menu's native RefreshButton calls (initial layout and after
    // the peer's CanDuel response): add the Honor button, then lay out natively.
    public static void SetupAndRefresh(InspectUserMenuPopup menu) {
        Setup(menu);
        if (RefreshButton != null) RefreshButton.Invoke(menu, null);
    }
    public static void Setup(InspectUserMenuPopup menu) {
        if (menu == null || menu.DuelButton == null) return;
        var victory = menu.DuelButton;
        Rename(victory.gameObject, "Victory Duel");
        var parent = victory.transform.parent;
        var existing = parent.Find(ButtonName);
        GameObject honor;
        if (existing != null) honor = existing.gameObject;
        else {
            honor = (GameObject)UnityEngine.Object.Instantiate(victory.gameObject, parent, false);
            honor.name = ButtonName;
            var button = honor.GetComponent<UIButton>();
            // The native OnClickButton compares against DuelButton; give the
            // clone its own action so it can never send a Victory invitation.
            if (button != null) { button.onClick.Clear(); EventDelegate.Add(button.onClick, delegate { Request(menu); }); }
        }
        honor.transform.SetSiblingIndex(victory.transform.GetSiblingIndex() + 1);
        Rename(honor, "Honor Duel");
        // Same visibility and peer eligibility as the native duel button.
        if (honor.activeSelf != victory.gameObject.activeSelf) honor.SetActive(victory.gameObject.activeSelf);
        // Same native enabled/grayscale state as Victory (the clone may come from a disabled, grey button).
        var clone = honor.GetComponent<UIButton>();
        if (clone != null) {
            NGame2.NUI.UIHelper.EnableButton(clone, victory.isEnabled, true, false);
            SyncLook(victory, clone);
        }
        if (honor.activeSelf != victory.gameObject.activeSelf) honor.SetActive(victory.gameObject.activeSelf);
    }

    // NGUI buttons capture their normal sprite/colour on first use, so a clone of
    // the disabled (grey) duel button keeps grey as its "normal" look. Copy the
    // Victory button's state sprites, colours and current background look.
    static void SyncLook(UIButton victory, UIButton clone) {
        clone.normalSprite = victory.normalSprite;
        clone.hoverSprite = victory.hoverSprite;
        clone.pressedSprite = victory.pressedSprite;
        clone.disabledSprite = victory.disabledSprite;
        clone.defaultColor = victory.defaultColor;
        clone.hover = victory.hover; clone.pressed = victory.pressed; clone.disabledColor = victory.disabledColor;
        var from = victory.tweenTarget != null ? victory.tweenTarget.GetComponent<UISprite>() : victory.GetComponent<UISprite>();
        var to = clone.tweenTarget != null ? clone.tweenTarget.GetComponent<UISprite>() : clone.GetComponent<UISprite>();
        if (from != null && to != null) {
            to.atlas = from.atlas; to.spriteName = from.spriteName; to.color = from.color; to.isGrayscale = from.isGrayscale;
        }
    }

    static bool IsFriendlyMatch() {
        return SingletonUpdater<MatchManager>.isValidInstance && (int)SingletonUpdater<MatchManager>.instance.DuelType == 1;
    }
    // MatchManager.OnRegistered_Pvp: a friendly Honor duel pairs the two invited
    // players at once, so skip the ranked Honor matching screen; the native
    // pairing opens the ban/pick draft directly.
    public static void OpenHonorWaiting(MatchManagement management) {
        if (management == null || IsFriendlyMatch()) return;
        management.OpenMatchWaiting_BanPickOnly();
    }
    // PvpResult.Init_BanPickOnly: like Victory duels, a friendly result has no
    // score/ranking section and no ranked Retry. The window is reused, so ranked
    // results set the section explicitly too.
    public static void FriendlyResult(NGame2.NUI.NWindow.NBattle.PvpResult result) {
        if (result == null) return;
        bool friendly = IsFriendlyMatch();
        if (result.BottomRoot != null) result.BottomRoot.gameObject.SetActive(!friendly);
        if (friendly && result.RetryButton != null) NGame2.NUI.UIHelper.EnableButton(result.RetryButton, false, false, false);
    }

    // MatchManager.UpdateRequestWaitMatch_Pvp polls every 0.5 s for League of
    // Victory but uses the ranked Honor interval otherwise. Both invited players
    // register at once, so a friendly Honor duel polls like Victory. Returns
    // non-zero only when the configured (ranked Honor) interval applies.
    public static int ConfiguredWaitInterval(MatchManager match) {
        return match != null && (int)match.ArenaType != 0 && (int)match.DuelType != 1 ? 1 : 0;
    }

    // ArenaContext.OnBattleBanPickStartCallback waits 5 s for the ranked
    // matching screen's "opponent found" reveal before opening the draft.
    // Friendly duels skip that screen, so open the draft at once.
    public static float BanPickRevealDelay() { return IsFriendlyMatch() ? 0f : 5f; }

    // Victory duels hide score/ranking on the result screen and enlarge the
    // win/lose effect (EffectRoot reset, effect scale 100). Do the same for
    // friendly Honor results before the effect plays. The result window is
    // reused, so ranked results restore the original layout.
    static NGame2.NUI.NWindow.NBattle.PvpResult layoutOwner;
    static Vector3 rootPosition, rootScale; static Quaternion rootRotation;
    static float winScale = -1, loseScale = -1;
    static NVespa.TransformData EffectTransform(NGame2.NUI.UIGame3DWidget widget) {
        return widget == null || widget.EntryList == null || widget.EntryList.Count == 0 ? null : widget.EntryList[0].ExtraTransform;
    }
    public static void PrepareResultEffect(NGame2.NUI.NWindow.NBattle.PvpResult result) {
        if (result == null || result.EffectRoot == null) return;
        var root = result.EffectRoot;
        var win = EffectTransform(result.EffectWin); var lose = EffectTransform(result.EffectLose);
        if (!ReferenceEquals(layoutOwner, result)) {
            layoutOwner = result;
            rootPosition = root.localPosition; rootRotation = root.localRotation; rootScale = root.localScale;
            winScale = win == null ? -1 : win.Scale; loseScale = lose == null ? -1 : lose.Scale;
        }
        if (IsFriendlyMatch()) {
            root.localPosition = Vector3.zero; root.localRotation = Quaternion.identity; root.localScale = Vector3.one;
            if (win != null) win.Scale = 100; if (lose != null) lose.Scale = 100;
        } else {
            root.localPosition = rootPosition; root.localRotation = rootRotation; root.localScale = rootScale;
            if (win != null && winScale >= 0) win.Scale = winScale; if (lose != null && loseScale >= 0) lose.Scale = loseScale;
        }
    }

    // An inviter can cancel while the friend's invitation popup is still open.
    // Accepting it afterwards used to open the Victory party screen briefly
    // before the server's CancelDuel reply closed it. Remember cancels received
    // while idle so that acceptance skips the party screen (the reply then shows
    // "Duel has been canceled."). A new invitation from that friend clears it.
    static readonly System.Collections.Generic.Dictionary<long, long> canceledInvites = new System.Collections.Generic.Dictionary<long, long>();
    public static void NoteCancel(long sender, NShared.NMessage.NProtocol.CancelDuel data) {
        if (data == null || !SingletonUpdater<DuelManager>.isValidInstance) return;
        if (SingletonUpdater<DuelManager>.instance.DuelState is NGame2.NMatch.NDuelState.None) canceledInvites[sender] = data.GroupId;
    }
    public static void ForgetCancel(long sender) { canceledInvites.Remove(sender); }
    public static bool WasCanceled(long sender, long groupId) {
        long canceled;
        return canceledInvites.TryGetValue(sender, out canceled) && canceled == groupId;
    }
    // Party states: Honor has no party screen; a withdrawn invitation gets none.
    public static bool SkipPartyScreen(NGame2.NMatch.NDuelState.StateBase state) {
        return state != null && (IsHonor(state.GroupId) || WasCanceled(state.VsPlayerId, state.GroupId));
    }

    // Receiver's invitation: keep "Lv {level} {nick}" as is (long nicknames)
    // and name the league in the message line below it.
    public static string InviteMessage(long groupId) {
        return (IsHonor(groupId) ? "Honor" : "Victory") + " Duel Request sent.";
    }
    public static void SetInviteMessage(DuelRequestNotify notify, NShared.NMessage.NProtocol.RequestDuel data) {
        if (notify == null || data == null) return;
        foreach (var label in notify.GetComponentsInChildren<UILabel>(true))
            if (label.name == "Label_Message") { Rename(label.gameObject, InviteMessage(data.GroupId)); return; }
    }

    static void Rename(GameObject go, string text) {
        foreach (var localize in go.GetComponentsInChildren<UILocalize>(true)) { localize.enabled = false; UnityEngine.Object.Destroy(localize); }
        foreach (var localize in go.GetComponentsInChildren<NVespa.NNGUIExtension.UILocalizeEx>(true)) { localize.enabled = false; UnityEngine.Object.Destroy(localize); }
        foreach (var label in go.GetComponentsInChildren<UILabel>(true)) label.text = text;
    }

    static void Request(InspectUserMenuPopup menu) {
        if (menu == null || InspectUserField == null) return;
        var target = InspectUserField.GetValue(menu);
        var accountId = target == null ? null : target.GetType().GetProperty("AccountId");
        if (accountId == null) return;
        long friend = (long)accountId.GetValue(target, null);
        // The inviter needs an eligible Honor collection too; native message on failure.
        int[] heroes;
        if (!Eligible(out heroes)) return;
        long me = GlobalAutoBehaviour<AccountManager>.instance.UserManager.Id;
        SingletonUpdater<DuelManager>.instance.RequestDuel(friend, me + GroupOffset);
        menu.Close();
    }

    static bool Eligible(out int[] heroes) {
        heroes = null;
        Action onFail = BanPickFail == null ? null : (Action)Delegate.CreateDelegate(typeof(Action), BanPickFail);
        return MatchManagement.CheckBanPickCondition(out heroes, onFail) && heroes != null && heroes.Length > 0;
    }

    // Duel BattlePartySetting state. Honor has no four-hero party screen: the
    // native Honor queue registers the eligible collection and drafts after
    // pairing. Submit it once; failing players cancel the duel natively.
    // A CompletePartySetting that reaches a peer still in its RequestDuel state
    // cancels the duel natively. Victory players take seconds to choose, but an
    // automatic submission would race the acceptance. So the inviter submits
    // first (BattlePartySetting); the acceptor submits only after receiving it
    // (CompleteVsPlayerSetting), when the inviter is known to be ready.
    public static bool IsInviter(long groupId) {
        return IsHonor(groupId) && groupId - GroupOffset == GlobalAutoBehaviour<AccountManager>.instance.UserManager.Id;
    }
    public static bool ShouldSubmit(string state, long groupId, bool inviter) {
        return IsHonor(groupId) && (state == "CompleteVsPlayerSetting" || (state == "BattlePartySetting" && inviter));
    }
    // Once per duel state: the same friend pair reuses the same GroupId later.
    static object submitted;
    public static void SubmitParty(NGame2.NMatch.NDuelState.StateBase state) {
        if (state == null || ReferenceEquals(submitted, state)) return;
        long groupId = state.GroupId;
        if (!ShouldSubmit(state.GetType().Name, groupId, IsInviter(groupId))) return;
        submitted = state;
        int[] heroes;
        if (!Eligible(out heroes)) {
            if (SceneAutoBehaviour<DuelManagement>.isValidInstance) SceneAutoBehaviour<DuelManagement>.instance.CancelDuel();
            SingletonUpdater<DuelManager>.instance.CancelCurrentDuel();
            return;
        }
        // Leader 1 matches the native Honor registration (StartBanPickArena).
        SingletonUpdater<DuelManager>.instance.CompletePartySetting(groupId, heroes, 1);
    }
}
