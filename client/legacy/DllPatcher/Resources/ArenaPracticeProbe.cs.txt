using System;
using System.Linq;
using System.Collections.Generic;
using System.Reflection;
using NGame2.NAccount;
using NGame2.NBattleContext;
using NGame2.NUI.NWindow;
using NGameAsset;
using NShared;
using NVespa;
using NVespa.NSingleton;
using UnityEngine;

public static class SprkArenaPracticeProbe
{
    static Transform visibleCard;
    static string[] victoryArtwork;
    static string[] Artwork(Transform root)
    {
        return root.GetComponentsInChildren<NGame2.NUI.NAtlasAssignComponent.UISpriteAtlasAssigner>(true)
            .Select(x => (x.transform == root ? "<root>" : x.name) + ":" + x.AtlasName + ":" + x.SpriteName).OrderBy(x => x).ToArray();
    }
    public static string VerifyLabels()
    {
        Check(visibleCard != null, "Run the native card fixture first");
        var title = visibleCard.GetComponentsInChildren<UILabel>(true).Single(l => l.name == "Label_SlotTitle");
        Check(title.text == "Arena Practice", "Localized Ordeals title replaced the practice title after Start/OnEnable");
        Check(visibleCard.GetComponentsInChildren<UILabel>(true).Any(l => l.text == "Practice"), "Practice button label was replaced");
        Check(visibleCard.GetComponentsInChildren<UILabel>(true).Single(l => l.name == "Label_Info").text.Contains("4 heroes"), "Practice description was replaced");
        Check(Artwork(visibleCard).SequenceEqual(victoryArtwork), "Practice artwork differs from League of Victory after OnEnable");
        return "PASS: practice title, description, button and League of Victory artwork survive native Start/OnEnable.";
    }
    static void Check(bool ok, string message) { if (!ok) throw new Exception(message); }
    public static string Run()
    {
        Check(UnityEngine.SceneManagement.SceneManager.GetActiveScene().name == "Login", "Probe requires Login");
        NShared.NJit.TableRoot.Instance.SetPatchDownLoadComplete(true);
        GameRoot.Instance.SetJsonLoader(new NGameAsset.JsonAssetAsyncLoader());
        GameRoot.Instance.SetAiScriptLoader(new NGameAsset.TextAssetSyncLoader());
        GameRoot.Instance.InitForClient();
        GameRoot.Instance.OnDataContainerLoaded(false);
        var prefab = VespaResource.Load<GameObject>(MatchSelect.PrefabPath);
        Check(prefab != null, "Arena selection prefab missing");
        var host = new GameObject("ArenaPracticeFixture"); host.SetActive(false);
        var uiRoot = UnityEngine.Object.FindObjectOfType<UIRoot>();
        if (uiRoot != null) host.transform.SetParent(uiRoot.transform, false);
        var go = UnityEngine.Object.Instantiate(prefab, host.transform, false);
        var view = go.GetComponent<MatchSelect>();
        try
        {
            view.SetArena(new[] { 1, 2, 5 });
            SprkArenaPractice.AddCard(view);
            view.SetArena(new[] { 1, 2, 5 });
            Check(view.Grid.transform.Cast<Transform>().Count(x => x.name == SprkArenaPractice.CardName) == 1, "Duplicate practice card after reopen");
            var card = view.Grid.transform.Find(SprkArenaPractice.CardName);
            var victory = view.PvpButtonGroups.First(x => x.ArenaType == MatchArenaType.Normal).Button.transform;
            while (victory.parent != view.Grid.transform) victory = victory.parent;
            victoryArtwork = Artwork(victory);
            System.IO.File.WriteAllLines(System.IO.Path.Combine(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR"), "card-components.txt"),
                victory.GetComponentsInChildren<Transform>(true).Select(t => t.name + ": " + string.Join(", ", t.GetComponents<Component>().Where(c => c != null).Select(c => c.GetType().FullName).ToArray())).ToArray());
            Check(victoryArtwork.Any(x => x.StartsWith("Sprite_Image:") && !x.EndsWith(":")), "League of Victory background binding unavailable");
            Check(Artwork(card).SequenceEqual(victoryArtwork), "Practice card must reuse League of Victory artwork");
            var ordeal = view.PvpButtonGroups.First(x => x.ArenaType == MatchArenaType.Ordeal).Button.transform;
            while (ordeal.parent != view.Grid.transform) ordeal = ordeal.parent;
            Check(card.GetSiblingIndex() == ordeal.GetSiblingIndex() + 1, "Practice card must follow Ordeals");
            var buttons = card.GetComponentsInChildren<UIButton>(true);
            Check(buttons.Any(b => b.gameObject.activeSelf && b.isEnabled && b.onClick.Count == 1), "Practice entry button unavailable");
            var labels = string.Join(" | ", card.GetComponentsInChildren<UILabel>(true).Select(l => l.name + "=" + l.text).ToArray());
            System.IO.File.WriteAllText(System.IO.Path.Combine(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR"), "labels.txt"), labels);
            var chapter = GameRoot.Instance.ConstantDataContainer.GetValueInt("HeroTryUseChapterIndex");
            var dungeon = GameRoot.Instance.ConstantDataContainer.GetValueInt("HeroTryUseDungeonIndex");
            var dummy = GameRoot.Instance.CampaignDungeonDataContainer.GetData(chapter, dungeon);
            Check(dummy != null, "Native dummy dungeon missing");
            Check(object.ReferenceEquals(SprkArenaPractice.CombatDungeon(dummy), dummy), "Regular See Motions was modified");
            var savedDamageText = SystemOptionData.Render_DamageText;
            Check(SprkArenaPractice.Menu(EBattleMenu.SystemButtonSpeed) == EBattleMenu.SystemButtonSpeed && SprkArenaPractice.Speed(1.5f) == 1.5f, "Nonpractice speed/menu settings were modified");
            SingletonAuto<BattleContext>.instance.SetCustomCampaign(chapter, dungeon, DungeonDifficultyType.Easy);
            typeof(SprkArenaPractice).GetProperty("Active").GetSetMethod(true).Invoke(null, new object[] { true });
            var provider = (NGame2.NBattleField.NManager.NPlay.IDataProvider)SingletonAuto<CustomCampaignContext>.instance;
            var menu = provider.GetBattleMenuType();
            Check((menu & EBattleMenu.Hud) != 0 && (menu & EBattleMenu.SystemButtonAutoSimple) != 0, "HUD or Auto-only widget missing from the actual provider menu");
            Check((menu & (EBattleMenu.SystemButtonAuto | EBattleMenu.SystemButtonSpeed | EBattleMenu.SystemButtonAutoFastContinue)) == 0, "Practice menu still exposes a speed control");
            Check(SprkArenaPractice.Speed(1.5f) == 1f && SprkArenaPractice.DamageText() == ERenderDamageText.ShowAll && SprkArenaPractice.ShowDamage(false), "Practice speed or damage-display policy missing");
            Check(SystemOptionData.Render_DamageText == savedDamageText, "Practice changed the saved damage-text preference");
            Check((string)provider.GetBattleDefine().Code == "Arena", "Presentation must use the Arena battle definition");
            var battleUi = SceneAutoBehaviour<NGame2.NUI.NManager.NBattle.BattleManagement>.instance;
            var uiFlags = BindingFlags.Instance | BindingFlags.NonPublic;
            var uiDungeon = (CampaignDungeonData)battleUi.GetType().GetMethod("GetCampaignDungeonData", uiFlags).Invoke(battleUi, new object[] { chapter, dungeon });
            Check(uiDungeon.BattleType == BattleType.Arena && (string)uiDungeon.BattleDefineCode == "Arena", "Battle UI still uses preview auto/speed rules");
            var speedField = battleUi.GetType().GetField("_basicGameSpeedRatio", uiFlags);
            speedField.SetValue(battleUi, 1.5f);
            var speedGetter = battleUi.GetType().GetMethod("GetGameSpeedRatio", uiFlags);
            Check((float)speedGetter.Invoke(battleUi, null) == 1f, "Native practice battle speed is not fixed to x1");
            speedField.SetValue(battleUi, 1f);
            var hudPrefab = VespaResource.Load<GameObject>(NGame2.NBattleView.NUIOn3D.HUD.PrefabPath);
            var hud = hudPrefab.GetComponent<NGame2.NBattleView.NUIOn3D.HUD>();
            Check(hud != null && hud.GetType().GetField("_spriteHp", uiFlags).GetValue(hud) != null && hud.GetType().GetField("_buffGrid", uiFlags).GetValue(hud) != null, "Native health/status HUD bindings are missing");
            var autoOnly = VespaResource.Load<GameObject>(NGame2.NUI.NWindow.NBattle.SystemButtonAutoSimple.PrefabPath);
            Check(autoOnly != null && !autoOnly.GetComponentsInChildren<UIButton>(true).Any(b => b.name.ToLowerInvariant().Contains("speed")), "Auto-only prefab unexpectedly exposes a speed button");
            var data = SprkArenaPractice.CombatDungeon(dummy);
            Check(data.BattleType == BattleType.Arena && (string)data.BattleDefineCode == "Arena", "Practice must use League of Victory combat rules");
            Check(data.BattleField == dummy.BattleField && !object.ReferenceEquals(data, dummy), "Practice must preserve the dummy field on a private copy");
            var instance = new BattleInstance(); instance.SetCampaignDungeon(chapter, dungeon, 0, false);
            Check(instance.DungeonData.BattleType == BattleType.Arena && (string)instance.BattleDefine.Code == "Arena", "Actual battle initialization bypassed Arena rules");
            Check(instance.InfiniteBattleTime && instance.InstanceConstant.MpRegenM == 0, "Practice must not inherit preview mana regeneration");
            GameRoot.Instance.BattleLogFactory.SetSerializer(new BattleLogFactory.BattleLogSerializer(JsonMarshaler.EncodeJsonString<IBattleLogRuntime>));
            GameRoot.Instance.BattleLogFactory.SetDeserializer(new BattleLogFactory.BattleLogDeserializer(JsonMarshaler.Parse));
            // Use table-defined playable heroes so this fixture follows the archive.
            var heroes = GameRoot.Instance.CreatureDataContainer.GetAllData().Where(c => c.Index < 10000 && c.Index > 0).Take(4)
                .Select(c => new CachedHeroInfo(new HeroInfo { HeroIndex = c.Index, Star = 5, Level = 90, SkillLevel1 = 20, SkillLevel2 = 20, SkillLevel3 = 20, SkillLevel4 = 20 })).ToArray();
            Check(heroes.Length == 4, "Four hero fixture unavailable");
            var baseStat = StatCalculator.GetEquippedHeroStat(heroes[0]).GetStat(StatType.PhysicalAttack);
            var weaponItem = GameRoot.Instance.ItemDataContainer.GetAllData().Select(i => i as ItemData<EquipItemData>)
                .First(i => i != null && i.SubData != null && (int)i.SubData.PartType == 1);
            heroes[0].EquipItemInfo1 = new EquipItemInfo { SlotIndex = 1, ItemIndex = weaponItem.Index, Star = 0, Level = 0 };
            var infos = heroes.Select(h => BattleCreatureCreateInfo.CreateMyCreatureCreateInfo(h, h == heroes[0], false, null, null)).ToArray();
            Check(infos.All(i => i != null), "Equipped hero stats missing");
            Check(infos[0].HeroInfo.EquipItemInfo1.ItemIndex == weaponItem.Index, "Equipped weapon was stripped");
            Check(infos[0].StatData.GetStat(StatType.PhysicalAttack) > baseStat, "Equipped weapon did not affect combat stats");
            var account = GlobalAutoBehaviour<AccountManager>.instance;
            var flags = BindingFlags.Instance | BindingFlags.NonPublic | BindingFlags.Public;
            var managers = account.GetType().GetProperty("AccountManagers", flags).GetValue(account, null);
            managers.GetType().GetField("UserManager").SetValue(managers, new UserManager(account,
                new UserInfo { AccountId = 123456 }, new PlayerMiscInfo(), new PlayerGuildPointInfo()));
            heroes[0].EquipItemSlotIndex1 = 1;
            account.InitEquipItemInfos(new[] { heroes[0].EquipItemInfo1 });
            account.InitHeroInfos(heroes.Cast<HeroInfo>().ToArray());
            var party = new BattlePartySetting.BattlePartyInfo { heroIndices = heroes.Select(h => h.HeroIndex).ToArray(), leaderHeroIndex = heroes[0].HeroIndex };
            var nativeParty = SprkArenaPractice.CreateParty(party);
            Check(nativeParty.CreatureCreateInfos.Length == 4 && nativeParty.CreatureCreateInfos[0].HeroInfo.EquipItemInfo1.ItemIndex == weaponItem.Index, "Native Arena party factory lost gear");
            infos = nativeParty.CreatureCreateInfos;
            foreach (var info in infos) { info.IsAi = true; info.IsSmartAi = true; }
            var builder = new BattleInstanceBuilder(); builder.SetCampaignDungeon(chapter, dungeon, 0, false);
            builder.SetEnterPlayerBattleLogList(new List<IBattleLogRuntime> {
                new PlayerEnterBattleLog { PlayerCreateInfo = new BattlePlayerCreateInfo { Id = 123456, TeamType = TeamType.Left }, CreatureCreateInfos = infos },
                new PlayerEnterBattleLog { PlayerCreateInfo = new BattlePlayerCreateInfo { Id = 0, TeamType = TeamType.Right } }
            });
            var updater = builder.BuildUpdater<BattleInstance>(); updater.StartBattle();
            for (int i = 0; i < 80; i++) { updater.Update(); updater.BattleInstance.UpdateTick(false); }
            var creatures = updater.BattleInstance.CreatureContainer.GetAllCreatures();
            Check(creatures.Count(c => c.PlayerId == 123456) >= 4, "Four selected heroes did not spawn: " + string.Join(",", creatures.Select(c => c.PlayerId + ":" + c.CreatureData.Index).ToArray()));
            Check(creatures.Any(c => c.PlayerId == 0), "Native dummy did not spawn");
            var dummyCreature = creatures.First(c => c.PlayerId == 0);
            Check(dummyCreature.CreatureData.Race == CreatureRaceType.Hero && dummyCreature.CreateInfo.ForceNoDie, "Dummy must activate PvP hero-target bonuses and survive testing");
            Check(GameRoot.Instance.CreatureDataContainer.GetData(999999).Race != CreatureRaceType.Hero, "Shared dummy table was changed");
            var damage = updater.BattleInstance.Snapshot.StatisticsManager.TotalDamagedMeter(dummyCreature.Uid);
            Check(damage > 0, "Four-hero combat did not record damage on the dummy");
            var target = "Dummy=" + dummyCreature.CreatureData.Index + ", race=" + dummyCreature.CreatureData.Race + ", noDie=" + dummyCreature.CreateInfo.ForceNoDie + ", damage=" + damage;
            SprkArenaPractice.Clear();
            Check(SprkArenaPractice.Speed(1.5f) == 1.5f && !SprkArenaPractice.ShowDamage(false) && SprkArenaPractice.DamageText() == savedDamageText, "Practice display overrides leaked after exit");
            var regular = new BattleInstance(); regular.SetCampaignDungeon(chapter, dungeon, 0, false);
            Check(regular.DungeonData.BattleType == dummy.BattleType && regular.BattleDefine.Code == dummy.BattleDefineCode, "PvP rules leaked into regular skill testing");
            view.enabled = false;
            go.SetActive(true); host.SetActive(true);
            foreach (var panel in go.GetComponentsInChildren<UIPanel>(true)) panel.alpha = 1;
            System.IO.File.WriteAllText(System.IO.Path.Combine(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR"), "layout.txt"),
                "Ordeals=" + ordeal.localPosition + ", practice=" + card.localPosition + ", sorting=" + view.Grid.sorting);
            // Present the two actual native cards to an isolated orthographic
            // camera for a visual check, without opening an authenticated UI.
            go.SetActive(false);
            var preview = new GameObject("ArenaCardPreview");
            var panelPreview = preview.AddComponent<UIPanel>(); panelPreview.depth = 1000;
            ordeal.SetParent(preview.transform, false); ordeal.localPosition = new UnityEngine.Vector3(-290, 0, 0);
            card.SetParent(preview.transform, false); card.localPosition = new UnityEngine.Vector3(290, 0, 0);
            visibleCard = card;
            ordeal.gameObject.SetActive(true); card.gameObject.SetActive(true);
            foreach (var t in preview.GetComponentsInChildren<Transform>(true)) t.gameObject.layer = 29;
            foreach (var tween in preview.GetComponentsInChildren<UITweener>(true)) tween.enabled = false;
            foreach (var widget in preview.GetComponentsInChildren<UIWidget>(true)) widget.alpha = 1;
            var cameraObject = new GameObject("ArenaCardPreviewCamera");
            var camera = cameraObject.AddComponent<Camera>(); camera.orthographic = true; camera.orthographicSize = 540;
            camera.transform.position = new UnityEngine.Vector3(0, 0, -1000); camera.nearClipPlane = 0.1f; camera.farClipPlane = 2000;
            camera.cullingMask = 1 << 29; camera.clearFlags = CameraClearFlags.SolidColor; camera.backgroundColor = new Color(.07f, .08f, .1f, 1);
            camera.depth = 1000; cameraObject.AddComponent<UICamera>();
            return "PASS: native Arena prefab, Victory artwork, card ordering/reuse, HUD/status bindings, Auto-only controls, native x1 speed, practice damage text, four-hero equipped combat and preview isolation. " + target + ". Labels: " + labels;
        }
        finally { SprkArenaPractice.Clear(); }
    }
}
