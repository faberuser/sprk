        Type type;if(types.TryGetValue(name,out type))return type;

        if(type==null)throw new InvalidOperationException("Unknown recovered scene type: "+name);
        types[name]=type;return type;
    }

    }

    void Fill(object target,JsonData data) {
        if(data==null || !data.IsObject())return;
        foreach(var pair in data.ToDictionary()) {
            string key=(string)pair.Key;var node=(JsonData)pair.Value;
            if(key=="m_Name" || key=="m_Enabled" || key=="m_Script" || key=="m_GameObject")continue;
            var field=Field(target.GetType(),key);
            if(field!=null && !field.IsInitOnly)field.SetValue(target,ConvertValue(node,field.FieldType));
        }
        var behaviour=target as Behaviour;if(behaviour!=null && Has(data,"m_Enabled"))behaviour.enabled=Number(data,"m_Enabled",1)!=0;
    }
    object ConvertValue(JsonData value,Type type) {
        if(value==null)return type.IsValueType?Activator.CreateInstance(type):null;
        if(Has(value,"ref"))return Ref(value);
        if(type==typeof(string))return value.IsString()?value.valueStr:value.ToJson();
        if(type.IsEnum)return Enum.ToObject(type,(int)NumberValue(value));
        if(type==typeof(bool))return value.IsBoolean()?value.valueBool:NumberValue(value)!=0;
        if(type.IsPrimitive || type==typeof(decimal))return Convert.ChangeType(NumberValue(value),type,CultureInfo.InvariantCulture);
        if(type.IsArray) {
            var element=type.GetElementType();int count=value.IsArray()?value.Count:0;var array=Array.CreateInstance(element,count);
            for(int i=0;i<count;i++)array.SetValue(ConvertValue(value.GetFromArray(i),element),i);return array;
        }
        if(typeof(IList).IsAssignableFrom(type)) {
            var list=(IList)Activator.CreateInstance(type);var element=type.GetGenericArguments()[0];
            if(value.IsArray())for(int i=0;i<value.Count;i++)list.Add(ConvertValue(value.GetFromArray(i),element));return list;
        }
        object instance;
        try{instance=Activator.CreateInstance(type,true);}catch{instance=System.Runtime.Serialization.FormatterServices.GetUninitializedObject(type);}
        Fill(instance,value);return instance;
    }

    // Split Color/Alpha effect textures were merged into RGBA at export and their
    // materials marked Sprites/Default, which alpha-blends. Additive glows must keep
    // the game's effect shader and blend factors or they render as dull/opaque quads.
    static bool AdditiveEffect(JsonData row){return Text(row,"shader")=="Sprites/Default" && Has(row,"floats") && Has(row["floats"],"_BlendModeSrc") && Number(row["floats"],"_BlendModeDst")==1;}
    // Old bundles name shaders without their current folders ("NormalNoZ",
    // "Unlit - Transparent Colored").
    static Shader FindShader(string name) {
        foreach(var candidate in new[]{name,name.Replace(" - ","/"),"Vespa/Effect2/"+name,"Vespa/"+name}){var shader=Shader.Find(candidate);if(shader!=null)return shader;}
        return null;
    }
    Material Material(JsonData row) {
        string shaderName=Text(row,"shader");
        if(AdditiveEffect(row) && Shader.Find(EffectShader)!=null)shaderName=EffectShader;
        Shader shader=FindShader(shaderName);
        if(shader==null || shaderName.IndexOf("SplitAlpha",StringComparison.OrdinalIgnoreCase)>=0)shader=Shader.Find("Sprites/Default");
        if(shader==null)shader=Shader.Find("Sprites/Default");
        if(shader==null)throw new InvalidOperationException("Login rendering shader unavailable");
        var result=Keep(new Material(shader){name=Text(row,"name")});
        if(Has(row,"keywords") && shader.name!="Sprites/Default") {
            var keywords=(string[])ConvertValue(row["keywords"],typeof(string[]));
            // Merged RGBA textures already carry their alpha.
            if(shaderName!=Text(row,"shader"))keywords=keywords.Select(k=>k=="ALPHATEX_ON"?"ALPHATEX_OFF":k).ToArray();
            result.shaderKeywords=keywords;
        }
        int queue=(int)Number(row,"renderQueue",-1);if(queue>=0)result.renderQueue=queue;
        return result;
    }
    Texture2D Readable(Texture2D source) {
        var copy=new Texture2D(2,2,TextureFormat.RGBA32,false);
        if(!ImageConversion.LoadImage(copy,File.ReadAllBytes(textureFiles[source]),false))throw new InvalidOperationException("Login texture failed: "+source.name);
        return copy;
    }
    // Android ETC effects keep alpha in a separate grayscale _AlphaTex. Bake it into
    // the main texture so the PC effect shader cuts out the particle shape (dust,
    // ash) instead of drawing the whole quad.
    Texture2D CombineAlpha(Texture2D main,Texture2D alpha) {
        string key=(main==null?"white":main.GetInstanceID().ToString())+"/"+alpha.GetInstanceID();Texture2D result;
        if(combinedAlpha.TryGetValue(key,out result))return result;
        var mask=Readable(alpha);var color=main!=null && textureFiles.ContainsKey(main)?Readable(main):null;
        int width=mask.width,height=mask.height;var a=mask.GetPixels32();var pixels=new Color32[a.Length];
        Color32[] rgb=color!=null && color.width==width && color.height==height?color.GetPixels32():null;
        for(int y=0;y<height;y++)for(int x=0;x<width;x++) {
            int i=y*width+x;
            Color32 c=rgb!=null?rgb[i]:color!=null?(Color32)color.GetPixelBilinear((x+.5f)/width,(y+.5f)/height):new Color32(255,255,255,255);
            c.a=a[i].r;pixels[i]=c;
        }
        UnityEngine.Object.Destroy(mask);if(color!=null)UnityEngine.Object.Destroy(color);
        result=Keep(new Texture2D(width,height,TextureFormat.RGBA32,true));result.name=alpha.name+" (alpha)";
        result.SetPixels32(pixels);result.wrapModeU=alpha.wrapModeU;result.wrapModeV=alpha.wrapModeV;result.filterMode=FilterMode.Trilinear;result.anisoLevel=2;result.Apply(true,true);
        combinedAlpha[key]=result;return result;
    }
    Texture2D AlphaFromColor(Texture2D source) {
        Texture2D result;if(colorAlpha.TryGetValue(source,out result))return result;
        var copy=Readable(source);var pixels=copy.GetPixels32();
        for(int i=0;i<pixels.Length;i++){var c=pixels[i];c.a=Math.Max(c.r,Math.Max(c.g,c.b));pixels[i]=c;}
        result=Keep(new Texture2D(copy.width,copy.height,TextureFormat.RGBA32,true));result.name=source.name+" (alpha)";
        result.SetPixels32(pixels);result.wrapModeU=source.wrapModeU;result.wrapModeV=source.wrapModeV;result.filterMode=FilterMode.Trilinear;result.anisoLevel=2;result.Apply(true,true);
        UnityEngine.Object.Destroy(copy);colorAlpha[source]=result;return result;
    }
    void FixEffectTextures(Material material,JsonData row) {
        string shaderName=material.shader.name;
        // Scrolling fog/aura textures tile; clamping smears their edge pixels into streaks.
        // Fallback for data without recorded wrap modes.
        bool scrolls=Has(row,"floats") && row["floats"].ToDictionary().Any(p=>((string)p.Key).Contains("Speed") && NumberValue((JsonData)p.Value)!=0);
        if(scrolls && shaderName.StartsWith("Vespa/") && shaderName.IndexOf("Clamp",StringComparison.OrdinalIgnoreCase)<0 && shaderName!=EffectShader)
            foreach(var name in material.GetTexturePropertyNames()){var texture=material.GetTexture(name) as Texture2D;if(texture!=null && !exactWrap.Contains(texture))texture.wrapMode=TextureWrapMode.Repeat;}
        // Android effect textures are ETC RGB with no alpha; the mobile shaders
        // derived coverage from color. On PC the effect shaders read texture alpha,
        // so blended (non-additive) smoke/fog would draw as opaque grey quads.
        bool effect=shaderName.StartsWith("Vespa/") && !shaderName.StartsWith("Vespa/NGUI");
        bool additive=shaderName.IndexOf("Addtive",StringComparison.OrdinalIgnoreCase)>=0 || shaderName.IndexOf("Additive",StringComparison.OrdinalIgnoreCase)>=0 ||
            Has(row,"floats") && Has(row["floats"],"_BlendModeDst") && Number(row["floats"],"_BlendModeDst")==1;
        if(effect && !additive && material.HasProperty("_MainTex") && !material.IsKeywordEnabled("ALPHATEX_ON")) {
            var main=material.GetTexture("_MainTex") as Texture2D;
            if(main!=null && rgbOnly.Contains(main))material.SetTexture("_MainTex",AlphaFromColor(main));
        }
        if(material.IsKeywordEnabled("ALPHATEX_ON") && material.HasProperty("_AlphaTex")) {
            var alpha=material.GetTexture("_AlphaTex") as Texture2D;
            if(alpha!=null && textureFiles.ContainsKey(alpha)) {
                var main=material.HasProperty("_MainTex")?material.GetTexture("_MainTex") as Texture2D:null;
                var scale=main!=null?material.GetTextureScale("_MainTex"):material.GetTextureScale("_AlphaTex");
                var offset=main!=null?material.GetTextureOffset("_MainTex"):material.GetTextureOffset("_AlphaTex");
                material.SetTexture("_MainTex",CombineAlpha(main,alpha));material.SetTextureScale("_MainTex",scale);material.SetTextureOffset("_MainTex",offset);
                material.SetTexture("_AlphaTex",null);material.DisableKeyword("ALPHATEX_ON");material.EnableKeyword("ALPHATEX_OFF");
            }
        }
    }
    void Build(JsonData scene,Transform parent) {
        var rows=scene["objects"];
        for(int i=0;i<rows.Count;i++) {
            var row=rows.GetFromArray(i);string id=Text(row,"id"),kind=Text(row,"kind");UnityEngine.Object item=null;
            stage="create "+kind+" "+id;
            if(kind=="GameObject") {
                var go=new GameObject(Text(row,"name"));go.transform.SetParent(bank.transform,false);go.layer=parent.gameObject.layer;item=go;
            } else if(kind=="Texture2D") {
                // Mipmaps keep small, distant particles soft instead of aliasing into blocks.
                var image=Keep(new Texture2D(2,2,TextureFormat.RGBA32,true));image.name=Text(row,"name");
                string file=Path.Combine(folder,Text(row,"file"));
                if(!ImageConversion.LoadImage(image,File.ReadAllBytes(file),true))throw new InvalidOperationException("Login texture failed: "+image.name);
                image.wrapMode=TextureWrapMode.Clamp;image.filterMode=FilterMode.Trilinear;image.anisoLevel=2;textureFiles[image]=file;item=image;
                // Original importer wrap modes (0 Repeat, 1 Clamp, 2 Mirror, 3 MirrorOnce).
                if(Has(row,"wrapU")){image.wrapModeU=(TextureWrapMode)(int)Number(row,"wrapU");image.wrapModeV=(TextureWrapMode)(int)Number(row,"wrapV",Number(row,"wrapU"));exactWrap.Add(image);}
                if(Flag(row,"rgbOnly"))rgbOnly.Add(image);
            } else if(kind=="TextAsset") {var raw=File.ReadAllBytes(Path.Combine(folder,Text(row,"file")));var text=Keep(new TextAsset(System.Text.Encoding.UTF8.GetString(raw)){name=Text(row,"name")});textBytes[text]=raw;item=text;}
            else if(kind=="Material")item=Material(row);
            else if(kind=="Mesh")item=Keep(new Mesh{name=Text(row,"name")});
            else if(kind=="AnimationClip")item=Keep(new AnimationClip{legacy=true,name=Text(row["data"],"m_Name")});
            if(item!=null)objects[id]=item;
        }
        for(int i=0;i<rows.Count;i++) {
            var row=rows.GetFromArray(i);if(Text(row,"kind")=="Transform")objects[Text(row,"id")]=Go(row["go"]).transform;
        }
        for(int i=0;i<rows.Count;i++) {
            var row=rows.GetFromArray(i);string kind=Text(row,"kind"),id=Text(row,"id");
            stage="component "+kind+" "+id;
            if(kind=="MonoBehaviour") {
                var type=FindType(Text(row,"type"));
                if(typeof(ScriptableObject).IsAssignableFrom(type))objects[id]=Keep(ScriptableObject.CreateInstance(type));
                else {var go=Go(row["go"]);if(go!=null)objects[id]=go.GetComponent(type)??go.AddComponent(type);}
            } else {
                Type type=kind=="MeshRenderer"?typeof(MeshRenderer):kind=="MeshFilter"?typeof(MeshFilter):kind=="ParticleSystem"?typeof(ParticleSystem):kind=="ParticleSystemRenderer"?typeof(ParticleSystemRenderer):kind=="Animation"?typeof(Animation):null;
                if(type!=null){var go=Go(row["go"]);if(go!=null)objects[id]=go.GetComponent(type)??go.AddComponent(type);}
            }
        }
        for(int i=0;i<rows.Count;i++) {
            var row=rows.GetFromArray(i);string id=Text(row,"id"),kind=Text(row,"kind");UnityEngine.Object item;if(!objects.TryGetValue(id,out item))continue;
            stage="fill "+kind+" "+id;
            if(kind=="Transform") {
                var tm=(Transform)item;var sourceParent=Ref(row["parent"]) as Transform;
                tm.SetParent(sourceParent!=null?sourceParent:bank.transform,false);tm.localPosition=Vector(row["position"]);tm.localRotation=Rotation(row["rotation"]);tm.localScale=Vector(row["scale"]);
            } else if(kind=="MonoBehaviour")Fill(item,row["data"]);
            else if(kind=="Material") {
                var material=(Material)item;
                foreach(var pair in row["textures"].ToDictionary()) {
                    var texture=(JsonData)pair.Value;var image=Ref(texture["m_Texture"]) as Texture;
                    if(image!=null){material.SetTexture((string)pair.Key,image);material.SetTextureOffset((string)pair.Key,new Vector2(Number(texture["m_Offset"],"x"),Number(texture["m_Offset"],"y")));material.SetTextureScale((string)pair.Key,new Vector2(Number(texture["m_Scale"],"x",1),Number(texture["m_Scale"],"y",1)));}
                }
                foreach(var pair in row["floats"].ToDictionary())if(material.HasProperty((string)pair.Key))material.SetFloat((string)pair.Key,(float)NumberValue((JsonData)pair.Value));
                // Historical effect shaders renamed their blend uniforms.
                foreach(var suffix in new[]{"Src","Dst"})if(Has(row["floats"],"_BlendMode"+suffix) && material.HasProperty("_"+suffix+"Blend"))material.SetFloat("_"+suffix+"Blend",Number(row["floats"],"_BlendMode"+suffix));
                if(Text(row,"shader").Contains("NoZ") && material.HasProperty("_ZWrite"))material.SetFloat("_ZWrite",0);
                foreach(var pair in row["colors"].ToDictionary()){var c=(JsonData)pair.Value;if(material.HasProperty((string)pair.Key))material.SetColor((string)pair.Key,new Color(Number(c,"r"),Number(c,"g"),Number(c,"b"),Number(c,"a",1)));}
                FixEffectTextures(material,row);
            } else if(kind=="Mesh") {
                var mesh=(Mesh)item;var vertices=row["vertices"];var v=new Vector3[vertices.Count/3];for(int n=0;n<v.Length;n++)v[n]=new Vector3((float)NumberValue(vertices.GetFromArray(n*3)),(float)NumberValue(vertices.GetFromArray(n*3+1)),(float)NumberValue(vertices.GetFromArray(n*3+2)));mesh.vertices=v;
                // Aura and fog meshes fade their edges with vertex alpha.
                var colors=Has(row,"colors")?row["colors"]:null;
                if(colors!=null && colors.Count==v.Length*4){var c=new Color[v.Length];for(int n=0;n<c.Length;n++)c[n]=new Color((float)NumberValue(colors.GetFromArray(n*4)),(float)NumberValue(colors.GetFromArray(n*4+1)),(float)NumberValue(colors.GetFromArray(n*4+2)),(float)NumberValue(colors.GetFromArray(n*4+3)));mesh.colors=c;}
                else mesh.colors=Enumerable.Repeat(Color.white,v.Length).ToArray();
                var uv=row["uv"];var u=new Vector2[uv.Count/2];for(int n=0;n<u.Length;n++)u[n]=new Vector2((float)NumberValue(uv.GetFromArray(n*2)),(float)NumberValue(uv.GetFromArray(n*2+1)));if(u.Length==v.Length)mesh.uv=u;
                mesh.subMeshCount=row["triangles"].Count;for(int n=0;n<mesh.subMeshCount;n++)mesh.SetTriangles((int[])ConvertValue(row["triangles"].GetFromArray(n),typeof(int[])),n);mesh.RecalculateBounds();
            } else if(kind=="MeshFilter")((MeshFilter)item).sharedMesh=Ref(row["data"]["m_Mesh"]) as Mesh;
            else if(kind=="AnimationClip")ConfigureAnimation((AnimationClip)item,row["data"]);
            else if(kind=="MeshRenderer" || kind=="ParticleSystemRenderer") {
                var renderer=(Renderer)item;var data=row["data"];renderer.sharedMaterials=(Material[])ConvertValue(data["m_Materials"],typeof(Material[]));renderer.sortingOrder=(int)Number(data,"m_SortingOrder");
                var particles=renderer as ParticleSystemRenderer;if(particles!=null)ConfigureParticleRenderer(particles,data);
            } else if(kind=="ParticleSystem")ConfigureParticle((ParticleSystem)item,row["data"]);
        }
        for(int i=0;i<rows.Count;i++) {
            var row=rows.GetFromArray(i);if(Text(row,"kind")!="Animation")continue;
            var animation=(Animation)objects[Text(row,"id")];var data=row["data"];
            for(int n=0;n<data["m_Animations"].Count;n++){var clip=Ref(data["m_Animations"].GetFromArray(n)) as AnimationClip;if(clip!=null)animation.AddClip(clip,clip.name);}
            animation.clip=Ref(data["m_Animation"]) as AnimationClip;animation.playAutomatically=Bool(data,"m_PlayAutomatically",true);animation.wrapMode=(WrapMode)(int)Number(data,"m_WrapMode");
        }
        foreach(var data in objects.Values.OfType<Spine.Unity.SkeletonDataAsset>()) {
            stage="Spine "+data.name;
            if(data.skeletonJSON==null || !data.skeletonJSON.name.ToLowerInvariant().Contains(".skel"))continue;
            var loader=new Spine.AtlasAttachmentLoader(data.atlasAssets.Select(a=>a.GetAtlas()).ToArray());
            var binary=new Spine.SkeletonBinary(loader){Scale=data.scale};
            var parsed=binary.ReadSkeletonData(new MemoryStream(textBytes[data.skeletonJSON],false));
            typeof(Spine.Unity.SkeletonDataAsset).GetMethod("InitializeWithData",Flags).Invoke(data,new object[]{parsed});
        }
        foreach(var spine in objects.Values.OfType<Spine.Unity.SkeletonAnimation>())if(spine.skeletonDataAsset==null)spine.enabled=false;
        // Historical window depths must stay behind the current login controls
        // and Settings modal, while preserving their internal layer ordering.
        foreach(var panel in objects.Values.OfType<UIPanel>())panel.depth-=1000;
        // Set activeSelf while the bank is inactive, then expose only the scene root.
        for(int i=0;i<rows.Count;i++)if(Text(rows.GetFromArray(i),"kind")=="GameObject")((GameObject)objects[Text(rows.GetFromArray(i),"id")]).SetActive(Bool(rows.GetFromArray(i),"active",true) && !Bool(rows.GetFromArray(i),"hidden"));
        Root=Go(scene["root"]);
        if(Has(scene,"attachments"))for(int i=0;i<scene["attachments"].Count;i++) {
            var child=Go(scene["attachments"].GetFromArray(i));
            if(child!=null){if(!child.transform.IsChildOf(Root.transform))child.transform.SetParent(Root.transform,false);child.SetActive(true);}
        }
        Root.transform.SetParent(parent,false);Root.SetActive(true);
        foreach(var widget in Root.GetComponentsInChildren<NGame2.NUI.UIGame3DWidgetFromRes>(true)) {
            if(widget.EntryList.Any(e=>e.ResourceObject is NGameAsset.NEffect.OpenDataFromResource && (((NGameAsset.NEffect.OpenDataFromResource)e.ResourceObject).Code??"").EndsWith("Bi",StringComparison.OrdinalIgnoreCase))){widget.gameObject.SetActive(false);continue;}
            if(widget.gameObject.activeInHierarchy && widget.GetComponent<NGame2.NUI.UIGame3DWidgetFromResAutoPlayer>()==null)widget.Play(null);
        }
        Root.AddComponent<SprkRecoveredLoginPhase>();
    }
    static float CurveValue(JsonData data,string key,float fallback) {

    static void ConfigureAnimation(AnimationClip clip,JsonData data) {
        clip.frameRate=Number(data,"m_SampleRate",30);clip.wrapMode=(WrapMode)(int)Number(data,"m_WrapMode");
        foreach(var spec in new[]{new[]{"m_PositionCurves","localPosition","xyz"},new[]{"m_ScaleCurves","localScale","xyz"},new[]{"m_RotationCurves","localRotation","xyzw"},new[]{"m_EulerCurves","localEulerAnglesRaw","xyz"}}) {
            if(!Has(data,spec[0]))continue;var entries=data[spec[0]];
            for(int i=0;i<entries.Count;i++){var entry=entries.GetFromArray(i);foreach(char axis in spec[2]){var curve=new AnimationCurve();var keys=entry["curve"]["m_Curve"];for(int n=0;n<keys.Count;n++){var key=keys.GetFromArray(n);curve.AddKey(new Keyframe(Number(key,"time"),Number(key["value"],axis.ToString()),Number(key["inSlope"],axis.ToString()),Number(key["outSlope"],axis.ToString())));}clip.SetCurve(Text(entry,"path"),typeof(Transform),spec[1]+"."+axis,curve);}}
        }
        clip.EnsureQuaternionContinuity();
    }
    static AnimationCurve Curve(JsonData data) {
        var curve=new AnimationCurve();if(!Has(data,"m_Curve"))return curve;
        for(int i=0;i<data["m_Curve"].Count;i++){var key=data["m_Curve"].GetFromArray(i);curve.AddKey(new Keyframe(Number(key,"time"),Number(key,"value"),Number(key,"inSlope"),Number(key,"outSlope")));}return curve;
    }
    static ParticleSystem.MinMaxCurve MinMax(JsonData data) {
        // Legacy (Unity 5) data stores some properties as plain constants.
        if(data==null)return new ParticleSystem.MinMaxCurve(0);
        if(!data.IsObject())return new ParticleSystem.MinMaxCurve(data.IsBoolean()?(data.valueBool?1:0):(float)NumberValue(data));
        int mode=(int)Number(data,"minMaxState");float max=Number(data,"scalar",1),min=Number(data,"minScalar",max);
        if(mode==1)return new ParticleSystem.MinMaxCurve(max,Curve(data["maxCurve"]));
        if(mode==2)return new ParticleSystem.MinMaxCurve(max,Curve(data["minCurve"]),Curve(data["maxCurve"]));
        if(mode==3)return new ParticleSystem.MinMaxCurve(min,max);
        return new ParticleSystem.MinMaxCurve(max);
    }

    static Gradient ReadGradient(JsonData data) {
        var colors=new GradientColorKey[(int)Number(data,"m_NumColorKeys",2)];var alpha=new GradientAlphaKey[(int)Number(data,"m_NumAlphaKeys",2)];
        for(int i=0;i<colors.Length;i++)colors[i]=new GradientColorKey(ReadColor(data["key"+i]),Number(data,"ctime"+i)/65535f);
        for(int i=0;i<alpha.Length;i++)alpha[i]=new GradientAlphaKey(Number(data["key"+i],"a",1),Number(data,"atime"+i)/65535f);
        var result=new Gradient();result.SetKeys(colors,alpha);return result;
    }
    static ParticleSystem.MinMaxGradient ReadMinMaxGradient(JsonData data) {
        int mode=(int)Number(data,"minMaxState");
        if(mode==1 || mode==4){var result=new ParticleSystem.MinMaxGradient(ReadGradient(data["maxGradient"]));result.mode=(ParticleSystemGradientMode)mode;return result;}
        if(mode==2)return new ParticleSystem.MinMaxGradient(ReadColor(data["minColor"]),ReadColor(data["maxColor"]));
        if(mode==3)return new ParticleSystem.MinMaxGradient(ReadGradient(data["minGradient"]),ReadGradient(data["maxGradient"]));
        return new ParticleSystem.MinMaxGradient(ReadColor(data["maxColor"]));
    }
    static Vector3 Vector3Of(JsonData data,string key,float fallback=0){return Has(data,key) && data[key]!=null?new Vector3(Number(data[key],"x",fallback),Number(data[key],"y",fallback),Number(data[key],"z",fallback)):new Vector3(fallback,fallback,fallback);}
    static JsonData Enabled(JsonData data,string name){return Has(data,name) && Flag(data[name],"enabled")?data[name]:null;}
    static ParticleSystemSimulationSpace Space(JsonData data,string key){return Flag(data,key)?ParticleSystemSimulationSpace.World:ParticleSystemSimulationSpace.Local;}
    // Mirror the serialized Unity particle modules used by the historical login
    // effects. Scaling mode, prewarm and shape scale matter most: without them glows
    // render far too small (pixel squares), start empty or spawn in a clump.
    void ConfigureParticle(ParticleSystem system,JsonData data) {
        system.Stop(true,ParticleSystemStopBehavior.StopEmittingAndClear);
        var main=system.main;main.duration=Mathf.Max(.05f,Number(data,"lengthInSec",5));main.loop=Flag(data,"looping",true);main.prewarm=Flag(data,"prewarm");main.playOnAwake=true;
        main.simulationSpeed=Has(data,"simulationSpeed")?Number(data,"simulationSpeed",1):Number(data,"speed",1);main.useUnscaledTime=Flag(data,"useUnscaledTime");
        main.scalingMode=(ParticleSystemScalingMode)(int)Number(data,"scalingMode",1);
        main.simulationSpace=(int)Number(data,"moveWithTransform")==1?ParticleSystemSimulationSpace.World:ParticleSystemSimulationSpace.Local;
        if(Has(data,"startDelay"))main.startDelay=MinMax(data["startDelay"]);
        if(Has(data,"autoRandomSeed") && !Flag(data,"autoRandomSeed",true)){system.useAutoRandomSeed=false;system.randomSeed=(uint)Number(data,"randomSeed");}
        if(Has(data,"InitialModule")) {
            var init=data["InitialModule"];
            main.startLifetime=MinMax(init["startLifetime"]);main.startSpeed=MinMax(init["startSpeed"]);main.startColor=ReadMinMaxGradient(init["startColor"]);
            main.startSize3D=Flag(init,"size3D");
            if(main.startSize3D){main.startSizeX=MinMax(init["startSize"]);main.startSizeY=MinMax(init["startSizeY"]);main.startSizeZ=MinMax(init["startSizeZ"]);}else main.startSize=MinMax(init["startSize"]);
            main.startRotation3D=Flag(init,"rotation3D");
            if(main.startRotation3D){main.startRotationX=MinMax(init["startRotationX"]);main.startRotationY=MinMax(init["startRotationY"]);main.startRotationZ=MinMax(init["startRotation"]);}else main.startRotation=MinMax(init["startRotation"]);
            main.flipRotation=Number(init,"randomizeRotationDirection");
            if(Has(init,"gravityModifier"))main.gravityModifier=MinMax(init["gravityModifier"]);
            main.maxParticles=(int)Number(init,"maxNumParticles",1000);
        }
        var emission=system.emission;var emit=Has(data,"EmissionModule")?data["EmissionModule"]:null;
        emission.enabled=emit!=null && Flag(emit,"enabled",true);
        if(emit!=null) {
            if(Has(emit,"rateOverTime"))emission.rateOverTime=MinMax(emit["rateOverTime"]);else emission.rateOverTime=CurveValue(emit,"rate",10);
            if(Has(emit,"rateOverDistance"))emission.rateOverDistance=MinMax(emit["rateOverDistance"]);
            var bursts=new List<ParticleSystem.Burst>();
            if(Has(emit,"m_Bursts") && emit["m_Bursts"]!=null)for(int i=0;i<emit["m_Bursts"].Count;i++) {
                var src=emit["m_Bursts"].GetFromArray(i);
                var burst=new ParticleSystem.Burst(Number(src,"time"),MinMax(src["countCurve"]),(int)Number(src,"cycleCount",1),Mathf.Max(.0001f,Number(src,"repeatInterval",.01f)));
                burst.probability=Number(src,"probability",1);bursts.Add(burst);
            }
            emission.SetBursts(bursts.ToArray());
        }
        var shape=system.shape;var shapeData=Has(data,"ShapeModule")?data["ShapeModule"]:null;shape.enabled=shapeData!=null && Flag(shapeData,"enabled");
        if(shape.enabled) {
            var src=shapeData;shape.shapeType=(ParticleSystemShapeType)(int)Number(src,"type");shape.angle=Number(src,"angle",25);shape.length=Number(src,"length",5);
            if(Has(src,"radius") && src["radius"].IsObject()){shape.radius=Number(src["radius"],"value",1);shape.radiusMode=(ParticleSystemShapeMultiModeValue)(int)Number(src["radius"],"mode");shape.radiusSpread=Number(src["radius"],"spread");}else shape.radius=Number(src,"radius",1);
            if(Has(src,"arc") && src["arc"].IsObject()){shape.arc=Number(src["arc"],"value",360);shape.arcMode=(ParticleSystemShapeMultiModeValue)(int)Number(src["arc"],"mode");shape.arcSpread=Number(src["arc"],"spread");}else shape.arc=Number(src,"arc",360);
            shape.radiusThickness=Number(src,"radiusThickness",1);shape.donutRadius=Number(src,"donutRadius",.2f);shape.boxThickness=Vector3Of(src,"boxThickness");
            shape.position=Vector3Of(src,"m_Position");shape.rotation=Vector3Of(src,"m_Rotation");
            shape.scale=Has(src,"m_Scale")?Vector3Of(src,"m_Scale",1):new Vector3(Number(src,"boxX",1),Number(src,"boxY",1),Number(src,"boxZ",1));
            shape.alignToDirection=Flag(src,"alignToDirection");shape.randomDirectionAmount=Has(src,"randomDirectionAmount")?Number(src,"randomDirectionAmount"):Flag(src,"randomDirection")?1:0;
            shape.sphericalDirectionAmount=Number(src,"sphericalDirectionAmount");shape.randomPositionAmount=Number(src,"randomPositionAmount");
        }
        var color=system.colorOverLifetime;var colorData=Enabled(data,"ColorModule");color.enabled=colorData!=null;if(colorData!=null)color.color=ReadMinMaxGradient(colorData["gradient"]);
        var size=system.sizeOverLifetime;var sizeData=Enabled(data,"SizeModule");size.enabled=sizeData!=null;
        if(sizeData!=null){size.separateAxes=Flag(sizeData,"separateAxes");if(size.separateAxes){size.x=MinMax(sizeData["curve"]);size.y=MinMax(sizeData["y"]);size.z=MinMax(sizeData["z"]);}else size.size=MinMax(sizeData["curve"]);}
        var rotation=system.rotationOverLifetime;var rotationData=Enabled(data,"RotationModule");rotation.enabled=rotationData!=null;
        if(rotationData!=null){rotation.separateAxes=Flag(rotationData,"separateAxes");if(rotation.separateAxes){rotation.x=MinMax(rotationData["x"]);rotation.y=MinMax(rotationData["y"]);}rotation.z=MinMax(rotationData["curve"]);}
        var velocity=system.velocityOverLifetime;var velocityData=Enabled(data,"VelocityModule");velocity.enabled=velocityData!=null;
        if(velocityData!=null) {
            velocity.space=Space(velocityData,"inWorldSpace");velocity.x=MinMax(velocityData["x"]);velocity.y=MinMax(velocityData["y"]);velocity.z=MinMax(velocityData["z"]);
            if(Has(velocityData,"orbitalX")){velocity.orbitalX=MinMax(velocityData["orbitalX"]);velocity.orbitalY=MinMax(velocityData["orbitalY"]);velocity.orbitalZ=MinMax(velocityData["orbitalZ"]);}
            if(Has(velocityData,"orbitalOffsetX")){velocity.orbitalOffsetX=MinMax(velocityData["orbitalOffsetX"]);velocity.orbitalOffsetY=MinMax(velocityData["orbitalOffsetY"]);velocity.orbitalOffsetZ=MinMax(velocityData["orbitalOffsetZ"]);}
            if(Has(velocityData,"radial"))velocity.radial=MinMax(velocityData["radial"]);
            if(Has(velocityData,"speedModifier"))velocity.speedModifier=MinMax(velocityData["speedModifier"]);
        }
        var limit=system.limitVelocityOverLifetime;var limitData=Enabled(data,"ClampVelocityModule");limit.enabled=limitData!=null;
        if(limitData!=null) {
            limit.separateAxes=Flag(limitData,"separateAxis");limit.space=Space(limitData,"inWorldSpace");
            if(limit.separateAxes){limit.limitX=MinMax(limitData["x"]);limit.limitY=MinMax(limitData["y"]);limit.limitZ=MinMax(limitData["z"]);}else limit.limit=MinMax(limitData["magnitude"]);
            limit.dampen=Number(limitData,"dampen");if(Has(limitData,"drag"))limit.drag=MinMax(limitData["drag"]);
            limit.multiplyDragByParticleSize=Flag(limitData,"multiplyDragByParticleSize",true);limit.multiplyDragByParticleVelocity=Flag(limitData,"multiplyDragByParticleVelocity",true);
        }
        var force=system.forceOverLifetime;var forceData=Enabled(data,"ForceModule");force.enabled=forceData!=null;
        if(forceData!=null){force.space=Space(forceData,"inWorldSpace");force.x=MinMax(forceData["x"]);force.y=MinMax(forceData["y"]);force.z=MinMax(forceData["z"]);force.randomized=Flag(forceData,"randomizePerFrame");}
        var noise=system.noise;var noiseData=Enabled(data,"NoiseModule");noise.enabled=noiseData!=null;
        if(noiseData!=null) {
            noise.separateAxes=Flag(noiseData,"separateAxes");
            if(noise.separateAxes){noise.strengthX=MinMax(noiseData["strength"]);noise.strengthY=MinMax(noiseData["strengthY"]);noise.strengthZ=MinMax(noiseData["strengthZ"]);}else noise.strength=MinMax(noiseData["strength"]);
            noise.frequency=Number(noiseData,"frequency",.5f);noise.damping=Flag(noiseData,"damping",true);noise.octaveCount=(int)Number(noiseData,"octaves",1);
            noise.octaveMultiplier=Number(noiseData,"octaveMultiplier",.5f);noise.octaveScale=Number(noiseData,"octaveScale",2);noise.quality=(ParticleSystemNoiseQuality)(int)Number(noiseData,"quality",2);
            if(Has(noiseData,"scrollSpeed"))noise.scrollSpeed=MinMax(noiseData["scrollSpeed"]);
            noise.remapEnabled=Flag(noiseData,"remapEnabled");if(noise.remapEnabled){noise.remap=MinMax(noiseData["remap"]);noise.remapY=MinMax(noiseData["remapY"]);noise.remapZ=MinMax(noiseData["remapZ"]);}
            if(Has(noiseData,"positionAmount"))noise.positionAmount=MinMax(noiseData["positionAmount"]);
            if(Has(noiseData,"rotationAmount"))noise.rotationAmount=MinMax(noiseData["rotationAmount"]);
            if(Has(noiseData,"sizeAmount"))noise.sizeAmount=MinMax(noiseData["sizeAmount"]);
        }
        var uv=system.textureSheetAnimation;var uvData=Enabled(data,"UVModule");uv.enabled=uvData!=null;
        if(uvData!=null) {
            uv.mode=ParticleSystemAnimationMode.Grid;uv.numTilesX=Mathf.Max(1,(int)Number(uvData,"tilesX",1));uv.numTilesY=Mathf.Max(1,(int)Number(uvData,"tilesY",1));
            uv.animation=(ParticleSystemAnimationType)(int)Number(uvData,"animationType");
            if(Has(uvData,"rowMode"))uv.rowMode=(ParticleSystemAnimationRowMode)(int)Number(uvData,"rowMode");else if(Has(uvData,"randomRow"))uv.rowMode=Flag(uvData,"randomRow",true)?ParticleSystemAnimationRowMode.Random:ParticleSystemAnimationRowMode.Custom;
            uv.rowIndex=(int)Number(uvData,"rowIndex");
            uv.timeMode=(ParticleSystemAnimationTimeMode)(int)Number(uvData,"timeMode");uv.fps=Number(uvData,"fps",30);
            uv.frameOverTime=MinMax(uvData["frameOverTime"]);uv.startFrame=MinMax(uvData["startFrame"]);uv.cycleCount=Mathf.Max(1,(int)Number(uvData,"cycles",1));
            uv.flipU=Number(uvData,"flipU");uv.flipV=Number(uvData,"flipV");
        }
    }
    void ConfigureParticleRenderer(ParticleSystemRenderer renderer,JsonData data) {
        var mode=(ParticleSystemRenderMode)(int)Number(data,"m_RenderMode");
        var mesh=Has(data,"m_Mesh")?Ref(data["m_Mesh"]) as Mesh:null;
        if(mode==ParticleSystemRenderMode.Mesh && mesh==null)mode=ParticleSystemRenderMode.Billboard;
        renderer.renderMode=mode;if(mesh!=null)renderer.mesh=mesh;
        renderer.minParticleSize=Number(data,"m_MinParticleSize");renderer.maxParticleSize=Number(data,"m_MaxParticleSize",.5f);
        renderer.velocityScale=Number(data,"m_VelocityScale");renderer.lengthScale=Number(data,"m_LengthScale",2);renderer.cameraVelocityScale=Number(data,"m_CameraVelocityScale");
        renderer.alignment=(ParticleSystemRenderSpace)(int)Number(data,"m_RenderAlignment");renderer.sortMode=(ParticleSystemSortMode)(int)Number(data,"m_SortMode");
        renderer.sortingFudge=Number(data,"m_SortingFudge");renderer.pivot=Vector3Of(data,"m_Pivot");renderer.normalDirection=Number(data,"m_NormalDirection",1);
    }

        for(int i=owned.Count-1;i>=0;i--)if(owned[i]!=null)UnityEngine.Object.DestroyImmediate(owned[i]);owned.Clear();objects.Clear();
    }
