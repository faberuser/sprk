    static RectInt SpriteRect(object atlas,string name) {
        var data=atlas.GetType().GetMethod("GetSprite",new[]{typeof(string)}).Invoke(atlas,new object[]{name});
        if(data==null)throw new InvalidOperationException("Logo sprite missing");
        // Members are fields or properties depending on the NGUI build.
        const System.Reflection.BindingFlags any=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.IgnoreCase;
        Func<string,int> get=f=>{
            var field=data.GetType().GetField(f,any);if(field!=null)return Convert.ToInt32(field.GetValue(data));
            var property=data.GetType().GetProperty(f,any);if(property!=null)return Convert.ToInt32(property.GetValue(data,null));
            throw new InvalidOperationException("Sprite data has no "+f+" on "+data.GetType().FullName);
        };
        return new RectInt(get("x"),get("y"),get("width"),get("height"));
    }

    static void RestoreLogoSubtitle(UISprite sprite,Texture2D target) {
        string folder=SystemOptionData.Spec==ESpecType.Low?"LowLargeSpec":"HighAndMidSpec";
        var bundle=AssetBundle.LoadFromFile(Path.Combine(Application.dataPath,"Documents/Patch/StandaloneWindows/Assetbundle",folder,"NguiAtlas_UI_Prefab_Login.unity3d"));
        if(bundle==null)throw new InvalidOperationException("Login atlas bundle could not be opened");
        try {
            var atlas=bundle.mainAsset as UIAtlas;
            if(atlas==null)atlas=bundle.LoadAllAssets<GameObject>().Select(g=>g.GetComponent<UIAtlas>()).FirstOrDefault(x=>x!=null);
            // The bundle material's shader is unresolved before preload; take the texture itself.
            var source=bundle.LoadAllAssets<Texture2D>().Where(t=>t.width==target.width && t.height==target.height).OrderByDescending(t=>t.name==target.name).FirstOrDefault();
            if(atlas==null)throw new InvalidOperationException("Login atlas bundle has no atlas");
            if(source==null)throw new InvalidOperationException("Login atlas bundle has no texture");
            var from=SpriteRect(atlas,CornerLogoSprite);var to=SpriteRect(sprite.atlas,CornerLogoSprite);
            if(from.width!=to.width || from.height!=to.height || source.format!=target.format || SystemInfo.copyTextureSupport==UnityEngine.Rendering.CopyTextureSupport.None)
                throw new InvalidOperationException("Login atlas layouts differ");
            // NGUI rects are top-left based; textures are bottom-left based.
            int mips=System.Math.Min(source.mipmapCount,target.mipmapCount);
            for(int mip=0;mip<mips;mip++) {
                int scale=1<<mip,w=from.width/scale,h=from.height/scale;if(w<1 || h<1)break;
                Graphics.CopyTexture(source,0,mip,from.x/scale,(source.height-from.y-from.height)/scale,w,h,target,0,mip,to.x/scale,(target.height-to.y-to.height)/scale);
            }
            restoredLogoTexture=target;
        } finally {bundle.Unload(true);}
    }

    void IncludeArt(Vector3 world) {
        var p=coverRoot.InverseTransformPoint(world);
        if(!coverHasArt){coverArt=new Rect(p.x,p.y,0,0);coverHasArt=true;return;}
        coverArt=Rect.MinMaxRect(Mathf.Min(coverArt.xMin,p.x),Mathf.Min(coverArt.yMin,p.y),Mathf.Max(coverArt.xMax,p.x),Mathf.Max(coverArt.yMax,p.y));
    }
    // Measures in root space, so the current zoom does not affect the result.
    // Spine meshes fill in over the first frames; keep the union of samples.
    void SampleArt() {
        foreach(var renderer in coverRoot.GetComponentsInChildren<Renderer>()) {
            if(!renderer.enabled || renderer is ParticleSystemRenderer || renderer.GetComponent<UIDrawCall>()!=null)continue;
            var b=renderer.bounds;if(b.size.x<=0 || b.size.y<=0)continue;
            IncludeArt(b.min);IncludeArt(b.max);
        }
        foreach(var widget in coverRoot.GetComponentsInChildren<UIWidget>()) {
            if(!(widget is UISprite || widget is UITexture) || !widget.enabled || widget.finalAlpha<.01f)continue;
            // Letterbox curtains are black edge masks, not art.
            if(widget.name.IndexOf("LetterBox",StringComparison.OrdinalIgnoreCase)>=0 || widget.color.maxColorComponent<.02f)continue;
            var corners=widget.worldCorners;IncludeArt(corners[0]);IncludeArt(corners[2]);
        }
    }
    // Renderer bounds include transparent padding and wide glow/fog layers, so
    // they overstate the art. Once rendered, read a few rows of the frame and
    // narrow the art to its non-black columns (black bars are exactly black).
    const int PixelChecks=3;
    int pixelChecks;
    bool pixelCheckRunning;
    public int PixelCorrections {get;private set;}
    IEnumerator CheckEdgePixels(Camera camera,Transform root) {
        pixelCheckRunning=true;
        yield return new WaitForEndOfFrame();
        try {
            if(root!=coverRoot || root==null || camera==null)yield break;
            int width=Screen.width,height=Screen.height;if(width<64 || height<64)yield break;
            // Rows clear of the logo, the bottom buttons and the status line.
            var rows=new[]{.22f,.30f,.38f,.46f,.54f,.62f,.70f,.78f};
            var strip=new Texture2D(width,rows.Length,TextureFormat.RGB24,false);
            try {
                for(int i=0;i<rows.Length;i++)strip.ReadPixels(new Rect(0,Mathf.RoundToInt(height*rows[i]),width,1),0,i,false);
                var pixels=strip.GetPixels32();
                Func<int,bool> art=x=>{int lit=0;for(int i=0;i<rows.Length;i++){var p=pixels[i*width+x];if(p.r>3 || p.g>3 || p.b>3)lit++;}return lit*2>=rows.Length;};
                int left=0;while(left<width && !art(left))left++;
                int right=width-1;while(right>=0 && !art(right))right--;
                // Nothing drawn yet (fade-in) or no bars: nothing to correct.
                if(right-left<width/3 || left<2 && right>width-3)yield break;
                var l=root.InverseTransformPoint(camera.ViewportToWorldPoint(new Vector3((left+.5f)/width,.5f,0))).x;
                var r=root.InverseTransformPoint(camera.ViewportToWorldPoint(new Vector3((right+.5f)/width,.5f,0))).x;
                float xMin=coverArt.xMin,xMax=coverArt.xMax;
                if(left>=2)xMin=Mathf.Max(xMin,Mathf.Min(l,r));
                if(right<=width-3)xMax=Mathf.Min(xMax,Mathf.Max(l,r));
                if(xMax-xMin>=coverArt.width*.5f && (xMin!=coverArt.xMin || xMax!=coverArt.xMax)){coverArt=Rect.MinMaxRect(xMin,coverArt.yMin,xMax,coverArt.yMax);PixelCorrections++;}
            } finally {Destroy(strip);}
        } finally {pixelCheckRunning=false;}
    }
    void CoverScreen(Camera camera) {
        if(recovered==null || recovered.Root==null){if(coverRoot!=null)ResetCover();return;}
        if(coverRoot!=recovered.Root.transform)BeginCover();
        if(coverSamples<CoverSampleFrames){coverSamples++;SampleArt();}
        // After sampling, check the rendered edges a few times, about every 20 frames.
        else if(coverHasArt && pixelChecks<PixelChecks && !pixelCheckRunning && (++coverSamples-CoverSampleFrames)%20==0){pixelChecks++;StartCoroutine(CheckEdgePixels(camera,coverRoot));}
        if(!coverHasArt || coverArt.width<1 || coverArt.height<1)return;
        var parent=coverRoot.parent;
        Vector3 a=parent.InverseTransformPoint(camera.ViewportToWorldPoint(new Vector3(0,0,0))),b=parent.InverseTransformPoint(camera.ViewportToWorldPoint(new Vector3(1,1,0)));
        var viewRect=Rect.MinMaxRect(Mathf.Min(a.x,b.x),Mathf.Min(a.y,b.y),Mathf.Max(a.x,b.x),Mathf.Max(a.y,b.y));CoverView=viewRect;
        // Art rectangle in parent space at the unzoomed transform.
        float xMin=coverBasePosition.x+coverBaseScale.x*coverArt.xMin,xMax=coverBasePosition.x+coverBaseScale.x*coverArt.xMax;
        float yMin=coverBasePosition.y+coverBaseScale.y*coverArt.yMin,yMax=coverBasePosition.y+coverBaseScale.y*coverArt.yMax;
        var c=viewRect.center;float scale=1;
        if(xMin<c.x)scale=Mathf.Max(scale,(c.x-viewRect.xMin)/(c.x-xMin));
        if(xMax>c.x)scale=Mathf.Max(scale,(viewRect.xMax-c.x)/(xMax-c.x));
        if(yMin<c.y)scale=Mathf.Max(scale,(c.y-viewRect.yMin)/(c.y-yMin));
        if(yMax>c.y)scale=Mathf.Max(scale,(viewRect.yMax-c.y)/(yMax-c.y));
        // A small overscan hides the art's anti-aliased edge.
        if(scale>1)scale=Mathf.Min(scale*1.004f,MaxCoverScale);
        CoverScale=scale;
        var targetScale=coverBaseScale*scale;
        var targetPosition=new Vector3(c.x+(coverBasePosition.x-c.x)*scale,c.y+(coverBasePosition.y-c.y)*scale,coverBasePosition.z);
        if(coverRoot.localScale!=targetScale)coverRoot.localScale=targetScale;
        if(coverRoot.localPosition!=targetPosition)coverRoot.localPosition=targetPosition;
    }

    void AlignLogo(Camera camera) {

        if(logoRightInset<0) {

            // widget; its serialized position is stale until NGUI applies that.
            if(logo.leftAnchor.target!=null && logo.rightAnchor.target!=null && logo.leftAnchor.relative==1 && logo.rightAnchor.relative==1)
                logoRightInset=-(logo.leftAnchor.absolute+logo.rightAnchor.absolute)*.5f;

        }

    }

    public static float TextCenterY(UILabel label) {
        float top=(1-label.pivotOffset.y)*label.height,bottom=-label.pivotOffset.y*label.height,printed=Mathf.Min(label.printedSize.y,label.height);
        switch(label.pivot) {

            default:return (top+bottom)*.5f;
        }
    }

    public void ApplyBackground() {

        ResetCover();

        recovered=new SprkRecoveredLoginScene(Path.Combine(ContentRoot,option.scene),view.transform);HideNativeArt();BeginCover();
        appliedBackground=selected;
        ApplyEraLogo(option);
    }
    public static AudioClip FilterMusic(AudioClip incoming) {
        if(!AtLogin)return incoming;
        if(incoming!=null && (activeView==null || incoming!=activeView.customMusic))nativeMusic=incoming;
        string selected=SelectedMusic;
        if(activeView!=null && activeView.customMusic!=null && activeView.customMusicId==selected)return activeView.customMusic;
        return incoming;
    }
    public static void RefreshMusic(){if(activeView!=null && AtLogin && activeView.gameObject.activeInHierarchy)activeView.ApplyMusic();}
    void ApplyMusic() {
        int generation=++musicGeneration;string selected=SelectedMusic;
        if(customMusic!=null && customMusicId==selected){SingletonAuto<BgmManager2>.instance.PlayFromResources(customMusic,1f);return;}
        StartCoroutine(LoadMusic(Options.music.First(x=>x.id==selected),generation));
    }
    IEnumerator LoadMusic(Music option,int generation) {
        string path=Path.Combine(ContentRoot,option.file);
        using(var request=UnityWebRequestMultimedia.GetAudioClip(new Uri(path).AbsoluteUri,path.EndsWith(".ogg",StringComparison.OrdinalIgnoreCase)?AudioType.OGGVORBIS:AudioType.WAV)) {
            yield return request.SendWebRequest();
            if(request.result!=UnityWebRequest.Result.Success)throw new IOException("Login music could not be loaded: "+request.error);
            var clip=DownloadHandlerAudioClip.GetContent(request);
            clip.name=option.label;
            if(generation!=musicGeneration || !AtLogin || SelectedMusic!=option.id){Destroy(clip);yield break;}
            var previous=customMusic;customMusic=clip;customMusicId=option.id;
            SingletonAuto<BgmManager2>.instance.PlayFromResources(customMusic,1f);
            if(previous!=null)Destroy(previous,2f);
        }
    }
    void OnDisable(){musicGeneration++;RemoveEraLogo();ResetCover();if(recovered!=null){recovered.Dispose();recovered=null;appliedBackground=null;RestoreNativeArt();}}
    void OnDestroy(){musicGeneration++;RemoveEraLogo();ResetCover();if(recovered!=null)recovered.Dispose();if(customMusic!=null)Destroy(customMusic);if(activeView==this)activeView=null;}

        menu.Setup(window);
    }

           String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_DIR")) || String.IsNullOrEmpty(Environment.GetEnvironmentVariable("SPRK_AUTOMATION_PROFILE")))return;
        probeStarted=true;var go=new GameObject("SPRK login presentation probe");DontDestroyOnLoad(go);go.AddComponent<SprkLoginPresentationProbe>();
    }
