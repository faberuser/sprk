    internal static Vector3 TextStart(UILabel label,Transform relativeTo) {
        // Use the first glyph, including the font's bearing, rather than the
        // label rectangle: its top edge is not the visible top of the text.
        var vertices=new RenderingList<Vector3>();
        label.OnFill(vertices,new RenderingList<Vector2>(),new RenderingList<Color32>());
        if(vertices.size<4)return relativeTo.InverseTransformPoint(label.worldCorners[1]);
        float left=float.MaxValue,top=float.MinValue;
        for(int i=0;i<4;i++) {
            var point=relativeTo.InverseTransformPoint(label.transform.TransformPoint(vertices[i]));
            left=Mathf.Min(left,point.x);top=Mathf.Max(top,point.y);
        }
        return new Vector3(left,top,0);
    }

    static Expression Binary(Expression a,Expression b,string op) {
        // The native decimal node returns its numerator on division by zero.
        if(op=="/" && b.Value==0)return a;
        decimal n=op=="+"?a.Value+b.Value:op=="-"?a.Value-b.Value:op=="*"?a.Value*b.Value:a.Value/b.Value;
        if(!a.Symbolic && !b.Symbolic)return Constant(n);
        if(op=="+" && b.Value==0 && !b.Symbolic)return a;
        if(op=="+" && a.Value==0 && !a.Symbolic)return b;
        if((op=="*" || op=="/") && b.Value==1 && !b.Symbolic)return a;
        int priority=(op=="+" || op=="-")?1:2;
        string left=a.Priority<priority?"("+a.Text+")":a.Text;
        string right=b.Priority<priority || ((op=="-" || op=="/") && b.Priority==priority)?"("+b.Text+")":b.Text;
        var result=Symbol(left+" "+op+" "+right,n);result.Priority=priority;
        result.Key=op+"("+a.Key+","+b.Key+")";return result;
    }
    static Expression Sum(IEnumerable<Expression> expressions) {
        // Keep each damage component intact. Flatten only sums created here,
        // never the additions inside a hit or a native rounding boundary.
        var terms=new List<Expression>();
        foreach(var expression in expressions) {
            var items=expression.Addends??new List<Expression>{expression};
            foreach(var item in items)if(item.Symbolic || item.Value!=0)terms.Add(item);
        }
        if(terms.Count==0)return Constant(0);
        if(terms.Count==1)return terms[0];
        var ordered=new List<Expression>();var counts=new Dictionary<string,int>(StringComparer.Ordinal);
        foreach(var term in terms) {
            string key=term.Key+"|value:"+Number(term.Value);
            if(counts.ContainsKey(key))counts[key]++;
            else {counts.Add(key,1);ordered.Add(term);}
        }
        var sum=Constant(0);
        foreach(var term in ordered) {
            int count=counts[term.Key+"|value:"+Number(term.Value)];
            sum=Binary(sum,count==1?term:Binary(Constant(count),term,"*"),"+");
        }
        sum.Addends=terms;return sum;
    }
    // Keep native rounding in the validation value; the player sees the clean

    static Expression Truncate(Expression a) {return a.Symbolic?new Expression(a.Text,decimal.Truncate(a.Value),true){Priority=a.Priority,Key="truncate("+a.Key+")"}:Constant(decimal.Truncate(a.Value));}
    static Expression Rounded(Expression a,string operation,int digits) {
        decimal value=operation=="Ceiling"?decimal.Ceiling(a.Value):decimal.Round(a.Value,digits);
        return a.Symbolic?new Expression(a.Text,value,true){Priority=a.Priority,Key=operation+":"+digits+"("+a.Key+")"}:Constant(value);
    }
    internal static void VerifyRepeatedTermGrouping() {
        var atk=Binary(Symbol("ATK",100),Constant(10),"+");
        var hp=Binary(Symbol("HP",100),Constant(10),"+");
        var repeated=Sum(new[]{atk,atk,atk,atk});
        if(repeated.Text!="4 * (ATK + 10)" || repeated.Value!=440)throw new Exception("Repeated hit grouping failed");
        var mixed=Sum(new[]{atk,hp,atk,hp});
        if(mixed.Text!="2 * (ATK + 10) + 2 * (HP + 10)" || mixed.Value!=440)throw new Exception("Mixed hit grouping failed");
        var nested=Sum(new[]{Sum(new[]{atk,atk}),atk});
        if(nested.Text!="3 * (ATK + 10)" || nested.Value!=330)throw new Exception("Nested grouping failed");
        var rounded=Truncate(Symbol("x",1.25m));var exact=Symbol("x",1);
        if(Sum(new[]{rounded,exact}).Text!="x + x")throw new Exception("Different rounding boundaries were grouped");
        var addition=Binary(Symbol("ATK",100.45m),Constant(10),"+");
        var roundedAddition=Rounded(addition,"Round",0);
        if(roundedAddition.Text!="ATK + 10" || roundedAddition.Value!=110 || Binary(roundedAddition,Constant(2),"*").Text!="(ATK + 10) * 2")throw new Exception("Hidden rounding lost value or precedence");
        var precision=Rounded(Symbol("x",1.25m),"Round",1);
        var ceiling=Rounded(Symbol("x",1.25m),"Ceiling",0);
        if(precision.Text!="x" || precision.Value!=1.2m || ceiling.Text!="x" || ceiling.Value!=2)throw new Exception("Hidden round/ceiling value mismatch");
        if(Sum(new[]{Rounded(Symbol("x",1),"Round",0),Rounded(Symbol("x",1),"Round",1)}).Text!="x + x")throw new Exception("Different rounding precision was grouped");
        if(Sum(new[]{ceiling,Symbol("x",2)}).Text!="x + x")throw new Exception("Ceiling boundary was grouped with an unrounded value");
        if(Sum(new[]{atk,hp}).Text.Contains("2 *"))throw new Exception("Different symbols were grouped by their current values");
    }

    static Expression LevelFactor(int type,Context c) {
        var table=GameRoot.Instance.SkillLevelFactorDataContainer;
        int value=table.GetLevelFactor(type,c.Level);
        if(type==9999)return Symbol(Words("Skill Level"),c.Level);
        // Infer only a fully verified affine lookup across the supported levels.
        int first=table.GetLevelFactor(type,1),delta=table.GetLevelFactor(type,2)-first;
        bool affine=true;for(int l=1;l<=100;l++)if(table.GetLevelFactor(type,l)!=first+(l-1)*delta){affine=false;break;}
        if(affine)return Binary(Constant((first-delta)/1000m),Binary(Constant(delta/1000m),Symbol(Words("Skill Level"),c.Level),"*"),"+");
        string name=Words("Level Scaling")+(type==5?"":" "+type);
        c.Result.Legends[name+"("+c.Level+")"]=Number(value/1000m);
        return Symbol(name+"("+Words("Skill Level")+")",value/1000m);
    }
    static Expression LevelValue(int[] factors,int at,Context c,bool awakening) {
        if(factors==null || at+1>=factors.Length || factors[at+1]==0)return Constant(0);
        var e=Truncate(Binary(Constant(factors[at+1]),LevelFactor(factors[at],c),"*"));
        if(awakening) {
            var hero=c.RuntimeValues?c.Operation.FiredCreature.CreateInfo.HeroInfo:c.Hero;
            int creatureIndex=c.RuntimeValues?c.Operation.FiredCreature.CreatureIndex:hero.HeroIndex;
            decimal factor=GameRoot.Instance.CreatureStatDataContainer.GetLevelStatFactor(creatureIndex,(int)hero.Star,(int)hero.Transcended)/1000m;
            c.Result.Legends[Words("Awakening Coefficient")]=Number(factor);
            e=Truncate(Binary(e,Symbol(Words("Awakening Coefficient"),factor),"*"));
        }
        return e;
    }
    static Expression OperationValue(OperationData op,int index,Context c) {
        decimal value=0;
        // These native nodes call GetOperationValueInt: non-integer strings,
        // including '$' expressions, parse to the native default zero here.
        var values=c.RuntimeValues?c.Operation.GetValues():op.Values;
        string raw=values!=null && index<values.Length?values[index]:"0";
        Expression result;
        if(c.RuntimeValues && raw!=null && raw.StartsWith("$"))result=OperationExpression(raw,c);
        else {int parsed=0;int.TryParse(raw,NumberStyles.Integer,CultureInfo.InvariantCulture,out parsed);value=parsed;result=Constant(value);}
        return Binary(result,LevelValue(op.ValueFactors,index*2,c,true),"+");
    }
    static Expression OperationExpression(string raw,Context c) {
        var expression=OperationValueExpression.Create(raw);
        if(expression==null)throw new NotSupportedException("Missing operation expression");
        var match=OperationValueExpression.GetMatches(raw)[0];
        var arguments=match.Groups["ARGUMENT"].Value.Split(new[]{',',';'});
        var stack=new Stack<Expression>();
        foreach(var token in expression.RpnTokens) {
            if(token is OperatorInfo) {
                var b=stack.Pop();var a=stack.Count>0?stack.Pop():Constant(0);
                string op=token.Token;
                if(op=="/" && b.Value==0)stack.Push(Constant(0));
                else if(new[]{"+","-","*","/"}.Contains(op))stack.Push(op=="/"?Truncate(Binary(a,b,op)):Binary(a,b,op));
                else throw new NotSupportedException("Operation expression operator "+op);
                continue;
            }
            var constant=token as ConstantInfo;if(constant!=null){stack.Push(Constant(constant.Value));continue;}
            string variable=token.Token;int arg;
            if(variable.StartsWith("$") && int.TryParse(variable.Substring(1),out arg) && arg>0 && arg<=arguments.Length)variable=arguments[arg-1].Trim();
            long number;if(long.TryParse(variable,out number)){stack.Push(Constant(number));continue;}
            decimal native=expression.VariableFunc(c.Operation,token.Token);
            if(variable=="$SkillLevel"){stack.Push(Symbol(Words("Skill Level"),native));continue;}
            if(variable=="$Fc.DistanceMm") {
                c.Result.Legends[Words("Distance")]=Words("caster-to-target distance in game units (preview: 0)");
                stack.Push(Symbol(Words("Distance"),native));continue;
            }
            string readerKey=variable.Split('(')[0];var stat=OperationValueReaderFactory.GetReadingStatType(readerKey);
            if(stat!=StatType.None) {
                bool caster=readerKey.StartsWith("$Fc.",StringComparison.OrdinalIgnoreCase);
                string label=stat==StatType.PhysicalAttack?"P.ATK":stat==StatType.MagicalAttack?"M.ATK":stat==StatType.MaxHp?Words("Max HP"):stat==StatType.PhysicalDefense?"P.DEF":stat==StatType.MagicalDefense?"M.DEF":stat.ToString();
                var reading=Symbol((caster?"":Words("Target")+" ")+label,native);
                reading.Key="reader:"+variable+":"+Number(native);stack.Push(reading);continue;
            }
            throw new NotSupportedException("Operation expression reader "+variable);
        }
        if(stack.Count!=1)throw new NotSupportedException("Operation expression shape");
        return stack.Pop();
    }

    static Expression Damage(OperationData op,SkillData skill,Context c,bool dot) {
        if(formulaLinker==null)formulaLinker=new NGame2.NNodeTree.NTypeValue.NOperator.NData.NOperationData.CalcDamage.FuncLinker(value=>value is GetDamage || value is GetHeal || value is GetDot || value is GetHot);
        var linker=formulaLinker;
        linker.InitCreature(GameRoot.Instance.CreatureDataContainer.GetData(c.Hero.HeroIndex),(int)c.Hero.Star,c.Stat);
        var runtime=(BaseOperationRuntime)linker.InitRuntime(skill,c.Level,op);
        if(runtime==null)return Constant(0);
        c.Operation=runtime.OperationInfo;c.RuntimeValues=true;c.FormulaStats=c.Operation.FiredCreature.StatController;
        string type=op.Type.ToString();
        bool heal=type=="GetHealR" || type=="GetHotR";
        bool ratio=heal || new[]{"GetDamageR","GetStateDamageR","GetPhysicalDamageR","GetMagicalDamageR","GetDotR","GetPhysicalDotR","GetMagicalDotR"}.Contains(type);
        Expression result;
        if(ratio) {
            var stat=heal || type.Contains("Magical")?StatType.MagicalAttack:type.Contains("Physical")?StatType.PhysicalAttack:
                skill.AttrType==SkillAttrType.Physical?StatType.PhysicalAttack:StatType.MagicalAttack;
            var atk=Symbol(stat==StatType.PhysicalAttack?"P.ATK":"M.ATK",c.FormulaStats==null?c.Stat.GetStat(stat):c.FormulaStats.GetFinalStat(stat));
            var power=OperationValue(op,0,c);
            if(type=="GetStateDamageR") {
                c.Result.Legends[Words("Stacks")]=Words("applicable effect stacks on the target (preview: 0)");
                var stacks=Symbol(Words("Stacks"),0);
                stacks.Key="stacks:"+c.Operation.GetValueString(3)+":"+c.Operation.GetValueBool(5);
                power=Binary(power,Binary(OperationValue(op,4,c),stacks,"*"),"+");
            }
            var percentage=Binary(power,Constant(10),"/");
            var scaled=Symbol((percentage.Symbolic?"("+percentage.Text+")":percentage.Text)+"% "+Words("of")+" "+atk.Text,System.Math.Max(decimal.Truncate(atk.Value*power.Value/1000),0));
            scaled.Priority=2;scaled.Key="attack("+percentage.Key+","+atk.Key+")";
            result=Binary(OperationValue(op,1,c),scaled,"+");
        } else if(new[]{"GetDamage","GetPhysicalDamage","GetMagicalDamage","GetHeal","GetDot","GetHot"}.Contains(type))result=OperationValue(op,0,c);
        else throw new NotSupportedException("Operation "+type);
        if(dot) {
            var ticks=Truncate(Binary(Timing(op,false,c),Timing(op,true,c),"/"));
            result=Binary(result,ticks,"*");
        }
        if(!dot && runtime is GetDamage) {
            long amount;DamageFlag flag;((GetDamage)runtime).GetInstantDamageInfo(out amount,out flag);
            if(result.Value!=amount)throw new InvalidOperationException("Operation parity "+skill.Index+"/"+op.OperationNo+" "+type+" values="+String.Join(",",op.Values)+" runtime="+String.Join(",",c.Operation.GetValues())+" native1="+c.Operation.GetValueInt(1)+" symbolic1="+OperationValue(op,1,c).Value+" expected="+amount+" got="+result.Value);
        }
        return result;
    }
    static Expression Numeric(ITypeValueBase node,Context c) {
        decimal native=Convert.ToDecimal(node.GetObjectValue(c.Variable),CultureInfo.InvariantCulture);
        try {
            var body=node.MakeBodyJson();string path=node.ValuePath;Expression e=null;c.RuntimeValues=false;
            if(path=="Data/OperationData/CalcDamage" || path=="Data/OperationData/CalcDot") {
                var skill=GameRoot.Instance.SkillDataContainer.GetData(Int(body,"SkillIndex"));var terms=new List<Expression>();
                var ids=body["OperationNoList"];
                for(int i=0;i<ids.Count;i++) {
                    var operation=skill==null?null:skill.GetOperationDataUnsafe(Convert.ToInt32(ids.GetFromArray(i).ToString()));
                    if(operation!=null && operation.Type!=OperationType.None)terms.Add(Damage(operation,skill,c,path.EndsWith("CalcDot")));
                }
                e=Sum(terms);
            } else if(path=="Data/ValueFromOperationData") {
                var skill=GameRoot.Instance.SkillDataContainer.GetData(Int(body,"SkillIndex"));var op=skill==null?null:skill.GetOperationDataUnsafe(Int(body,"OperationNo"));
                string type=Str(body,"ValueType");
                e=op==null?Constant(0):type=="Value"?OperationValue(op,Int(body,"ValueIndex"),c):type=="Duration"?Timing(op,false,c):type=="TickTime"?Timing(op,true,c):Constant(native);
            } else if(path.StartsWith("Decimal/Calculate/")) {
                var args=node.GetArgumentContainer();string op=path.Substring(path.LastIndexOf('/')+1);
                var a=Numeric(args.GetTypeValueBase(0),c);
                if(op=="Add") {
                    var terms=new List<Expression>{a};for(int i=1;i<args.HolderList.Count;i++)terms.Add(Numeric(args.GetTypeValueBase(i),c));e=Sum(terms);
                } else if(new[]{"Sub","Multiply","Divide"}.Contains(op)) {
                    e=a;for(int i=1;i<args.HolderList.Count;i++)e=Binary(e,Numeric(args.GetTypeValueBase(i),c),op=="Add"?"+":op=="Sub"?"-":op=="Multiply"?"*":"/");
                } else if(op=="Round" || op=="Ceiling") {
                    int digits=args.HolderList.Count>1?(int)Numeric(args.GetTypeValueBase(1),c).Value:0;
                    e=Rounded(a,op,digits);
                }
            } else if(path=="Data/ValueFromStatData") {
                string name=Str(body,"StatType");e=Symbol(name=="PhysicalAttack"?"P.ATK":name=="MagicalAttack"?"M.ATK":name=="MaxHp"?Words("Max HP"):name,native);
            } else if(path=="Decimal/Convert/ToFloat")e=Numeric(node.GetArgumentContainer().GetTypeValueBase(0),c);
            else if(path.EndsWith("/Constant") || path=="Data/ValueFromSkillData" || path=="Data/SkillData/OperationFireCount")e=Constant(native);
            if(e==null)throw new NotSupportedException(path);
            // Guard each symbolic quantity against the actual native evaluator.
            if(e.Value!=native)throw new InvalidOperationException("Native parity: "+path+" expected "+native+" got "+e.Value);
            if(e.Symbolic)c.Result.Formulas++;
            return e;
        } catch(Exception e) { c.Result.Unsupported.Add(node.ValuePath+": "+e.Message);return Constant(native); }
    }
    static string Text(ITypeValueBase node,Context c) {
        if(node.ValuePath!="String/Localize")return Convert.ToString(node.GetObjectValue(c.Variable),CultureInfo.CurrentCulture);
        var json=node.MakeBodyJson();string key=Str(json["LocalizeLink"],"Key");
        var args=node.GetArgumentContainer();var values=new List<object>();
        if(args!=null && args.HolderList!=null)foreach(var holder in args.HolderList) {
            var child=holder.Value;
            values.Add(child.GetValueType()==typeof(string)?(object)Text(child,c):Numeric(child,c));
        }
        if(c.Localize==null)return LocalizationManager.GetStringDirect(key,values.ToArray());
        // The game uses one-based replacement tokens, not String.Format.
        string template=c.Localize(key);
        for(int i=0;i<values.Count;i++)template=template.Replace("{"+(i+1)+"}",Convert.ToString(values[i],CultureInfo.CurrentCulture));
        for(int i=0;i<10 && template.Contains("{@");i++)
            template=System.Text.RegularExpressions.Regex.Replace(template,@"\{@([^}]+)\}",m=>c.Localize(m.Groups[1].Value));
        return template;
    }
    public static Result Render(int index,HeroInfo hero,Func<string,string> localize) {
        var container=SingletonAuto<SkillDescContainer>.instance;
        var desc=container.SkillDescDic.GETVALUE(new SkillDescContainer.DescKey(index,hero.CostumeIndex),null)
            ??container.SkillDescDic.GETVALUE(new SkillDescContainer.DescKey(index,0),null);
        if(desc==null) {
            var skill=GameRoot.Instance.SkillDataContainer.GetData(index);
            if(skill==null)throw new NotSupportedException("Missing skill");
            if(localize==null)return new Result{Text=skill.Description};
            string template=localize("SKILL_DESC_"+index);
            if(skill.DescParam!=null)for(int i=0;i<skill.DescParam.Length;i++)template=template.Replace("{"+(i+1)+"}",skill.DescParam[i]);
            for(int i=0;i<10 && template.Contains("{@");i++)template=System.Text.RegularExpressions.Regex.Replace(template,@"\{@([^}]+)\}",m=>localize(m.Groups[1].Value));
            return new Result{Text=template};
        }
        var variable=SmallPooling<SkillDescVariable>.CreateInstance();
        try {
            var creature=GameRoot.Instance.CreatureDataContainer.GetData(hero.HeroIndex);
            int level=hero.GetSkillLevelBySkillIndex(creature,index,true,false);
            var stat=hero is CachedHeroInfo?SceneAutoBehaviour<StatDataCache>.instance.GetCachedHeroStat((CachedHeroInfo)hero,StatCalculator.HeroStatCalculationFlag.All):SceneAutoBehaviour<StatDataCache>.instance.GetHeroStat(hero);
            variable.Open(stat,hero.HeroIndex,(int)hero.Star,(int)hero.Transcended,index,level);
            var c=new Context{Hero=hero,Stat=stat,Level=level+StatCalculator.GetEquipOptionAddedSkillLevel(stat,creature.GetSkillSlotByIndex(index,true)),Variable=variable,Localize=localize};
            var argument=(ArgumentSingleGeneric<string>)typeof(SkillDesc).GetField("_skillDesc",Private).GetValue(desc);
            c.Result.Text=Text(argument.GetTypeValue(),c);
            if(c.Result.Unsupported.Count>0)c.Result.Text+="\n\n"+Words("Some values remain calculated.");
            return c.Result;
        } finally { SmallPooling<SkillDescVariable>.ReleaseInstance(variable); }
    }
