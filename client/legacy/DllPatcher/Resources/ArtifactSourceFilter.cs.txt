using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using NNguiExtension;
using NGame2.NUI;
using NGame2.NUI.NWindow;
using NShared;
using UnityEngine;

public sealed class SprkArtifactSourceState : MonoBehaviour {
    public bool Active;
    public bool Fresh = true;
}

public static class SprkArtifactSourceFilter {
    // The same archived reward groups used by artifact boxes and selectors.
    // Resolve item codes from native tables rather than guessing from names or IDs.
    static readonly string[] Groups = {
        "GACHA_ALL_IN_ONE_ALL_220920_NORMAL_ARTIFACT_GROUP",
        "GACHA_ALL_IN_ONE_ALL_220920_EVENT_ARTIFACT_GROUP",
        "GACHA_ALL_IN_ONE_ALL_220920_WORLD_1_ARTIFACT_GROUP",
        "GACHA_ALL_IN_ONE_ALL_220920_VELKAZAR_1_ARTIFACT_GROUP",
        "GACHA_ALL_IN_ONE_ALL_220920_SHAKMEH_1_ARTIFACT_GROUP",
        "GACHA_ALL_IN_ONE_ALL_220920_KINGSPASS_ARTIFACT_GROUP",
        "GACHA_ALL_IN_ONE_ALL_220920_PUNISHMENT_1_ARTIFACT_GROUP"
    };
    static readonly string[] Labels = { "Normal", "Event", "World Boss", "Velkazar", "Shakmeh", "King's Pass", "Divine Punishment", "Other" };
    const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    static readonly FieldInfo TicketField = typeof(WeaponUniqueSelect).GetField("_useItemData",Private);
    static readonly FieldInfo SelectionField = typeof(WeaponUniqueSelect).GetField("_buyItemData",Private);
    static readonly MethodInfo ClearDescription = typeof(WeaponUniqueSelect).GetMethod("SetItemDescription",Private);
    static Dictionary<string,int> categories;

    static void AddGroup(Dictionary<string,int> result,string group,int category,HashSet<string> visiting) {
        if(!visiting.Add(group)) return;
        var data=GameRoot.Instance.ItemGroupDataContainer.GetData(group);
        if(data!=null && data.SubDataList!=null) {
            foreach(var row in data.SubDataList) {
                var child=GameRoot.Instance.ItemGroupDataContainer.GetData(row.ItemCode);
                if(child!=null) AddGroup(result,row.ItemCode,category,visiting);
                else if(!String.IsNullOrEmpty(row.ItemCode)) result[row.ItemCode]=category;
            }
        }
        visiting.Remove(group);
    }
    public static int Category(string code) {
        if(categories==null) {
            var result=new Dictionary<string,int>(StringComparer.Ordinal);
            for(int i=0;i<Groups.Length;i++) AddGroup(result,Groups[i],101+i,new HashSet<string>());
            categories=result;
        }
        int category;
        return code!=null && categories.TryGetValue(code,out category)?category:108;
    }
    public static void Reset(SelectFilter filter) {
        var state=filter.GetComponent<SprkArtifactSourceState>();
        if(state!=null) {state.Active=false;state.Fresh=true;}
    }
    public static bool Build(SelectFilter filter,UIComboBox sub,int type,SelectFilter.Callback callback) {
        var state=filter.GetComponent<SprkArtifactSourceState>();
        var window=filter.GetComponentInParent<WeaponUniqueSelect>(true);
        if(type!=(int)SelectFilter.SelectType.SELECT_EQUIP_ARTIFACT || window==null || sub==null) {
            if(state!=null)state.Active=false;
            return false;
        }
        var ticket=TicketField.GetValue(window) as ItemData;
        var pool=ticket==null?null:ticket.GetSubData<WeaponUniqueSelectItemData>();
        if(pool==null || pool.ItemIndices==null) return false;
        if(state==null)state=filter.gameObject.AddComponent<SprkArtifactSourceState>();
        int previous=state.Fresh?0:sub.CurrentKey;
        state.Active=true;state.Fresh=false;
        var present=new HashSet<int>();
        foreach(int index in pool.ItemIndices) {
            var item=GameRoot.Instance.ItemDataContainer.GetData(index);
            if(item!=null)present.Add(Category(item.Code));
        }
        sub.Clear();
        sub.AddItem(0,NVespa.NGlobal.LocalizationManager.GetString(NShared.NLocalization.FilterSelect.FILTER_SELECT_ALL));
        for(int i=0;i<Labels.Length;i++)if(present.Contains(101+i))sub.AddItem(101+i,Labels[i]);
        sub.Reposition();
        if(!sub.SelectItem(previous,false))sub.SelectFirstItem(false);
        if(callback!=null)callback();
        return true;
    }
    public static bool Matches(SelectFilter filter,EquipItemData equip) {
        var state=filter.GetComponent<SprkArtifactSourceState>();
        if(state==null || !state.Active || filter.SubComboBoxCurrentKey<=0)return true;
        return equip!=null && Category(equip.EquipCode)==filter.SubComboBoxCurrentKey;
    }
    static bool IsActive(WeaponUniqueSelect window) {
        return window.GetComponentsInChildren<SprkArtifactSourceState>(true).Any(s=>s.Active);
    }
    public static void BeforeList(WeaponUniqueSelect window) {
        if(!IsActive(window))return;
        SelectionField.SetValue(window,null);
        ClearDescription.Invoke(window,new object[]{null});
        UIHelper.EnableButton(window.Button_Buy,false,true,false);
    }
    public static void AfterList(WeaponUniqueSelect window) {
        if(!IsActive(window) || window.ItemGrid==null || window.ItemGrid.ScrollView==null)return;
        var scroll=window.ItemGrid.ScrollView;
        scroll.DisableSpring();scroll.currentMomentum=Vector3.zero;scroll.ResetPosition();
    }
}
