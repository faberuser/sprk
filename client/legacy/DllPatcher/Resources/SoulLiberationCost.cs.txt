using NGame2.NAccount;
using NGame2.NUI.NWindow;
using NGame2.NUI.NWindow.NSoulWeapon;
using NShared;
using NVespa.NGlobal;
using NVespa.NSingleton;
using UnityEngine;

public sealed class SprkSoulLiberationCost : MonoBehaviour
{
    SoulWeaponLiberationWindow window;
    RequirePrice price;
    float nextRefresh;
    bool opened;

    public static void Show(SoulWeaponLiberationWindow window)
    {
        var view = window.LiberationTransform.GetComponent<SprkSoulLiberationCost>();
        if (view == null) view = window.LiberationTransform.gameObject.AddComponent<SprkSoulLiberationCost>();
        view.window = window;
        view.Refresh();
    }

    GameObject Child(string name, float x)
    {
        var child = new GameObject(name);
        child.layer = gameObject.layer;
        child.transform.SetParent(transform, false);
        child.transform.localPosition = new Vector3(x, -68, 0);
        return child;
    }

    void Refresh()
    {
        if (window == null) return;
        if (price == null)
        {
            var template = window.SoulWeaponMaterialInfo.MaterialCountLabel;
            var label = Child("SprkLiberationGoldPrice", 22).AddComponent<UILabel>();
            label.bitmapFont = template.bitmapFont;
            label.trueTypeFont = template.trueTypeFont;
            label.fontSize = 28;
            label.width = 230;
            label.height = 40;
            label.depth = template.depth + 1;
            label.supportEncoding = false;
            label.overflowMethod = UILabel.Overflow.ShrinkContent;
            label.alignment = NGUIText.Alignment.Center;
            var icon = Child("SprkLiberationGoldIcon", -120).AddComponent<UIWidget>();
            icon.width = icon.height = 36;
            icon.depth = label.depth;
            price = new RequirePrice {
                ReqPriceLabel = label, IconProxyRootWidget = icon,
                ReqPriceCurrencyType = CurrencyType.Gold, UsePriceColor = true
            };
            // Make room between the material counts and the existing button.
            window.LiberationButton.transform.localPosition += new Vector3(0, -26, 0);
        }
        var soul = window.EquipItemInfo == null ? null : window.EquipItemInfo.SoulWeaponInfo;
        var data = soul == null ? null : soul.GetSoulWeaponUpgradePriceData();
        var gold = GlobalAutoBehaviour<AccountManager>.instance.UserManager.Gold;
        if (!opened) { price.Open(data == null ? -1 : data.ReqGold, gold); opened = true; }
        else price.SetPrice(data == null ? -1 : data.ReqGold, gold);
        nextRefresh = Time.unscaledTime + .25f;
    }

    void Update() { if (Time.unscaledTime >= nextRefresh) Refresh(); }
    void OnDisable() { if (price != null) price.Close(); opened = false; }
}
