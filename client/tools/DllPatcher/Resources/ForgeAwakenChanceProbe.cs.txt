using System;
using System.Linq;
using System.Collections.Generic;
using System.Reflection;
using NShared;
using NGame2.NAccount;
using NGame2.NUI.NWindow;
using UnityEngine;

public static class SprkForgeAwakenChanceProbe {
    const BindingFlags Any = BindingFlags.Instance | BindingFlags.NonPublic | BindingFlags.Public;
    static void Check(bool ok, string message) { if(!ok) throw new Exception("Forge chance: " + message); }
    public static string Run(string mode) {
        NShared.NJit.TableRoot.Instance.SetPatchDownLoadComplete(true);
        var account = NVespa.NSingleton.GlobalAutoBehaviour<AccountManager>.instance;
        var managers = account.GetType().GetProperty("AccountManagers", Any).GetValue(account, null);
        var userField = managers.GetType().GetField("UserManager");
        var equipField = managers.GetType().GetField("EquipItemManager");
        var itemField = managers.GetType().GetField("ItemManager");
        var heroField = managers.GetType().GetField("HeroManager");
        var oldUser = userField.GetValue(managers); var oldEquip = equipField.GetValue(managers);
        var oldItems = itemField.GetValue(managers); var oldHeroes = heroField.GetValue(managers);
        var host = new GameObject("ForgeChanceFixture"); host.SetActive(false);
        bool preview = mode == "preview";
        var ui = UnityEngine.Object.FindObjectOfType<UIRoot>();
        if(ui != null) host.transform.SetParent(ui.transform, false);
        try {
            userField.SetValue(managers, new UserManager(account, new UserInfo { AccountId = 123456, Gold = 1000000000 }, new PlayerMiscInfo(), new PlayerGuildPointInfo()));
            equipField.SetValue(managers, new NGame2.NAccount.UserEquipItemManager());
            account.InitItemInfos(new ItemInfo[0]);
            account.InitHeroInfos(new HeroInfo[0]);
            var data = GameRoot.Instance.ItemDataContainer.GetAllData().OfType<ItemData<EquipItemData>>().Where(d => d.SubData != null).ToArray();
            var report = new List<string>();
            foreach(int category in new[]{0,1,2}) {
                var row = data.First(d => {
                    var test = new EquipItemInfo { ItemIndex = d.Index, SlotIndex = 1, Star = 3, Level = 90 };
                    return (category == 1 ? test.IsUniqueWeaponWithoutClassWeapon : category == 2 ? test.IsUniqueTreasure : (!test.IsUniqueWeaponWithoutClassWeapon && !test.IsUniqueTreasure)) &&
                        __Extension_Method_ContentsInfo_Item_Equip_MaxStar__.CLI_CanAwaken(test) && __Extension_Method_ContentsInfo_Item_Equip_MaxStar__.CLI_CanUse_as_AwakenMaterial(test);
                });
                var gear = Enumerable.Range(1, 13).Select(i => new EquipItemInfo { ItemIndex = row.Index, SlotIndex = i, Star = i == 1 ? 3 : 0, Level = 90 }).ToArray();
                gear[12].Locked = 1;
                account.InitEquipItemInfos(gear);
                var prefab = VespaResource.Load<GameObject>(HeroEquipAwaken.PrefabPath);
                Check(prefab != null, "native prefab missing");
                var view = UnityEngine.Object.Instantiate(prefab, host.transform, false).GetComponent<HeroEquipAwaken>();
                typeof(HeroEquipAwaken).GetMethod("Init", Any).Invoke(view, null);
                view.SetTargetItem(gear[0]);
                if(mode == "inspect-layout") {
                    var lines = new List<string>();
                    foreach(var t in view.GetComponentsInChildren<Transform>(true)) {
                        string path = t.name;
                        var parent = t.parent;
                        int depth = 0;
                        while(parent != null && parent != view.transform) { path = parent.name + "/" + path; parent = parent.parent; depth++; }
                        if(depth > 6) continue;
                        var widget = t.GetComponent<UIWidget>();
                        var panel = t.GetComponent<UIPanel>();
                        lines.Add(path + " pos=" + t.localPosition + " scale=" + t.localScale +
                            (widget == null ? "" : " widget=" + widget.width + "x" + widget.height + " pivot=" + widget.pivot + " anchors=" + widget.isAnchored) +
                            (panel == null ? "" : " clip=" + panel.baseClipRegion));
                    }
                    return string.Join("\n", lines.ToArray());
                }
                var state = view.GetComponent<SprkForgeAwakenChance>();
                Check(state != null, "helper absent");
                var slider = state.GetComponentInChildren<UISlider>(true);
                var chances = (int[])typeof(SprkForgeAwakenChance).GetField("chances", Any).GetValue(state);
                Check(slider != null && slider.numberOfSteps == chances.Length && chances.Length > 0, "dynamic slider steps");
                Check(chances.SequenceEqual(chances.Distinct().OrderBy(x => x)) && chances.All(x => x > 0), "stops must be unique positive chances in ascending order");
                var rate = typeof(HeroEquipAwaken).GetMethod("GetTotalAwakenSuccessRate", Any);
                int maxRate = Math.Min(100, (int)rate.Invoke(view, null));
                Check(maxRate > 0 && view.GetMaterialEquipSlotIndexArray().Length > 0, "maximum default selected no materials");
                for(int step = 0; step < chances.Length; step++) {
                    slider.value = chances.Length == 1 ? 1 : (float)step / (chances.Length - 1);
                    EventDelegate.Execute(slider.onChange);
                    int actual = Math.Min(100, (int)rate.Invoke(view, null));
                    var selected = view.GetMaterialEquipSlotIndexArray() ?? new int[0];
                    Check(!selected.Contains(1) && !selected.Contains(13) && selected.Distinct().Count() == selected.Length, "target, locked or duplicate selected");
                    Check(selected.Length > 0 && actual == chances[step], "stop differs from actual native chance");
                }
                var label = state.GetComponentsInChildren<UILabel>(true).First(l => l.name == "SprkSuccessRateLabel");
                Check(label.text == "Success: " + maxRate + "%", "readout differs from native chance");
                // Native manual removal must remain effective; the Sync hook must
                // update the readout without restoring the maximum selection.
                var icons = (List<NGame2.NUI.NComponent2.ItemIconComponent>)typeof(HeroEquipAwaken).GetField("_ListMaterialItems", Any).GetValue(view);
                typeof(HeroEquipAwaken).GetMethod("OnClickEquipItem", Any).Invoke(view, new object[]{icons[0]});
                int manualCount = view.GetMaterialEquipSlotIndexArray() == null ? 0 : view.GetMaterialEquipSlotIndexArray().Length;
                typeof(HeroEquipAwaken).GetMethod("UpdateAwakeUITask", Any).Invoke(view, null);
                Check((view.GetMaterialEquipSlotIndexArray() ?? new int[0]).Length == manualCount, "manual selection overwritten");
                // Same slot/star with an updated failure bonus rebuilds the plan.
                gear[0].UpgradeStarFailBonus = 10;
                view.SetTargetItem(gear[0]);
                Check(slider.value == 1, "failure refresh did not reset maximum");
                view.SetTargetItem(null);
                Check(!slider.transform.parent.gameObject.activeSelf, "cleared target remains interactive");
                // The requested one-star target / two zero-star copies example.
                gear[0].Star = 1; gear[0].UpgradeStarFailBonus = 0;
                account.InitEquipItemInfos(gear.Take(3).ToArray());
                view.RefreshItemList(); view.SetTargetItem(gear[0]);
                chances = (int[])typeof(SprkForgeAwakenChance).GetField("chances", Any).GetValue(state);
                Check(slider.numberOfSteps == 2 && chances.SequenceEqual(new[]{50,100}) && slider.value == 1, "two copies must give exactly 50% and 100% stops");
                slider.value = 0; EventDelegate.Execute(slider.onChange);
                Check((int)rate.Invoke(view, null) == 50 && view.GetMaterialEquipSlotIndexArray().Length == 1, "50% stop should select one copy");
                slider.value = 1; EventDelegate.Execute(slider.onChange);
                Check((int)rate.Invoke(view, null) == 100 && view.GetMaterialEquipSlotIndexArray().Length == 2, "100% stop should select two copies");
                // Repeat at the next stars, verifying native rates and limited
                // supplies rather than imposing fixed percentage intervals.
                for(int targetStar = 2; targetStar <= 4; targetStar++) {
                    gear[0].Star = targetStar;
                    view.SetTargetItem(gear[0]);
                    chances = (int[])typeof(SprkForgeAwakenChance).GetField("chances", Any).GetValue(state);
                    Check(chances.Length == 2 && slider.numberOfSteps == 2 && slider.value == 1, "next star should retain just two achievable material counts");
                    slider.value = 0; EventDelegate.Execute(slider.onChange);
                    Check((int)rate.Invoke(view, null) == chances[0] && view.GetMaterialEquipSlotIndexArray().Length == 1, "next-star first stop");
                    slider.value = 1; EventDelegate.Execute(slider.onChange);
                    Check((int)rate.Invoke(view, null) == chances[1] && view.GetMaterialEquipSlotIndexArray().Length == 2, "next-star maximum stop");
                }
                account.InitEquipItemInfos(gear.Take(2).ToArray());
                view.RefreshItemList(); view.SetTargetItem(gear[0]);
                Check(slider.numberOfSteps == 1 && !slider.enabled && slider.value == 1 && view.GetMaterialEquipSlotIndexArray().Length == 1, "single stop must select the only available copy without dragging");
                account.InitEquipItemInfos(new[]{gear[0]});
                view.RefreshItemList(); view.SetTargetItem(gear[0]);
                Check(slider.numberOfSteps == 0 && !slider.enabled && label.text == "Success: 0%" && view.GetMaterialEquipSlotIndexArray() == null, "empty inventory must disable dragging and select nothing");
                if(category != 0) {
                    var stone = GameRoot.Instance.ItemDataContainer.GetAllData().First(d =>
                        d.Type == (category == 1 ? ItemType.PublicUniqueEquipAwaken : ItemType.PublicTreasureAwaken) &&
                        d.GetSubData<AwakenStoneData>() != null && (d.GetSubData<AwakenStoneData>().ItemIndices == null || d.GetSubData<AwakenStoneData>().ItemIndices.Length == 0));
                    account.InitEquipItemInfos(new[]{gear[0],gear[12]});
                    account.InitItemInfos(new[]{new ItemInfo { ItemIndex = stone.Index, Count = 5 }});
                    view.RefreshItemList();
                    view.SetTargetItem(gear[0]);
                    var used = view.GetMaterialPublicAwakenItemIndexArray();
                    Check(used != null && used.Length == 2 && used[0] == stone.Index && used[1] > 0 && used[1] <= 5, "stone quantities missing from native request");
                    Check(view.GetMaterialEquipSlotIndexArray().Length == 0, "stone plan selected locked gear");
                    slider.value = 0; EventDelegate.Execute(slider.onChange);
                    Check(view.GetMaterialPublicAwakenItemIndexArray()[1] == 1, "leftmost stone stop must select one material");
                    slider.value = 1; EventDelegate.Execute(slider.onChange);
                    account.InitItemInfos(new ItemInfo[0]);
                }
                report.Add((category == 0 ? "normal" : category == 1 ? "UW" : "UT") + "=" + row.Index + ": " + maxRate + "%");
                if(preview && category == 2) host.SetActive(true);
                else UnityEngine.Object.Destroy(view.gameObject);
            }
            return "PASS: native Forge dynamic stops, exact 50%/100% two-copy example, next-star rates, single/empty inventories, maximum defaults, native request arrays, stones, locks, manual removal and failure refresh for " + string.Join(", ", report.ToArray());
        } finally {
            userField.SetValue(managers, oldUser); equipField.SetValue(managers, oldEquip);
            itemField.SetValue(managers, oldItems); heroField.SetValue(managers, oldHeroes);
            if(!preview) UnityEngine.Object.Destroy(host);
        }
    }
}
