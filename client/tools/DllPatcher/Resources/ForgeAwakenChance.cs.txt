using System;
using System.Collections.Generic;
using System.Reflection;
using NGame2.NUI.NComponent2;
using NGame2.NUI.NWindow;
using NShared;
using UnityEngine;

public sealed class SprkForgeAwakenChance : MonoBehaviour {
    const BindingFlags Any = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
    static readonly FieldInfo Inventory = typeof(HeroEquipAwaken).GetField("_ListAwakeInvenItems", Any);
    static readonly FieldInfo Stones = typeof(HeroEquipAwaken).GetField("_ListPublicAwakenItemComponents", Any);
    static readonly MethodInfo ClearMaterials = typeof(HeroEquipAwaken).GetMethod("ClearMaterialEquipIcon", Any);
    static readonly MethodInfo AddMaterial = typeof(HeroEquipAwaken).GetMethod("SetupMaterialEquipIcon", Any);
    static readonly MethodInfo CanUseStone = typeof(HeroEquipAwaken).GetMethod("CanUseAwakenStone", Any);
    static readonly MethodInfo Rate = typeof(HeroEquipAwaken).GetMethod("GetTotalAwakenSuccessRate", Any);
    static readonly MethodInfo[] Refresh = {
        typeof(HeroEquipAwaken).GetMethod("UpdateAwakeUITask", Any),
        typeof(HeroEquipAwaken).GetMethod("UpdateAwakeEquipItemListTask", Any),
        typeof(HeroEquipAwaken).GetMethod("UpdatePublicAwakeItemListTask", Any)
    };
    HeroEquipAwaken window;
    GameObject root;
    UISlider slider;
    UILabel label;
    int slot = -1, star = -1, bonus = -1;
    bool updating;
    readonly List<ItemIconComponent> candidates = new List<ItemIconComponent>();
    int[][] plans;
    int[] chances;

    // Every native UI refresh sees target changes, manual selections and result
    // refreshes. The guard lets our native refresh calls run without recursion.
    public static void Sync(HeroEquipAwaken window) {
        var state = window.GetComponent<SprkForgeAwakenChance>();
        if(state == null) state = window.gameObject.AddComponent<SprkForgeAwakenChance>();
        if(state.updating) return;
        state.window = window;
        state.updating = true;
        try { state.SyncTarget(); } finally { state.updating = false; }
    }

    void SyncTarget() {
        var info = window.TargetItem == null ? null : window.TargetItem.EquipItemInfo;
        if(info == null || !__Extension_Method_ContentsInfo_Item_Equip_MaxStar__.CLI_CanAwaken(info)) {
            slot = star = bonus = -1;
            if(root != null) root.SetActive(false);
            return;
        }
        EnsureSlider();
        root.SetActive(true);
        if(slot != info.SlotIndex || star != info.Star || bonus != info.UpgradeStarFailBonus) {
            slot = info.SlotIndex; star = info.Star; bonus = info.UpgradeStarFailBonus;
            BuildPlans(info);
            Apply(plans.Length - 1);
        } else UpdateLabel();
    }

    void BuildPlans(EquipItemInfo target) {
        candidates.Clear();
        var seen = new HashSet<int>();
        foreach(var icon in (List<ItemIconComponent>)Inventory.GetValue(window)) {
            var info = icon.EquipItemInfo;
            if(info != null && info.SlotIndex != target.SlotIndex && info.ItemIndex == target.ItemIndex &&
                __Extension_Method_ContentsInfo_Item_Equip_MaxStar__.CLI_CanUse_as_AwakenMaterial(info) && seen.Add(info.SlotIndex))
                candidates.Add(icon);
        }
        foreach(var icon in (List<ItemIconComponent>)Stones.GetValue(window)) {
            var data = icon.ItemData;
            if(data != null && ((data.Type == ItemType.PublicUniqueEquipAwaken && target.IsUniqueWeaponWithoutClassWeapon) ||
                (data.Type == ItemType.PublicTreasureAwaken && target.IsUniqueTreasure)) && (bool)CanUseStone.Invoke(window, new object[]{icon}))
                candidates.Add(icon);
        }
        // Deterministic ties prefer lower-star duplicates before universal stones.
        candidates.Sort((a,b) => {
            int sa = a.EquipItemInfo == null ? a.ItemData.AwakenStar : a.EquipItemInfo.Star;
            int sb = b.EquipItemInfo == null ? b.ItemData.AwakenStar : b.EquipItemInfo.Star;
            int cmp = sa.CompareTo(sb);
            if(cmp != 0) return cmp;
            if(a.EquipItemInfo == null) return b.EquipItemInfo == null ? a.ItemData.Index.CompareTo(b.ItemData.Index) : 1;
            return b.EquipItemInfo == null ? -1 : a.EquipItemInfo.SlotIndex.CompareTo(b.EquipItemInfo.SlotIndex);
        });
        int max = GameRoot.Instance.EquipAwakenRatioDataContainer.GetMaxAwakenPoint(star + 1);
        var points = new int[candidates.Count];
        for(int i = 0; i < points.Length; i++) {
            var icon = candidates[i];
            points[i] = GameRoot.Instance.EquipAwakenPointDataContainer.GetData(icon.EquipItemInfo == null ? icon.ItemData.AwakenStar : icon.EquipItemInfo.Star).AwakenPoint;
        }
        var rates = new int[max + 1];
        for(int i = 1; i <= max; i++) {
            var row = GameRoot.Instance.EquipAwakenRatioDataContainer.GetData(star + 1, i);
            rates[i] = row == null ? rates[i-1] : row.SuccessRatio;
        }
        plans = Plan(points, rates, bonus);
        chances = new int[plans.Length];
        for(int i = 0; i < plans.Length; i++) {
            long total = 0;
            foreach(int index in plans[i]) total += points[index];
            chances[i] = Math.Min(100, rates[(int)Math.Min(max, total)] + bonus);
        }
        slider.numberOfSteps = plans.Length;
        slider.enabled = plans.Length > 1;
        foreach(var collider in slider.GetComponentsInChildren<Collider>(true)) collider.enabled = plans.Length > 1;
        foreach(var collider in slider.GetComponentsInChildren<Collider2D>(true)) collider.enabled = plans.Length > 1;
    }

    sealed class Choice {
        public int Index, Count;
        public long Points;
        public Choice Previous;
    }

    // Bounded subset selection: least excess awakening points, then fewest
    // materials. Immutable links prevent a later DP update reusing an item.
    public static int[][] Plan(int[] points, int[] rates, int failBonus) {
        int cap = rates.Length - 1;
        var reachable = new Choice[cap + 1];
        reachable[0] = new Choice { Index = -1 };
        for(int i = 0; i < points.Length; i++) {
            if(points[i] <= 0) continue;
            for(int p = cap - 1; p >= 0; p--) {
                var before = reachable[p];
                if(before == null) continue;
                long actual = before.Points + points[i];
                int next = (int)Math.Min(cap, actual);
                var old = reachable[next];
                if(old == null || actual < old.Points || (actual == old.Points && before.Count + 1 < old.Count))
                    reachable[next] = new Choice { Index = i, Previous = before, Points = actual, Count = before.Count + 1 };
            }
        }
        // Each distinct attainable positive success rate gets exactly one stop.
        // No empty-material stop and no repeated targets that select the same
        // materials. In particular two 50% copies produce just 50%, 100%.
        var stops = new SortedDictionary<int, Choice>();
        for(int p = 1; p <= cap; p++) {
            var choice = reachable[p];
            if(choice == null) continue;
            int rate = Math.Min(100, rates[p] + failBonus);
            if(rate <= 0) continue;
            Choice old;
            if(!stops.TryGetValue(rate, out old) || choice.Points < old.Points ||
                (choice.Points == old.Points && choice.Count < old.Count)) stops[rate] = choice;
        }
        var result = new List<int[]>();
        foreach(var best in stops.Values) {
            var selected = new List<int>();
            for(var node = best; node != null && node.Index >= 0; node = node.Previous) selected.Add(node.Index);
            selected.Reverse();
            result.Add(selected.ToArray());
        }
        return result.ToArray();
    }

    void Apply(int step) {
        ClearMaterials.Invoke(window, null);
        // Lower-point items first: native selection stops accepting at 100%.
        if(plans != null && step >= 0 && step < plans.Length)
            foreach(int i in plans[step]) AddMaterial.Invoke(window, new object[]{candidates[i]});
        foreach(var refresh in Refresh) refresh.Invoke(window, null);
        slider.value = plans.Length == 0 ? 0 : plans.Length == 1 ? 1 : (float)step / (plans.Length - 1);
        UpdateLabel();
    }

    void UpdateLabel() {
        int actual = window.GetMaterialEquipSlotIndexArray() == null ? 0 : Math.Min(100, (int)Rate.Invoke(window, null));
        label.text = "Success: " + actual + "%";
        if(chances != null && chances.Length > 1) {
            int index = Array.IndexOf(chances, actual);
            slider.value = index < 0 ? 0 : (float)index / (chances.Length - 1);
        }
    }

    void OnSliderChanged() {
        if(updating || plans == null || plans.Length < 2) return;
        updating = true;
        try { Apply(Mathf.RoundToInt(slider.value * (plans.Length - 1))); } finally { updating = false; }
    }

    void EnsureSlider() {
        if(root != null) return;
        var group = window.Label_UpperMsg.transform.parent;
        // Move complete prefab groups so labels, slots, material clipping,
        // stat scrolling and the arrow retain their native relative layout.
        // This runs only once per pooled window, avoiding drift on reopen.
        foreach(var name in new[]{"WidgetAwakeItems", "StatChange", "Sprite_Arrow", "Label_UpperMsg", "Sprite_Bg"}) {
            var section = group.Find(name);
            if(section == null) throw new InvalidOperationException("Missing Forge layout section: " + name);
            section.localPosition += new Vector3(0, 55, 0);
        }
        // A discount banner normally occupies the space above the instruction.
        // When present, fit it alongside the instruction in the raised row.
        if(window.EventRoot != null && window.EventRoot.gameObject.activeSelf) {
            var position = window.Label_UpperMsg.transform.localPosition;
            window.Label_UpperMsg.transform.localPosition = new Vector3(-240, position.y, position.z);
            window.Label_UpperMsg.width = 500;
            window.Label_UpperMsg.fontSize = 20;
            window.EventRoot.localPosition = new Vector3(245, position.y, window.EventRoot.localPosition.z);
            window.EventRoot.localScale *= .5f;
        }
        root = new GameObject("SprkForgeAwakenChance");
        root.transform.SetParent(group, false);
        root.transform.localPosition = new Vector3(-16, -332, 0);
        var host = new GameObject("SprkForgeSliderSource");
        host.SetActive(false);
        try {
            var source = Instantiate(GameOptionSliderComponent.PrefabAuto.Value.LoadGo, host.transform, false);
            slider = source.GetComponent<GameOptionSliderComponent>().Slider;
            slider.transform.SetParent(root.transform, false);
        } finally { Destroy(host); }
        foreach(var assigner in root.GetComponentsInChildren<NGame2.NUI.NAtlasAssignComponent.AtlasAssignerBase>(true)) assigner.Assign();
        slider.name = "SprkSuccessRateSlider";
        slider.onChange.Clear();
        slider.numberOfSteps = 0;
        EventDelegate.Add(slider.onChange, OnSliderChanged);
        int depth = window.Label_UpperMsg.depth + 1, lowest = int.MaxValue;
        var widgets = slider.GetComponentsInChildren<UIWidget>(true);
        foreach(var widget in widgets) lowest = Math.Min(lowest, widget.depth);
        foreach(var widget in widgets) {
            widget.leftAnchor.target = null; widget.rightAnchor.target = null;
            widget.topAnchor.target = null; widget.bottomAnchor.target = null;
            widget.depth = depth + widget.depth - lowest;
        }
        // Match the complete Gold/button row, reserving a fixed readout at its
        // right edge. Resize track widgets rather than stretching the thumb or
        // increasing the height of this deliberately narrow row.
        var gold = group.Find("WidgetGold").GetComponent<UIWidget>();
        var button = window.Button_Awaken.GetComponent<UIWidget>();
        float left = root.transform.InverseTransformPoint(gold.worldCorners[0]).x;
        float right = root.transform.InverseTransformPoint(button.worldCorners[2]).x;
        const float scale = .5f, labelWidth = 180, labelGap = 24;
        float thumbWidth = slider.thumb.GetComponent<UIWidget>().width * scale;
        // The Settings prefab leaves the optional background reference empty.
        // Bind the lifted track explicitly before sizing or moving its thumb.
        slider.backgroundWidget = slider.transform.Find("GaugeBg").GetComponent<UIWidget>();
        slider.foregroundWidget = slider.transform.Find("Gauge").GetComponent<UIWidget>();
        int extraBorder = slider.backgroundWidget.width - slider.foregroundWidget.width;
        int trackWidth = Mathf.FloorToInt((right - left - labelWidth - labelGap - thumbWidth) / scale);
        slider.foregroundWidget.width = trackWidth;
        slider.backgroundWidget.width = trackWidth + extraBorder;
        slider.GetComponent<UIWidget>().width = trackWidth;
        slider.transform.localScale = new Vector3(scale, scale, 1);
        slider.transform.localPosition = new Vector3(left + (thumbWidth + trackWidth * scale) * .5f, 0, 0);
        foreach(var widget in widgets) NGUITools.UpdateWidgetCollider(widget.gameObject);
        slider.ForceUpdate();
        var go = Instantiate(window.Label_UpperMsg.gameObject, root.transform, false);
        label = go.GetComponent<UILabel>();
        label.name = "SprkSuccessRateLabel";
        label.leftAnchor.target = null; label.rightAnchor.target = null;
        label.topAnchor.target = null; label.bottomAnchor.target = null;
        label.pivot = UIWidget.Pivot.Right;
        label.alignment = NGUIText.Alignment.Right;
        label.fontSize = 24;
        label.width = (int)labelWidth;
        label.height = 30;
        label.overflowMethod = UILabel.Overflow.ShrinkContent;
        label.depth = depth;
        go.transform.localScale = Vector3.one;
        go.transform.localPosition = new Vector3(right, 0, 0);
        go.SetActive(true);
        NGUITools.SetLayer(root, window.gameObject.layer);
        NGUITools.MarkParentAsChanged(root);
    }
}
