using System;
using System.Collections.Generic;
using System.Linq;
using NGame2.NUI.NComponent2;
using NGame2.NUI.NWindow;
using NGame2.NUI.NManager;
using NGame2.NUI.NHelper;
using NShared;
using NVespa;
using UnityEngine;

public sealed class SprkHeroRuneFilter : MonoBehaviour, RuneSelectItemFilterOptionList.IListener {
    HeroRuneListPanel panel;
    readonly List<Enum> selected = new List<Enum>();
    UIButton button;
    UILabel buttonLabel;
    RuneSelectItemFilterOptionList popup;
    bool opened;

    public static void Open(HeroRuneListPanel panel) {
        var state = panel.GetComponent<SprkHeroRuneFilter>();
        if(state == null) state = panel.gameObject.AddComponent<SprkHeroRuneFilter>();
        state.panel = panel;
        state.selected.Clear();
        state.opened = true;
        state.CreateButton();
        state.UpdateLabel();
    }
    void CreateButton() {
        if(button != null) return;
        var original = panel.Label_SelectedCategory.GetComponentInParent<UIButton>(true);
        var clone = Instantiate(original.gameObject,original.transform.parent,false);
        clone.name = "SprkRuneSortOption";
        button = clone.GetComponent<UIButton>();
        button.onClick.Clear();
        EventDelegate.Add(button.onClick, OnClick);
        // Keep the native dropdown, artwork and collider while sharing the header.
        Layout(original.gameObject,432,-120,false);
        Layout(clone,220,226,true);
        buttonLabel = clone.GetComponentInChildren<UILabel>(true);
        buttonLabel.fontSize = 26;
        buttonLabel.text = "Sort Option";
        clone.SetActive(true);
    }
    static void Layout(GameObject go,int width,float x,bool sort) {
        foreach(var widget in go.GetComponentsInChildren<UIWidget>(true)) {
            widget.leftAnchor.target=null; widget.rightAnchor.target=null;
            widget.topAnchor.target=null; widget.bottomAnchor.target=null;
            if(widget.transform==go.transform) widget.width=width;
            else if(widget is UILabel) {widget.width=width-(sort?16:84);widget.transform.localPosition=new Vector3(sort?0:28,0,0);}
            else if(widget.name=="Sprite_Bg") widget.width=width+16;
            else if(widget.name=="Sprite_Arrow") {
                if(sort) widget.gameObject.SetActive(false);
                else widget.transform.localPosition=new Vector3(-width*.5f+60,0,0);
            }
        }
        go.transform.localPosition=new Vector3(x,0,0);
        NGUITools.AddWidgetCollider(go);
    }
    void UpdateLabel() {
        if(buttonLabel!=null)buttonLabel.text=selected.Count==0?"Sort Option":"Sort Option ("+selected.Count+")";
    }
    public void OnClick() {
        if(!opened || panel==null || !panel.gameObject.activeInHierarchy)return;
        var manager=NVespa.NSingleton.SceneAutoBehaviour<MyHeroManagement>.instance;
        popup=manager.GetInstance<RuneSelectItemFilterOptionList>();
        if(popup==null) {
            manager.AddInstance<RuneSelectItemFilterOptionList>(RuneSelectItemFilterOptionList.PrefabPath);
            popup=manager.GetInstance<RuneSelectItemFilterOptionList>();
        }
        if(popup!=null)popup.Open(this,new List<Enum>(selected),2);
    }
    public static void Close(HeroRuneListPanel panel) {
        var state=panel.GetComponent<SprkHeroRuneFilter>();
        if(state==null)return;
        state.opened=false;
        state.selected.Clear();state.UpdateLabel();
        if(state.popup!=null && state.popup.IsOpen())state.popup.Close();
        state.popup=null;
    }
    public static bool Matches(HeroRuneListPanel panel, IItemInfo item) {
        var state = panel.GetComponent<SprkHeroRuneFilter>();
        if(state == null || state.selected.Count == 0) return true;
        var data = GameRoot.Instance.ItemDataContainer.GetData(item.ItemIndex);
        if(data == null) return false;
        var equip=item as EquipItemInfo;
        return ContainsAll(RuneHelper.GetRuneStat(data,equip==null?null:equip.PunishmentRuneOption), state.selected);
    }
    public static bool ContainsAll(Dictionary<Enum,long> stats, List<Enum> filters) {
        return filters == null || filters.All(stat => stats != null && stats.ContainsKey(stat));
    }
    public void OnClickConfirm(List<Enum> filters, Action complete) {
        if(!opened) {if(complete!=null)complete();return;}
        selected.Clear();
        if(filters != null) selected.AddRange(filters);
        UpdateLabel();
        if(complete != null) complete();
        if(panel != null && panel.gameObject.activeInHierarchy) { panel.Refresh(); panel.ResetScroll(); }
    }
    public static string Probe() {
        NShared.NJit.TableRoot.Instance.SetPatchDownLoadComplete(true);
        var prefab = VespaResource.Load<GameObject>(HeroRunehengeWindow.PrefabPath);
        var host = new GameObject("RuneFilterProbe"); host.SetActive(false);
        var go = Instantiate(prefab,host.transform,false);
        var panel = go.GetComponent<HeroRunehengeWindow>().RuneListPanel;
        var lines = new List<string>();
        Open(panel);
        var state=panel.GetComponent<SprkHeroRuneFilter>();
        var category=panel.Label_SelectedCategory.GetComponentInParent<UIButton>(true);
        Check(state.button!=null && state.button.onClick.Count==1,"native button binding");
        Check(category.onClick.Count>0,"native category dropdown preserved");
        var left=category.GetComponent<UIWidget>();var right=state.button.GetComponent<UIWidget>();
        Check(category.transform.localPosition.x+left.width*.5f < state.button.transform.localPosition.x-right.width*.5f,"separate header hit areas");
        Open(panel);
        Check(panel.GetComponentsInChildren<UIButton>(true).Count(b=>b.name=="SprkRuneSortOption")==1,"reopen does not duplicate button");
        var ticketHost=Instantiate(VespaResource.Load<GameObject>(RuneSelectItemListWindow.PrefabPath),host.transform,false);
        var ticket=ticketHost.GetComponent<RuneSelectItemListWindow>();
        var ticketMatch=typeof(RuneSelectItemListWindow).GetMethod("IsSatisfyCondition",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance);
        var selector=typeof(HeroRuneListPanel).GetMethod("Selector",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance);
        var runes=GameRoot.Instance.ItemDataContainer.GetAllData().Where(d=>d.Type==ItemType.Rune).ToArray();
        Check(runes.Length>0,"native rune tables loaded");
        int comparisons=0;
        var choices=runes.SelectMany(d=>d.GetSubData<RuneItemData>().GetStatTypeAndValue().Keys).Distinct().ToArray();
        foreach(var stat in choices) {
            state.selected.Clear();state.selected.Add(stat);
            foreach(var data in runes) {
                var item=new ItemInfo{ItemIndex=data.Index};
                bool expected=(bool)ticketMatch.Invoke(ticket,new object[]{state.selected,data});
                Check(Matches(panel,item)==expected,"ticket stat matching "+data.Index+" "+stat);
                comparisons++;
            }
        }
        var dual=runes.First(d=>d.GetSubData<RuneItemData>().GetStatTypeAndValue().Count==2);
        state.selected.Clear();state.selected.AddRange(dual.GetSubData<RuneItemData>().GetStatTypeAndValue().Keys);
        foreach(var data in runes) {
            Check(Matches(panel,new ItemInfo{ItemIndex=data.Index})==(bool)ticketMatch.Invoke(ticket,new object[]{state.selected,data}),"two-stat AND semantics");
        }
        var optionPrefab=VespaResource.Load<GameObject>(RuneSelectItemFilterOptionList.PrefabPath);
        var option=Instantiate(optionPrefab,host.transform,false).GetComponent<RuneSelectItemFilterOptionList>();
        var init=typeof(RuneSelectItemFilterOptionList).GetMethod("Init",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance);
        init.Invoke(option,new object[]{state,new List<Enum>(state.selected),2});
        Check(option.GridOptionList.GetComponentsInChildren<RuneSelectItemFilterMenuComponent>(true).Length>0,"native stat picker populated");
        var sample=dual;
        Check((bool)selector.Invoke(panel,new object[]{new ItemInfo{ItemIndex=sample.Index}}),"matching rune accepted by native selector");
        var providerField=typeof(HeroRuneListPanel).GetField("_dataProvider",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance);
        var wrongSlot=RuneSlotType.None;
        providerField.SetValue(panel,new ProbeSlots(wrongSlot));
        Check(!(bool)selector.Invoke(panel,new object[]{new ItemInfo{ItemIndex=sample.Index}}),"slot restrictions preserved");
        providerField.SetValue(panel,null);
        option.OnClickResetFilter();option.OnClickConfirm();
        Check(state.selected.Count==0 && state.buttonLabel.text=="Sort Option","native Deselect and OK restore full list");
        foreach(var data in runes)Check(Matches(panel,new ItemInfo{ItemIndex=data.Index}),"unfiltered restore");
        var crafted=GameRoot.Instance.ItemDataContainer.GetAllData().First(d=>d.Type==ItemType.PunishmentRune);
        var roll=GameRoot.Instance.PunishmentRuneOptionDataContainer.GetAllData().First(d=>d.StatTypes!=null && d.StatTypes.Length>0);
        var equip=new EquipItemInfo{ItemIndex=crafted.Index,PunishmentRuneOption=new PunishmentRuneOptionInfo{MainOptionIndex=roll.Index,MainOptionValue=100}};
        var rolledStats=RuneHelper.GetRuneStat(crafted,equip.PunishmentRuneOption);
        Check(rolledStats.Count>0,"crafted main stat fixture");
        state.selected.Add(rolledStats.Keys.First());
        Check(Matches(panel,equip),"crafted rolled stat matches");
        state.selected.Clear();state.selected.Add(choices.First(s=>!rolledStats.ContainsKey(s)));
        Check(!Matches(panel,equip),"crafted nonmatching stat hidden");
        Close(panel);
        Check(!state.opened && state.selected.Count==0,"close clears filter state");
        state.OnClickConfirm(new List<Enum>{choices[0]},delegate{});
        Check(state.selected.Count==0,"closed page rejects stale picker callback");
        Open(panel);
        Check(state.opened && state.selected.Count==0,"reopen starts unfiltered");
        // Exercise the real management-owned popup, including its stacking and
        // close lifecycle, without opening a hero or touching player inventory.
        var manager=NVespa.NSingleton.SceneAutoBehaviour<MyHeroManagement>.instance;
        var preview=manager.GetInstance<HeroRunehengeWindow>();
        ((BaseWindow)preview).Open();
        var previewList=preview.RuneListPanel;
        previewList.gameObject.SetActive(true);
        Open(previewList);
        previewList.Label_SelectedCategory.text="Normal Rune";
        var previewState=previewList.GetComponent<SprkHeroRuneFilter>();
        previewState.OnClick();
        Check(previewState.popup!=null && previewState.popup.IsOpen(),"management-owned stat picker opens");
        Close(previewList);
        Check(previewState.popup==null,"owned stat picker closed with page");
        Open(previewList);
        // Render the native header on a dedicated camera above Login's loading
        // overlay; no login or account is needed for this visual fixture.
        var visual=new GameObject("RuneFilterVisual");visual.layer=30;
        visual.transform.position=new Vector3(10000,0,0);
        visual.AddComponent<UIPanel>().depth=5000;
        var cameraObject=new GameObject("RuneFilterVisualCamera");
        var camera=cameraObject.AddComponent<Camera>();camera.orthographic=true;
        camera.orthographicSize=240;camera.nearClipPlane=1;camera.farClipPlane=2000;
        camera.cullingMask=1<<30;camera.clearFlags=CameraClearFlags.SolidColor;
        camera.backgroundColor=new Color(.15f,.12f,.10f,1);camera.depth=100000;
        camera.transform.position=new Vector3(10000,0,-1000);
        panel.Label_SelectedCategory.text="Normal Rune";
        var header=Instantiate(category.transform.parent.gameObject,visual.transform,false);
        header.transform.localPosition=Vector3.zero;header.transform.localScale=Vector3.one;
        NGUITools.SetLayer(header,30);
        lines.Add("PASS: native header button/dropdown layout and reuse; "+runes.Length+" runes, "+choices.Length+" stat choices, "+comparisons+" ticket comparisons; two-stat matching, native picker Deselect/OK, slot eligibility, crafted rolled main stats and close/reopen lifecycle.");
        // Leave the header visible for the external screenshot check. The
        // isolated test process owns and discards this preview.
        Destroy(host);
        return String.Join("\n",lines.ToArray());
    }
    sealed class ProbeSlots : HeroRuneListPanel.IDataProvider {
        readonly RuneSlotType slot;
        public ProbeSlots(RuneSlotType value){slot=value;}
        public List<RuneSlotType> GetActivedRuneSlotTypes(){return new List<RuneSlotType>{slot};}
    }
    static void Check(bool condition,string message){if(!condition)throw new Exception("Rune filter: "+message);}
}
