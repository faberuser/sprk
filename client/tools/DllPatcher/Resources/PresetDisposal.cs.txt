using System.Linq;
using NGame2.NAccount;
using NVespa.NSingleton;

public static class SprkPresetDisposal
{
    public static bool Clear(int[] slots, string[] uids, int[] pages, int removedSlot)
    {
        bool changed = false;
        if (slots == null || removedSlot <= 0) return false;
        for (int i = 0; i < slots.Length; i++)
        {
            if (slots[i] != removedSlot) continue;
            slots[i] = 0;
            if (uids != null && i < uids.Length) uids[i] = "";
            if (pages != null && i < pages.Length) pages[i] = 0;
            changed = true;
        }
        return changed;
    }

    // Called only after native equipment removal has succeeded, before return.
    public static bool AfterRemove(bool removed, int slot)
    {
        if (!removed || slot <= 0 || !GlobalAutoBehaviour<AccountManager>.isValidInstance) return removed;
        var account = GlobalAutoBehaviour<AccountManager>.instance;
        var gear = account.EquipPresetSotrageManger;
        if (gear != null)
            foreach (var preset in gear.ToArray())
                if (preset != null && Clear(preset.EquipItemSlotIndex, preset.EquipItemUid, null, slot))
                    gear.Apply(preset);
        var heroes = account.HeroPresetStorageManager;
        if (heroes != null)
            foreach (var storage in heroes.ToArray())
            {
                bool changed = false;
                if (storage == null || storage.HeroPresetInfos == null) continue;
                foreach (var preset in storage.HeroPresetInfos)
                    if (preset != null) changed |= Clear(preset.EquipItemSlotIndex, preset.EquipItemUid, preset.EquipItemRunePage, slot);
                if (changed) heroes.Apply(storage);
            }
        return removed;
    }
}
