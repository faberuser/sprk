// Adds "Upgrade All" to the Skills tab's skill-book row. One native extend
// request (with ExtendAll) raises every learned skill's II-IV tier as far as
// its own slot book allows; the server returns every book result and tier.
public sealed class SprkSkillUpgradeAll : MonoBehaviour {
    // SkillExtend value that marks an upgrade-all request for the encoder.
    public const int ExtendAllMarker = -1;
    // Layout, in the panel root's units: gap between counters, gap before the
    // button, margin inside the panel frame, button height relative to a
    // counter and the largest enlargement of the native row.
    const string ButtonText = "Upgrade";
    const float ItemGap = 2f;
    const float Gap = 12f;
    const float Margin = 20f;
    // Button height relative to a native counter; fixed, not scaled with them.
    const float ButtonHeight = .88f;
    const float LabelSize = .42f;
    const float MaxScale = 1.8f;
    // Count box width relative to native (five digits still fit).
    const float CountBox = .72f;
    // Count digits: 0 = centred beside the icon, 1 = centred on the box.
    const float LabelShift = .5f;
    // Nudge the button down about one screen pixel at 1280x720.
    const float ButtonDrop = 1.5f;
    static readonly FieldInfo IsMine = typeof(HeroSkillListPanel).GetField("_isMine", BindingFlags.Instance | BindingFlags.NonPublic);

    sealed class Extras {
        public ItemResultInfo[] Items;
        public readonly List<KeyValuePair<int, int>> Extends = new List<KeyValuePair<int, int>>();
    }
    static readonly ConditionalWeakTable<NShared.ExtendHeroSkill.Response, Extras> Captured =
        new ConditionalWeakTable<NShared.ExtendHeroSkill.Response, Extras>();

    // Tickets: books this step takes from selection tickets (converted before the request).
    struct Step { public int Slot; public string Name; public int From; public int To; public int Cost; public int Tickets; public string Book; }

    HeroSkillListPanel panel;
    UIButton button;
    GameObject buttonRoot;
    Vector3 gridScale;
    Vector3 gridPosition;
    float cellWidth;
    bool laidOut;
    bool requesting;

    // Called after native HeroSkillListPanel.Open.
    public static void Attach(HeroSkillListPanel panel) {
        var state = panel.GetComponent<SprkSkillUpgradeAll>();
        if(state == null) state = panel.gameObject.AddComponent<SprkSkillUpgradeAll>();
        state.panel = panel;
        state.Refresh();
    }

    // Called after native book-count and skill-tier updates.
    public static void Changed(HeroSkillListPanel panel) {
        var state = panel.GetComponent<SprkSkillUpgradeAll>();
        if(state != null) state.Refresh();
        // Counters include convertible tickets; ticket-only changes skip the native book path.
        if(panel.SkillBookGrid != null) SprkSkillBookTickets.RefreshCounters(panel.SkillBookGrid.GetComponentsInChildren<UpgradeBookItemIconComponent>(true));
    }

    // Called by JM_NShared_ExtendHeroSkill_Request.EncodeJsonObject.
    public static void Encode(Dictionary<string, object> json, NShared.ExtendHeroSkill.Request request) {
        if(json != null && request != null && request.SkillExtend == ExtendAllMarker) json["ExtendAll"] = "1";
    }

    // Called at the start of JM_NShared_ExtendHeroSkill_Response.Parse(IDictionary, Response).
    public static void Capture(IDictionary json, NShared.ExtendHeroSkill.Response response) {
        if(json == null || response == null || !json.Contains("SkillExtends")) return;
        var extras = new Extras();
        var items = json["ItemResults"] as IList;
        if(items != null) {
            var list = new List<ItemResultInfo>();
            foreach(var item in items) {
                var dict = item as IDictionary;
                if(dict != null) list.Add(JM_NShared_ItemResultInfo.Parse(dict));
            }
            extras.Items = list.ToArray();
        }
        var extends = json["SkillExtends"] as IList;
        if(extends != null) foreach(var value in extends) {
            var dict = value as IDictionary;
            if(dict != null && dict.Contains("SkillSlotIndex") && dict.Contains("SkillExtend"))
                extras.Extends.Add(new KeyValuePair<int, int>(Convert.ToInt32(dict["SkillSlotIndex"]), Convert.ToInt32(dict["SkillExtend"])));
        }
        Captured.Remove(response);
        Captured.Add(response, extras);
    }

    List<Step> Plan() {
        SprkSkillBookTickets.Pool tickets;
        return Plan(out tickets);
    }

    // Books first, then owned selection tickets from one shared pool (slot order),
    // so a class or general ticket is never counted for two skills.
    List<Step> Plan(out SprkSkillBookTickets.Pool tickets) {
        var steps = new List<Step>();
        tickets = new SprkSkillBookTickets.Pool();
        var hero = panel == null ? null : panel.CachedHeroInfo;
        if(hero == null) return steps;
        for(int slot = 1; slot <= 4; slot++) {
            var detail = Helper.GetSkillDetailInfo(hero, slot);
            if(detail.ExtendInfos == null || detail.BasicInfo.IsLocked || detail.BasicInfo.Level <= 0 || detail.BookInfo.BookItemData == null) continue;
            int books = detail.BookInfo.CurItemCount, book = detail.BookInfo.BookItemData.Index;
            int from = hero.GetSkillExtend(slot), to = from, cost = 0;
            long have = (long)books + tickets.Available(book);
            while(to < detail.ExtendInfos.Length) {
                int need = detail.ExtendInfos[to].ExtendItemCount;
                if(need <= 0 || cost + need > have) break;
                cost += need;
                to++;
            }
            if(to <= from) continue;
            int fromTickets = tickets.Take(book, System.Math.Max(0, cost - books));
            steps.Add(new Step {
                Slot = slot, From = from, To = to, Cost = cost, Tickets = fromTickets,
                Name = detail.BasicInfo.TableData == null ? "Skill " + slot : NGame2.NUI.NDescription.SkillDescContainer.GetSkillName(hero, detail.BasicInfo.TableData),
                Book = LocalizationManager.GetStringDirect(detail.BookInfo.BookItemData.Name)
            });
        }
        return steps;
    }

    void Refresh() {
        bool mine = panel != null && (bool)IsMine.GetValue(panel) && panel.CachedHeroInfo != null;
        if(mine) EnsureButton();
        if(buttonRoot != null) buttonRoot.SetActive(mine);
        Layout(mine && buttonRoot != null);
        if(button != null && mine) UIHelper.EnableButton(button, !requesting && Plan().Count > 0, true);
    }

    // Clone the native blue "All Heroes" button from the same HeroList window.
    void EnsureButton() {
        if(buttonRoot != null) return;
        Transform window = panel.transform;
        while(window != null && !window.name.StartsWith("HeroList")) window = window.parent;
        var source = window == null ? null : FindChild(window, "Button_AllHero");
        if(source == null) return;
        var parent = panel.SkillBookGrid.transform.parent;
        buttonRoot = Instantiate(source.gameObject, parent, false);
        buttonRoot.name = "SprkSkillUpgradeAll";
        foreach(var behaviour in buttonRoot.GetComponentsInChildren<MonoBehaviour>(true))
            if(behaviour.GetType().Name == "UILocalizeEx") DestroyImmediate(behaviour);
        var label = buttonRoot.GetComponentInChildren<UILabel>(true);
        if(label != null) label.text = ButtonText;
        button = buttonRoot.GetComponentInChildren<UIButton>(true);
        button.onClick.Clear();
        EventDelegate.Add(button.onClick, OnClick);
        int depth = 0;
        foreach(var widget in panel.SkillBookGrid.GetComponentsInChildren<UIWidget>(true)) depth = System.Math.Max(depth, widget.depth);
        foreach(var widget in buttonRoot.GetComponentsInChildren<UIWidget>(true)) {
            widget.leftAnchor.target = null; widget.rightAnchor.target = null;
            widget.topAnchor.target = null; widget.bottomAnchor.target = null;
            widget.depth += depth + 1;
        }
        NGUITools.SetLayer(buttonRoot, panel.gameObject.layer);
        NGUITools.MarkParentAsChanged(buttonRoot);
    }

    // Native geometry of a narrowed book counter, restored before the pooled
    // component is released so other windows keep the native layout.
    sealed class NarrowedCounter : MonoBehaviour {
        public int RootWidth, BgWidth;
        public Vector3 Bg, Icon, Label, Update;
        public UIWidget.Pivot LabelPivot;
    }

    // Called at the start of native HeroSkillListPanel.Reset.
    public static void Release(HeroSkillListPanel panel) {
        if(panel == null || panel.SkillBookGrid == null) return;
        foreach(var item in panel.SkillBookGrid.GetComponentsInChildren<UpgradeBookItemIconComponent>(true)) Restore(item.transform);
    }

    // Narrow the count box to what five digits need, keeping the icon, the
    // digits and the box height native; the left edge stays fixed.
    static void Narrow(Transform item) {
        var root = item.GetComponent<UIWidget>();
        var bgT = FindChild(item, "Sprite_Bg");
        var bg = bgT == null ? null : bgT.GetComponent<UIWidget>();
        var labelT = FindChild(item, "Label_Count");
        var label = labelT == null ? null : labelT.GetComponent<UILabel>();
        var icon = FindChild(item, "Texture_ItemIcon");
        var update = FindChild(item, "Label_UpdateCount");
        if(root == null || bg == null || label == null || icon == null) return;
        var mark = item.GetComponent<NarrowedCounter>();
        if(mark == null) {
            mark = item.gameObject.AddComponent<NarrowedCounter>();
            mark.RootWidth = root.width; mark.BgWidth = bg.width;
            mark.Bg = bgT.localPosition; mark.Icon = icon.localPosition; mark.Label = labelT.localPosition;
            mark.Update = update == null ? Vector3.zero : update.localPosition; mark.LabelPivot = label.pivot;
        } else Restore(item);
        var before = NGUIMath.CalculateRelativeWidgetBounds(item, bgT, true);
        var iconBounds = NGUIMath.CalculateRelativeWidgetBounds(item, icon, true);
        bg.width = Mathf.RoundToInt(mark.BgWidth * CountBox);
        var after = NGUIMath.CalculateRelativeWidgetBounds(item, bgT, true);
        bgT.localPosition = mark.Bg + new Vector3(before.min.x - after.min.x, 0, 0);
        float right = before.min.x + after.size.x, shift = right - before.max.x;
        label.pivot = UIWidget.Pivot.Center;
        // Shift the digits left, halfway between the space beside the icon and
        // the box centre; the full box centre would touch the icon.
        float besideIcon = (Mathf.Max(iconBounds.max.x, before.min.x) + right) * .5f, boxCentre = (before.min.x + right) * .5f;
        labelT.localPosition = new Vector3(Mathf.Lerp(besideIcon, boxCentre, LabelShift), mark.Label.y, mark.Label.z);
        if(update != null) update.localPosition = mark.Update + new Vector3(shift, 0, 0);
        var rootBefore = NGUIMath.CalculateRelativeWidgetBounds(item.parent, item, false);
        root.width = Mathf.Max(1, Mathf.RoundToInt(mark.RootWidth + shift));
        var rootAfter = NGUIMath.CalculateRelativeWidgetBounds(item.parent, item, false);
        // Keep the counter's left edge in place when its own pivot is centred.
        bgT.localPosition += new Vector3(rootBefore.min.x - rootAfter.min.x, 0, 0);
        labelT.localPosition += new Vector3(rootBefore.min.x - rootAfter.min.x, 0, 0);
        icon.localPosition += new Vector3(rootBefore.min.x - rootAfter.min.x, 0, 0);
        if(update != null) update.localPosition += new Vector3(rootBefore.min.x - rootAfter.min.x, 0, 0);
        NGUITools.UpdateWidgetCollider(item.gameObject);
    }

    static void Restore(Transform item) {
        var mark = item.GetComponent<NarrowedCounter>();
        if(mark == null) return;
        var root = item.GetComponent<UIWidget>();
        var bgT = FindChild(item, "Sprite_Bg");
        var labelT = FindChild(item, "Label_Count");
        var icon = FindChild(item, "Texture_ItemIcon");
        var update = FindChild(item, "Label_UpdateCount");
        if(root != null) root.width = mark.RootWidth;
        if(bgT != null) {bgT.GetComponent<UIWidget>().width = mark.BgWidth; bgT.localPosition = mark.Bg;}
        if(labelT != null) {labelT.GetComponent<UILabel>().pivot = mark.LabelPivot; labelT.localPosition = mark.Label;}
        if(icon != null) icon.localPosition = mark.Icon;
        if(update != null) update.localPosition = mark.Update;
        NGUITools.UpdateWidgetCollider(item.gameObject);
    }

    static Transform FindChild(Transform root, string name) {
        foreach(var child in root.GetComponentsInChildren<Transform>(true)) if(child.name == name) return child;
        return null;
    }

    // Lays the four book counters out with a tight, even gap, then scales the
    // counters and button together to fill the panel width and centres the
    // whole group in the header row. Restored when the button is hidden.
    void Layout(bool show) {
        var grid = panel.SkillBookGrid.transform;
        if(!laidOut) {gridScale = grid.localScale; gridPosition = grid.localPosition; laidOut = true;
            if(panel.SkillBookGrid.Grid != null) cellWidth = panel.SkillBookGrid.Grid.cellWidth;}
        grid.localScale = gridScale;
        grid.localPosition = gridPosition;
        if(panel.SkillBookGrid.Grid != null && cellWidth > 0) panel.SkillBookGrid.Grid.cellWidth = cellWidth;
        if(!show) return;
        var parent = grid.parent;
        var items = new List<Transform>();
        foreach(var item in panel.SkillBookGrid.GetComponentsInChildren<UpgradeBookItemIconComponent>(false)) items.Add(item.transform);
        if(items.Count == 0) return;
        items.Sort((a, b) => a.localPosition.x.CompareTo(b.localPosition.x));
        foreach(var item in items) Narrow(item);
        // Even spacing through the native grid itself (it repositions its
        // children later using cellWidth). Hidden widgets, such as the "+N"
        // count tween, are excluded from every measurement.
        var nativeGrid = panel.SkillBookGrid.Grid;
        // Space by the visible count box, not the wider component container.
        // The visible extent is the box plus its icon, which overhangs the box.
        var box = FindChild(items[0], "Sprite_Bg");
        var bookIcon = FindChild(items[0], "Texture_ItemIcon");
        var itemBounds = NGUIMath.CalculateRelativeWidgetBounds(items[0].parent, box != null ? box : items[0], false);
        if(box != null && bookIcon != null) itemBounds.Encapsulate(NGUIMath.CalculateRelativeWidgetBounds(items[0].parent, bookIcon, false));
        if(itemBounds.size.x <= 0 || nativeGrid == null) return;
        nativeGrid.cellWidth = itemBounds.size.x + ItemGap;
        nativeGrid.Reposition();
        var row = NGUIMath.CalculateRelativeWidgetBounds(parent, grid, false);
        if(row.size.x <= 0 || row.size.y <= 0) return;
        // Size the sliced native button sprite directly: taller than a counter
        // with a larger label, as wide as its text plus padding.
        float height = row.size.y * ButtonHeight;
        var sprite = button.GetComponent<UIWidget>();
        var label = buttonRoot.GetComponentInChildren<UILabel>(true);
        float width = height * 2f;
        if(label != null) {
            label.overflowMethod = UILabel.Overflow.ResizeFreely;
            label.pivot = UIWidget.Pivot.Center;
            label.fontSize = Mathf.RoundToInt(height * LabelSize);
            label.text = ButtonText;
            label.transform.localScale = Vector3.one;
            label.transform.localPosition = new Vector3(0, height * .04f, 0);
            width = Mathf.Max(width, label.printedSize.x + height * .9f);
        }
        foreach(var widget in new[]{buttonRoot.GetComponent<UIWidget>(), sprite})
            if(widget != null) {widget.pivot = UIWidget.Pivot.Center; widget.width = Mathf.RoundToInt(width); widget.height = Mathf.RoundToInt(height);}
        if(sprite != null) {
            sprite.transform.localScale = Vector3.one;
            sprite.transform.localPosition = Vector3.zero;
            NGUITools.UpdateWidgetCollider(sprite.gameObject);
        }
        // Available width: the panel frame (or the native row when absent).
        var frame = FindChild(panel.transform, "Sprite_MainFrameSmall");
        Bounds area = row;
        if(frame != null && frame.GetComponent<UIWidget>() != null) area = NGUIMath.CalculateRelativeWidgetBounds(parent, frame, false);
        float available = area.size.x - 2 * Margin;
        // The button keeps its own size; the counters take the remaining width.
        float scale = Mathf.Min(MaxScale, (available - Gap - width) / row.size.x);
        float total = row.size.x * scale + Gap + width;
        float left = area.center.x - total * .5f;
        float centerY = row.center.y;
        grid.localScale = new Vector3(gridScale.x * scale, gridScale.y * scale, gridScale.z);
        grid.localPosition = new Vector3(
            left - (row.min.x - gridPosition.x) * scale,
            centerY - (row.center.y - gridPosition.y) * scale, gridPosition.z);
        buttonRoot.transform.localScale = Vector3.one;
        buttonRoot.transform.localPosition = new Vector3(left + row.size.x * scale + Gap + width * .5f, centerY - ButtonDrop, 0);
    }

    public void OnClick() {
        if(requesting || panel == null || panel.CachedHeroInfo == null) return;
        SprkSkillBookTickets.Pool tickets;
        var plan = Plan(out tickets);
        if(plan.Count == 0) return;
        var text = new StringBuilder();
        foreach(var step in plan) {
            text.Append(step.Name).Append("  ").Append(Roman(step.From)).Append(" → ").Append(Roman(step.To))
                .Append("  (").Append(step.Book).Append(" ×").Append(step.Cost);
            if(step.Tickets > 0) text.Append(", ").Append(step.Tickets).Append(" from tickets");
            text.Append(")\n");
        }
        // Totals per ticket, as the inventory names them.
        var used = new Dictionary<int, int>();
        foreach(var c in tickets.Conversions) {int n; used.TryGetValue(c.Ticket, out n); used[c.Ticket] = n + c.Count;}
        if(used.Count > 0) {
            text.Append("Uses:");
            foreach(var pair in used) {
                var data = GameRoot.Instance.ItemDataContainer.GetData(pair.Key);
                text.Append("\n  ").Append(data == null ? pair.Key.ToString() : LocalizationManager.GetStringDirect(data.Name)).Append(" ×").Append(pair.Value);
            }
        }
        int hero = panel.CachedHeroInfo.HeroIndex;
        var conversions = tickets.Conversions;
        MessagePopup.Open(LocalizationManager.GetString(NShared.NLocalization.HeroSkill.LEARN_EXTEND_SKILL),
            "Upgrade all skills with Skill Books?\n" + text.ToString().TrimEnd(), () => RedeemThenSend(hero, conversions), null);
    }

    // Redeem the planned tickets first; the native upgrade-all request then sees enough books.
    void RedeemThenSend(int heroIndex, List<SprkSkillBookTickets.Conversion> conversions) {
        if(requesting) return;
        if(conversions.Count == 0) {Send(heroIndex); return;}
        requesting = true;
        Refresh();
        SprkSkillBookTickets.Redeem(conversions, ok => {
            requesting = false;
            if(this == null) return;
            // On failure the books already converted stay owned; the button replans from them.
            if(ok && panel != null && panel.CachedHeroInfo != null && panel.CachedHeroInfo.HeroIndex == heroIndex) Send(heroIndex);
            else Refresh();
        });
    }

    static string Roman(int extend) {
        // Native tiers: no extend = I, then II, III, IV.
        switch(extend) {case 0: return "I"; case 1: return "II"; case 2: return "III"; case 3: return "IV"; default: return (extend + 1).ToString();}
    }

    void Send(int heroIndex) {
        if(requesting) return;
        requesting = true;
        Refresh();
        var request = new NShared.ExtendHeroSkill.Request {HeroIndex = heroIndex, SkillIndex = 0, SkillExtend = ExtendAllMarker};
        SingletonUpdater<WebServiceManager>.instance.Requester.Request<NShared.ExtendHeroSkill.Request, NShared.ExtendHeroSkill.Response>(request, response => {
            requesting = false;
            try {
                if(response == null) return;
                if(response.BaseResult != BaseResultType.Success) {MessagePopup.OpenError(response.BaseResult);return;}
                if(response.Result != NShared.ExtendHeroSkill.ResultType.Success) {MessagePopup.OpenError(response.Result);return;}
                Extras extras;
                var account = GlobalAutoBehaviour<AccountManager>.instance;
                if(Captured.TryGetValue(response, out extras)) {
                    if(extras.Items != null) account.ItemManager.Apply(extras.Items);
                    foreach(var pair in extras.Extends) account.HeroManager.SetSkillExtend(heroIndex, pair.Key, pair.Value);
                } else if(response.ItemResult != null) {
                    account.ItemManager.Apply(response.ItemResult);
                }
            } finally {
                if(this != null) Refresh();
            }
        });
    }
}
