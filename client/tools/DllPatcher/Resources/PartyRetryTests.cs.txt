using System;
using System.Linq;

public static class PartyRetryTests
{
    static void Check(bool value, string message) { if (!value) throw new Exception(message); }
    public static void Run()
    {
        var allowed = new[] { 1, 10, 12, 29, 30, 33, 40, 41, 42, 45, 46 };
        for (int kind = 0; kind <= 48; kind++)
        {
            Check(SprkPartyRetryPolicy.Eligible(true, kind) == allowed.Contains(kind), "Mode " + kind);
            Check(!SprkPartyRetryPolicy.Eligible(false, kind), "Solo battle gained party exit " + kind);
        }
        foreach (int competitive in new[] { 4, 15, 19, 20, 26, 31, 32, 36, 43, 44 })
            Check(!SprkPartyRetryPolicy.Eligible(true, competitive), "Competitive mode allowed " + competitive);

        Check(SprkPartyRetryPolicy.Propose() == "sprk-retry:propose", "Propose wire format");
        Check(SprkPartyRetryPolicy.Answer(3, true) == "sprk-retry:yes:3" && SprkPartyRetryPolicy.Answer(3, false) == "sprk-retry:no:3", "Answer wire format");

        var vote = SprkPartyRetryPolicy.Parse("SprkRetry", "Vote|2|77|15|Ann|ie");
        Check(vote.Kind == SprkPartyRetryPolicy.Kind.Vote && vote.Vote == 2 && vote.Proposer == 77 && vote.Seconds == 15 && vote.Who == "Ann|ie", "Vote notice");
        var cancel = SprkPartyRetryPolicy.Parse("SprkRetry", "Cancel|2|Timeout|Bo, Cy");
        Check(cancel.Kind == SprkPartyRetryPolicy.Kind.Cancel && cancel.Reason == "Timeout" && cancel.Who == "Bo, Cy", "Cancel notice");
        var restart = SprkPartyRetryPolicy.Parse("SprkRetry", "Restart|2|{\"RaidIndex\":31,\"BattleServerAddress\":\"a|b\"}");
        Check(restart.Kind == SprkPartyRetryPolicy.Kind.Restart && restart.Start.Contains("a|b"), "Restart keeps JSON intact");
        foreach (var bad in new[] { "", "Vote|x|1|15|A", "Vote|1|2", "Restart|1|nope", "Other|1", "Cancel|1|Left" })
            Check(SprkPartyRetryPolicy.Parse("SprkRetry", bad).Kind == SprkPartyRetryPolicy.Kind.None, "Malformed accepted: " + bad);
        Check(SprkPartyRetryPolicy.Parse("Emoticon", "Vote|1|2|15|A").Kind == SprkPartyRetryPolicy.Kind.None, "Emotes are not votes");

        Func<string, string, string> text = (key, fallback) => key == "RAID_USER_OUT_CENTER_MSG" ? "Bo left the party." : fallback;
        Func<string, string> failure = code => code == "NotEnoughStamina" ? "Not enough stamina." : code;
        Check(SprkPartyRetryPolicy.CancelMessage(SprkPartyRetryPolicy.Parse("SprkRetry", "Cancel|1|Declined|Bo"), text, failure) == "Bo declined the retry.", "Declined text");
        Check(SprkPartyRetryPolicy.CancelMessage(SprkPartyRetryPolicy.Parse("SprkRetry", "Cancel|1|Left|Bo"), text, failure).StartsWith("Bo left the party."), "Native leave text");
        Check(SprkPartyRetryPolicy.CancelMessage(SprkPartyRetryPolicy.Parse("SprkRetry", "Cancel|1|Failed|NotEnoughStamina"), text, failure).EndsWith("Not enough stamina."), "Native failure text");
        Check(SprkPartyRetryPolicy.CancelMessage(SprkPartyRetryPolicy.Parse("SprkRetry", "Cancel|0|Unavailable|"), text, failure) == "Retry is not available right now.", "Unavailable text");
    }
}
