using System;
using System.Collections.Generic;
using System.Security.Cryptography;
using System.Text;
using System.Text.RegularExpressions;
using System.Runtime.CompilerServices;
using NVespa.NWeb;

// Optional capabilities: older servers ignore these headers and keep working.
public static class SprkNetworkPerformance {
    static readonly object gate = new object();
    static string scope, token, catalogToken;
    static readonly string[] sectionNames={"Achievement","MainQuest","ClearMission","PlayRecord"};
    static readonly string[] sectionFields={"AchievementInfos","ReservedMainQuestInfo","ReservedClearMissionInfos","ReservedPlayRecordInfos"};
    static readonly string[] sectionHeaders={"X-SPRK-Achievements","X-SPRK-Main-Quest","X-SPRK-Clear-Missions","X-SPRK-Play-Records"};
    static readonly string[] sectionTokens=new string[4];
    static readonly Regex[] sectionPatterns=Array.ConvertAll(sectionNames,n=>new Regex("\"Sprk"+n+"Token\"\\s*:\\s*\"([0-9a-f]{64})\""));
    static readonly Regex tokenField = new Regex("\"SprkProgressionToken\"\\s*:\\s*\"([0-9a-f]{64})\"");
    static readonly Regex catalogField = new Regex("\"SprkShopCatalogToken\"\\s*:\\s*\"([0-9a-f]{64})\"");
    sealed class Delivery { public string Context, Progression, Catalog; public bool FullProgression, FullCatalog; public string[] Sections; public bool[] FullSections; public WebServiceRequest Request; }
    sealed class Timing { public long Queued, Sent, Processing, Parsed; }
    static readonly ConditionalWeakTable<WebServiceRequest,Timing> timings=new ConditionalWeakTable<WebServiceRequest,Timing>();
    static readonly bool trace=Environment.GetEnvironmentVariable("SPRK_NETWORK_TRACE")=="1";
    static long Now { get { return System.Diagnostics.Stopwatch.GetTimestamp(); } }
    static double Ms(long ticks) {return ticks*1000.0/System.Diagnostics.Stopwatch.Frequency;}
    static void TraceLog(string message) {
        UnityEngine.Debug.Log(message);
        try {
            string path=Environment.GetEnvironmentVariable("SPRK_NETWORK_TRACE_FILE");
            if(String.IsNullOrEmpty(path))path=System.IO.Path.Combine(UnityEngine.Application.persistentDataPath,"sprk-network.log");
            // The game's release logger can suppress Debug.Log. Keep the opt-in
            // trace available without enabling unrelated verbose game logging.
            lock(gate)System.IO.File.AppendAllText(path,message+Environment.NewLine);
        } catch { /* Diagnostics must never interrupt a gameplay callback. */ }
    }

    public static void Started(WebServiceRequest request) {
        if(!trace || request==null)return;
        lock(gate) {Timing timing;if(timings.TryGetValue(request,out timing))timing.Sent=Now;}
    }
    public static void BeginResponse(WebServiceRequest request) {
        if(!trace || request==null)return;
        lock(gate) {Timing timing;if(timings.TryGetValue(request,out timing))timing.Processing=Now;}
    }
    public static void TransportComplete(UnityEngine.Networking.UnityWebRequest request) {
        if(!trace || request==null || !request.isDone || !String.IsNullOrEmpty(request.error))return;
        Uri uri;string path=Uri.TryCreate(request.url,UriKind.Absolute,out uri)?uri.AbsolutePath:"unknown";
        // Paths only: never log request bodies, hosts, query strings or sessions.
        TraceLog("SPRK network transport path="+path+" downloadedBytes="+request.downloadedBytes
            +" encoding="+(request.GetResponseHeader("Content-Encoding")??"identity")
            +" serverTiming="+(request.GetResponseHeader("Server-Timing")??"unknown"));
    }
    static readonly ConditionalWeakTable<object,Delivery> deliveries = new ConditionalWeakTable<object,Delivery>();
    static readonly ConditionalWeakTable<object,HashSet<int>> noStaminaClock = new ConditionalWeakTable<object,HashSet<int>>();

    // Zero means the server has no scheduled recharge (e.g. unlimited World Boss
    // attempts). Preserve native timer/UI fields and stop only automatic polling.
    // Positive results re-enable polling; negative results retain native state.
    public static void SetStaminaClock(object manager,int type,int next) {
        if(manager==null || next<0)return;
        lock(gate) {
            var disabled=noStaminaClock.GetOrCreateValue(manager);
            if(next==0)disabled.Add(type);else disabled.Remove(type);
        }
    }
    public static bool HasStaminaClock(object manager,int type) {
        if(manager==null)return true;
        lock(gate) {HashSet<int> disabled;return !noStaminaClock.TryGetValue(manager,out disabled) || !disabled.Contains(type);}
    }

    public static void Prepare(WebServiceRequest request, string session) {
        string context;
        using (var hash = SHA256.Create())
            context = BitConverter.ToString(hash.ComputeHash(Encoding.UTF8.GetBytes(request.Host + "\n" + session))).Replace("-", "");
        // Headers normally reference the requester's shared dictionary. Copy it
        // so queued/retried requests keep their own acknowledgement and scope.
        var headers = request.Headers == null ? new Dictionary<string,string>()
            : new Dictionary<string,string>(request.Headers);
        headers.Remove("X-SPRK-Progression");
        headers.Remove("X-SPRK-Compact-Lobby");
        headers.Remove("X-SPRK-Shop-Catalog");
        foreach(var header in sectionHeaders)headers.Remove(header);
        headers["X-SPRK-Sections"]="1";
        headers["X-SPRK-Context"] = context;
        lock (gate) {
            if (scope != context) { scope = context; token = catalogToken = null; Array.Clear(sectionTokens,0,sectionTokens.Length); }
            if (!String.IsNullOrEmpty(token)) headers["X-SPRK-Progression"] = token;
            if (!String.IsNullOrEmpty(catalogToken)) headers["X-SPRK-Shop-Catalog"] = catalogToken;
            for(int i=0;i<sectionTokens.Length;i++)if(sectionTokens[i]!=null)headers[sectionHeaders[i]]=sectionTokens[i];
        }
        string service = request.ServiceName.Trim('/');
        if (service == "enter/lobby" || service == "lobby/enter_lobby")
            headers["X-SPRK-Compact-Lobby"] = "1";
        request.Headers = headers;
        if(trace) {lock(gate){timings.Remove(request);timings.Add(request,new Timing{Queued=Now});}}
    }

    // Capture on the response worker, acknowledge only after the corresponding
    // native managers apply that exact response. Weak keys avoid retaining replies.
    public static void Capture(string text, WebServiceRequest request, object response) {
        if (text == null || request == null || request.Headers == null || response == null) return;
        string context;
        if (!request.Headers.TryGetValue("X-SPRK-Context", out context)) return;
        var match = tokenField.Match(text);
        var catalog = catalogField.Match(text);
        var sections=new string[4];var fullSections=new bool[4];
        for(int i=0;i<sections.Length;i++) {
            var section=sectionPatterns[i].Match(text);sections[i]=section.Success?section.Groups[1].Value:null;
            fullSections[i]=text.Contains("\""+sectionFields[i]+"\"");
        }
        lock (gate) {
            deliveries.Remove(response);
            deliveries.Add(response, new Delivery {Context=context,
                Request=request,
                Sections=sections,FullSections=fullSections,
                Progression=match.Success?match.Groups[1].Value:null, Catalog=catalog.Success?catalog.Groups[1].Value:null,
                FullCatalog=text.Contains("\"PayShopProductInfos\""),
                FullProgression=text.Contains("\"AchievementInfos\"") || text.Contains("\"ReservedMainQuestInfo\"")
                    || text.Contains("\"ReservedClearMissionInfos\"") || text.Contains("\"ReservedPlayRecordInfos\"")});
            Timing timing;if(trace && timings.TryGetValue(request,out timing))timing.Parsed=Now;
        }
    }
    public static void ProgressionApplied(object response) {
        if(response==null)return;
        lock(gate) {
            Delivery delivery;
            if(!deliveries.TryGetValue(response,out delivery) || delivery.Context!=scope)return;
            Timing timing;
            if(trace && timings.TryGetValue(delivery.Request,out timing)) {
                TraceLog(String.Format(System.Globalization.CultureInfo.InvariantCulture,
                    "SPRK network apply path={0} queueMs={1:F2} transportAndWorkerMs={2:F2} parseMs={3:F2} dispatchAndApplyMs={4:F2}",
                    delivery.Request.ServiceName.Trim('/'),Ms(timing.Sent-timing.Queued),Ms(timing.Processing-timing.Sent),Ms(timing.Parsed-timing.Processing),Ms(Now-timing.Parsed)));
                timings.Remove(delivery.Request);
            }
            if(delivery.Progression!=null)token=delivery.Progression;
            else if(delivery.FullProgression)token=null;
            if(delivery.FullCatalog && delivery.Catalog==null)catalogToken=null;
            for(int i=0;i<sectionTokens.Length;i++) {
                if(delivery.Sections[i]!=null)sectionTokens[i]=delivery.Sections[i];
                else if(delivery.FullSections[i])sectionTokens[i]=null;
            }
        }
    }
    public static void CatalogApplied(object response) {
        if(response==null)return;
        lock(gate) {
            Delivery delivery;
            if(deliveries.TryGetValue(response,out delivery) && delivery.Context==scope)catalogToken=delivery.Catalog;
        }
    }

    // A restock response already contains the authoritative purchase counters.
    // Feed that same result into the native count cache before the shop opens,
    // avoiding its second GetAllShopItemPurchaseCount request.
    public static void ShopListApplied(NShared.ShopListItemInfo[] items) {
        if(items==null || !NVespa.NSingleton.SceneAutoBehaviour<NGame2.NUI.NManager.ShopManagement>.isValidInstance)return;
        var grouped=new Dictionary<int,List<NShared.ShopItemInfo>>();
        foreach(var item in items) {
            if(item==null)continue;
            List<NShared.ShopItemInfo> list;
            if(!grouped.TryGetValue(item.ShopIndex,out list))grouped[item.ShopIndex]=list=new List<NShared.ShopItemInfo>();
            list.Add(new NShared.ShopItemInfo {ShopIndex=item.ShopIndex,ListNo=item.ListNo,Sold=item.Sold,Purchased=item.Purchased});
        }
        var manager=NVespa.NSingleton.SceneAutoBehaviour<NGame2.NUI.NManager.ShopManagement>.instance;
        foreach(var pair in grouped)manager.SetShopItemInfos(pair.Key,pair.Value.ToArray());
    }

    // ---- Request queue ----
    // The native queue sends one request at a time and spends frames between
    // them. These reads match the server's observation() list: they skip
    // mission notifications, carry no progression snapshot and only perform
    // idempotent self-healing writes, so their replies cannot be reordered
    // against gameplay state. Every other request remains exclusive and FIFO:
    // it starts only when nothing is in flight, and nothing overtakes it.
    static readonly HashSet<string> parallelReads=new HashSet<string>(StringComparer.Ordinal) {
        "ping","ping/idle","dispatch/get_dispatch_list","mail/check_new_mail",
        "mail/get_mail_list","mail/get_mail_list_by_page","mail/get_global_mail_list","mail/list",
        "friend/list","friend/get_friend_list","friend/search_friend",
        "match/get_season_info","global_arena/get_season_info","chat/info","chat/recent","query/session",
        "guild/get_guildbasicinfo","guild/get_all_guild_member_info"};
    public const int MaxParallelReads=4;
    static List<WebServiceRequest> queue;
    static string Service(WebServiceRequest request) {return request==null || request.ServiceName==null?"":request.ServiceName.Trim('/');}
    public static bool ParallelRead(WebServiceRequest request) {return parallelReads.Contains(Service(request));}

    // Replaces WebServiceRequester.ChainUpdate. Unlike the native loop it starts
    // the next request in the same frame the previous one finished.
    public static void Pump(List<WebServiceRequest> list) {
        if(list==null)return;
        queue=list;
        double expiry=WebServiceRequest.GetTimeout()+5.0;
        var now=DateTime.UtcNow;
        for(int i=list.Count-1;i>=0;i--) {
            var request=list[i];
            if(request==null || request.Started && (request.Ended || (now-request.RequestedTime).TotalSeconds>expiry))list.RemoveAt(i);
        }
        int active=0;
        for(int i=0;i<list.Count;i++) {
            var request=list[i];
            bool read=ParallelRead(request);
            if(request.Started) {
                if(!read)return;
                active++;continue;
            }
            if(read) {
                if(active>=MaxParallelReads)return;
                request.Start();active++;continue;
            }
            if(active==0)request.Start();
            return;
        }
    }
    // True when an identical service call is already queued or in flight.
    public static bool Pending(string service) {
        var list=queue;if(list==null)return false;
        foreach(var request in list)
            if(request!=null && (!request.Started || !request.Ended) && Service(request)==service)return true;
        return false;
    }
    // Two lobby paths request the same mail check back to back. The queued
    // request's callback updates the shared mail state for both callers.
    public static bool SkipMailCheck() {return Pending("mail/check_new_mail");}

    // query_market_url only drives store-upgrade popups, which the SPRK
    // launcher replaces; the server answers it with a generic success. Answer
    // locally with the same result and keep the native asynchronous callback.
    static readonly System.Reflection.MethodInfo marketDone=typeof(NGame2.NUtil.QueryMarketInfo).GetProperty("IsDone").GetSetMethod(true);
    static readonly System.Reflection.MethodInfo marketResponse=typeof(NGame2.NUtil.QueryMarketInfo).GetProperty("Response").GetSetMethod(true);
    public static bool AnswerMarketQuery(NGame2.NUtil.QueryMarketInfo query,Action onComplete) {
        if(query==null)return false;
        var response=new NShared.QueryMarketUrl.Response{BaseResult=NShared.BaseResultType.Success,Result=NShared.QueryMarketUrl.ResultType.Success};
        marketResponse.Invoke(query,new object[]{response});
        marketDone.Invoke(query,new object[]{true});
        if(onComplete!=null)NVespa.SingletonUpdater<NVespa.NUtil.Executer>.instance.NextRun(onComplete);
        return true;
    }
}
