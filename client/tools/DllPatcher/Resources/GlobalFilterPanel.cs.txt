    public const string GlobalFilterRowName="SPRK Global Filter";
    static void AddGlobalFilterRow(GameOptionWindow.TabUIInfo tab) {
        var existing=tab.GameOptionGrid.GetComponentsInChildren<GameOptionBaseComponent>(true).Where(r=>r.name==GlobalFilterRowName).ToArray();

        row.name=GlobalFilterRowName;row.enabled=false;

            if(row.ParentRoot.Root!=null)row.ParentRoot.Root.SetActive(true);

        }
        if(row.ChildRoot!=null && row.ChildRoot.Root!=null && row.ChildRoot.Root!=(row.ParentRoot==null?null:row.ParentRoot.Root))row.ChildRoot.Root.SetActive(false);
        RefreshGlobalFilterRow(row);

    }

    }

    // ---- Thumbnails: an ungraded snapshot of the world, graded per preset ----
    public static bool FilterPanelOpen;
    static int thumbFilterLayer=-1;
    public static int ThumbFilterMask { get { return thumbFilterLayer<0?0:1<<thumbFilterLayer; } }
    RenderTexture filterSnapshot;
    UnityEngine.Rendering.CommandBuffer snapshotCommand;
    Camera snapshotOwner;
    int snapshotFrame=-100;
    FilterStack thumbFilter;
    Camera thumbCamera;
    Texture2D fallbackSnapshot;
    void UpdateFilterSnapshot(Camera worldCompanion) {
        if(!FilterPanelOpen || worldCompanion==null){ReleaseFilterSnapshotCommand();return;}
        int width=Mathf.Max(64,Screen.width/4),height=Mathf.Max(36,Screen.height/4);
        if(filterSnapshot==null || filterSnapshot.width!=width || filterSnapshot.height!=height) {
            ReleaseFilterSnapshotCommand();
            if(filterSnapshot!=null){filterSnapshot.Release();Destroy(filterSnapshot);}
            filterSnapshot=new RenderTexture(width,height,0,RenderTextureFormat.ARGB32){name="SPRK Global Filter Snapshot",filterMode=FilterMode.Bilinear};
            filterSnapshot.Create();
        }
        if(snapshotOwner!=worldCompanion) {
            ReleaseFilterSnapshotCommand();
            snapshotCommand=new UnityEngine.Rendering.CommandBuffer{name="SPRK Global Filter snapshot"};
            snapshotCommand.Blit(UnityEngine.Rendering.BuiltinRenderTextureType.CameraTarget,filterSnapshot);
            // Before the companion's own image effects: the unfiltered world.
            worldCompanion.AddCommandBuffer(UnityEngine.Rendering.CameraEvent.BeforeImageEffectsOpaque,snapshotCommand);
            snapshotOwner=worldCompanion;
        }
        snapshotFrame=Time.frameCount;
    }
    void ReleaseFilterSnapshotCommand() {
        if(snapshotCommand!=null){if(snapshotOwner!=null)snapshotOwner.RemoveCommandBuffer(UnityEngine.Rendering.CameraEvent.BeforeImageEffectsOpaque,snapshotCommand);snapshotCommand.Release();}
        snapshotCommand=null;snapshotOwner=null;
    }
    void ReleaseFilterThumbnails() {
        ReleaseFilterSnapshotCommand();
        if(filterSnapshot!=null){filterSnapshot.Release();Destroy(filterSnapshot);filterSnapshot=null;}
        snapshotFrame=-100;
    }
    public static bool HasFilterSnapshot { get { return instance!=null && instance.filterSnapshot!=null && Time.frameCount-instance.snapshotFrame<5; } }
    public static float SnapshotAspect { get { return Screen.height<=0?16f/9f:Screen.width/(float)Screen.height; } }
    Texture FallbackSnapshot() {
        if(fallbackSnapshot!=null)return fallbackSnapshot;
        // Shown when no 3D world is on screen (login, full-screen menus).
        const int w=256,h=144;
        fallbackSnapshot=new Texture2D(w,h,TextureFormat.RGBA32,false){wrapMode=TextureWrapMode.Clamp,hideFlags=HideFlags.DontSave};
        for(int y=0;y<h;y++)for(int x=0;x<w;x++) {
            float u=x/(w-1f),v=y/(h-1f);
            Color sky=Color.Lerp(new Color(.95f,.78f,.55f),new Color(.32f,.58f,.86f),Mathf.SmoothStep(.35f,1f,v));
            float far=.46f+.06f*Mathf.Sin(u*9f+1.3f)+.03f*Mathf.Sin(u*23f);
            float near=.3f+.07f*Mathf.Sin(u*5f+.4f)+.025f*Mathf.Sin(u*17f+2f);
            Color c=sky;
            c=Color.Lerp(c,new Color(.42f,.5f,.62f),Mathf.Clamp01((far-v)*h*.5f));
            c=Color.Lerp(c,Color.Lerp(new Color(.18f,.3f,.17f),new Color(.4f,.56f,.3f),v/.35f),Mathf.Clamp01((near-v)*h*.5f));
            fallbackSnapshot.SetPixel(x,y,c);
        }
        fallbackSnapshot.Apply();return fallbackSnapshot;
    }
    public static void RenderFilterThumbnail(RenderTexture target,int preset,FilterValues values) { if(instance!=null)instance.RenderThumbnail(target,preset,values); }
    void RenderThumbnail(RenderTexture target,int preset,FilterValues values) {
        if(target==null || !FindResources() || !EnsureFilterLayers())return;
        if(thumbFilter==null)thumbFilter=CreateFilterStack("SPRK Global Filter (Thumbnail)",thumbFilterLayer,true);
        if(thumbCamera==null) {
            var host=new GameObject("SPRK Global Filter Thumbnail Camera");host.transform.SetParent(transform,false);host.SetActive(false);
            thumbCamera=host.AddComponent<Camera>();thumbCamera.enabled=false;
            thumbCamera.clearFlags=CameraClearFlags.Nothing;thumbCamera.cullingMask=0;thumbCamera.orthographic=true;thumbCamera.nearClipPlane=.01f;thumbCamera.farClipPlane=1;
            thumbCamera.allowHDR=false;thumbCamera.allowMSAA=false;thumbCamera.useOcclusionCulling=false;
            var layer=host.AddComponent<PostProcessLayer>();layer.Init(resources);layer.volumeTrigger=host.transform;layer.volumeLayer=ThumbFilterMask;
            layer.antialiasingMode=PostProcessLayer.Antialiasing.None;layer.fog.enabled=false;layer.stopNaNPropagation=false;
            thumbCamera.AddCommandBuffer(UnityEngine.Rendering.CameraEvent.AfterImageEffects,OpaqueAlphaCommand());
            filterCameraSet.Add(thumbCamera);
            host.SetActive(true);
        }
        Graphics.Blit(HasFilterSnapshot?(Texture)filterSnapshot:FallbackSnapshot(),target);
        ApplyFilterValues(thumbFilter,preset,values);
        thumbCamera.targetTexture=target;thumbCamera.Render();thumbCamera.targetTexture=null;
    }
}

    // The window's root object can sit on a layer no UI camera renders (lobby windows do);
    // use the layer its visible widgets are drawn on.
    static int RenderedLayer(GameOptionWindow window) {
        foreach(var widget in window.GetComponentsInChildren<UIWidget>(false))
            if(NGUITools.FindCameraForLayer(widget.gameObject.layer)!=null)return widget.gameObject.layer;
        foreach(var widget in window.GetComponentsInChildren<UIWidget>(true))
            if(NGUITools.FindCameraForLayer(widget.gameObject.layer)!=null)return widget.gameObject.layer;
        return window.gameObject.layer;
    }
    // Hide the settings window and the other UI windows on its layer (lobby HUD, menus) so the
    // scene is visible, and let the lobby world map render live instead of its blurred stand-in.
    void HideSceneUi() {
        var scope=window.transform.parent!=null?window.transform.parent:window.transform;
        foreach(var p in scope.GetComponentsInChildren<UIPanel>(true)) {
            bool own=p.transform.IsChildOf(window.transform);
            if(!own && (p.gameObject.layer!=windowLayer ||p.GetComponentInParent<NGame2.NUI.NWindow.LoginBackground>()!=null))continue;
            hiddenPanels.Add(new KeyValuePair<UIPanel,float>(p,p.alpha));p.alpha=0;
        }
        foreach(var w in scope.GetComponentsInChildren<NGame2.NUI.NWindow.BaseWindow>(true))
            if(w.HideWorldMap){w.HideWorldMap=false;worldMapHiders.Add(w);}
    }
    void RestoreSceneUi() {
        foreach(var pair in hiddenPanels)if(pair.Key!=null)pair.Key.alpha=pair.Value;
        hiddenPanels.Clear();

        worldMapHiders.Clear();

            bool chosen=dx==0 && dy==0,near=Mathf.Abs(dx)+Mathf.Abs(dy)==1,line=dx==0 || dy==0;
            int size=chosen?26:near?16:line?12:10;
            var dot=dots[x,y];dot.width=size;dot.height=size;
            dot.color=new Color(1,1,1,chosen||near?1f:line?.85f:.45f);
        }
        originRing.alpha=values.X==0 && values.Y==0?0f:.9f;
        foreach(var slider in sliders)UpdateSlider(slider,values,true);
        UpdateSlider(intensity,values,hasIntensity);
        bool dirty=Dirty;
        // Disabled until something changes, the way the game greys out unavailable buttons.
        applyButton.spriteName=dirty?EmphasisSprite:DisabledSprite;applyLabel.alpha=dirty?1f:.7f;
        if(thumbnails)thumbTimer=0;
    }
    void RenderThumbs() {
        int count=SprkGraphicsOptions.FilterPresetNames.Length;
        for(int i=0;i<3;i++) {
            int preset=((selected+i-1)%count+count)%count;
            SprkGraphicsOptions.RenderFilterThumbnail(thumbs[i],preset,working[preset]);
        }
    }
    void Update() {
        if(window==null || !window.gameObject.activeInHierarchy || !window.IsOpen()){Close();return;}
        if(Screen.width!=(int)screenSize.x || Screen.height!=(int)screenSize.y) {
            var root=NGUITools.FindInParents<UIRoot>(gameObject);if(root!=null){Layout(root);Sync(true);}
        }
        foreach(var pair in hiddenPanels)if(pair.Key!=null && pair.Key.alpha>0)pair.Key.alpha=0;
        if(Input.GetKeyDown(KeyCode.Escape)){Close();return;}
        if(Input.GetKeyDown(KeyCode.A))Step(-1);
        if(Input.GetKeyDown(KeyCode.D))Step(1);
        if(thumbTimer--<=0){RenderThumbs();thumbTimer=15;}
    }

    // ---- Sliders and palette ----------------------------------------------
    Slider CreateSlider(Transform parent,int field,Vector2 center,float width,bool scroll) {
        var info=SprkGraphicsOptions.FilterFields[field];
        var s=new Slider{field=field,width=width,bipolar=info.Min<0};
        // Each slider sits in a native option-row box, with the game's sound-volume slider parts.
        Sprite(parent,info.Name+" Box",BoxSprite,(int)width+90,88,center,3);
        s.name=Label(parent,info.Name,28,center+new Vector2(-width*.5f,20f),8,UIWidget.Pivot.Left,nameStyle);
        s.value=Label(parent,"0",28,center+new Vector2(width*.5f,20f),8,UIWidget.Pivot.Right,valueStyle);
        bool temperature=field==2;
        var trackCenter=center+new Vector2(0,-18f);
        s.track=Sprite(parent,info.Name+" Track",TrackSprite,(int)width,24,trackCenter,4);
        if(temperature)Tex(parent,info.Name+" Gradient",temperatureTrack,(int)width-10,14,trackCenter,Color.white,5);
        var fill=Sprite(parent,info.Name+" Fill",FillSprite,10,16,trackCenter,5);fill.pivot=UIWidget.Pivot.Left;s.fill=fill;
        if(temperature)s.fill.alpha=0;
        s.knob=Sprite(parent,info.Name+" Knob",KnobSprite,32,44,trackCenter,6,false);
        s.hit=Widget<UIWidget>(parent,info.Name+" Input",(int)width+30,54,trackCenter,9);
        NGUITools.AddWidgetCollider(s.hit.gameObject);
        var listener=UIEventListener.Get(s.hit.gameObject);
        listener.onPress=delegate(GameObject g,bool down){if(down)Drag(s);};
        listener.onDrag=delegate(GameObject g,Vector2 d){Drag(s);};
        if(scroll) {
            listener.onScroll=delegate(GameObject g,float delta){if(list!=null)list.Scroll(delta);};
            // The row outside the track drags the list.
            var rowArea=Widget<UIWidget>(parent,info.Name+" Row",(int)width+90,88,center,2);
            NGUITools.AddWidgetCollider(rowArea.gameObject);rowArea.gameObject.AddComponent<UIDragScrollView>().scrollView=list;
        }
        return s;
    }
    void Drag(Slider s) {
        if(UICamera.currentCamera==null)return;
        var info=SprkGraphicsOptions.FilterFields[s.field];
        var world=UICamera.currentCamera.ScreenToWorldPoint(UICamera.lastTouchPosition);
        float x=s.track.transform.InverseTransformPoint(world).x;
        float t=Mathf.Clamp01(x/s.width+.5f);
        SetField(s.field,Mathf.RoundToInt(Mathf.Lerp(info.Min,info.Max,t)));
    }
    void UpdateSlider(Slider s,SprkGraphicsOptions.FilterValues values,bool enabled) {
        var info=SprkGraphicsOptions.FilterFields[s.field];
        int value=values.Get(s.field);
        float t=(value-info.Min)/(float)(info.Max-info.Min);
        s.value.text=SprkGraphicsOptions.FilterValueLabel(s.field,value);
        var p=s.track.transform.localPosition;
        // Two-sided sliders fill outward from the centre; at 0 nothing is filled.
        float from=s.bipolar?Mathf.Min(t,.5f):0f,to=s.bipolar?Mathf.Max(t,.5f):t;
        bool filled=to-from>.001f;
        s.fill.width=Mathf.Max(10,Mathf.RoundToInt((to-from)*(s.width-8f)));
        s.fill.transform.localPosition=new Vector3(p.x-s.width*.5f+4f+from*(s.width-8f),p.y,0);
        s.knob.transform.localPosition=new Vector3(p.x-s.width*.5f+t*s.width,p.y,0);
        float a=enabled?1f:.35f;
        s.name.alpha=a;s.value.alpha=a;s.knob.alpha=a;s.track.alpha=a;if(s.field!=2)s.fill.alpha=filled?a:0f;
        var collider=s.hit.GetComponent<Collider>();if(collider!=null)collider.enabled=enabled;
    }
    void PickPalette(UITexture pad) {
        if(UICamera.currentCamera==null)return;
        var local=pad.transform.InverseTransformPoint(UICamera.currentCamera.ScreenToWorldPoint(UICamera.lastTouchPosition));
        SetPalette(Mathf.Clamp(Mathf.RoundToInt(local.x/DotSpacing),-5,5),Mathf.Clamp(Mathf.RoundToInt(local.y/DotSpacing),-5,5));
    }

    UISprite Sprite(Transform parent,string name,string sprite,int width,int height,Vector2 position,int depth,bool sliced=true) {
        var s=Widget<UISprite>(parent,name,width,height,position,depth);
        var atlas=atlases.FirstOrDefault(a=>a.GetSprite(sprite)!=null);
        if(atlas==null){if(!missingSprites.Contains(sprite))missingSprites.Add(sprite);}
        else{s.atlas=atlas;s.spriteName=sprite;}
        s.type=sliced?UIBasicSprite.Type.Sliced:UIBasicSprite.Type.Simple;
        s.width=width;s.height=height;return s;
    }

    // Copy a native label's look (colour, gradient, outline) so text matches the game.
    static void Style(UILabel label,UILabel source) {
        if(label==null)return;
        if(source==null){label.color=Color.white;label.effectStyle=UILabel.Effect.Outline8;label.effectColor=Color.black;label.applyGradient=false;return;}
        label.color=source.color;label.applyGradient=source.applyGradient;label.gradientTop=source.gradientTop;label.gradientBottom=source.gradientBottom;
        label.effectStyle=source.effectStyle;label.effectColor=source.effectColor;label.effectDistance=source.effectDistance;
        label.spacingX=source.spacingX;label.fontStyle=source.fontStyle;
    }
    UILabel Label(Transform parent,string text,int size,Vector2 position,int depth,UIWidget.Pivot pivot,UILabel style) {
        var go=Child(parent,"Label "+text);go.transform.localPosition=position;
        var label=go.AddComponent<UILabel>();
        var font=style!=null && (style.bitmapFont!=null || style.trueTypeFont!=null)?style:null;
        if(font!=null){label.bitmapFont=font.bitmapFont;if(font.bitmapFont==null)label.trueTypeFont=font.trueTypeFont;}
        else if(bitmapFont!=null)label.bitmapFont=bitmapFont;else if(trueTypeFont!=null)label.trueTypeFont=trueTypeFont;
        label.fontSize=size;label.overflowMethod=UILabel.Overflow.ResizeFreely;label.pivot=pivot;label.depth=depth;
        label.alignment=pivot==UIWidget.Pivot.Left?NGUIText.Alignment.Left:pivot==UIWidget.Pivot.Right?NGUIText.Alignment.Right:NGUIText.Alignment.Center;
        Style(label,style);
        label.text=text;
        go.transform.localPosition=position;
        return label;
    }

    static Texture2D NewTexture(int w,int h) { return new Texture2D(w,h,TextureFormat.RGBA32,false){wrapMode=TextureWrapMode.Clamp,filterMode=FilterMode.Bilinear,hideFlags=HideFlags.DontSave}; }
    static void EnsureTextures() {
        if(white!=null)return;
        white=NewTexture(4,4);for(int y=0;y<4;y++)for(int x=0;x<4;x++)white.SetPixel(x,y,Color.white);white.Apply();
        circle=Disc(64,0f);ring=Disc(64,.72f);
        eyeIcon=NewTexture(96,60);
        for(int y=0;y<60;y++)for(int x=0;x<96;x++) {
            float u=(x-47.5f)/46f,v=(y-29.5f)/29f;
            float lid=(1f-u*u)*.92f,edge=Mathf.Abs(Mathf.Abs(v)-lid)*29f;
            float a=Mathf.Abs(u)<=1f?Mathf.Clamp01(3.2f-edge):0f;
            float r=Vector2.Distance(new Vector2(x,y),new Vector2(47.5f,29.5f));
            a=Mathf.Max(a,Mathf.Clamp01(15.5f-r));
            if(r<6f)a=Mathf.Min(a,Mathf.Clamp01(r-4.5f));
            eyeIcon.SetPixel(x,y,new Color(1,1,1,a));
        }
        eyeIcon.Apply();
        // Measured from the reference: X = saturation (grey -> cyan), Y = lightness.
        padGradient=NewTexture(64,64);
        Color tl=new Color32(175,185,189,255),tr=new Color32(119,185,207,255),bl=new Color32(34,38,39,255),br=new Color32(27,43,63,255);
        for(int y=0;y<64;y++)for(int x=0;x<64;x++){float u=x/63f,v=y/63f;padGradient.SetPixel(x,y,Color.Lerp(Color.Lerp(bl,br,u),Color.Lerp(tl,tr,u),v));}
        padGradient.Apply();
        temperatureTrack=NewTexture(256,4);
        for(int x=0;x<256;x++){float u=x/255f;Color c=u<.5f?Color.Lerp(new Color(.36f,.66f,.86f),new Color(.6f,.8f,.75f),u*2f):Color.Lerp(new Color(.6f,.8f,.75f),new Color(.88f,.6f,.24f),(u-.5f)*2f);for(int y=0;y<4;y++)temperatureTrack.SetPixel(x,y,c);}
        temperatureTrack.Apply();
    }
    static Texture2D Disc(int size,float hole) {
        var t=NewTexture(size,size);float c=(size-1)*.5f,r=size*.5f;
        for(int y=0;y<size;y++)for(int x=0;x<size;x++) {
            float d=Vector2.Distance(new Vector2(x,y),new Vector2(c,c));
            float a=Mathf.Clamp01(r-d);if(hole>0)a=Mathf.Min(a,Mathf.Clamp01(d-r*hole));
            t.SetPixel(x,y,new Color(1,1,1,a));
        }
        t.Apply();return t;
    }
