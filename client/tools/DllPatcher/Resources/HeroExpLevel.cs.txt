using System;
using System.Reflection;
using UnityEngine;

using NVespa.NGlobal;

// Which potion counts reach which hero levels. Pure arithmetic; the patcher
// executes it outside Unity. levelTotal(L) is the cumulative EXP at level L.
public sealed class SprkHeroExpLevelPlan {
    readonly long current, max, perPotion;
    readonly int maxPotions, fromLevel, maxLevel;
    readonly Func<int,long> levelTotal;
    public readonly int MinLevel, TopLevel;

    public SprkHeroExpLevelPlan(long current, long max, long perPotion, int maxPotions, int fromLevel, int maxLevel, Func<int,long> levelTotal) {
        this.current = current; this.max = max; this.perPotion = Math.Max(1, perPotion);
        this.maxPotions = Math.Max(1, maxPotions); this.fromLevel = fromLevel; this.maxLevel = Math.Max(fromLevel, maxLevel);
        this.levelTotal = levelTotal;
        MinLevel = LevelAfter(Potions(Math.Min(this.maxLevel, fromLevel + 1)));
        TopLevel = LevelAfter(this.maxPotions);
    }
    // At least one level must be reachable for a level choice to make sense.
    public bool CanChoose { get { return TopLevel > fromLevel; } }
    public long TotalAfter(int potions) { return Math.Min(max, current + perPotion * (long)potions); }
    public int LevelAfter(int potions) {
        long total = TotalAfter(potions);
        int level = fromLevel;
        while (level < maxLevel && levelTotal(level + 1) <= total) level++;
        return level;
    }
    // Fewest potions that reach at least this level.
    public int Potions(int level) {
        long need = levelTotal(level) - current;
        long count = need <= 0 ? 1 : (need + perPotion - 1) / perPotion;
        return (int)Math.Max(1, Math.Min(maxPotions, count));
    }
    // The next reachable level in the step's direction. One potion can cross
    // several levels, so not every level is reachable.
    public int Step(int level, int delta) {
        int wanted = Math.Max(MinLevel, Math.Min(TopLevel, level + delta));
        int count = Potions(wanted);
        int reached = LevelAfter(count);
        if (delta < 0 && reached > wanted) reached = count > 1 ? LevelAfter(count - 1) : MinLevel;
        return Math.Max(MinLevel, Math.Min(TopLevel, reached));
    }
}

// Tapping a hero slot in the EXP Potion window opens the native "use EXP
// potions" confirmation with level steppers. The native Max confirmation
// closure keeps the potion count and target EXP; Yes still runs native UseMax.
public sealed class SprkHeroExpLevel : MonoBehaviour {
    static readonly FieldInfo YesField = typeof(LevelUpMaxPopup).GetField("_onYes", BindingFlags.Instance | BindingFlags.NonPublic);
    static readonly int[] Steps = { -10, -1, 1, 10 };
    static SprkHeroExpLevel runner;

    HeroGetExpComponent pendingSlot; int pendingPerPotion; float pendingUntil;

    LevelUpMaxPopup popup; Action boundYes; object closure;
    FieldInfo countField, nextTotalField;
    SprkHeroExpLevelPlan plan; int level; long perPotion; string itemName = "";
    GameObject[] buttons;

    // Called by the patched tap path right after native OnClickMax.
    public static void Attach(HeroGetExpComponent slot, int perPotion) {
        if (slot == null || perPotion <= 0) return;
        if (runner == null) {
            var go = new GameObject("SPRK Hero EXP Level");
            DontDestroyOnLoad(go);
            runner = go.AddComponent<SprkHeroExpLevel>();
        }
        runner.pendingSlot = slot; runner.pendingPerPotion = perPotion; runner.pendingUntil = Time.realtimeSinceStartup + 3f;
        runner.TryBind();
    }

    void Update() {
        if (pendingSlot != null) {
            if (Time.realtimeSinceStartup > pendingUntil) pendingSlot = null;
            else TryBind();
        }
        // Closed or reopened by another caller (e.g. the Max button): hide.
        if (popup != null && (!popup.gameObject.activeInHierarchy || !ReferenceEquals(YesField.GetValue(popup), boundYes))) Unbind();
    }

    void TryBind() {
        var slot = pendingSlot;
        if (slot == null || YesField == null) return;
        foreach (var candidate in Resources.FindObjectsOfTypeAll<LevelUpMaxPopup>()) {
            if (candidate == null || !candidate.gameObject.activeInHierarchy) continue;
            var yes = YesField.GetValue(candidate) as Action;
            if (yes == null || yes.Target == null) continue;
            var type = yes.Target.GetType();
            var owner = type.GetField("<>4__this");
            if (owner == null || !ReferenceEquals(owner.GetValue(yes.Target), slot)) continue;
            pendingSlot = null;
            Bind(candidate, yes, slot);
            return;
        }
    }

    void Bind(LevelUpMaxPopup target, Action yes, HeroGetExpComponent slot) {
        Unbind();
        var type = yes.Target.GetType();
        countField = type.GetField("usablePotion"); nextTotalField = type.GetField("nextHeroTotalExp");
        var currentField = type.GetField("curHeroTotalExp"); var maxField = type.GetField("MaxHeroExp"); var itemField = type.GetField("itemInfo");
        var hero = slot.HeroInfo; var levels = GameRoot.Instance == null ? null : GameRoot.Instance.CreatureLevelDataContainer;
        if (countField == null || nextTotalField == null || currentField == null || maxField == null || itemField == null || hero == null || levels == null) return;
        perPotion = pendingPerPotion;
        plan = new SprkHeroExpLevelPlan((long)currentField.GetValue(yes.Target), (long)maxField.GetValue(yes.Target), perPotion,
            (int)countField.GetValue(yes.Target), hero.Level, hero.GetMaxLevel(), l => levels.GetTotalExp(l, 0));
        // Not enough potions for any level: keep the plain native confirmation.
        if (!plan.CanChoose) { plan = null; return; }
        var item = itemField.GetValue(yes.Target) as ItemInfo;
        var data = item == null || GameRoot.Instance.ItemDataContainer == null ? null : GameRoot.Instance.ItemDataContainer.GetData(item.ItemIndex);
        itemName = data == null ? "" : data.Name;
        popup = target; boundYes = yes; closure = yes.Target;
        BuildButtons();
        level = plan.TopLevel;
        Apply();
    }

    void Unbind() {
        if (buttons != null) foreach (var b in buttons) if (b != null) Destroy(b);
        buttons = null; popup = null; boundYes = null; closure = null; plan = null;
    }

    void BuildButtons() {
        var root = popup.transform;
        var minus = Find(root, "Button_No"); var plus = Find(root, "Button_Yes");
        var after = popup.Label_AfterLevel; var before = popup.Label_BeforeLevel;
        if (minus == null || plus == null || after == null || before == null) return;
        var parent = minus.parent;
        // Lay out in the popup's units: the before/after level labels set the scale.
        var afterGroup = after.transform.parent != null ? after.transform.parent : after.transform;
        var beforeGroup = before.transform.parent != null ? before.transform.parent : before.transform;
        Vector3 center = parent.InverseTransformPoint(Middle(afterGroup));
        float span = Mathf.Abs(center.x - parent.InverseTransformPoint(Middle(beforeGroup)).x);
        if (span < 1f) span = 240f;
        buttons = new GameObject[Steps.Length];
        for (int i = 0; i < Steps.Length; i++) {
            int delta = Steps[i];
            var go = (GameObject)Instantiate((delta < 0 ? minus : plus).gameObject, parent, false);
            go.name = "SprkLevelStep" + (delta < 0 ? "Minus" : "Plus") + Math.Abs(delta);
            foreach (var behaviour in go.GetComponentsInChildren<MonoBehaviour>(true))
                // Drop native click forwarding and localization that would reset "Yes"/"No".
                if (behaviour is UIButtonMessage || behaviour is UIEventTrigger || behaviour is UIEventListener || behaviour is UIForwardEvents
                    || behaviour.GetType().Name.IndexOf("Locali", StringComparison.Ordinal) >= 0) DestroyImmediate(behaviour);
            var button = go.GetComponent<UIButton>();
            if (button != null) { button.onClick.Clear(); EventDelegate.Add(button.onClick, delegate { Change(delta); }); }
            float offset = (Math.Abs(delta) == 1 ? 0.36f : 0.64f) * span * Math.Sign(delta);
            go.transform.localPosition = new Vector3(center.x + offset, center.y, go.transform.localPosition.z);
            var widget = go.GetComponent<UIWidget>();
            if (widget != null) { widget.width = Mathf.RoundToInt(span * 0.25f); widget.height = Mathf.RoundToInt(span * 0.16f); }
            var label = go.GetComponentInChildren<UILabel>(true);
            if (label != null) {
                label.text = (delta > 0 ? "+" : "−") + Math.Abs(delta);
                label.overflowMethod = UILabel.Overflow.ShrinkContent;
                label.width = Mathf.RoundToInt(span * 0.22f); label.height = Mathf.RoundToInt(span * 0.13f);
                label.transform.localPosition = Vector3.zero;
            }
            NGUITools.UpdateWidgetCollider(go);
            go.SetActive(true);
            buttons[i] = go;
        }
    }

    static Vector3 Middle(Transform group) {
        var labels = group.GetComponentsInChildren<UILabel>(true);
        if (labels.Length == 0) return group.position;
        Vector3 sum = Vector3.zero;
        foreach (var l in labels) sum += l.transform.position;
        return sum / labels.Length;
    }

    static Transform Find(Transform root, string name) {
        foreach (var t in root.GetComponentsInChildren<Transform>(true)) if (t.name == name) return t;
        return null;
    }

    void Change(int delta) {
        if (plan == null) return;
        level = plan.Step(level, delta);
        Apply();
    }

    void Apply() {
        if (plan == null || closure == null) return;
        int count = plan.Potions(level);
        countField.SetValue(closure, count);
        nextTotalField.SetValue(closure, plan.TotalAfter(count));
        UIHelper.SetLabelText(popup.Label_AfterLevel, plan.LevelAfter(count), false);
        UIHelper.SetLabelText(popup.Label_Content1, LocalizationManager.GetStringDirect("EXP_USE_POPUP_MESSAGE_1", new object[] { itemName, count }), false);
        UIHelper.SetLabelText(popup.Label_Content2, LocalizationManager.GetStringDirect("EXP_USE_POPUP_MESSAGE_2", new object[] { perPotion * (long)count }), false);
        if (buttons != null)
            for (int i = 0; i < buttons.Length; i++) {
                var button = buttons[i] == null ? null : buttons[i].GetComponent<UIButton>();
                if (button != null) button.isEnabled = Steps[i] < 0 ? level > plan.MinLevel : level < plan.TopLevel;
            }
    }
}
