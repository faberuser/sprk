using System;
using System.Linq;
using System.Reflection;
using NGame2.NAccount;
using NGame2.NUI.NComponent2;
using NGame2.NUI.NWindow;
using NShared;
using NVespa.NSingleton;
using UnityEngine;

public static class SprkDispatchSelection {
    static readonly MethodInfo ForcedHero=typeof(BattlePartySetting).GetMethod("IsForcedHero",BindingFlags.Instance|BindingFlags.NonPublic);
    public static bool IsDispatched(int hero) {
        var account=GlobalAutoBehaviour<AccountManager>.instance;
        var manager=account==null?null:account.DispatchBattleInfoManager;
        return manager!=null && manager.GetByHeroIndex(hero)!=null;
    }
    public static int[] AvailableHeroes(int[] heroes) {
        return heroes==null?null:heroes.Where(id=>!IsDispatched(id)).ToArray();
    }
    public static bool CanUseHeroes(int[] heroes) {
        if(heroes==null || !heroes.Any(IsDispatched))return true;
        MessagePopup.Open("Hero dispatched", "A selected Hero is dispatched and cannot enter this battle. Select another Hero or cancel the dispatch first.");
        return false;
    }
    public static bool CanRegister(NShared.RegisterMatch.Request request) {
        // Friendly readiness is checked before notifying the partner. If the
        // dispatch changes afterward, let the server reject registration and
        // cancel the peer's queue together; silently dropping it here strands it.
        return request==null || request.DuelType==MatchDuelType.Friendly || CanUseHeroes(request.HeroIndices);
    }
    public static bool CanUseParty(BattlePartySetting window) {
        var account=GlobalAutoBehaviour<AccountManager>.instance;
        long mine=account.UserInfo==null?0:account.UserInfo.AccountId;
        var heroes=window.GetAllSquardHeroPortraitAnimList().Where(a=>a!=null && a.Component!=null)
            .Select(a=>a.Component).Where(p=>p.HeroInfo!=null && (p.OwnerId==0 || p.OwnerId==mine)
                && !(bool)ForcedHero.Invoke(window,new object[]{p.HeroInfo.HeroIndex}))
            .Select(p=>p.HeroInfo.HeroIndex).ToArray();
        return CanUseHeroes(heroes);
    }
    public static bool CanClick(HeroPortraitComponent portrait) {
        // Dispatch still permits viewing a Hero's information outside party prep.
        if(!portrait.IsDispatched || portrait.HeroInfo==null)return true;
        var window=portrait.GetComponentInParent<BattlePartySetting>();
        return window==null || window.UserHeroListPanel==null
            || window.UserHeroListPanel.GetUserHeroPortraitComponent(portrait.OwnerId,portrait.HeroInfo.HeroIndex)!=portrait;
    }
    public static void RefreshList(UserHeroListPanel panel) {panel.SetDispatch(true);}
    public static void FitList(UserHeroListPanel panel) {
        var account=GlobalAutoBehaviour<AccountManager>.instance;
        long mine=account.UserInfo==null?0:account.UserInfo.AccountId;
        panel.ForeachAction(p=>{
            if(p==null || p.HeroInfo==null)return;
            var window=p.GetComponentInParent<BattlePartySetting>();
            bool loaned=p.OwnerId!=0 && p.OwnerId!=mine;
            bool forced=window!=null && (bool)ForcedHero.Invoke(window,new object[]{p.HeroInfo.HeroIndex});
            if(p.IsDispatched && (loaned || forced))p.SetDispatched(false,false,"");
            RestoredPortal.FitHeroDeckDispatchLabel(p);
        });
    }
    public static void RefreshOpenLists() {
        foreach(var window in UnityEngine.Object.FindObjectsOfType<BattlePartySetting>())
            if(window.UserHeroListPanel!=null)window.UserHeroListPanel.SetDispatch(true);
    }
}
